   193→function useReportQuery<T>(type: ReportType, date?: string) {
   194→  return useQuery<T>({
   195→    queryKey: ['front-desk-reports', type, date],
   196→    queryFn: async () => {
   197→      const params = new URLSearchParams({ type })
   198→      if (date) params.set('date', date)
   199→      return apiFetch(`/api/front-desk/reports?${params.toString()}`)
   200→    },
   201→    refetchInterval: 30000,
   202→  })
   203→}

   328→      {source.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}

   330→  )

   461→// 2. Arrivals Report
   462→function ArrivalsReport({ date }: { date?: string }) {
   463→  const { data, isLoading, error } = useReportQuery<ArrivalsData>('arrivals', date)
   464→
   465→  if (isLoading) return <TableSkeleton cols={6} rows={6} />
   466→  if (error) return <EmptyState icon={ArrowDownToLine} message="Failed to load arrivals report" />
   467→
   468→  const arrivalsData = data as ArrivalsData
   469→  const arrivals = arrivalsData.reservations || []
   470→
   471→  if (arrivals.length === 0) {
   472→    return <EmptyState icon={ArrowDownToLine} message="No arrivals scheduled for this date" />
   473→  }
   474→
   475→  return (
   476→    <Card className="py-0">
   477→      <CardContent className="p-0">
   478→        <Table>
   479→          <TableHeader>
   480→            <TableRow>
   481→              <TableHead className="text-xs">Guest</TableHead>
   482→              <TableHead className="text-xs">Confirmation</TableHead>
   483→              <TableHead className="text-xs">Room</TableHead>
   484→              <TableHead className="text-xs hidden md:table-cell">Check-In</TableHead>
   485→              <TableHead className="text-xs hidden lg:table-cell">Check-Out</TableHead>
   486→              <TableHead className="text-xs hidden sm:table-cell">Source</TableHead>
   487→              <TableHead className="text-xs">Status</TableHead>
   488→            </TableRow>
   489→          </TableHeader>
   490→          <TableBody>
   491→            {arrivals.map((res, idx) => {
   492→              const guestName = res.guest
   493→                ? `${res.guest.firstName} ${res.guest.lastName}`
   494→                : 'Unknown Guest'
   495→              const isVip = res.guest?.vipLevel && res.guest.vipLevel !== 'none'
   496→
   497→              return (
   498→                <TableRow
   499→                  key={res.id}
   500→                  className={cn(idx % 2 !== 0 && 'bg-muted/30')}
   501→                >
   502→                  <TableCell>
   503→                    <div className="flex items-center gap-2">
   504→                      <span className="font-medium text-xs">{guestName}</span>
   505→                      {isVip && (
   506→                        <Badge className="text-[9px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700">
   507→                          VIP
   508→                        </Badge>
   509→                      )}
   510→                    </div>
   511→                    {res.specialRequests && (
   512→                      <p className="text-[10px] text-muted-foreground mt-0.5 max-w-[200px] truncate">
   513→                        {res.specialRequests}
   514→                      </p>
   515→                    )}
   516→                  </TableCell>
   517→                  <TableCell>
   518→                    <span className="font-mono text-xs">{res.confirmationNo}</span>
   519→                  </TableCell>
   520→                  <TableCell>
   521→                    {res.room ? (
   522→                      <span className="text-xs">{res.room.number} <span className="text-muted-foreground text-[10px]">({res.room.type.code})</span></span>
   523→                    ) : (
   524→                      <Badge variant="outline" className="text-[10px] px-1 py-0 border-amber-400 text-amber-600 dark:text-amber-400">
   525→                        Unassigned
   526→                      </Badge>
   527→                    )}
   528→                  </TableCell>
   529→                  <TableCell className="hidden md:table-cell">
   530→                    <span className="text-xs text-muted-foreground flex items-center gap-1">

   548→                          : 'bg-sky-50 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   549→                      )}
