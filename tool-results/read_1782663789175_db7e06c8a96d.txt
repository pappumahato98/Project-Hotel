    41→// Use shared formatDateLong and formatDateTime from @/lib/format (en-GB/DD-MM-YYYY)
    42→// Re-export for convenience since component references them as formatDate/formatDateTime
    43→const formatDate = (dateStr: string | null | undefined): string => {
    44→  if (!dateStr) return '—'
    45→  try {
    46→    const d = new Date(dateStr)
    47→    if (isNaN(d.getTime())) return dateStr ?? ''
    48→    return d.toLocaleDateString('en-GB', { year: 'numeric', month: 'long', day: 'numeric' })
    49→  } catch {
    50→    return dateStr ?? ''
    51→  }
    52→}
    53→
    54→const formatDateTime = (dateStr: string | null | undefined): string => {
    55→  if (!dateStr) return '—'
    56→  try {
    57→    const d = new Date(dateStr)
    58→    if (isNaN(d.getTime())) return dateStr ?? ''
    59→    return d.toLocaleDateString('en-GB', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    60→  } catch {
    61→    return dateStr ?? ''
    62→  }
    63→}
    64→
    65→function toDateInputValue(dateStr: string | null | undefined): string {
    66→  if (!dateStr) return ''
    67→  try {
    68→    const d = new Date(dateStr)
    69→    if (isNaN(d.getTime())) return ''
    70→    return d.toISOString().split('T')[0]
    71→  } catch {
    72→    return ''
    73→  }
    74→}
