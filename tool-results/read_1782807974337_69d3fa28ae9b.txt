     1→'use client'
     2→
     3→import { useState, useMemo } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { apiFetch } from '@/lib/api'
     6→import { invalidate, qk } from '@/lib/queryKeys'
     7→import { toast } from 'sonner'
     8→import {
     9→  Clock, AlertTriangle, Crown, Phone, Plus, Search, Trash2,
    10→  BedDouble, UserPlus, Users, CheckCircle2, Filter, X, Loader2,
    11→} from 'lucide-react'
    12→
    13→import { Button } from '@/components/ui/button'
    14→import { Badge } from '@/components/ui/badge'
    15→import { Card, CardContent } from '@/components/ui/card'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Textarea } from '@/components/ui/textarea'
    19→import { Skeleton } from '@/components/ui/skeleton'
    20→import {
    21→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    22→} from '@/components/ui/dialog'
    23→import {
    24→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    25→} from '@/components/ui/select'
    26→import {
    27→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    28→} from '@/components/ui/table'

    32→// ─── Types ──────────────────────────────────────────────────────────────
    33→
    34→type Priority = 'High' | 'Normal' | 'Low'
    35→type WaitlistStatus = 'Waiting' | 'Assigned' | 'Cancelled'
    36→
    37→interface WaitlistEntry {
    38→  id: string
    39→  guestName: string
    40→  contactPhone: string | null
    41→  contactEmail: string | null
    42→  roomPreference: string | null
    43→  checkInDate: string
    44→  checkOutDate: string
    45→  adults: number
    46→  children: number
    47→  priority: string
    48→  status: string
    49→  notes: string | null
    50→  createdAt: string
    51→}
    52→
    53→interface ApiResult {
    54→  entries: WaitlistEntry[]
    55→}
    56→
    57→// ─── Priority / Status Helpers ──────────────────────────────────────────
    58→
    59→const PRIORITY_ORDER: Record<string, number> = { High: 0, Normal: 1, Low: 2 }
    60→
    61→function getPriorityBadge(priority: string) {
    62→  switch (priority) {
    63→    case 'High':
    64→      return <Badge variant="destructive" className="text-[10px] gap-0.5"><AlertTriangle className="size-3" />High</Badge>
    65→    case 'Low':
    66→      return <Badge variant="outline" className="text-[10px] text-muted-foreground">Low</Badge>
    67→    default:
    68→      return <Badge className="text-[10px] bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300">Normal</Badge>
    69→  }
    70→}
    71→
    72→function getStatusBadge(status: string) {
    73→  switch (status) {
    74→    case 'Assigned':
    75→      return <Badge className="text-[10px] bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 gap-0.5"><CheckCircle2 className="size-3" />Assigned</Badge>
    76→    case 'Cancelled':
    77→      return <Badge variant="outline" className="text-[10px] text-muted-foreground line-through">Cancelled</Badge>
    78→    default:
    79→      return <Badge className="text-[10px] bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 gap-0.5"><Clock className="size-3" />Waiting</Badge>
    80→  }
    81→}

    83→const ROOM_TYPE_OPTIONS = [
    84→  { value: 'Deluxe King', label: 'Deluxe King' },
    85→  { value: 'Deluxe Twin', label: 'Deluxe Twin' },
    86→  { value: 'Standard Double', label: 'Standard Double' },
    87→  { value: 'Standard Single', label: 'Standard Single' },
    88→  { value: 'Suite', label: 'Suite' },
    89→  { value: 'Family Room', label: 'Family Room' },
    90→]

    92→// ─── Main Component ────────────────────────────────────────────────────
    93→
    94→export function WaitlistView() {
    95→  const queryClient = useQueryClient()
    96→
    97→  // Fetch waitlist
    98→  const { data: apiData, isLoading, error, refetch } = useQuery<ApiResult>({
    99→    queryKey: qk.waitlist(),
   100→    queryFn: () => apiFetch('/api/waitlist'),
   101→  })
   102→
   103→  const entries = apiData?.entries ?? []
   104→
   105→  // ─── Local state ──────────────────────────────────────────────
   106→  const [searchQuery, setSearchQuery] = useState('')
   107→  const [statusFilter, setStatusFilter] = useState<string>('all')
   108→  const [priorityFilter, setPriorityFilter] = useState<string>('all')
   109→
   110→  // Dialog states
   111→  const [addOpen, setAddOpen] = useState(false)
   112→  const [editOpen, setEditOpen] = useState(false)
   113→  const [deleteOpen, setDeleteOpen] = useState(false)
   114→  const [assignOpen, setAssignOpen] = useState(false)
   115→  const [selected, setSelected] = useState<WaitlistEntry | null>(null)
   116→
   117→  // Form state
   118→  const [formGuestName, setFormGuestName] = useState('')
   119→  const [formPhone, setFormPhone] = useState('')
   120→  const [formEmail, setFormEmail] = useState('')
   121→  const [formRoomPref, setFormRoomPref] = useState('')
   122→  const [formCheckIn, setFormCheckIn] = useState('')
   123→  const [formCheckOut, setFormCheckOut] = useState('')
   124→  const [formAdults, setFormAdults] = useState(2)
   125→  const [formChildren, setFormChildren] = useState(0)
   126→  const [formPriority, setFormPriority] = useState<string>('Normal')
   127→  const [formNotes, setFormNotes] = useState('')
   128→  const [assignRoom, setAssignRoom] = useState('')
   129→
   130→  // ─── Mutations ────────────────────────────────────────────────
   131→  const createMutation = useMutation({
   132→    mutationFn: (data: Record<string, unknown>) => apiFetch('/api/waitlist', {
   133→      method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data),
   134→    }),
   135→    onSuccess: () => {
   136→      queryClient.invalidateQueries({ queryKey: qk.waitlist() })
   137→      toast.success('Added to waitlist')
   138→      setAddOpen(false)
   139→      resetForm()
   140→    },
   141→    onError: (err: Error) => toast.error(err.message),
   142→  })
   143→
   144→  const updateMutation = useMutation({
   145→    mutationFn: ({ id, data }: { id: string; data: Record<string, unknown> }) =>
   146→      apiFetch(`/api/waitlist/${id}`, {
   147→        method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data),
   148→      }),
   149→    onSuccess: () => {
   150→      queryClient.invalidateQueries({ queryKey: qk.waitlist() })
   151→      toast.success('Waitlist updated')
   152→      setEditOpen(false)
   153→      setAssignOpen(false)
   154→    },
   155→    onError: (err: Error) => toast.error(err.message),
   156→  })
   157→
   158→  const deleteMutation = useMutation({
   159→    mutationFn: (id: string) => apiFetch(`/api/waitlist/${id}`, { method: 'DELETE' }),
   160→    onSuccess: () => {
   161→      queryClient.invalidateQueries({ queryKey: qk.waitlist() })
   162→      toast.success('Removed from waitlist')
   163→      setDeleteOpen(false)
   164→      setSelected(null)
   165→    },
   166→    onError: (err: Error) => toast.error(err.message),
   167→  })
   168→
   169→  // ─── Form helpers ─────────────────────────────────────────────
   170→  function resetForm() {
   171→    setFormGuestName(''); setFormPhone(''); setFormEmail('')
   172→    setFormRoomPref(''); setFormCheckIn(''); setFormCheckOut('')
   173→    setFormAdults(2); setFormChildren(0); setFormPriority('Normal')
   174→    setFormNotes('')
   175→  }
   176→
   177→  function openAdd() { resetForm(); setAddOpen(true) }
   178→
   179→  function openEdit(entry: WaitlistEntry) {
   180→    setSelected(entry)
   181→    setFormGuestName(entry.guestName)
   182→    setFormPhone(entry.contactPhone || '')
   183→    setFormEmail(entry.contactEmail || '')
   184→    setFormRoomPref(entry.roomPreference || '')
   185→    setFormCheckIn(entry.checkInDate)
   186→    setFormCheckOut(entry.checkOutDate)
   187→    setFormAdults(entry.adults)
   188→    setFormChildren(entry.children)
   189→    setFormPriority(entry.priority)
   190→    setFormNotes(entry.notes || '')
   191→    setEditOpen(true)
   192→  }
   193→
   194→  function openAssign(entry: WaitlistEntry) {
   195→    setSelected(entry)
   196→    setAssignRoom(entry.roomPreference || '')
   197→    setAssignOpen(true)
   198→  }
   199→
   200→  function openDelete(entry: WaitlistEntry) {
   201→    setSelected(entry)
   202→    setDeleteOpen(true)
   203→  }
   204→
   205→  function handleCreate() {
   206→    if (!formGuestName || !formCheckIn || !formCheckOut) {
   207→      toast.error('Guest name, check-in, and check-out dates are required')
   208→      return
   209→    }
   210→    createMutation.mutate({
   211→      guestName: formGuestName,
   212→      contactPhone: formPhone || undefined,
   213→      contactEmail: formEmail || undefined,
   214→      roomPreference: formRoomPref || undefined,
   215→      checkInDate: formCheckIn,
   216→      checkOutDate: formCheckOut,
   217→      adults: formAdults,
   218→      children: formChildren,
   219→      priority: formPriority,
   220→      notes: formNotes || undefined,
   221→    })
   222→  }
   223→
   224→  function handleEditSave() {
   225→    if (!selected) return
   226→    updateMutation.mutate({
   227→      id: selected.id,
   228→      data: {
   229→        guestName: formGuestName,
   230→        contactPhone: formPhone || null,
   231→        contactEmail: formEmail || null,
   232→        roomPreference: formRoomPref || null,
   233→        checkInDate: formCheckIn,
   234→        checkOutDate: formCheckOut,
   235→        adults: formAdults,
   236→        children: formChildren,
   237→        priority: formPriority,
   238→        notes: formNotes || null,
   239→      },
   240→    })
   241→  }
   242→
   243→  function handleAssign() {
   244→    if (!selected) return
   245→    updateMutation.mutate({
   246→      id: selected.id,
   247→      data: { status: 'Assigned', roomPreference: assignRoom || null },
   248→    })
   249→  }
   250→
   251→  function handleCancel(entry: WaitlistEntry) {
   252→    updateMutation.mutate({
   253→      id: entry.id,
   254→      data: { status: 'Cancelled' },
   255→    })
   256→  }
   257→
   258→  function handleDelete() {
   259→    if (!selected) return
   260→    deleteMutation.mutate(selected.id)
   261→  }
   262→
   263→  // ─── Filtered list ────────────────────────────────────────────
   264→  const filtered = useMemo(() => {
   265→    let result = entries
   266→    if (statusFilter !== 'all') result = result.filter(e => e.status === statusFilter)
   267→    if (priorityFilter !== 'all') result = result.filter(e => e.priority === priorityFilter)
   268→    if (searchQuery.trim()) {
   269→      const q = searchQuery.toLowerCase()
   270→      result = result.filter(e =>
   271→        e.guestName.toLowerCase().includes(q) ||
   272→        (e.contactPhone || '').includes(q) ||
   273→        (e.contactEmail || '').toLowerCase().includes(q) ||
   274→        (e.roomPreference || '').toLowerCase().includes(q),
   275→      )
   276→    }
   277→    // Assign position numbers based on filtered order
   278→    return result.map((e, i) => ({ ...e, position: i + 1 }))
   279→  }, [entries, statusFilter, priorityFilter, searchQuery])
   280→
   281→  // Stats
   282→  const totalWaiting = entries.filter(e => e.status === 'Waiting').length
   283→  const totalAssigned = entries.filter(e => e.status === 'Assigned').length
   284→  const highPriority = entries.filter(e => e.status === 'Waiting' && e.priority === 'High').length
   285→
   286→  // ─── Render ───────────────────────────────────────────────────
   287→
   288→  if (isLoading) {
   289→    return (
   290→      <div className="flex flex-col gap-4">
   291→        <Skeleton className="h-8 w-64" />
   292→        <div className="grid grid-cols-3 gap-3">
   293→          <Skeleton className="h-20 rounded-lg" />
   294→          <Skeleton className="h-20 rounded-lg" />
   295→          <Skeleton className="h-20 rounded-lg" />
   296→        </div>
   297→        <Skeleton className="h-64 rounded-lg" />
   298→      </div>
   299→    )
   300→  }
   301→
   302→  if (error) {
   303→    return (
   304→      <Card>
   305→        <CardContent className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   306→          <AlertTriangle className="size-8 mb-2" />
   307→          <p className="text-sm font-medium">Failed to load waitlist</p>
   308→          <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   309→            <Users className="size-3.5 mr-1.5" />Retry
   310→          </Button>
   311→        </CardContent>
   312→      </Card>
   313→    )
   314→  }
   315→
   316→  return (
   317→    <div className="flex flex-col gap-4">
   318→      {/* Header */}
   319→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   320→        <div>
   321→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Room Waitlist</h2>
   322→          <p className="text-xs text-muted-foreground">
   323→            Manage guests waiting for room availability
   324→          </p>
   325→        </div>
   326→        <Button onClick={openAdd} className="gap-2 shrink-0">
   327→          <Plus className="size-3.5" />
   328→          Add to Waitlist
   329→        </Button>
   330→      </div>
   331→
   332→      {/* Stats */}
   333→      <div className="grid grid-cols-3 gap-3">
   334→        <Card className="border-amber-200 dark:border-amber-800">
   335→          <CardContent className="flex items-center gap-3 p-3">
   336→            <div className="flex size-9 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-950">
   337→              <Clock className="size-4 text-amber-600 dark:text-amber-400" />
   338→            </div>
   339→            <div>
   340→              <p className="text-lg font-bold text-amber-700 dark:text-amber-300">{totalWaiting}</p>
   341→              <p className="text-[10px] text-muted-foreground">Waiting</p>
   342→            </div>
   343→          </CardContent>
   344→        </Card>
   345→        <Card className="border-emerald-200 dark:border-emerald-800">
   346→          <CardContent className="flex items-center gap-3 p-3">
   347→            <div className="flex size-9 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-950">
   348→              <CheckCircle2 className="size-4 text-emerald-600 dark:text-emerald-400" />
   349→            </div>
   350→            <div>
   351→              <p className="text-lg font-bold text-emerald-700 dark:text-emerald-300">{totalAssigned}</p>
   352→              <p className="text-[10px] text-muted-foreground">Assigned</p>
   353→            </div>
   354→          </CardContent>
   355→        </Card>
   356→        <Card className="border-red-200 dark:border-red-800">
   357→          <CardContent className="flex items-center gap-3 p-3">
   358→            <div className="flex size-9 items-center justify-center rounded-lg bg-red-100 dark:bg-red-950">
   359→              <AlertTriangle className="size-4 text-red-600 dark:text-red-400" />
   360→            </div>
   361→            <div>
   362→              <p className="text-lg font-bold text-red-700 dark:text-red-300">{highPriority}</p>
   363→              <p className="text-[10px] text-muted-foreground">High Priority</p>
   364→            </div>
   365→          </CardContent>
   366→        </Card>
   367→      </div>
   368→
   369→      {/* Filters */}
   370→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-2">
   371→        <div className="flex items-center gap-2">
   372→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   373→            <SelectTrigger className="w-[130px] h-8 text-xs">
   374→              <SelectValue placeholder="Status" />
   375→            </SelectTrigger>
   376→            <SelectContent>
   377→              <SelectItem value="all">All Status</SelectItem>
   378→              <SelectItem value="Waiting">Waiting</SelectItem>
   379→              <SelectItem value="Assigned">Assigned</SelectItem>
   380→              <SelectItem value="Cancelled">Cancelled</SelectItem>
   381→            </SelectContent>
   382→          </Select>
   383→          <Select value={priorityFilter} onValueChange={setPriorityFilter}>
   384→            <SelectTrigger className="w-[130px] h-8 text-xs">
   385→              <SelectValue placeholder="Priority" />
   386→            </SelectTrigger>
   387→            <SelectContent>
   388→              <SelectItem value="all">All Priority</SelectItem>
   389→              <SelectItem value="High">High</SelectItem>
   390→              <SelectItem value="Normal">Normal</SelectItem>
   391→              <SelectItem value="Low">Low</SelectItem>
   392→            </SelectContent>
   393→          </Select>
   394→        </div>
   395→        <div className="relative flex-1 max-w-xs">
   396→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3 text-muted-foreground" />
   397→          <Input
   398→            placeholder="Search guest, phone, room..."
   399→            value={searchQuery}
   400→            onChange={(e) => setSearchQuery(e.target.value)}
   401→            className="pl-8 h-8 text-xs"
   402→          />
   403→          {searchQuery && (
   404→            <button onClick={() => setSearchQuery('')} className="absolute right-2 top-1/2 -translate-y-1/2">
   405→              <X className="size-3 text-muted-foreground hover:text-foreground" />
   406→            </button>
   407→          )}
   408→        </div>
   409→      </div>
   410→
   411→      {/* Table */}
   412→      <Card>
   413→        <CardContent className="p-0">
   414→          {filtered.length === 0 ? (
   415→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   416→              <Users className="size-8 mb-2 opacity-40" />
   417→              <p className="text-sm">No waitlist entries found</p>
   418→            </div>
   419→          ) : (
   420→            <Table>
   421→              <TableHeader>
   422→                <TableRow>
   423→                  <TableHead className="w-10 text-xs">#</TableHead>
   424→                  <TableHead className="text-xs">Guest</TableHead>
   425→                  <TableHead className="text-xs">Room Pref</TableHead>
   426→                  <TableHead className="text-xs">Dates</TableHead>
   427→                  <TableHead className="text-xs">Guests</TableHead>
   428→                  <TableHead className="text-xs">Priority</TableHead>
   429→                  <TableHead className="text-xs">Status</TableHead>
   430→                  <TableHead className="text-xs">Added</TableHead>
   431→                  <TableHead className="w-[100px] text-xs">Actions</TableHead>
   432→                </TableRow>
   433→              </TableHeader>
   434→              <TableBody>
   435→                {filtered.map((entry) => (
   436→                  <TableRow key={entry.id} className={cn(
   437→                    'group',
   438→                    entry.status === 'Cancelled' && 'opacity-50',
   439→                  )}>
   440→                    <TableCell className="text-xs font-mono text-muted-foreground">{entry.position}</TableCell>
   441→                    <TableCell>
   442→                      <div>
   443→                        <p className="text-xs font-medium">{entry.guestName}</p>
   444→                        {entry.contactPhone && (
   445→                          <p className="text-[10px] text-muted-foreground flex items-center gap-0.5"><Phone className="size-2.5" />{entry.contactPhone}</p>
   446→                        )}
   447→                      </div>
   448→                    </TableCell>
   449→                    <TableCell className="text-xs">{entry.roomPreference || '—'}</TableCell>
   450→                    <TableCell>
   451→                      <div className="text-[10px]">
   452→                        <p>{formatDate(entry.checkInDate)}</p>
   453→                        <p className="text-muted-foreground">→ {formatDate(entry.checkOutDate)}</p>
   454→                      </div>
   455→                    </TableCell>
   456→                    <TableCell className="text-xs text-center">{entry.adults + entry.children}</TableCell>
   457→                    <TableCell>{getPriorityBadge(entry.priority)}</TableCell>
   458→                    <TableCell>{getStatusBadge(entry.status)}</TableCell>
   459→                    <TableCell className="text-[10px] text-muted-foreground">{formatDate(entry.createdAt)}</TableCell>
   460→                    <TableCell>
   461→                      <div className="flex items-center gap-1">
   462→                        {entry.status === 'Waiting' && (
   463→                          <>
   464→                            <Button size="sm" variant="outline" className="h-7 text-[10px] px-1.5" onClick={() => openAssign(entry)} title="Assign Room">
   465→                              <BedDouble className="size-3" />
   466→                            </Button>
   467→                            <Button size="sm" variant="ghost" className="h-7 text-[10px] px-1.5 text-amber-600" onClick={() => openEdit(entry)} title="Edit">
   468→                              <UserPlus className="size-3" />
   469→                            </Button>
   470→                          </>
   471→                        )}
   472→                        {entry.status !== 'Cancelled' && (
   473→                          <Button size="sm" variant="ghost" className="h-7 text-[10px] px-1.5 text-red-500" onClick={() => openDelete(entry)} title="Remove">
   474→                            <Trash2 className="size-3" />
   475→                          </Button>
   476→                        )}
   477→                      </div>
   478→                    </TableCell>
   479→                  </TableRow>
   480→                ))}
   481→              </TableBody>
   482→            </Table>
   483→          )}
   484→        </CardContent>
   485→      </Card>
   486→
   487→      {/* ─── Add Dialog ─────────────────────────────────────── */}
   488→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   489→        <DialogContent className="max-w-md">
   490→          <DialogHeader>
   491→            <DialogTitle className="flex items-center gap-2"><UserPlus className="size-5" />Add to Waitlist</DialogTitle>
   492→            <DialogDescription>Add a guest to the room waitlist when no suitable rooms are available.</DialogDescription>
   493→          </DialogHeader>
   494→          <div className="grid gap-3">
   495→            <div className="grid gap-1.5">
   496→              <Label className="text-xs">Guest Name *</Label>
   497→              <Input value={formGuestName} onChange={(e) => setFormGuestName(e.target.value)} placeholder="Full name" className="h-8 text-sm" />
   498→            </div>
   499→            <div className="grid grid-cols-2 gap-3">
   500→              <div className="grid gap-1.5">
   501→                <Label className="text-xs">Phone</Label>
   502→                <Input value={formPhone} onChange={(e) => setFormPhone(e.target.value)} placeholder="+977-..." className="h-8 text-sm" />
   503→              </div>
   504→              <div className="grid gap-1.5">
   505→                <Label className="text-xs">Email</Label>
   506→                <Input value={formEmail} onChange={(e) => setFormEmail(e.target.value)} placeholder="email@example.com" className="h-8 text-sm" />
   507→              </div>
   508→            </div>
   509→            <div className="grid gap-1.5">
   510→              <Label className="text-xs">Room Preference</Label>
   511→              <Select value={formRoomPref} onValueChange={setFormRoomPref}>
   512→                <SelectTrigger className="h-8 text-sm"><SelectValue placeholder="Any room type" /></SelectTrigger>
