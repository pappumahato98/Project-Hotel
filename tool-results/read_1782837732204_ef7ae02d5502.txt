   100→   100→
   101→   101→export function DeparturesView() {
   102→   102→  const queryClient = useQueryClient()
   103→   103→  const { settings } = useSettingsStore()
   104→   104→  const { preferences } = usePreferencesStore()
   105→   105→  const { navigateTo } = useNavigationStore()
   106→   106→  const today = getTodayString()
   107→   107→
   108→   108→  // Dialog states
   109→   109→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   110→   110→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   111→   111→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   112→   112→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   113→   113→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   114→   114→
   115→   115→  // Selected departure
   116→   116→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   117→   117→
   118→   118→  // Late checkout
   119→   119→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   120→   120→
   121→   121→  // Payment
   122→   122→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   123→   123→  const [paymentAmount, setPaymentAmount] = useState('')
   124→   124→  const [paymentReference, setPaymentReference] = useState('')
   125→   125→
   126→   126→  // Receipt data
   127→   127→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   128→   128→
   129→   129→  // Fetch today's departures (checked_in with checkOut = today)
   130→   130→  const { data, isLoading } = useQuery({
   131→   131→    queryKey: ['departures', today],
   132→   132→    queryFn: async () => {
   133→   133→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   134→   134→      return apiFetch(`/api/reservations?${params.toString()}`)
   135→   135→    },
   136→   136→    refetchInterval: 30000,
   137→   137→  })
   138→   138→
   139→   139→  const departures: Departure[] = data?.reservations || []
   140→   140→
   141→   141→  // Stats
   142→   142→  const totalDepartures = departures.length
   143→   143→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   144→   144→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   145→   145→  const outstandingBalance = departures.reduce((sum, d) => {
   146→   146→    return sum + (d.folios[0]?.balance || 0)
   147→   147→  }, 0)
   148→   148→
   149→   149→  // Zero-balance pending guests (for express & batch checkout)
   150→   150→  const zeroBalancePending = departures.filter(
   151→   151→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   152→   152→  )
   153→   153→
   154→   154→  // ─── Mutations ─────────────────────────────────────────────────────
   155→   155→
   156→   156→  // Checkout mutation
   157→   157→  const checkoutMutation = useMutation({
   158→   158→    mutationFn: async (reservationId: string) => {
   159→   159→      return apiFetch(`/api/reservations/${reservationId}`, {
   160→   160→        method: 'PATCH',
   161→   161→        headers: { 'Content-Type': 'application/json' },
   162→   162→        body: JSON.stringify({ status: 'checked_out' }),
   163→   163→      })
   164→   164→    },
   165→   165→    onSuccess: () => {
   166→   166→      invalidate.afterCheckout(queryClient)
   167→   167→    },
   168→   168→  })
   169→   169→
   170→   170→  // Payment mutation
   171→   171→  const paymentMutation = useMutation({
   172→   172→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   173→   173→      return apiFetch(`/api/folio/${folioId}`, {
   174→   174→        method: 'POST',
   175→   175→        headers: { 'Content-Type': 'application/json' },
   176→   176→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   177→   177→      })
   178→   178→    },
   179→   179→    onSuccess: () => {
   180→   180→      invalidate.afterFolioChange(queryClient)
   181→   181→      setPaymentDialogOpen(false)
   182→   182→      setPaymentAmount('')
   183→   183→      setPaymentReference('')
   184→   184→      toast.success('Payment recorded successfully')
   185→   185→    },
   186→   186→    onError: () => {
   187→   187→      toast.error('Failed to record payment')
   188→   188→    },
   189→   189→  })
   190→   190→
   191→   191→  // Late checkout mutation
   192→   192→  const lateCheckoutMutation = useMutation({
   193→   193→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   194→   194→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   195→   195→        method: 'PATCH',
   196→   196→        headers: { 'Content-Type': 'application/json' },
   197→   197→        body: JSON.stringify({ checkOut: newCheckOut }),
   198→   198→      })
   199→   199→      // Add surcharge to folio
   200→   200→      if (surcharge > 0) {
   201→   201→        await apiFetch(`/api/folio`, {
   202→   202→          method: 'POST',
   203→   203→          headers: { 'Content-Type': 'application/json' },
   204→   204→          body: JSON.stringify({
   205→   205→            reservationId,
   206→   206→            type: 'charge',
   207→   207→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   208→   208→            amount: surcharge,
   209→   209→          }),
   210→   210→        })
   211→   211→      }
   212→   212→      return result
   213→   213→    },
   214→   214→    onSuccess: () => {
   215→   215→      invalidate.afterReservationChange(queryClient)
   216→   216→      invalidate.afterFolioChange(queryClient)
   217→   217→      setLateCheckoutDialogOpen(false)
   218→   218→      toast.success('Late checkout request processed successfully')
   219→   219→    },
   220→   220→    onError: () => {
   221→   221→      toast.error('Failed to process late checkout request')
   222→   222→    },
   223→   223→  })
   224→   224→
   225→   225→  // Batch checkout mutation
   226→   226→  const batchCheckoutMutation = useMutation({
   227→   227→    mutationFn: async (ids: string[]) => {
   228→   228→      const results = []
   229→   229→      for (let i = 0; i < ids.length; i++) {
   230→   230→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   231→   231→          method: 'PATCH',
   232→   232→          headers: { 'Content-Type': 'application/json' },
   233→   233→          body: JSON.stringify({ status: 'checked_out' }),
   234→   234→        })
   235→   235→        results.push(result)
   236→   236→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   237→   237→      }
   238→   238→      return results
   239→   239→    },
   240→   240→    onSuccess: (_data, ids) => {
   241→   241→      invalidate.afterCheckout(queryClient)
   242→   242→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   243→   243→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   244→   244→    },
   245→   245→    onError: () => {
   246→   246→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   247→   247→    },
   248→   248→  })
   249→   249→
   250→   250→  // ─── Handlers ─────────────────────────────────────────────────────
   251→   251→
   252→   252→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   253→   253→
   254→   254→  const handleExpressCheckout = (dep: Departure) => {
   255→   255→    checkoutMutation.mutate(dep.id, {
   256→   256→      onSuccess: () => {
   257→   257→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   258→   258→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   259→   259→      },
   260→   260→    })
   261→   261→  }
   262→   262→
   263→   263→  const handleReviewFolio = (dep: Departure) => {
   264→   264→    setSelectedDeparture(dep)
   265→   265→    setFolioDialogOpen(true)
   266→   266→  }
   267→   267→
   268→   268→  const handleNavigateToFolio = (dep: Departure) => {
   269→   269→    useFolioContextStore.getState().setFolioContext({
   270→   270→      reservationId: dep.id,
   271→   271→      guestId: dep.guest?.id || '',
   272→   272→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   273→   273→      roomNumber: dep.room?.number || '',
   274→   274→      confirmationNo: dep.confirmationNo,
   275→   275→    })
   276→   276→    navigateTo('front-desk', 'folio')
   277→   277→  }
   278→   278→
   279→   279→  const handleViewLedger = (dep: Departure) => {
   280→   280→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   281→   281→      guestId: dep.guest?.id || '',
   282→   282→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   283→   283→    })
   284→   284→    navigateTo('front-desk', 'guest-ledger')
   285→   285→  }
   286→   286→
   287→   287→  const handleFolioSettleCheckout = () => {
   288→   288→    setFolioDialogOpen(false)
   289→   289→    setCheckoutDialogOpen(true)
   290→   290→  }
   291→   291→
   292→   292→  const handleQuickCheckout = (dep: Departure) => {
   293→   293→    setSelectedDeparture(dep)
   294→   294→    setCheckoutDialogOpen(true)
   295→   295→  }
   296→   296→
   297→   297→  const handleRequestLateCheckout = (dep: Departure) => {
   298→   298→    setSelectedDeparture(dep)
   299→   299→    setLateCheckoutOption('14:00')
   300→   300→    setLateCheckoutDialogOpen(true)
   301→   301→  }
   302→   302→
   303→   303→  const handleConfirmLateCheckout = () => {
   304→   304→    if (!selectedDeparture) return
   305→   305→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   306→   306→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   307→   307→
   308→   308→    // Build new checkout datetime with the selected time
   309→   309→    const currentCheckout = new Date(selectedDeparture.checkOut)
   310→   310→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   311→   311→    currentCheckout.setHours(hours, minutes, 0, 0)
   312→   312→
   313→   313→    lateCheckoutMutation.mutate({
   314→   314→      reservationId: selectedDeparture.id,
   315→   315→      newCheckOut: currentCheckout.toISOString(),
   316→   316→      surcharge,
   317→   317→    })
   318→   318→  }
   319→   319→
   320→   320→  const handleOpenPaymentDialog = () => {
   321→   321→    if (!selectedDeparture) return
   322→   322→    const balance = getBalance(selectedDeparture)
   323→   323→    setPaymentAmount(String(balance))
   324→   324→    setPaymentMethod('cash')
   325→   325→    setPaymentReference('')
   326→   326→    setPaymentDialogOpen(true)
   327→   327→  }

   330→   330→    if (!selectedDeparture) return
   331→   331→    const folioId = selectedDeparture.folios[0]?.id

   333→   333→      toast.error('No folio found for this reservation')
   334→   334→      return
   335→   335→    }
   336→   336→    const amount = Number(paymentAmount)
   337→   337→    if (isNaN(amount) || amount <= 0) {
   338→   338→      toast.error('Please enter a valid payment amount')
   339→   339→      return
   340→   340→    }
   341→   341→    paymentMutation.mutate({
   342→   342→      folioId,
   343→   343→      amount,
   344→   344→      method: paymentMethod,
   345→   345→      reference: paymentReference,
   346→   346→    })
   347→   347→  }
   348→   348→

   352→   352→    checkoutMutation.mutate(selectedDeparture.id, {
   353→   353→      onSuccess: () => {

   355→   355→        setReceiptData(selectedDeparture)
   356→   356→        setReceiptDialogOpen(true)
   357→   357→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   358→   358→      },
   359→   359→    })
   360→   360→  }

   367→   367→    }
   368→   368→    batchCheckoutMutation.mutate(ids)
   369→   369→  }
   370→   370→
   371→   371→  const handlePrintReceipt = () => {
   372→   372→    toast.success('Receipt sent to printer')
   373→   373→  }
   374→   374→
   375→   375→  const handleEmailReceipt = () => {
   376→   376→    toast.success('Receipt emailed to guest')
   377→   377→  }
   378→   378→
   379→   379→  // ─── Computed values ──────────────────────────────────────────────
   380→   380→
   381→   381→  const lateCheckoutFee = selectedDeparture
   382→   382→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   383→   383→    : 0
   384→   384→
   385→   385→  // ─── Render ────────────────────────────────────────────────────────
   386→   386→
   387→   387→  return (

   389→   389→      {/* Header */}
   390→   390→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">

   394→   394→            Guest check-outs scheduled for {formatDate(today)}
   395→   395→          </p>
   396→   396→        </div>
   397→   397→        {zeroBalancePending.length > 0 && (
   398→   398→          <Button
   399→   399→            onClick={handleBatchCheckout}
   400→   400→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   401→   401→            disabled={batchCheckoutMutation.isPending}
   402→   402→          >
   403→   403→            <Zap className="size-4 mr-1.5" />
   404→   404→            {batchCheckoutMutation.isPending
   405→   405→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   406→   406→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   407→   407→          </Button>
   408→   408→        )}
   409→   409→      </div>
   410→   410→
   411→   411→      {/* Departures List */}
   412→   412→      <Card className="py-0">
   413→   413→        <CardContent className="p-0">
   414→   414→          <div className="max-h-[600px] overflow-y-auto">
   415→   415→            <Table>
   416→   416→              <TableHeader>
   417→   417→                <TableRow>
   418→   418→                  <TableHead className="w-[70px]">Room</TableHead>
   419→   419→                  <TableHead>Guest</TableHead>
   420→   420→                  <TableHead className="w-[100px]">Check-out</TableHead>
   421→   421→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   422→   422→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   423→   423→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   424→   424→                </TableRow>
   425→   425→              </TableHeader>
   426→   426→              <TableBody>
   427→   427→                {isLoading ? (
   428→   428→                  Array.from({ length: 5 }).map((_, i) => (
   429→   429→                    <TableRow key={i}>
   430→   430→                      {Array.from({ length: 6 }).map((_, j) => (
   431→   431→                        <TableCell key={j}>
   432→   432→                          <Skeleton className="h-4 w-full" />
   433→   433→                        </TableCell>
   434→   434→                      ))}
   435→   435→                    </TableRow>
   436→   436→                  ))
   437→   437→                ) : departures.length === 0 ? (
   438→   438→                  <TableRow>
   439→   439→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   440→   440→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   441→   441→                      No departures scheduled for today
   442→   442→                    </TableCell>
   443→   443→                  </TableRow>
   444→   444→                ) : (
   445→   445→                  departures.map((dep) => {
   446→   446→                    const balance = getBalance(dep)
   447→   447→                    const isPending = dep.status === 'checked_in'
   448→   448→                    const isZeroBalance = balance === 0
   449→   449→                    return (
   450→   450→                      <TableRow key={dep.id}>
   451→   451→                        <TableCell className="font-bold font-mono">{dep.room.number} <RoomTypeBedBadge typeName={dep.room.type.name} bedConfig={dep.room.type.bedConfig} typeCode={dep.room.type.code} pax={(dep.adults ?? 1) + (dep.children ?? 0)} inline /></TableCell>
   452→   452→                        <TableCell>
   453→   453→                          <div className="flex items-center gap-2">
   454→   454→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   455→   455→                            {dep.guest.vipLevel !== 'none' && (
   456→   456→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   457→   457→                                VIP
   458→   458→                              </Badge>
