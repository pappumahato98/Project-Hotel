   275→// ── Inline Row Action Buttons ─────────────────────────────────
   276→function InlineRowActions({
   277→  row,
   278→  onStatusChange,
   279→  onViewDetail,
   280→  isForceMutated,
   281→}: {
   282→  row: RoomRow
   283→  onStatusChange: (roomId: string, taskId: string | null, newStatus: string) => void
   284→  onViewDetail: (roomId: string) => void
   285→  isForceMutated: boolean
   286→}) {
   287→  const s = row.hkDisplayStatus
   288→  return (
   289→    <div className="flex items-center gap-0.5 flex-nowrap">
   290→      {/* Occupied indicator */}
   291→      {row.reservationStatus === 'occupied' && (
   292→        <div className="flex items-center gap-0.5 text-amber-500" title={`Occupied by ${row.guestName || 'Guest'}`}>
   293→          <User className="h-3 w-3" />
   294→        </div>
   295→      )}
   296→      {/* View Details — always visible as separate button */}
   297→      <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground"
   298→        onClick={() => {
   299→          if (row.reservationStatus === 'occupied') {
   300→            toast.info(`Room ${row.roomNumber} — Occupied By "${row.guestName || 'Guest'}"`, { duration: 3000 })
   301→          }
   302→          onViewDetail(row.roomId)
   303→        }} title="View Details">
   304→        <Eye className="h-3.5 w-3.5" />
   305→      </Button>
   306→
   307→      {/* Three-dot menu for status actions */}
   308→      <DropdownMenu>
   309→        <DropdownMenuTrigger asChild>
   310→          <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground">
   311→            <MoreHorizontal className="h-3.5 w-3.5" />
   312→          </Button>
   313→        </DropdownMenuTrigger>
   314→        <DropdownMenuContent align="end" className="w-44">
   315→          {(s === 'pending' || s === 'assigned') && (
   316→            <DropdownMenuItem
   317→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'in_progress') }}
   318→              disabled={isForceMutated}
   319→              className="gap-2 text-xs text-sky-600 focus:text-sky-600 focus:bg-sky-50 dark:text-sky-400 dark:focus:bg-sky-950/40"
   320→            >
   321→              <PlayCircle className="h-3.5 w-3.5" /> Start Cleaning
   322→            </DropdownMenuItem>
   323→          )}
   324→
   325→          {(s === 'cleaning' || s === 'in_progress') && (
   326→            <DropdownMenuItem
   327→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'cleaned') }}
   328→              disabled={isForceMutated}
   329→              className="gap-2 text-xs text-emerald-600 focus:text-emerald-600 focus:bg-emerald-50 dark:text-emerald-400 dark:focus:bg-emerald-950/40"
   330→            >
   331→              <CheckCircle className="h-3.5 w-3.5" /> Mark Cleaned
   332→            </DropdownMenuItem>
   333→          )}
   334→
   335→          {s === 'cleaned' && (
   336→            <DropdownMenuItem
   337→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'inspected') }}
   338→              disabled={isForceMutated}
   339→              className="gap-2 text-xs text-purple-600 focus:text-purple-600 focus:bg-purple-50 dark:text-purple-400 dark:focus:bg-purple-950/40"
   340→            >
   341→              <ShieldCheck className="h-3.5 w-3.5" /> Mark Inspected
   342→            </DropdownMenuItem>
   343→          )}
   344→
   345→          {s === 'failed' && (
   346→            <DropdownMenuItem
   347→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'pending') }}
   348→              disabled={isForceMutated}
   349→              className="gap-2 text-xs text-amber-600 focus:text-amber-600 focus:bg-amber-50 dark:text-amber-400 dark:focus:bg-amber-950/40"
   350→            >
   351→              <RotateCcw className="h-3.5 w-3.5" /> Reset to Pending
   352→            </DropdownMenuItem>
   353→          )}
   354→
   355→          {/* Rush — always available (not blocked by force-mutated) */}
   356→          <DropdownMenuItem
   357→            onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'rush') }}
   358→            className="gap-2 text-xs text-red-500 focus:text-red-500 focus:bg-red-50 dark:text-red-400 dark:focus:bg-red-950/40"
   359→          >
   360→            <AlertTriangle className="h-3.5 w-3.5" /> Set Rush Priority
   361→          </DropdownMenuItem>
   362→
   363→          {/* Mark Failed — show for in_progress/cleaning/cleaned */}
   364→          {(s === 'cleaning' || s === 'in_progress' || s === 'cleaned') && (
   365→            <>
   366→              <DropdownMenuSeparator />
   367→              <DropdownMenuItem
   368→                onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'failed') }}
   369→                disabled={isForceMutated}
   370→                className="gap-2 text-xs text-red-600 focus:text-red-600 focus:bg-red-50 dark:text-red-400 dark:focus:bg-red-950/40"
   371→              >
   372→                <XCircle className="h-3.5 w-3.5" /> Mark Failed
   373→              </DropdownMenuItem>
   374→            </>
   375→          )}
   376→        </DropdownMenuContent>
   377→      </DropdownMenu>
   378→
   379→      {/* Locked badge */}
   380→      {isForceMutated && (
   381→        <Badge variant="outline" className="text-[9px] text-red-500 border-red-300 dark:border-red-700 shrink-0">
   382→          <Ban className="h-2.5 w-2.5 mr-0.5" /> Locked
   383→        </Badge>
   384→      )}
   385→    </div>
   386→  )
   387→}

   428→        </div>

   434→// ── Room Number Filter ────────────────────────────────────────
   435→function RoomFilterDropdown({ value, onChange }: { value: string; onChange: (v: string) => void }) {
   436→  const [open, setOpen] = useState(false)
   437→  const [input, setInput] = useState(value)
   438→  return (
   439→    <Popover open={open} onOpenChange={setOpen}>
   440→      <PopoverTrigger asChild>
   441→        <Button variant="ghost" size="sm" className="h-8 gap-1 text-xs font-semibold">
   442→          Room
   443→          {value && <span className="w-1.5 h-1.5 rounded-full bg-primary shrink-0" />}
   444→        </Button>
   445→      </PopoverTrigger>
   446→      <PopoverContent className="w-36 p-2" align="start">
   447→        <Input
   448→          placeholder="Room #..."
   449→          value={input}
   450→          onChange={(e) => setInput(e.target.value)}
   451→          onKeyDown={(e) => { if (e.key === 'Enter') { onChange(input); setOpen(false) } }}
   452→          className="h-7 text-xs"
   453→          autoFocus
   454→        />
   455→        <div className="flex gap-1 mt-2">
   456→          <Button size="sm" className="h-6 flex-1 text-[10px]" onClick={() => { onChange(input); setOpen(false) }}>Apply</Button>
   457→          <Button size="sm" variant="ghost" className="h-6 text-[10px]" onClick={() => { setInput(''); onChange(''); setOpen(false) }}>Clear</Button>
   458→        </div>
   459→      </PopoverContent>
   460→    </Popover>
