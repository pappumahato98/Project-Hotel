   103→// ─── Helper ──────────────────────────────────────────────────────────────
   104→
   105→function nightsBetween(start: string | Date, end: string | Date): number {
   106→  const s = new Date(start)
   107→  const e = new Date(end)
   108→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   109→}
   110→
   111→function formatDateValue(date: Date): string {
   112→  return toDateOnly(date)
   113→}
