     4→import { useMutation, useQueryClient, useQuery } from '@tanstack/react-query'
     5→import { toast } from 'sonner'
     6→import { apiFetch } from '@/lib/api'
     7→import {
     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,
     9→  Zap, Eye, Clock, Banknote, Mail, X, ArrowRight, BedSingle, FileText, BookOpen,
    10→} from 'lucide-react'
    11→
    12→import { Button } from '@/components/ui/button'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Card, CardContent } from '@/components/ui/card'
    15→import { Skeleton } from '@/components/ui/skeleton'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Separator } from '@/components/ui/separator'
    19→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'

    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select'

    28→} from '@/components/ui/dialog'

    34→// ─── Types ──────────────────────────────────────────────────────────────
    35→
    36→interface DepartureGuest {
    37→  id: string
    38→  firstName: string
    39→  lastName: string
    40→  vipLevel: string
    41→}
    42→
    43→interface DepartureRoom {
    44→  id: string
    45→  number: string
    46→  floor: number
    47→  type: { name: string; code: string }
    48→}
    49→
    50→interface FolioLineItem {
    51→  id: string
    52→  description: string
    53→  amount: number
    54→  type: 'charge' | 'payment'
    55→  date: string
    56→}
    57→
    58→interface DepartureFolio {
    59→  id: string
    60→  balance: number
    61→  status: string
    62→  items?: FolioLineItem[]
    63→}
    64→
    65→interface Departure {
    66→  id: string
    67→  confirmationNo: string
    68→  status: string
    69→  checkIn: string
    70→  checkOut: string
    71→  roomRate: number
    72→  totalAmount: number
    73→  paidAmount: number
    74→  creditLimit: number
    75→  guest: DepartureGuest
    76→  room: DepartureRoom
    77→  folios: DepartureFolio[]
    78→}
    79→
    80→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    81→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    82→
    83→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    84→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    85→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    86→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    87→}
    88→
    89→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    90→  cash: 'Cash',
    91→  card: 'Credit/Debit Card',
    92→  bank_transfer: 'Bank Transfer',
    93→}
    94→
    95→// ─── Component ──────────────────────────────────────────────────────────
    96→
    97→export function DeparturesView() {
    98→  const queryClient = useQueryClient()
    99→  const { settings } = useSettingsStore()
   100→  const { preferences } = usePreferencesStore()
   101→  const { navigateTo } = useNavigationStore()
   102→  const today = getTodayString()
   103→
   104→  // Dialog states
   105→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   106→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   107→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   108→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   109→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   110→
   111→  // Selected departure
   112→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   113→
   114→  // Late checkout
   115→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   116→
   117→  // Payment
   118→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   119→  const [paymentAmount, setPaymentAmount] = useState('')
   120→  const [paymentReference, setPaymentReference] = useState('')
   121→
   122→  // Receipt data
   123→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   124→
   125→  // Fetch today's departures (checked_in with checkOut = today)
   126→  const { data, isLoading } = useQuery({
   127→    queryKey: ['departures', today],
   128→    queryFn: async () => {
   129→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   130→      return apiFetch(`/api/reservations?${params.toString()}`)
   131→    },
   132→    refetchInterval: 30000,
   133→  })
   134→
   135→  const departures: Departure[] = data?.reservations || []
   136→
   137→  // Stats
   138→  const totalDepartures = departures.length
   139→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   140→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   141→  const outstandingBalance = departures.reduce((sum, d) => {
   142→    return sum + (d.folios[0]?.balance || 0)
   143→  }, 0)
   144→
   145→  // Zero-balance pending guests (for express & batch checkout)
   146→  const zeroBalancePending = departures.filter(
   147→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   148→  )
   149→
   150→  // ─── Mutations ─────────────────────────────────────────────────────
   151→
   152→  // Checkout mutation
   153→  const checkoutMutation = useMutation({
   154→    mutationFn: async (reservationId: string) => {
   155→      return apiFetch(`/api/reservations/${reservationId}`, {
   156→        method: 'PATCH',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ status: 'checked_out' }),
   159→      })
   160→    },
   161→    onSuccess: () => {
   162→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   163→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   164→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   165→    },
   166→  })
   167→
   168→  // Payment mutation
   169→  const paymentMutation = useMutation({
   170→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   171→      return apiFetch(`/api/folio/${folioId}`, {
   172→        method: 'POST',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   175→      })
   176→    },
   177→    onSuccess: () => {
   178→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   179→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   180→      setPaymentDialogOpen(false)
   181→      setPaymentAmount('')
   182→      setPaymentReference('')
   183→      toast.success('Payment recorded successfully')
   184→    },
   185→    onError: () => {
   186→      toast.error('Failed to record payment')
   187→    },
   188→  })
   189→
   190→  // Late checkout mutation
   191→  const lateCheckoutMutation = useMutation({
   192→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   193→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   194→        method: 'PATCH',
   195→        headers: { 'Content-Type': 'application/json' },
   196→        body: JSON.stringify({ checkOut: newCheckOut }),
   197→      })
   198→      // Add surcharge to folio
   199→      if (surcharge > 0) {
   200→        await apiFetch(`/api/folio`, {
   201→          method: 'POST',
   202→          headers: { 'Content-Type': 'application/json' },
   203→          body: JSON.stringify({
   204→            reservationId,
   205→            type: 'charge',
   206→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   207→            amount: surcharge,
   208→          }),
   209→        })
   210→      }
   211→      return result
   212→    },
   213→    onSuccess: () => {
   214→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   215→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   216→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   217→      setLateCheckoutDialogOpen(false)
   218→      toast.success('Late checkout request processed successfully')
   219→    },
   220→    onError: () => {
   221→      toast.error('Failed to process late checkout request')
   222→    },
   223→  })
   224→
   225→  // Batch checkout mutation
   226→  const batchCheckoutMutation = useMutation({
   227→    mutationFn: async (ids: string[]) => {
   228→      const results = []
   229→      for (let i = 0; i < ids.length; i++) {
   230→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   231→          method: 'PATCH',
   232→          headers: { 'Content-Type': 'application/json' },
   233→          body: JSON.stringify({ status: 'checked_out' }),
   234→        })
   235→        results.push(result)
   236→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   237→      }
   238→      return results
   239→    },
   240→    onSuccess: (_data, ids) => {
   241→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   242→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   243→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   244→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   245→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   246→    },
   247→    onError: () => {
   248→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   249→    },
   250→  })
   251→
   252→  // ─── Handlers ─────────────────────────────────────────────────────
   253→
   254→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   255→
   256→  const handleExpressCheckout = (dep: Departure) => {
   257→    checkoutMutation.mutate(dep.id, {
   258→      onSuccess: () => {
   259→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   260→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   261→      },
   262→    })
   263→  }
   264→
   265→  const handleReviewFolio = (dep: Departure) => {
   266→    setSelectedDeparture(dep)
   267→    setFolioDialogOpen(true)
   268→  }
   269→
   270→  const handleNavigateToFolio = (dep: Departure) => {
   271→    useFolioContextStore.getState().setFolioContext({
   272→      reservationId: dep.id,
   273→      guestId: dep.guest?.id || '',
   274→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   275→      roomNumber: dep.room?.number || '',
   276→      confirmationNo: dep.confirmationNo,
   277→    })
   278→    navigateTo('front-desk', 'folio')
   279→  }
   280→
   281→  const handleViewLedger = (dep: Departure) => {
   282→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   283→      guestId: dep.guest?.id || '',
   284→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   285→    })
   286→    navigateTo('front-desk', 'guest-ledger')
   287→  }
   288→
   289→  const handleFolioSettleCheckout = () => {
   290→    setFolioDialogOpen(false)
   291→    setCheckoutDialogOpen(true)
   292→  }
   293→
   294→  const handleQuickCheckout = (dep: Departure) => {
   295→    setSelectedDeparture(dep)
   296→    setCheckoutDialogOpen(true)
   297→  }
   298→
   299→  const handleRequestLateCheckout = (dep: Departure) => {
   300→    setSelectedDeparture(dep)
   301→    setLateCheckoutOption('14:00')
   302→    setLateCheckoutDialogOpen(true)
   303→  }
   304→
   305→  const handleConfirmLateCheckout = () => {
   306→    if (!selectedDeparture) return
   307→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   308→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   309→
   310→    // Build new checkout datetime with the selected time
   311→    const currentCheckout = new Date(selectedDeparture.checkOut)
   312→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   313→    currentCheckout.setHours(hours, minutes, 0, 0)
   314→
   315→    lateCheckoutMutation.mutate({
   316→      reservationId: selectedDeparture.id,
   317→      newCheckOut: currentCheckout.toISOString(),
   318→      surcharge,
   319→    })
   320→  }
   321→
   322→  const handleOpenPaymentDialog = () => {
   323→    if (!selectedDeparture) return
   324→    const balance = getBalance(selectedDeparture)
   325→    setPaymentAmount(String(balance))
   326→    setPaymentMethod('cash')
   327→    setPaymentReference('')
   328→    setPaymentDialogOpen(true)
   329→  }
   330→
   331→  const handleRecordPayment = () => {
   332→    if (!selectedDeparture) return
   333→    const folioId = selectedDeparture.folios[0]?.id
   334→    if (!folioId) {
   335→      toast.error('No folio found for this reservation')
   336→      return
   337→    }
   338→    const amount = Number(paymentAmount)
   339→    if (isNaN(amount) || amount <= 0) {
   340→      toast.error('Please enter a valid payment amount')
   341→      return
   342→    }
   343→    paymentMutation.mutate({
   344→      folioId,
   345→      amount,
   346→      method: paymentMethod,
   347→      reference: paymentReference,
   348→    })
   349→  }
   350→
   351→  const confirmCheckout = () => {
   352→    if (!selectedDeparture) return
   353→    const roomNumber = selectedDeparture.room.number
   354→    checkoutMutation.mutate(selectedDeparture.id, {
   355→      onSuccess: () => {
   356→        setCheckoutDialogOpen(false)
   357→        setReceiptData(selectedDeparture)
   358→        setReceiptDialogOpen(true)
   359→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   360→      },
   361→    })
   362→  }
   363→
   364→  const handleBatchCheckout = () => {
   365→    const ids = zeroBalancePending.map((d) => d.id)
   366→    if (ids.length === 0) {
   367→      toast.info('No guests with zero balance to batch checkout')
   368→      return
   369→    }
   370→    batchCheckoutMutation.mutate(ids)
   371→  }
   372→
   373→  const handlePrintReceipt = () => {
   374→    toast.success('Receipt sent to printer')
   375→  }
   376→
   377→  const handleEmailReceipt = () => {
   378→    toast.success('Receipt emailed to guest')
   379→  }
   380→
   381→  // ─── Computed values ──────────────────────────────────────────────
   382→
   383→  const lateCheckoutFee = selectedDeparture
   384→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   385→    : 0
   386→
   387→  // ─── Render ────────────────────────────────────────────────────────
   388→
   389→  return (
   390→    <div className="flex flex-col gap-2">
   391→      {/* Header */}
   392→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   393→        <div>
   394→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   395→          <p className="text-xs text-muted-foreground">
   396→            Guest check-outs scheduled for {formatDate(today)}
   397→          </p>
   398→        </div>
   399→        {zeroBalancePending.length > 0 && (
   400→          <Button
   401→            onClick={handleBatchCheckout}
   402→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   403→            disabled={batchCheckoutMutation.isPending}
   404→          >
   405→            <Zap className="size-4 mr-1.5" />
   406→            {batchCheckoutMutation.isPending
   407→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   408→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   409→          </Button>
   410→        )}
   411→      </div>
   412→
   413→      {/* Departures List */}
   414→      <Card className="py-0">
   415→        <CardContent className="p-0">
   416→          <div className="max-h-[600px] overflow-y-auto">
   417→            <Table>
   418→              <TableHeader>
   419→                <TableRow>
   420→                  <TableHead className="w-[70px]">Room</TableHead>
   421→                  <TableHead>Guest</TableHead>
   422→                  <TableHead className="w-[100px]">Check-out</TableHead>
   423→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   424→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   425→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   426→                </TableRow>
   427→              </TableHeader>
   428→              <TableBody>
   429→                {isLoading ? (
   430→                  Array.from({ length: 5 }).map((_, i) => (
   431→                    <TableRow key={i}>
   432→                      {Array.from({ length: 6 }).map((_, j) => (
   433→                        <TableCell key={j}>
   434→                          <Skeleton className="h-4 w-full" />
   435→                        </TableCell>
   436→                      ))}
   437→                    </TableRow>
   438→                  ))
   439→                ) : departures.length === 0 ? (
   440→                  <TableRow>
   441→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   442→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   443→                      No departures scheduled for today
   444→                    </TableCell>
   445→                  </TableRow>
   446→                ) : (
   447→                  departures.map((dep) => {
   448→                    const balance = getBalance(dep)
   449→                    const isPending = dep.status === 'checked_in'
   450→                    const isZeroBalance = balance === 0
   451→                    return (
   452→                      <TableRow key={dep.id}>
   453→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   454→                        <TableCell>
   455→                          <div className="flex items-center gap-2">
   456→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   457→                            {dep.guest.vipLevel !== 'none' && (
   458→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   459→                                VIP
   460→                              </Badge>
   461→                            )}
   462→                          </div>
   463→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   464→                        </TableCell>
   465→                        <TableCell className="text-xs">
   466→                          <div>{formatDate(dep.checkOut)}</div>
   467→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   468→                        </TableCell>
   469→                        <TableCell className="text-right">
   470→                          <div className="flex items-center justify-end gap-1.5">
   471→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   472→                              {formatCurrency(balance)}
   473→                            </span>
   474→                            {/* Outstanding Balance Alert Badges */}
   475→                            {isPending && balance > 5000 && (
   476→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   477→                                HIGH
   478→                              </Badge>
   479→                            )}
   480→                            {isPending && balance > 0 && balance <= 5000 && (
   481→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   482→                                DUE
   483→                              </Badge>
   484→                            )}
   485→                          </div>
   486→                        </TableCell>
   487→                        <TableCell className="text-right">
   488→                          {dep.status === 'checked_out' ? (
   489→                            <div className="flex items-center justify-end gap-1">
   490→                              <BedSingle className="size-3 text-yellow-500" />
   491→                              <StatusBadge status="vacant_dirty" />
   492→                            </div>
   493→                          ) : (
   494→                            <StatusBadge status="checked_in" />
   495→                          )}
   496→                        </TableCell>
   497→                        <TableCell className="text-right">
   498→                          <div className="flex justify-end gap-1 flex-wrap">
   499→                            {isPending && (
   500→                              <>
   501→                                {/* Express Checkout (zero balance) */}
   502→                                {isZeroBalance && (
   503→                                  <Button
   504→                                    size="sm"
   505→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   506→                                    onClick={() => handleExpressCheckout(dep)}
   507→                                    disabled={checkoutMutation.isPending}
   508→                                  >
   509→                                    <Zap className="size-3 mr-0.5" /> Express
   510→                                  </Button>
   511→                                )}
   512→                                {/* Review Folio */}
   513→                                <Button
   514→                                  size="sm"
   515→                                  variant="outline"
   516→                                  className="text-xs h-7"
   517→                                  onClick={() => handleReviewFolio(dep)}
   518→                                >
   519→                                  <Eye className="size-3 mr-0.5" /> Review Folio
   520→                                </Button>
   521→                                <Button
   522→                                  size="sm"
   523→                                  variant="outline"
   524→                                  className="text-xs h-7"
   525→                                  onClick={() => handleNavigateToFolio(dep)}
   526→                                >
   527→                                  <FileText className="size-3 mr-0.5" /> Full Folio
   528→                                </Button>
   529→                                <Button
   530→                                  size="sm"
   531→                                  variant="outline"
   532→                                  className="text-xs h-7"
   533→                                  onClick={() => handleViewLedger(dep)}
   534→                                >
   535→                                  <BookOpen className="size-3 mr-0.5" /> View Ledger
   536→                                </Button>
   537→                                {/* Late Checkout */}
   538→                                <Button
   539→                                  size="sm"
   540→                                  variant="outline"
   541→                                  className="text-xs h-7"
   542→                                  onClick={() => handleRequestLateCheckout(dep)}
   543→                                >
   544→                                  <Clock className="size-3 mr-0.5" /> Late
   545→                                </Button>
   546→                                {/* Regular Checkout */}
   547→                                <Button
   548→                                  size="sm"
   549→                                  className="text-xs h-7"
   550→                                  onClick={() => handleQuickCheckout(dep)}
   551→                                >
