   111→function isSameDay(dateStr: string, todayStr: string): boolean {
   112→  const d = new Date(dateStr)
   113→  const t = new Date(todayStr)
   114→  return d.getFullYear() === t.getFullYear() && d.getMonth() === t.getMonth() && d.getDate() === t.getDate()
   115→}
   116→
   117→function isBeforeDay(dateStr: string, todayStr: string): boolean {
   118→  const d = new Date(dateStr)
   119→  d.setHours(0, 0, 0, 0)
   120→  const t = new Date(todayStr)
   121→  t.setHours(0, 0, 0, 0)
   122→  return d.getTime() < t.getTime()
   123→}

   378→  // Late checkout

   381→  // Payment

   386→  // Receipt data

   391→  // Pending departures: checked_in with checkOut <= today (includes overdue)

   406→    queryKey: ['departures-checked-out', today],
   407→    queryFn: async () => {
   408→      const params = new URLSearchParams({
   409→        status: 'checked_out',
   410→        checkOutDate: today,
   411→      })
   412→      return apiFetch(`/api/reservations?${params.toString()}`)

   497→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   498→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   499→        method: 'PATCH',
   500→        headers: { 'Content-Type': 'application/json' },
   501→        body: JSON.stringify({ checkOut: newCheckOut }),
   502→      })

   506→          method: 'POST',
   507→          headers: { 'Content-Type': 'application/json' },

   514→        })

   524→    onError: () => {
   525→      toast.error('Failed to process late checkout request')
   526→    },
   527→  })
   528→
   529→  // Batch checkout mutation
   530→  const batchCheckoutMutation = useMutation({
   531→    mutationFn: async (ids: string[]) => {
   532→      const results = []
   533→      for (let i = 0; i < ids.length; i++) {
   534→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   535→          method: 'PATCH',
   536→          headers: { 'Content-Type': 'application/json' },
   537→          body: JSON.stringify({ status: 'checked_out' }),
   538→        })
   539→        results.push(result)
   540→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   541→      }
   542→      return results
   543→    },
   544→    onSuccess: (_data, ids) => {
   545→      invalidate.afterCheckout(queryClient)
   546→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   547→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   548→    },
   549→    onError: () => {
   550→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   551→    },
   552→  })
   553→
   554→  // ─── Handlers ─────────────────────────────────────────────────────
   555→
   556→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   557→
   558→  const handleExpressCheckout = (dep: Departure) => {
   559→    checkoutMutation.mutate(dep.id, {
   560→      onSuccess: () => {
   561→        toast.success(`Express checkout complete for Room ${dep.room?.number || '—'}`)
   562→        toast.info(`Room ${dep.room?.number || '—'} marked for housekeeping - Vacant Dirty`)
