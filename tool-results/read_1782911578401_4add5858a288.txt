   257→   257→function TaskDetailDialog({ task, open, onOpenChange }: { task: HkTask | null; open: boolean; onOpenChange: (open: boolean) => void }) {
   258→   258→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   259→   259→  const queryClient = useQueryClient()
   260→   260→
   261→   261→  const updateStatusMutation = useMutation({
   262→   262→    mutationFn: (body: { id: string; status: string; inspectedBy?: string }) =>
   263→   263→      apiFetch('/api/housekeeping', { method: 'POST', body }),
   264→   264→    onSuccess: () => {
   265→   265→      toast.success('Task status updated')
   266→   266→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   267→   267→      queryClient.invalidateQueries({ queryKey: ['housekeeping-rooms'] })
   268→   268→      onOpenChange(false)
   269→   269→    },
   270→   270→  })
   271→   271→
   272→   272→  if (!task) return null
   273→   273→
   274→   274→  const toggleCheck = (item: string) => {
   275→   275→    setCheckedItems((prev) => ({ ...prev, [item]: !prev[item] }))
   276→   276→  }
   277→   277→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   278→   278→
   279→   279→  return (
   280→   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→   281→      <DialogContent className="sm:max-w-lg">
   282→   282→        <DialogHeader>
   283→   283→          <DialogTitle className="flex items-center gap-2">
   284→   284→            Room {task.room.number} — {formatTaskType(task.taskType)}
   285→   285→          </DialogTitle>
   286→   286→          <DialogDescription>
   287→   287→            Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   288→   288→          </DialogDescription>
   289→   289→        </DialogHeader>
   290→   290→
   291→   291→        <div className="space-y-4">
   292→   292→          <div className="grid grid-cols-2 gap-3">
   293→   293→            <div className="rounded-lg border p-3 space-y-1">
   294→   294→              <p className="text-xs text-muted-foreground">Priority</p>
   295→   295→              <Badge variant="outline" className={cn(priorityColor(task.priority))}>
   296→   296→                {task.priority.charAt(0).toUpperCase() + task.priority.slice(1)}
   297→   297→              </Badge>
   298→   298→            </div>
   299→   299→            <div className="rounded-lg border p-3 space-y-1">

   302→   302→            </div>
   303→   303→            <div className="rounded-lg border p-3 space-y-1">
   304→   304→              <p className="text-xs text-muted-foreground">Assigned To</p>
   305→   305→              <p className="text-sm font-medium">{task.assignedTo || 'Unassigned'}</p>
   306→   306→            </div>
   307→   307→            <div className="rounded-lg border p-3 space-y-1">
   308→   308→              <p className="text-xs text-muted-foreground">Est. Time</p>
   309→   309→              <p className="text-sm font-medium">{task.estimatedMinutes} min</p>
   310→   310→            </div>
   311→   311→          </div>
   312→   312→
   313→   313→          <div className="space-y-2">
   314→   314→            <div className="flex items-center justify-between">
   315→   315→              <p className="text-sm font-semibold">Cleaning Checklist</p>

   384→   384→function AttendantView({ tasks }: { tasks: HkTask[] }) {
   385→   385→  const attendants = new Map<string, HkTask[]>()
   386→   386→  tasks.forEach((task) => {
   387→   387→    const name = task.assignedTo || 'Unassigned'
   388→   388→    if (!attendants.has(name)) attendants.set(name, [])
   389→   389→    attendants.get(name)!.push(task)
   390→   390→  })
   391→   391→
   392→   392→  return (
   393→   393→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   394→   394→      {Array.from(attendants.entries()).map(([name, attTasks]) => (
   395→   395→        <Card key={name}>
   396→   396→          <CardContent className="p-3 space-y-2">
   397→   397→            <div className="flex items-center gap-3">
   398→   398→              <div className="flex h-9 w-9 items-center justify-center rounded-full bg-muted text-sm font-medium">
   399→   399→                {name.charAt(0)}
   400→   400→              </div>
