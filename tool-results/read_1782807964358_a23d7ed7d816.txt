   194→function useReportQuery<T>(type: ReportType, date?: string) {
   195→  return useQuery<T>({
   196→    queryKey: ['front-desk-reports', type, date],
   197→    queryFn: async () => {
   198→      const params = new URLSearchParams({ type })
   199→      if (date) params.set('date', date)
   200→      return apiFetch(`/api/front-desk/reports?${params.toString()}`)
   201→    },
   202→    refetchInterval: 30000,
   203→  })
   204→}

   312→function SourceBadge({ source }: { source: string | null }) {

   317→    walk_in: 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-300 border-violet-300 dark:border-violet-700',
   318→    phone: 'bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   319→    email: 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-300 border-cyan-300 dark:border-cyan-700',
   320→    agent: 'bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300 border-orange-300 dark:border-orange-700',
   321→    corporate: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300 border-slate-300 dark:border-slate-600',
   322→    group: 'bg-pink-100 text-pink-700 dark:bg-pink-950 dark:text-pink-300 border-pink-300 dark:border-pink-700',

   324→
   325→  const colorClass = colors[source] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300 border-gray-300 dark:border-gray-600'

   327→  return (
   328→    <Badge variant="outline" className={cn('text-[10px] px-1.5 py-0', colorClass)}>
   329→      {source.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}
   330→    </Badge>
   331→  )
   332→}

   462→// 2. Arrivals Report
   463→function ArrivalsReport({ date }: { date?: string }) {
   464→  const { data, isLoading, error } = useReportQuery<ArrivalsData>('arrivals', date)
   465→
   466→  if (isLoading) return <TableSkeleton cols={6} rows={6} />
   467→  if (error) return <EmptyState icon={ArrowDownToLine} message="Failed to load arrivals report" />
   468→
   469→  const arrivalsData = data as ArrivalsData
   470→  const arrivals = arrivalsData.reservations || []
   471→
   472→  if (arrivals.length === 0) {
   473→    return <EmptyState icon={ArrowDownToLine} message="No arrivals scheduled for this date" />
   474→  }
   475→
   476→  return (
   477→    <Card className="py-0">
   478→      <CardContent className="p-0">
   479→        <Table>
   480→          <TableHeader>
   481→            <TableRow>
   482→              <TableHead className="text-xs">Guest</TableHead>
   483→              <TableHead className="text-xs">Confirmation</TableHead>
   484→              <TableHead className="text-xs">Room</TableHead>
   485→              <TableHead className="text-xs hidden md:table-cell">Check-In</TableHead>
   486→              <TableHead className="text-xs hidden lg:table-cell">Check-Out</TableHead>
   487→              <TableHead className="text-xs hidden sm:table-cell">Source</TableHead>
   488→              <TableHead className="text-xs">Status</TableHead>
   489→            </TableRow>
   490→          </TableHeader>
   491→          <TableBody>
   492→            {arrivals.map((res, idx) => {
   493→              const guestName = res.guest
   494→                ? `${res.guest.firstName} ${res.guest.lastName}`
   495→                : 'Unknown Guest'
   496→              const isVip = res.guest?.vipLevel && res.guest.vipLevel !== 'none'
   497→
   498→              return (
   499→                <TableRow
   500→                  key={res.id}
   501→                  className={cn(idx % 2 !== 0 && 'bg-muted/30')}
   502→                >
   503→                  <TableCell>
   504→                    <div className="flex items-center gap-2">
   505→                      <span className="font-medium text-xs">{guestName}</span>
   506→                      {isVip && (
   507→                        <Badge className="text-[9px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700">
   508→                          VIP
   509→                        </Badge>
   510→                      )}
   511→                    </div>
   512→                    {res.specialRequests && (
   513→                      <p className="text-[10px] text-muted-foreground mt-0.5 max-w-[200px] truncate">
   514→                        {res.specialRequests}
   515→                      </p>
   516→                    )}
   517→                  </TableCell>
   518→                  <TableCell>
   519→                    <span className="font-mono text-xs">{res.confirmationNo}</span>
   520→                  </TableCell>
   521→                  <TableCell>
   522→                    {res.room ? (
   523→                      <span className="text-xs flex items-center gap-1">{res.room.number} <RoomTypeBedBadge typeName={res.room.type.name} bedConfig={res.room.type.bedConfig} typeCode={res.room.type.code} inline /></span>
   524→                    ) : (
   525→                      <Badge variant="outline" className="text-[10px] px-1 py-0 border-amber-400 text-amber-600 dark:text-amber-400">
   526→                        Unassigned
   527→                      </Badge>
   528→                    )}
   529→                  </TableCell>
   530→                  <TableCell className="hidden md:table-cell">
   531→                    <span className="text-xs text-muted-foreground flex items-center gap-1">
   532→                      <Clock className="size-3" />
   533→                      {formatDate(res.checkIn)}
   534→                    </span>
   535→                  </TableCell>

   537→                    <span className="text-xs text-muted-foreground">{formatDate(res.checkOut)}</span>
   538→                  </TableCell>
   539→                  <TableCell className="hidden sm:table-cell">
   540→                    <SourceBadge source={res.source} />

   542→                  <TableCell>
   543→                    <Badge
   544→                      variant="outline"
   545→                      className={cn(
   546→                        'text-[10px] px-1.5 py-0',
   547→                        res.status === 'checked_in'
   548→                          ? 'bg-green-50 text-green-700 dark:bg-green-950 dark:text-green-300 border-green-300 dark:border-green-700'
   549→                          : 'bg-sky-50 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   550→                      )}
   551→                    >
   552→                      {res.status.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}
   553→                    </Badge>
   554→                  </TableCell>
