   235→// ── Component ────────────────────────────────────────────────
   236→export function PurchaseOrdersView() {
   237→  const queryClient = useQueryClient()
   238→  const [purchaseOrders, setPurchaseOrders] = useState<PurchaseOrder[]>(MOCK_PURCHASE_ORDERS)
   239→  const [searchQuery, setSearchQuery] = useState('')
   240→  const [statusFilter, setStatusFilter] = useState('all')
   241→  const [vendorFilter, setVendorFilter] = useState('all')
   242→
   243→  // Dialog states
   244→  const [createOpen, setCreateOpen] = useState(false)
   245→  const [viewOpen, setViewOpen] = useState(false)
   246→  const [approveOpen, setApproveOpen] = useState(false)
   247→  const [rejectOpen, setRejectOpen] = useState(false)
   248→  const [statusChangeOpen, setStatusChangeOpen] = useState(false)
   249→  const [selectedPO, setSelectedPO] = useState<PurchaseOrder | null>(null)
   250→  const [nextStatus, setNextStatus] = useState('')
   251→  const [form, setForm] = useState<POForm>(EMPTY_PO_FORM)
   252→
   253→  // ── Queries ─────────────────────────────────────────────────
   254→  const { data: vendorsData } = useQuery({
   255→    queryKey: ['vendors'],
   256→    queryFn: fetchVendors,
   257→  })
   258→
   259→  const vendors: Vendor[] = useMemo(() => {
   260→    const apiVendors = vendorsData?.vendors?.map((v: { id: string; name: string }) => ({ id: v.id, name: v.name })) || []
   261→    // Merge mock vendor names not in API
   262→    const mockVendorNames = ['CleanPro Nepal', 'Thamel Tech Solutions', 'Pokhara Paints & Hardware']
   263→    const existingNames = new Set(apiVendors.map((v: Vendor) => v.name))
   264→    mockVendorNames.forEach((name) => {
   265→      if (!existingNames.has(name)) {
   266→        apiVendors.push({ id: `mock-${name.toLowerCase().replace(/\s+/g, '-')}`, name })
   267→      }
   268→    })
   269→    return apiVendors
   270→  }, [vendorsData])
   271→
   272→  // ── Filtering ────────────────────────────────────────────────
   273→  const filteredPOs = useMemo(() => {
   274→    return purchaseOrders.filter((po) => {
   275→      if (statusFilter !== 'all' && po.status !== statusFilter) return false
   276→      if (vendorFilter !== 'all' && po.vendorId !== vendorFilter && po.vendor !== vendorFilter) return false
   277→      if (!searchQuery) return true
   278→      const q = searchQuery.toLowerCase()
   279→      return po.poNumber.toLowerCase().includes(q) || po.vendor.toLowerCase().includes(q)
   280→    })
   281→  }, [purchaseOrders, searchQuery, statusFilter, vendorFilter])
   282→
   283→  // ── Summary stats ──────────────────────────────────────────
   284→  const summaryStats = useMemo(() => {
   285→    const totalPOs = purchaseOrders.length
   286→    const pendingApproval = purchaseOrders.filter((po) => po.status === 'pending').length
   287→    const inTransit = purchaseOrders.filter((po) => po.status === 'ordered' || po.status === 'partial').length
   288→    const totalValue = purchaseOrders.reduce((sum, po) => sum + po.totalAmount, 0)
   289→    return { totalPOs, pendingApproval, inTransit, totalValue }
   290→  }, [purchaseOrders])
   291→
   292→  // ── Helpers ─────────────────────────────────────────────────
   293→  const openCreate = () => {
   294→    setForm(EMPTY_PO_FORM)
   295→    setCreateOpen(true)
   296→  }
   297→
   298→  const openView = (po: PurchaseOrder) => {
   299→    setSelectedPO(po)
   300→    setViewOpen(true)
   301→  }
   302→
   303→  const openApprove = (po: PurchaseOrder) => {
   304→    setSelectedPO(po)
   305→    setApproveOpen(true)
   306→  }
   307→
   308→  const openReject = (po: PurchaseOrder) => {
   309→    setSelectedPO(po)
   310→    setRejectOpen(true)
   311→  }
   312→
   313→  const openStatusChange = (po: PurchaseOrder, status: string) => {
   314→    setSelectedPO(po)
   315→    setNextStatus(status)
   316→    setStatusChangeOpen(true)
   317→  }
   318→
   319→  const updateFormItems = (newItems: POItem[]) => {
   320→    setForm((f) => ({ ...f, items: newItems }))
   321→  }
   322→
   323→  const updatePOStatus = (poId: string, newStatus: string, approvedBy?: string) => {
   324→    setPurchaseOrders((prev) =>
   325→      prev.map((po) =>
   326→        po.id === poId
   327→          ? {
   328→              ...po,
   329→              status: newStatus,
   330→              ...(approvedBy ? { approvedBy, approvedAt: new Date().toISOString() } : {}),
   331→            }
   332→          : po
   333→      )
   334→    )
   335→    toast.success(`PO updated to ${newStatus.charAt(0).toUpperCase() + newStatus.slice(1)}`)
   336→  }
   337→
   338→  const getNextStatuses = (po: PurchaseOrder): string[] => {
   339→    return STATUS_WORKFLOW[po.status] || []
   340→  }
   341→
   342→  // ── Handle Create ──────────────────────────────────────────
   343→  const handleCreate = () => {
   344→    const vendor = vendors.find((v) => v.id === form.vendorId)
   345→    if (!vendor || form.items.every((i) => !i.name)) return
   346→
   347→    const newPO: PurchaseOrder = {
   348→      id: `po-${Date.now()}`,
   349→      poNumber: `PO-2024-${String(purchaseOrders.length + 1).padStart(3, '0')}`,
   350→      vendor: vendor.name,
   351→      vendorId: vendor.id,
   352→      date: new Date().toISOString(),
   353→      expectedDelivery: form.expectedDelivery || '',
   354→      items: form.items.filter((i) => i.name),
   355→      totalAmount: form.items.reduce((sum, item) => sum + item.quantity * item.unitPrice, 0),
   356→      priority: form.priority.toLowerCase(),
   357→      status: 'draft',
   358→      notes: form.notes || undefined,
   359→      terms: form.terms || undefined,
   360→      createdAt: new Date().toISOString(),
   361→    }
   362→
   363→    setPurchaseOrders((prev) => [newPO, ...prev])
   364→    setCreateOpen(false)
   365→    setForm(EMPTY_PO_FORM)
   366→    toast.success('Purchase Order created successfully')
   367→  }
   368→
   369→  // ── Render ─────────────────────────────────────────────────
   370→  return (
   371→    <div className="space-y-2">
   372→      {/* Header */}
   373→      <div className="flex items-center justify-between">
   374→        <div>
   375→          <h1 className="text-sm font-semibold text-gray-700 dark:text-gray-200 tracking-tight">Purchase Orders</h1>
   376→          <p className="text-xs text-muted-foreground">Manage purchase orders and track deliveries</p>
   377→        </div>
   378→        <div className="flex items-center gap-2">
   379→          <Badge variant="outline" className="text-[11px]">
   380→            {summaryStats.totalPOs} orders
   381→          </Badge>
   382→          <Button className="gap-1" onClick={openCreate}>
   383→            <Plus className="h-4 w-4" />
   384→            Create New PO
   385→          </Button>
   386→        </div>
   387→      </div>
   388→
   389→      {/* Summary Cards */}
   390→      <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-4">
   391→        <Card className="p-2.5">
   392→          <div className="flex items-center gap-2">
   393→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-blue-100 dark:bg-blue-950">
   394→              <FileText className="h-5 w-5 text-blue-600" />
   395→            </div>
   396→            <div>
   397→              <p className="text-xs text-muted-foreground">Total POs</p>
   398→              <p className="text-lg font-bold">{summaryStats.totalPOs}</p>
   399→            </div>
   400→          </div>
   401→        </Card>
   402→        <Card className="p-2.5">
   403→          <div className="flex items-center gap-2">
   404→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-950">
   405→              <Clock className="h-5 w-5 text-amber-600" />
   406→            </div>
   407→            <div>
   408→              <p className="text-xs text-muted-foreground">Pending Approval</p>
   409→              <p className="text-lg font-bold text-amber-600">{summaryStats.pendingApproval}</p>
   410→            </div>
   411→          </div>
   412→        </Card>
   413→        <Card className="p-2.5">
   414→          <div className="flex items-center gap-2">
   415→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-purple-100 dark:bg-purple-950">
   416→              <Truck className="h-5 w-5 text-purple-600" />
   417→            </div>
   418→            <div>
   419→              <p className="text-xs text-muted-foreground">In Transit</p>
   420→              <p className="text-lg font-bold text-purple-600">{summaryStats.inTransit}</p>
   421→            </div>
   422→          </div>
   423→        </Card>
   424→        <Card className="p-2.5">
   425→          <div className="flex items-center gap-2">
   426→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-green-100 dark:bg-green-950">
   427→              <DollarSign className="h-5 w-5 text-green-600" />
   428→            </div>
   429→            <div>
   430→              <p className="text-xs text-muted-foreground">Total PO Value</p>
   431→              <p className="text-lg font-bold">{formatNPR(summaryStats.totalValue)}</p>
   432→            </div>
   433→          </div>
   434→        </Card>
   435→      </div>
   436→
   437→      {/* Filters */}
   438→      <div className="flex flex-wrap items-center gap-2">
   439→        <div className="relative flex-1 min-w-[200px] max-w-sm">
   440→          <Search className="absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   441→          <Input
