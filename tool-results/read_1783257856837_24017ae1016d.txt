    82→// ── Format helpers ────────────────────────────────────────────
    83→function formatAuditAction(action: string): { label: string; color: string } {
    84→  switch (action) {
    85→    case 'reject_reassign': return { label: 'Rejected & Reassigned', color: 'text-red-600 bg-red-50 border-red-200 dark:text-red-400 dark:bg-red-950/30 dark:border-red-800' }
    86→    case 'approve': return { label: 'Approved', color: 'text-green-600 bg-green-50 border-green-200 dark:text-green-400 dark:bg-green-950/30 dark:border-green-800' }
    87→    case 'force_mutation': return { label: 'Force Mutated', color: 'text-amber-600 bg-amber-50 border-amber-200 dark:text-amber-400 dark:bg-amber-950/30 dark:border-amber-800' }
    88→    default: return { label: action, color: 'text-muted-foreground bg-muted border-border' }
    89→  }
    90→}

    92→function formatDateTime(iso: string): string {
    93→  const d = new Date(iso)
    94→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
    95→    + ' ' + d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
    96→}

   141→// ── Photo Capture Component ──────────────────────────────────
   142→function PhotoCaptureSection({
   143→  photos,
   144→  onCapture,
   145→  onRemove,
   146→}: {
   147→  photos: string[]
   148→  onCapture: (dataUrl: string) => void
   149→  onRemove: (index: number) => void
   150→}) {
   151→  const videoRef = useRef<HTMLVideoElement>(null)
   152→  const streamRef = useRef<MediaStream | null>(null)
   153→  const [cameraOpen, setCameraOpen] = useState(false)
   154→  const [cameraError, setCameraError] = useState<string | null>(null)
   155→
   156→  const startCamera = useCallback(async () => {
   157→    try {
   158→      setCameraError(null)
   159→      const stream = await navigator.mediaDevices.getUserMedia({
   160→        video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
   161→      })
   162→      streamRef.current = stream
   163→      if (videoRef.current) {
   164→        videoRef.current.srcObject = stream
   165→      }
   166→      setCameraOpen(true)
   167→    } catch (err) {
   168→      console.error('Camera error:', err)
   169→      setCameraError('Camera access denied or unavailable')
   170→    }
   171→  }, [])
   172→
   173→  const stopCamera = useCallback(() => {
   174→    if (streamRef.current) {
   175→      streamRef.current.getTracks().forEach((t) => t.stop())
   176→      streamRef.current = null
   177→    }
   178→    setCameraOpen(false)
   179→  }, [])
   180→
   181→  const capturePhoto = useCallback(() => {
   182→    const video = videoRef.current
   183→    if (!video) return
   184→    const canvas = document.createElement('canvas')
   185→    canvas.width = video.videoWidth
   186→    canvas.height = video.videoHeight
   187→    const ctx = canvas.getContext('2d')
   188→    if (!ctx) return
   189→    ctx.drawImage(video, 0, 0)
   190→    const dataUrl = canvas.toDataURL('image/jpeg', 0.85)
   191→    onCapture(dataUrl)
   192→    stopCamera()
   193→  }, [onCapture, stopCamera])
   194→
   195→  return (
   196→    <div className="space-y-2">
   197→      <p className="text-sm font-semibold">Photo Evidence</p>
   198→
   199→      {/* Camera Preview */}
   200→      {cameraOpen && (
   201→        <div className="relative rounded-lg border overflow-hidden bg-black">
   202→          <video
   203→            ref={videoRef}
   204→            autoPlay
   205→            playsInline
   206→            muted
   207→            className="w-full h-36 sm:h-48 object-cover"
   208→          />
   209→          <div className="absolute bottom-2 left-1/2 -translate-x-1/2 flex gap-2">
   210→            <Button
   211→              size="sm"
   212→              className="gap-1.5 bg-white text-black hover:bg-gray-200"
   213→              onClick={capturePhoto}
   214→            >
   215→              <Camera className="h-4 w-4" /> Capture
   216→            </Button>
   217→            <Button
   218→              size="sm"
   219→              variant="outline"
   220→              className="gap-1.5 bg-white/90 text-black hover:bg-gray-200 border-gray-300"
   221→              onClick={stopCamera}
   222→            >
   223→              <X className="h-4 w-4" /> Cancel
   224→            </Button>
   225→          </div>
   226→        </div>
   227→      )}
   228→
   229→      {cameraError && (
   230→        <p className="text-xs text-red-500 flex items-center gap-1">
   231→          <AlertCircle className="h-3 w-3" /> {cameraError}
   232→        </p>
   233→      )}
   234→
   235→      {/* Photo Grid */}
   236→      {photos.length > 0 && (
   237→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   238→          {photos.map((photo, idx) => (
   239→            <div key={idx} className="relative group rounded-md border overflow-hidden aspect-square">
   240→              <img src={photo} alt={`Evidence ${idx + 1}`} className="w-full h-full object-cover" />
   241→              <button
   242→                onClick={() => onRemove(idx)}
   243→                className="absolute top-1 right-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-70 sm:opacity-0 sm:group-hover:opacity-100 transition-opacity"
   244→              >
   245→                <Trash2 className="h-3 w-3" />
   246→              </button>
   247→            </div>
   248→          ))}
   249→        </div>
   250→      )}
   251→
   252→      {/* Open Camera Button */}
   253→      {!cameraOpen && (
   254→        <Button variant="outline" className="w-full gap-2" onClick={startCamera}>
   255→          <Camera className="h-4 w-4" />
   256→          {photos.length > 0 ? `Add More Photo (${photos.length} captured)` : 'Capture Photo Evidence'}
   257→        </Button>
   258→      )}
   259→    </div>
   260→  )
   261→}

   263→// ── Audit Trail Section ──────────────────────────────────────
   264→function AuditTrailSection({ roomId }: { roomId: string }) {
   265→  const { data: audits, isLoading } = useQuery<AuditRecord[]>({
   266→    queryKey: ['inspection-audit', roomId],
   267→    queryFn: () => apiFetch(`/api/housekeeping?section=inspection-audit&roomId=${roomId}`) as Promise<AuditRecord[]>,
   268→  })
   269→
   270→  if (isLoading) {
   271→    return (
   272→      <div className="flex items-center gap-2 py-3 text-xs text-muted-foreground">
   273→        <Loader2 className="h-3 w-3 animate-spin" /> Loading audit history...
   274→      </div>
   275→    )
   276→  }
   277→
   278→  if (!audits || audits.length === 0) {
   279→    return (
   280→      <div className="py-3 text-center">
   281→        <History className="h-6 w-6 mx-auto text-muted-foreground/30" />
   282→        <p className="mt-1 text-xs text-muted-foreground">No inspection history for this room</p>
   283→      </div>
   284→    )
   285→  }
   286→
   287→  return (
   288→    <div className="space-y-2 max-h-48 overflow-y-auto">
   289→      {audits.map((audit) => {
   290→        const { label, color } = formatAuditAction(audit.action)
   291→        return (
   292→          <div key={audit.id} className="rounded-md border p-2.5 text-xs space-y-1">
   293→            <div className="flex items-center justify-between flex-wrap gap-1">
   294→              <Badge variant="outline" className={cn('text-[10px] font-semibold', color)}>
   295→                {label}
   296→              </Badge>
   297→              <span className="text-muted-foreground">{formatDateTime(audit.createdAt)}</span>
   298→            </div>
   299→            <div className="flex items-center gap-2 flex-wrap text-muted-foreground">
   300→              <span className="flex items-center gap-1"><User className="h-3 w-3" /> {audit.performedBy}</span>
   301→            </div>
   302→            {audit.reason && (
   303→              <div className="flex items-start gap-1 text-muted-foreground">
   304→                <FileText className="h-3 w-3 mt-0.5 shrink-0" />
   305→                <span className="italic">&quot;{audit.reason}&quot;</span>
   306→              </div>
   307→            )}
   308→          </div>
   309→        )
   310→      })}
   311→    </div>
   312→  )
   313→}

   315→// ── Inspection Dialog ────────────────────────────────────────
   316→function InspectionDialog({
   317→  task,
   318→  open,
   319→  onOpenChange,
   320→}: {
   321→  task: HkTask | null
   322→  open: boolean
   323→  onOpenChange: (open: boolean) => void
   324→}) {
   325→  const queryClient = useQueryClient()
   326→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   327→  const [photos, setPhotos] = useState<string[]>([])
   328→  const [rejectReason, setRejectReason] = useState('')
   329→  const [showRejectForm, setShowRejectForm] = useState(false)
   330→  const [showAuditTrail, setShowAuditTrail] = useState(false)
   331→
   332→  const approveMutation = useMutation({
   333→    mutationFn: async () => {
   334→      if (!task) return
   335→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   336→        if (v) acc[k] = true
   337→        return acc
   338→      }, {} as Record<string, boolean>)
   339→      return apiFetch('/api/housekeeping', {
   340→        method: 'POST',
   341→        headers: { 'Content-Type': 'application/json' },
   342→        body: JSON.stringify({
   343→          action: 'approve-inspection',
   344→          id: task.id,
   345→          performedBy: 'Inspector',
   346→          checklist,
   347→          photos: photos.length > 0 ? photos : undefined,
   348→        }),
   349→      })
   350→    },
   351→    onSuccess: () => {
   352→      invalidate.afterRoomStatusChange(queryClient)
   353→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   354→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   355→      toast.success(`Room ${task?.room.number} inspection passed`)
   356→      resetAndClose()
   357→    },
   358→    onError: (err: Error) => {
   359→      toast.error(err.message || 'Failed to approve inspection')
   360→    },
   361→  })
   362→
   363→  const rejectMutation = useMutation({
   364→    mutationFn: async () => {
   365→      if (!task) return
   366→      if (!rejectReason.trim()) {
   367→        toast.error('Please provide a reason for rejection')
   368→        throw new Error('Reason required')
   369→      }
   370→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   371→        if (v) acc[k] = true
   372→        return acc
   373→      }, {} as Record<string, boolean>)
   374→      return apiFetch('/api/housekeeping', {
   375→        method: 'POST',
   376→        headers: { 'Content-Type': 'application/json' },
   377→        body: JSON.stringify({
   378→          action: 'reject-inspection',
   379→          id: task.id,
   380→          performedBy: 'Inspector',
   381→          reason: rejectReason.trim(),
   382→          checklist,
   383→          photos: photos.length > 0 ? photos : undefined,
   384→        }),
   385→      })
   386→    },
   387→    onSuccess: () => {
   388→      invalidate.afterRoomStatusChange(queryClient)
   389→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   390→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   391→      toast.info(`Room ${task?.room.number} rejected & reassigned for cleaning`)
   392→      resetAndClose()
   393→    },
   394→    onError: (err: Error) => {
   395→      if (err.message !== 'Reason required') {
   396→        toast.error(err.message || 'Failed to reject inspection')
   397→      }
   398→    },
   399→  })
   400→
   401→  const resetAndClose = () => {
   402→    setCheckedItems({})
   403→    setPhotos([])
   404→    setRejectReason('')
   405→    setShowRejectForm(false)
   406→    setShowAuditTrail(false)
   407→    onOpenChange(false)
   408→  }
   409→
   410→  if (!task) return null
   411→
   412→  const toggleCheck = (id: string) => {
   413→    setCheckedItems((prev) => ({ ...prev, [id]: !prev[id] }))
   414→  }
   415→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   416→  const allChecked = checkedCount === INSPECTION_ITEMS.length
   417→  const passCount = INSPECTION_ITEMS.length
   418→  const passPercent = Math.round((checkedCount / passCount) * 100)
   419→
   420→  const handleCapturePhoto = (dataUrl: string) => {
   421→    setPhotos((prev) => [...prev, dataUrl])
   422→    toast.success('Photo captured')
   423→  }
   424→
   425→  const handleRemovePhoto = (index: number) => {
   426→    setPhotos((prev) => prev.filter((_, i) => i !== index))
   427→  }
   428→
   429→  const isSubmitting = approveMutation.isPending || rejectMutation.isPending
   430→
   431→  return (
   432→    <Dialog open={open} onOpenChange={(v) => { if (!v) resetAndClose(); else onOpenChange(true) }}>
   433→      <DialogContent className="w-[calc(100vw-1rem)] sm:w-auto sm:max-w-lg max-h-[100dvh] sm:max-h-[92vh] flex flex-col p-0 gap-0 rounded-2xl sm:rounded-lg">
   434→        {/* Fixed Header */}
   435→        <div className="shrink-0 px-4 sm:px-6 pt-4 sm:pt-5 pb-2">
   436→          <DialogHeader>
   437→            <DialogTitle className="flex items-center gap-2 text-base sm:text-lg">
   438→              <ShieldCheck className="h-5 w-5 text-purple-600 shrink-0" />
   439→              <span>Inspect Room {task.room.number}</span>
   440→            </DialogTitle>
   441→            <DialogDescription className="mt-1.5 text-xs sm:text-sm leading-relaxed">
   442→              <span className="inline-flex flex-wrap gap-x-1.5 gap-y-0.5">
   443→                <span>Floor {task.room.floor}</span>
   444→                {task.room.wing && <span>· {task.room.wing} Wing</span>}
   445→                <span>· {task.room.type.name}</span>
   446→                {task.assignedTo && <span>· by {task.assignedTo}</span>}
   447→              </span>
   448→            </DialogDescription>
   449→          </DialogHeader>
   450→        </div>
   451→
   452→        {/* Scrollable Body */}
   453→        <ScrollArea className="flex-1 min-h-0 px-4 sm:px-6">
   454→          <div className="space-y-3 sm:space-y-4 pb-2 pr-1">
   455→            {/* Progress */}
   456→            <div className="rounded-lg border p-3">
   457→              <div className="flex items-center justify-between mb-2">
   458→                <p className="text-sm font-medium">Inspection Progress</p>
   459→                <span className="text-sm text-muted-foreground">{checkedCount}/{passCount} items</span>
   460→              </div>
   461→              <div className="h-2 rounded-full bg-muted overflow-hidden">
   462→                <div
   463→                  className={cn(
   464→                    'h-full rounded-full transition-all',
   465→                    allChecked ? 'bg-green-500' : passPercent >= 50 ? 'bg-amber-500' : 'bg-red-500'
   466→                  )}
   467→                  style={{ width: `${passPercent}%` }}
   468→                />
   469→              </div>
   470→            </div>
   471→
   472→            {/* Checklist */}
   473→            <div className="space-y-1.5">
   474→              <p className="text-sm font-semibold">Inspection Checklist</p>
   475→              <div className="rounded-lg border divide-y">
   476→                {INSPECTION_ITEMS.map((item) => (
   477→                  <label
   478→                    key={item.id}
   479→                    className="flex items-center gap-2.5 sm:gap-3 px-3 sm:px-4 py-2.5 sm:py-3 text-sm hover:bg-muted/50 cursor-pointer transition-colors"
   480→                  >
   481→                    <Checkbox
   482→                      checked={checkedItems[item.id] || false}
   483→                      onCheckedChange={() => toggleCheck(item.id)}
   484→                    />
   485→                    <span className={cn(
   486→                      'flex-1',
   487→                      checkedItems[item.id] && 'text-green-700 dark:text-green-400'
   488→                    )}>
   489→                      {item.label}
   490→                    </span>
   491→                    {checkedItems[item.id] && (
   492→                      <Check className="h-4 w-4 text-green-600" />
   493→                    )}
   494→                  </label>
   495→                ))}
   496→              </div>
   497→            </div>
   498→
   499→            {/* Photo Capture */}
   500→            <PhotoCaptureSection
   501→              photos={photos}
   502→              onCapture={handleCapturePhoto}
   503→              onRemove={handleRemovePhoto}
   504→            />
   505→
   506→            {/* Reject Reason (shown when Reject is clicked) */}
   507→            {showRejectForm && (
   508→              <div className="rounded-lg border border-red-200 bg-red-50 dark:border-red-800 dark:bg-red-950/20 p-3 space-y-2">
   509→                <p className="text-sm font-semibold text-red-700 dark:text-red-400">
   510→                  <AlertCircle className="h-4 w-4 inline mr-1.5" />
   511→                  Reason for Rejection (required)
   512→                </p>
   513→                <Textarea
   514→                  placeholder="Describe why this room failed inspection and needs to be recleaned..."
   515→                  value={rejectReason}
   516→                  onChange={(e) => setRejectReason(e.target.value)}
   517→                  className="min-h-[80px] sm:min-h-[60px] text-sm"
   518→                  autoFocus
   519→                />
   520→              </div>
   521→            )}
   522→
   523→            {/* Audit Trail Toggle */}
   524→            <div>
   525→              <button
   526→                className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground transition-colors w-full"
   527→                onClick={() => setShowAuditTrail(!showAuditTrail)}
   528→              >
   529→                <History className="h-4 w-4" />
   530→                <span className="font-medium">Inspection History</span>
   531→                <ChevronRight className={cn('h-4 w-4 transition-transform', showAuditTrail && 'rotate-90')} />
   532→              </button>
   533→              {showAuditTrail && (
   534→                <div className="mt-2 rounded-lg border p-3">
   535→                  <AuditTrailSection roomId={task.roomId} />
   536→                </div>
   537→              )}
   538→            </div>
   539→          </div>
   540→        </ScrollArea>
   541→
   542→        {/* Fixed Footer Actions */}
   543→        <div className="shrink-0 px-4 sm:px-6 pb-4 sm:pb-5 pt-2">
   544→          <Separator className="mb-3" />
   545→          <div className="flex flex-col sm:flex-row gap-2 sm:gap-3">
   546→            {!showRejectForm ? (
   547→              <>
   548→                <Button
   549→                  variant="outline"
   550→                  className="flex-1 gap-2 border-red-200 text-red-600 hover:bg-red-50 dark:border-red-800 dark:text-red-400 dark:hover:bg-red-950/40"
   551→                  onClick={() => setShowRejectForm(true)}
   552→                  disabled={isSubmitting}
   553→                >
   554→                  <X className="h-4 w-4" />
   555→                  Reject &amp; Reassign
   556→                </Button>
   557→                <Button
   558→                  className="flex-1 gap-2"
   559→                  onClick={() => approveMutation.mutate()}
   560→                  disabled={isSubmitting}
   561→                >
   562→                  {approveMutation.isPending && <Loader2 className="h-4 w-4 animate-spin" />}
   563→                  <Check className="h-4 w-4" />
   564→                  Approve
   565→                </Button>
   566→              </>
   567→            ) : (
   568→              <>
   569→                <Button
   570→                  variant="outline"
   571→                  className="flex-1 gap-2"
   572→                  onClick={() => { setShowRejectForm(false); setRejectReason('') }}
   573→                  disabled={isSubmitting}
   574→                >
   575→                  <X className="h-4 w-4" />
   576→                  Cancel
   577→                </Button>
   578→                <Button
   579→                  variant="destructive"
   580→                  className="flex-1 gap-2"
   581→                  onClick={() => rejectMutation.mutate()}
   582→                  disabled={isSubmitting || !rejectReason.trim()}
   583→                >
   584→                  {rejectMutation.isPending && <Loader2 className="h-4 w-4 animate-spin" />}
   585→                  <AlertCircle className="h-4 w-4" />
   586→                  Confirm Reject
   587→                </Button>
   588→              </>
   589→            )}
   590→          </div>
   591→        </div>
   592→      </DialogContent>
   593→    </Dialog>
   594→  )
   595→}

   597→// ── Main Component ───────────────────────────────────────────
   598→export function InspectionView() {
   599→  const [selectedTask, setSelectedTask] = useState<HkTask | null>(null)
   600→
   601→  const { data, isLoading } = useQuery<{
   602→    tasks: HkTask[]
   603→    summary: HkSummary
   604→  }>({
   605→    queryKey: ['housekeeping-tasks'],
   606→    queryFn: () => apiFetch('/api/housekeeping'),
   607→  })
   608→
   609→  const tasks = data?.tasks || []
   610→  const summary = data?.summary
   611→  const cleanedTasks = tasks.filter((t) => t.status === 'cleaned')
   612→  const inspectedTasks = tasks.filter((t) => t.status === 'inspected')
   613→  const totalInspectable = cleanedTasks.length + inspectedTasks.length
   614→  const inspectionRate = totalInspectable > 0
   615→    ? Math.round((inspectedTasks.length / totalInspectable) * 100)
   616→    : 0
   617→
   618→  if (isLoading) {
   619→    return (
   620→      <div className="space-y-4">
   621→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   622→          {Array.from({ length: 4 }).map((_, i) => (
   623→            <div key={i} className="h-24 rounded-lg bg-muted animate-pulse" />
   624→          ))}
   625→        </div>
   626→        <div className="h-64 rounded-lg bg-muted animate-pulse" />
   627→      </div>
   628→    )
   629→  }
   630→
   631→  return (
   632→    <div className="space-y-4">
   633→      {/* Stats Bar */}
   634→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   635→        {[
   636→          { label: 'Awaiting Inspection', value: cleanedTasks.length, icon: ShieldCheck, color: 'text-green-600' },
   637→          { label: 'Inspected Today', value: inspectedTasks.length, icon: Check, color: 'text-purple-600' },
   638→          { label: 'Inspection Rate', value: `${inspectionRate}%`, icon: BadgePercent, color: 'text-amber-600' },
   639→          { label: 'Failed Today', value: summary?.failed ?? 0, icon: X, color: 'text-red-600' },
   640→        ].map((stat) => (
   641→          <Card key={stat.label} className="p-2.5">
   642→            <div className="flex items-center justify-between">
   643→              <div>
   644→                <p className="text-xs font-medium text-muted-foreground">{stat.label}</p>
   645→                <p className="text-lg font-bold tracking-tight">{stat.value}</p>
   646→              </div>
   647→              <stat.icon className={cn('h-7 w-7 sm:h-8 sm:w-8 opacity-20', stat.color)} />
   648→            </div>
   649→          </Card>
   650→        ))}
   651→      </div>
   652→
   653→      {/* Awaiting Inspection */}
   654→      <div className="space-y-3">
   655→        <h3 className="text-sm font-semibold flex items-center gap-2">
   656→          <ShieldCheck className="h-4 w-4 text-purple-600" />
   657→          Rooms Awaiting Inspection
   658→          <Badge variant="secondary" className="text-xs">{cleanedTasks.length}</Badge>
   659→        </h3>
   660→        {cleanedTasks.length === 0 ? (
   661→          <Card className="p-8 text-center text-muted-foreground">
   662→            <ShieldCheck className="h-10 w-10 mx-auto opacity-20" />
   663→            <p className="mt-2 text-sm">No rooms awaiting inspection</p>
   664→          </Card>
   665→        ) : (
   666→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   667→            {cleanedTasks.map((task) => (
   668→              <RoomInspectionCard
   669→                key={task.id}
   670→                task={task}
   671→                onInspect={setSelectedTask}
   672→              />
   673→            ))}
   674→          </div>
   675→        )}
   676→      </div>
   677→
   678→      {/* Recently Inspected */}
   679→      {inspectedTasks.length > 0 && (
   680→        <div className="space-y-3">
   681→          <h3 className="text-sm font-semibold flex items-center gap-2">
   682→            <ClipboardCheck className="h-4 w-4 text-green-600" />
   683→            Recently Inspected
   684→            <Badge variant="secondary" className="text-xs">{inspectedTasks.length}</Badge>
   685→          </h3>
   686→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   687→            {inspectedTasks.slice(0, 6).map((task) => (
   688→              <RoomInspectionCard
   689→                key={task.id}
   690→                task={task}
   691→                onInspect={setSelectedTask}
   692→              />
   693→            ))}
   694→          </div>
   695→        </div>
   696→      )}
   697→
   698→      <InspectionDialog
   699→        task={selectedTask}
   700→        open={!!selectedTask}
   701→        onOpenChange={(open) => !open && setSelectedTask(null)}
   702→      />
   703→    </div>
   704→  )
