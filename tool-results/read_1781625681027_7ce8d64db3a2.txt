   394→/** Aging tab content */
   395→function AgingTab({ summary, stays }: { summary: LedgerSummary; stays: LedgerStay[] }) {
   396→  const totalAging =
   397→    summary.aging.current +
   398→    summary.aging.days30 +
   399→    summary.aging.days60 +
   400→    summary.aging.days90 +
   401→    summary.aging.over90
   402→
   403→  return (
   404→    <div className="space-y-6">
   405→      {/* Horizontal bar chart */}
   406→      <Card className="rounded-xl border bg-card shadow-sm">
   407→        <CardHeader className="pb-3">
   408→          <CardTitle className="text-sm font-semibold">Aging Summary</CardTitle>
   409→        </CardHeader>
   410→        <CardContent className="space-y-4">
   411→          {/* Bar visualization */}
   412→          <div className="flex h-8 rounded-md overflow-hidden bg-muted">
   413→            {totalAging > 0
   414→              ? AGING_COLORS.map(({ key, color }) => {
   415→                  const val = summary.aging[key as keyof typeof summary.aging]
   416→                  const pct = (val / totalAging) * 100
   417→                  if (pct === 0) return null
   418→                  return (
   419→                    <div
   420→                      key={key}
   421→                      className={cn('h-full', color)}
   422→                      style={{ width: `${pct}%` }}
   423→                      title={`${AGING_COLORS.find((c) => c.key === key)?.label}: ${formatCurrency(val)}`}
   424→                    />
   425→                  )
   426→                })
   427→              : <div className="flex-1 flex items-center justify-center text-xs text-muted-foreground">No outstanding balance</div>
   428→            }
   429→          </div>
   430→
   431→          {/* Legend + values */}
   432→          <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   433→            {AGING_COLORS.map(({ key, label, color }) => {
   434→              const val = summary.aging[key as keyof typeof summary.aging]
   435→              return (
   436→                <div key={key} className="flex items-center gap-2 text-xs">
   437→                  <div className={cn('h-2.5 w-2.5 rounded-sm shrink-0', color)} />
   438→                  <span className="text-muted-foreground">{label}</span>
   439→                </div>
   440→              )
   441→            })}
   442→          </div>
   443→          <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   444→            {AGING_COLORS.map(({ key }) => {
   445→              const val = summary.aging[key as keyof typeof summary.aging]

   448→              )
   449→            })}
   450→          </div>
   451→
   452→          <Separator />
   453→
   454→          {/* Per-stay aging table */}
   455→          <div>
   456→            <h4 className="text-xs font-semibold text-muted-foreground mb-3 uppercase tracking-wide">
   457→              Balance by Stay
   458→            </h4>
   459→            <div className="max-h-64 overflow-y-auto">
   460→              <Table>
   461→                <TableHeader>
   462→                  <TableRow>
   463→                    <TableHead className="text-xs">Confirmation #</TableHead>

   466→                    <TableHead className="text-xs">Check-Out</TableHead>
   467→                    <TableHead className="text-xs">Status</TableHead>

   473→                    <TableRow>
   474→                      <TableCell colSpan={6} className="text-xs text-center py-4 text-muted-foreground">
   475→                        No stays found
   476→                      </TableCell>

   481→                        <TableCell className="text-xs font-mono">{stay.confirmationNo}</TableCell>

   486→                          <Badge variant="outline" className="text-[10px] px-1.5 py-0">{stay.status}</Badge>

   489→                          'text-xs text-right font-medium',
