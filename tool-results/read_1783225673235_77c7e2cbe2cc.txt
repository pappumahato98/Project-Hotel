   164→    default: return 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400 border-gray-200 dark:border-gray-700'

   256→// ── Task Detail Dialog ───────────────────────────────────────
   257→function TaskDetailDialog({ task, open, onOpenChange }: { task: HkTask | null; open: boolean; onOpenChange: (open: boolean) => void }) {
   258→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   259→  const queryClient = useQueryClient()
   260→
   261→  const updateStatusMutation = useMutation({
   262→    mutationFn: (body: { id: string; status: string; inspectedBy?: string }) =>
   263→      apiFetch('/api/housekeeping', {
   264→        method: 'POST',
   265→        headers: { 'Content-Type': 'application/json' },
   266→        body: JSON.stringify({ ...body, action: 'update-task-status' }),
   267→      }),
   268→    onSuccess: () => {
   269→      toast.success('Task status updated')
   270→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   271→      queryClient.invalidateQueries({ queryKey: ['housekeeping-rooms'] })
   272→      onOpenChange(false)
   273→    },
   274→  })
   275→
   276→  if (!task) return null
   277→
   278→  const toggleCheck = (item: string) => {
   279→    setCheckedItems((prev) => ({ ...prev, [item]: !prev[item] }))
   280→  }
   281→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   282→
   283→  return (
   284→    <Dialog open={open} onOpenChange={onOpenChange}>
   285→      <DialogContent className="sm:max-w-lg">
   286→        <DialogHeader>
   287→          <DialogTitle className="flex items-center gap-2">
   288→            Room {task.room.number} — {formatTaskType(task.taskType)}
   289→          </DialogTitle>
   290→          <DialogDescription>
   291→            Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   292→          </DialogDescription>
   293→        </DialogHeader>
   294→
   295→        <div className="space-y-4">
   296→          <div className="grid grid-cols-2 gap-3">
   297→            <div className="rounded-lg border p-3 space-y-1">
   298→              <p className="text-xs text-muted-foreground">Priority</p>
   299→              <Badge variant="outline" className={cn(priorityColor(task.priority))}>
   300→                {task.priority.charAt(0).toUpperCase() + task.priority.slice(1)}
   301→              </Badge>
   302→            </div>
   303→            <div className="rounded-lg border p-3 space-y-1">
   304→              <p className="text-xs text-muted-foreground">Status</p>
   305→              <StatusBadge status={task.status} />
   306→            </div>
   307→            <div className="rounded-lg border p-3 space-y-1">
   308→              <p className="text-xs text-muted-foreground">Assigned To</p>
   309→              <p className="text-sm font-medium">{task.assignedTo || 'Unassigned'}</p>
   310→            </div>
   311→            <div className="rounded-lg border p-3 space-y-1">
   312→              <p className="text-xs text-muted-foreground">Est. Time</p>
   313→              <p className="text-sm font-medium">{task.estimatedMinutes} min</p>
   314→            </div>
   315→          </div>
   316→
   317→          <div className="space-y-2">
   318→            <div className="flex items-center justify-between">
   319→              <p className="text-sm font-semibold">Cleaning Checklist</p>
   320→              <span className="text-xs text-muted-foreground">{checkedCount}/{CLEANING_CHECKLIST.length}</span>
   321→            </div>
   322→            <div className="rounded-lg border divide-y max-h-64 overflow-y-auto">
   323→              {CLEANING_CHECKLIST.map((item) => (
   324→                <label
   325→                  key={item}
   326→                  className="flex items-center gap-3 px-3 py-2 text-sm hover:bg-muted/50 cursor-pointer transition-colors"
   327→                >
   328→                  <Checkbox
   329→                    checked={checkedItems[item] || false}
   330→                    onCheckedChange={() => toggleCheck(item)}
   331→                  />
   332→                  <span className={checkedItems[item] ? 'line-through text-muted-foreground' : ''}>
   333→                    {item}
   334→                  </span>
   335→                </label>
   336→              ))}
   337→            </div>
   338→          </div>
   339→
   340→          {task.notes && (
   341→            <div className="rounded-lg border p-3">
   342→              <p className="text-xs text-muted-foreground mb-1">Notes</p>
   343→              <p className="text-sm">{task.notes}</p>
   344→            </div>
   345→          )}
   346→
   347→          {/* Status Actions */}
   348→          <div className="flex gap-2 pt-2">
   349→            {task.status === 'pending' && (
   350→              <Button
   351→                size="sm"
   352→                variant="outline"
   353→                className="flex-1"
   354→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'in_progress' })}
   355→                disabled={updateStatusMutation.isPending}
   356→              >
   357→                <Clock className="w-3.5 h-3.5 mr-1.5" /> Start Cleaning
   358→              </Button>
   359→            )}
   360→            {task.status === 'in_progress' && (
   361→              <Button
   362→                size="sm"
   363→                className="flex-1"
   364→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'cleaned' })}
   365→                disabled={updateStatusMutation.isPending}
   366→              >
   367→                <CheckCircle2 className="w-3.5 h-3.5 mr-1.5" /> Mark Cleaned
   368→              </Button>
   369→            )}
   370→            {task.status === 'cleaned' && (
   371→              <Button
   372→                size="sm"
   373→                className="flex-1"
   374→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'inspected', inspectedBy: 'Admin' })}
   375→                disabled={updateStatusMutation.isPending}
   376→              >
   377→                <Eye className="w-3.5 h-3.5 mr-1.5" /> Mark Inspected
   378→              </Button>
   379→            )}
   380→          </div>
   381→        </div>
   382→      </DialogContent>
   383→    </Dialog>
   384→  )
   385→}

   387→// ── Attendant View ───────────────────────────────────────────
   388→function AttendantView({ tasks }: { tasks: HkTask[] }) {
   389→  const attendants = new Map<string, HkTask[]>()
   390→  tasks.forEach((task) => {
   391→    const name = task.assignedTo || 'Unassigned'
   392→    if (!attendants.has(name)) attendants.set(name, [])
   393→    attendants.get(name)!.push(task)
   394→  })
   395→
   396→  return (
   397→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   398→      {Array.from(attendants.entries()).map(([name, attTasks]) => (
   399→        <Card key={name}>
   400→          <CardContent className="p-3 space-y-2">
   401→            <div className="flex items-center gap-3">
   402→              <div className="flex h-9 w-9 items-center justify-center rounded-full bg-muted text-sm font-medium">
   403→                {name.charAt(0)}
   404→              </div>
   405→              <div>
   406→                <p className="text-sm font-semibold">{name}</p>
   407→                <p className="text-xs text-muted-foreground">{attTasks.length} tasks assigned</p>
   408→              </div>
   409→            </div>
   410→            {attTasks.map((task) => (
   411→              <div key={task.id} className="flex items-center justify-between rounded-md border p-2 text-sm">
   412→                <div className="flex items-center gap-2">
   413→                  <BedDouble className="h-3.5 w-3.5 text-muted-foreground" />
   414→                  <span className="font-medium">Room {task.room.number}</span>
   415→                </div>
   416→                <StatusBadge status={task.status} />
   417→              </div>
   418→            ))}
   419→          </CardContent>
   420→        </Card>
   421→      ))}
   422→    </div>
   423→  )
   424→}

   981→            className={cn('rounded-lg border-2 border-t-4 bg-card', column.color)}

   986→            >

   988→                <span className="font-semibold text-sm">{column.label}</span>

   991→            </button>
   992→
   993→            {isExpanded && (
   994→              <ScrollArea className="h-[500px] px-3 pb-3">
   995→                <div className="space-y-2">
   996→                  {columnTasks.length === 0 ? (
   997→                    <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   998→                      <ClipboardCheck className="h-8 w-8 opacity-20" />
   999→                      <p className="mt-2 text-xs">No tasks</p>
  1000→                    </div>
  1001→                  ) : (
  1002→                    columnTasks.map((task) => (
  1003→                      <TaskCard key={task.id} task={task} onClick={() => setSelectedTask(task)} />
  1004→                    ))
  1005→                  )}
  1006→                </div>
  1007→              </ScrollArea>
  1008→            )}
  1009→          </div>
  1010→        )
  1011→      })}
  1012→    </div>
  1013→  )
  1014→
  1015→  return (
  1016→    <div className="space-y-3">
  1017→      {/* View Toggle */}
  1018→      <div className="flex items-center gap-2">
  1019→        <Button
  1020→          variant={viewMode === 'table' ? 'default' : 'outline'}
  1021→          size="sm"
  1022→          onClick={() => setViewMode('table')}
  1023→          className="gap-1.5"
  1024→        >
  1025→          <List className="h-4 w-4" />

  1029→          variant={viewMode === 'kanban' ? 'default' : 'outline'}
  1030→          size="sm"
  1031→          onClick={() => setViewMode('kanban')}
  1032→          className="gap-1.5"
  1033→        >
  1034→          <LayoutGrid className="h-4 w-4" />
  1035→          Kanban
  1036→        </Button>
  1037→        <Button
  1038→          variant={viewMode === 'attendant' ? 'default' : 'outline'}
  1039→          size="sm"
  1040→          onClick={() => setViewMode('attendant')}
  1041→          className="gap-1.5"
  1042→        >
  1043→          <Users className="h-4 w-4" />
  1044→          Attendant
  1045→        </Button>
  1046→      </div>
  1047→
  1048→      {/* Views */}
  1049→      {viewMode === 'table' && renderTableView()}
  1050→      {viewMode === 'kanban' && renderKanbanView()}
  1051→      {viewMode === 'attendant' && <AttendantView tasks={kanbanTasks} />}
  1052→
  1053→      {/* Task Detail Dialog (for Kanban/Attendant) */}
  1054→      <TaskDetailDialog
  1055→        task={selectedTask}
  1056→        open={!!selectedTask}
  1057→        onOpenChange={(open) => !open && setSelectedTask(null)}
  1058→      />
  1059→    </div>
  1060→  )
