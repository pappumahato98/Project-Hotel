    97→    97→
    98→    98→export function DeparturesView() {
    99→    99→  const queryClient = useQueryClient()
   100→   100→  const { settings } = useSettingsStore()
   101→   101→  const { preferences } = usePreferencesStore()
   102→   102→  const { navigateTo } = useNavigationStore()
   103→   103→  const today = getTodayString()
   104→   104→
   105→   105→  // Dialog states
   106→   106→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   107→   107→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   108→   108→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   109→   109→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   110→   110→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   111→   111→
   112→   112→  // Selected departure
   113→   113→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   114→   114→
   115→   115→  // Late checkout
   116→   116→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   117→   117→
   118→   118→  // Payment
   119→   119→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   120→   120→  const [paymentAmount, setPaymentAmount] = useState('')
   121→   121→  const [paymentReference, setPaymentReference] = useState('')
   122→   122→
   123→   123→  // Receipt data
   124→   124→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   125→   125→
   126→   126→  // Fetch today's departures (checked_in with checkOut = today)
   127→   127→  const { data, isLoading } = useQuery({
   128→   128→    queryKey: ['departures', today],
   129→   129→    queryFn: async () => {
   130→   130→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   131→   131→      return apiFetch(`/api/reservations?${params.toString()}`)
   132→   132→    },
   133→   133→    refetchInterval: 30000,
   134→   134→  })
   135→   135→
   136→   136→  const departures: Departure[] = data?.reservations || []
   137→   137→
   138→   138→  // Stats
   139→   139→  const totalDepartures = departures.length
   140→   140→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length

   142→   142→  const outstandingBalance = departures.reduce((sum, d) => {
   143→   143→    return sum + (d.folios[0]?.balance || 0)
   144→   144→  }, 0)
   145→   145→
   146→   146→  // Zero-balance pending guests (for express & batch checkout)
   147→   147→  const zeroBalancePending = departures.filter(
   148→   148→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   149→   149→  )

   152→   152→

   155→   155→    mutationFn: async (reservationId: string) => {
   156→   156→      return apiFetch(`/api/reservations/${reservationId}`, {
   157→   157→        method: 'PATCH',
   158→   158→        headers: { 'Content-Type': 'application/json' },
   159→   159→        body: JSON.stringify({ status: 'checked_out' }),
   160→   160→      })
   161→   161→    },
   162→   162→    onSuccess: () => {
   163→   163→      invalidate.afterCheckout(queryClient)
   164→   164→    },
   165→   165→  })
   166→   166→
   167→   167→  // Payment mutation
   168→   168→  const paymentMutation = useMutation({
   169→   169→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   170→   170→      return apiFetch(`/api/folio/${folioId}`, {
   171→   171→        method: 'POST',
   172→   172→        headers: { 'Content-Type': 'application/json' },
   173→   173→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   174→   174→      })
   175→   175→    },
   176→   176→    onSuccess: () => {
   177→   177→      invalidate.afterFolioChange(queryClient)
   178→   178→      setPaymentDialogOpen(false)
   179→   179→      setPaymentAmount('')
   180→   180→      setPaymentReference('')

   182→   182→    },
   183→   183→    onError: () => {
   184→   184→      toast.error('Failed to record payment')
   185→   185→    },
   186→   186→  })
   187→   187→
   188→   188→  // Late checkout mutation
   189→   189→  const lateCheckoutMutation = useMutation({
   190→   190→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   191→   191→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   192→   192→        method: 'PATCH',
   193→   193→        headers: { 'Content-Type': 'application/json' },
   194→   194→        body: JSON.stringify({ checkOut: newCheckOut }),
   195→   195→      })
   196→   196→      // Add surcharge to folio
   197→   197→      if (surcharge > 0) {
   198→   198→        await apiFetch(`/api/folio`, {
   199→   199→          method: 'POST',
   200→   200→          headers: { 'Content-Type': 'application/json' },
   201→   201→          body: JSON.stringify({
   202→   202→            reservationId,
   203→   203→            type: 'charge',
   204→   204→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   205→   205→            amount: surcharge,
   206→   206→          }),
   207→   207→        })
   208→   208→      }
   209→   209→      return result
   210→   210→    },
   211→   211→    onSuccess: () => {
   212→   212→      invalidate.afterReservationChange(queryClient)
   213→   213→      invalidate.afterFolioChange(queryClient)
   214→   214→      setLateCheckoutDialogOpen(false)
   215→   215→      toast.success('Late checkout request processed successfully')
   216→   216→    },
   217→   217→    onError: () => {
   218→   218→      toast.error('Failed to process late checkout request')
   219→   219→    },
   220→   220→  })
   221→   221→
   222→   222→  // Batch checkout mutation
   223→   223→  const batchCheckoutMutation = useMutation({
   224→   224→    mutationFn: async (ids: string[]) => {
   225→   225→      const results = []
   226→   226→      for (let i = 0; i < ids.length; i++) {
   227→   227→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   228→   228→          method: 'PATCH',
   229→   229→          headers: { 'Content-Type': 'application/json' },
   230→   230→          body: JSON.stringify({ status: 'checked_out' }),
   231→   231→        })
   232→   232→        results.push(result)
   233→   233→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   234→   234→      }
   235→   235→      return results
   236→   236→    },
   237→   237→    onSuccess: (_data, ids) => {
   238→   238→      invalidate.afterCheckout(queryClient)
   239→   239→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   240→   240→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   241→   241→    },
   242→   242→    onError: () => {
   243→   243→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   244→   244→    },
   245→   245→  })
   246→   246→
   247→   247→  // ─── Handlers ─────────────────────────────────────────────────────
   248→   248→
   249→   249→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   250→   250→
   251→   251→  const handleExpressCheckout = (dep: Departure) => {
   252→   252→    checkoutMutation.mutate(dep.id, {
   253→   253→      onSuccess: () => {
   254→   254→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   255→   255→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   256→   256→      },
   257→   257→    })
   258→   258→  }
   259→   259→
   260→   260→  const handleReviewFolio = (dep: Departure) => {
   261→   261→    setSelectedDeparture(dep)
   262→   262→    setFolioDialogOpen(true)
   263→   263→  }
   264→   264→
   265→   265→  const handleNavigateToFolio = (dep: Departure) => {
   266→   266→    useFolioContextStore.getState().setFolioContext({
   267→   267→      reservationId: dep.id,
   268→   268→      guestId: dep.guest?.id || '',
   269→   269→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   270→   270→      roomNumber: dep.room?.number || '',
   271→   271→      confirmationNo: dep.confirmationNo,
   272→   272→    })
   273→   273→    navigateTo('front-desk', 'folio')
   274→   274→  }
   275→   275→
   276→   276→  const handleViewLedger = (dep: Departure) => {
   277→   277→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   278→   278→      guestId: dep.guest?.id || '',
   279→   279→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   280→   280→    })
   281→   281→    navigateTo('front-desk', 'guest-ledger')
   282→   282→  }
   283→   283→
   284→   284→  const handleFolioSettleCheckout = () => {
   285→   285→    setFolioDialogOpen(false)
   286→   286→    setCheckoutDialogOpen(true)
   287→   287→  }
   288→   288→
   289→   289→  const handleQuickCheckout = (dep: Departure) => {
   290→   290→    setSelectedDeparture(dep)
   291→   291→    setCheckoutDialogOpen(true)
   292→   292→  }
   293→   293→
   294→   294→  const handleRequestLateCheckout = (dep: Departure) => {
   295→   295→    setSelectedDeparture(dep)
   296→   296→    setLateCheckoutOption('14:00')
   297→   297→    setLateCheckoutDialogOpen(true)
   298→   298→  }
   299→   299→
   300→   300→  const handleConfirmLateCheckout = () => {
   301→   301→    if (!selectedDeparture) return
   302→   302→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   303→   303→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   304→   304→
   305→   305→    // Build new checkout datetime with the selected time
   306→   306→    const currentCheckout = new Date(selectedDeparture.checkOut)
   307→   307→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   308→   308→    currentCheckout.setHours(hours, minutes, 0, 0)
   309→   309→
   310→   310→    lateCheckoutMutation.mutate({
   311→   311→      reservationId: selectedDeparture.id,
   312→   312→      newCheckOut: currentCheckout.toISOString(),
   313→   313→      surcharge,
   314→   314→    })
   315→   315→  }
   316→   316→
   317→   317→  const handleOpenPaymentDialog = () => {
   318→   318→    if (!selectedDeparture) return
   319→   319→    const balance = getBalance(selectedDeparture)
   320→   320→    setPaymentAmount(String(balance))
   321→   321→    setPaymentMethod('cash')
   322→   322→    setPaymentReference('')
   323→   323→    setPaymentDialogOpen(true)
   324→   324→  }
   325→   325→
   326→   326→  const handleRecordPayment = () => {
   327→   327→    if (!selectedDeparture) return
   328→   328→    const folioId = selectedDeparture.folios[0]?.id
   329→   329→    if (!folioId) {
   330→   330→      toast.error('No folio found for this reservation')
   331→   331→      return
   332→   332→    }
   333→   333→    const amount = Number(paymentAmount)
   334→   334→    if (isNaN(amount) || amount <= 0) {
   335→   335→      toast.error('Please enter a valid payment amount')
   336→   336→      return
   337→   337→    }
   338→   338→    paymentMutation.mutate({
   339→   339→      folioId,
   340→   340→      amount,
   341→   341→      method: paymentMethod,
   342→   342→      reference: paymentReference,
   343→   343→    })
   344→   344→  }
   345→   345→

   348→   348→    const roomNumber = selectedDeparture.room.number
   349→   349→    checkoutMutation.mutate(selectedDeparture.id, {
   350→   350→      onSuccess: () => {
   351→   351→        setCheckoutDialogOpen(false)
   352→   352→        setReceiptData(selectedDeparture)
   353→   353→        setReceiptDialogOpen(true)
   354→   354→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   355→   355→      },
   356→   356→    })
   357→   357→  }
   358→   358→
   359→   359→  const handleBatchCheckout = () => {
   360→   360→    const ids = zeroBalancePending.map((d) => d.id)
   361→   361→    if (ids.length === 0) {
   362→   362→      toast.info('No guests with zero balance to batch checkout')
   363→   363→      return
   364→   364→    }
   365→   365→    batchCheckoutMutation.mutate(ids)
   366→   366→  }
   367→   367→
   368→   368→  const handlePrintReceipt = () => {
   369→   369→    toast.success('Receipt sent to printer')
   370→   370→  }
   371→   371→
   372→   372→  const handleEmailReceipt = () => {
   373→   373→    toast.success('Receipt emailed to guest')
   374→   374→  }
   375→   375→
   376→   376→  // ─── Computed values ──────────────────────────────────────────────
   377→   377→
   378→   378→  const lateCheckoutFee = selectedDeparture
   379→   379→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   380→   380→    : 0
   381→   381→
   382→   382→  // ─── Render ────────────────────────────────────────────────────────
   383→   383→
   384→   384→  return (

   386→   386→      {/* Header */}
   387→   387→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">

   390→   390→          <p className="text-xs text-muted-foreground">
   391→   391→            Guest check-outs scheduled for {formatDate(today)}
   392→   392→          </p>
   393→   393→        </div>
   394→   394→        {zeroBalancePending.length > 0 && (
   395→   395→          <Button
   396→   396→            onClick={handleBatchCheckout}
   397→   397→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   398→   398→            disabled={batchCheckoutMutation.isPending}
   399→   399→          >
   400→   400→            <Zap className="size-4 mr-1.5" />
   401→   401→            {batchCheckoutMutation.isPending
   402→   402→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   403→   403→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   404→   404→          </Button>
   405→   405→        )}
   406→   406→      </div>
   407→   407→
   408→   408→      {/* Departures List */}
   409→   409→      <Card className="py-0">
   410→   410→        <CardContent className="p-0">
   411→   411→          <div className="max-h-[600px] overflow-y-auto">
   412→   412→            <Table>
   413→   413→              <TableHeader>
   414→   414→                <TableRow>
   415→   415→                  <TableHead className="w-[70px]">Room</TableHead>
   416→   416→                  <TableHead>Guest</TableHead>
   417→   417→                  <TableHead className="w-[100px]">Check-out</TableHead>
   418→   418→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   419→   419→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   420→   420→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   421→   421→                </TableRow>
   422→   422→              </TableHeader>
   423→   423→              <TableBody>
   424→   424→                {isLoading ? (
   425→   425→                  Array.from({ length: 5 }).map((_, i) => (
   426→   426→                    <TableRow key={i}>
   427→   427→                      {Array.from({ length: 6 }).map((_, j) => (
   428→   428→                        <TableCell key={j}>
   429→   429→                          <Skeleton className="h-4 w-full" />
   430→   430→                        </TableCell>
   431→   431→                      ))}
   432→   432→                    </TableRow>
   433→   433→                  ))
   434→   434→                ) : departures.length === 0 ? (
   435→   435→                  <TableRow>
   436→   436→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   437→   437→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   438→   438→                      No departures scheduled for today
   439→   439→                    </TableCell>
   440→   440→                  </TableRow>
   441→   441→                ) : (
   442→   442→                  departures.map((dep) => {
   443→   443→                    const balance = getBalance(dep)
   444→   444→                    const isPending = dep.status === 'checked_in'
   445→   445→                    const isZeroBalance = balance === 0
   446→   446→                    return (
   447→   447→                      <TableRow key={dep.id}>
   448→   448→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   449→   449→                        <TableCell>
   450→   450→                          <div className="flex items-center gap-2">
   451→   451→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   452→   452→                            {dep.guest.vipLevel !== 'none' && (
   453→   453→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   454→   454→                                VIP
   455→   455→                              </Badge>
   456→   456→                            )}
   457→   457→                          </div>
   458→   458→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   459→   459→                        </TableCell>
   460→   460→                        <TableCell className="text-xs">
   461→   461→                          <div>{formatDate(dep.checkOut)}</div>
