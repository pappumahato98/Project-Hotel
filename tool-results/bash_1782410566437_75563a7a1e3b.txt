    63	// ─── Quick Search Dialog ────────────────────────────────────────────
    64	function QuickSearchDialog() {
    65	  const { searchOpen, setSearchOpen, navigateTo } = useNavigationStore()
    66	  const [query, setQuery] = React.useState('')
    67	
    68	  const filteredItems = React.useMemo(() => {
    69	    if (!query.trim()) return NAV_ITEMS
    70	    const lower = query.toLowerCase()
    71	    return NAV_ITEMS.filter(
    72	      (item) =>
    73	        item.label.toLowerCase().includes(lower) ||
    74	        item.children?.some((c) => c.label.toLowerCase().includes(lower))
    75	    )
    76	  }, [query])
    77	
    78	  React.useEffect(() => {
    79	    const down = (e: KeyboardEvent) => {
    80	      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
    81	        e.preventDefault()
    82	        setSearchOpen(!searchOpen)
    83	      }
    84	    }
    85	    document.addEventListener('keydown', down)
    86	    return () => document.removeEventListener('keydown', down)
    87	  }, [searchOpen, setSearchOpen])
    88	
    89	  return (
    90	    <CommandDialog open={searchOpen} onOpenChange={setSearchOpen}>
    91	      <CommandInput
    92	        placeholder="Search modules, pages..."
    93	        value={query}
    94	        onValueChange={setQuery}
    95	      />
    96	      <CommandList>
    97	        <CommandEmpty>No results found.</CommandEmpty>
    98	        {filteredItems.map((item) => (
    99	          <React.Fragment key={item.id}>
   100	            <CommandGroup heading={item.label}>
   101	              <CommandItem
   102	                onSelect={() => {
   103	                  navigateTo(item.id)
   104	                  setSearchOpen(false)
   105	                  setQuery('')
   106	                }}
   107	              >
   108	                <item.icon className="mr-2 size-4 text-muted-foreground" />
   109	                <span>{item.label}</span>
   110	              </CommandItem>
   111	              {item.children?.map((child) => (
   112	                <CommandItem
   113	                  key={child.id}
   114	                  onSelect={() => {
   115	                    navigateTo(item.id, child.id)
   116	                    setSearchOpen(false)
   117	                    setQuery('')
   118	                  }}
   119	                  className="pl-8"
   120	                >
   121	                  <span>{child.label}</span>
   122	                </CommandItem>
   123	              ))}
   124	            </CommandGroup>
   125	            <CommandSeparator />
   126	          </React.Fragment>
   127	        ))}
   128	      </CommandList>
   129	    </CommandDialog>
   130	  )
   131	}
