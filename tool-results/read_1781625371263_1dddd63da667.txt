   156→// ─── Component ──────────────────────────────────────────────────────────
   157→
   158→export function ArrivalsView() {
   159→  const queryClient = useQueryClient()
   160→  const { settings } = useSettingsStore()
   161→  const today = getTodayString()
   162→
   163→  // Dialog states
   164→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   165→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   166→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   167→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   168→
   169→  // Selection states
   170→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   171→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   172→
   173→  // Check-in form states
   174→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   175→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   176→  const [roomPreference, setRoomPreference] = useState('')
   177→  const [pillowType, setPillowType] = useState('')
   178→  const [wakeupCall, setWakeupCall] = useState('')
   179→
   180→  // Key card states
   181→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   182→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   183→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   184→
   185→  // Walk-in form state
   186→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   187→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   188→
   189→  // ─── Queries ────────────────────────────────────────────────────────
   190→
   191→  const { data, isLoading } = useQuery({
   192→    queryKey: ['arrivals', today],
   193→    queryFn: async () => {
   194→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   195→      return apiFetch(`/api/reservations?${params.toString()}`)
   196→    },
   197→    refetchInterval: 30000,
   198→  })
   199→
   200→  const { data: roomsData } = useQuery({
   201→    queryKey: ['rooms', 'vacant'],
   202→    queryFn: () => apiFetch('/api/rooms'),
   203→  })
   204→
   205→  const arrivals: Arrival[] = data?.reservations || []
   206→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   207→  const availableRooms = allRooms.filter(
   208→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   209→  )
   210→
   211→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   212→
   213→  const sortedArrivals = useMemo(() => {
   214→    return [...arrivals].sort((a, b) => {
   215→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   216→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   217→      if (aVip && !bVip) return -1
   218→      if (!aVip && bVip) return 1
   219→      return 0
   220→    })
   221→  }, [arrivals])
   222→
   223→  // ─── Stats ──────────────────────────────────────────────────────────
   224→
   225→  const totalArrivals = arrivals.length
   226→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   227→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   228→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   229→
   230→  // ─── Mutations ──────────────────────────────────────────────────────
   231→
   232→  const assignRoomMutation = useMutation({
   233→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   234→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   235→      if (specialRequests !== undefined) {
   236→        payload.specialRequests = specialRequests
   237→      }
   238→      return apiFetch(`/api/reservations/${reservationId}`, {
   239→        method: 'PATCH',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify(payload),
   242→      })
   243→    },
   244→    onSuccess: (_data, variables) => {
   245→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   246→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   247→      setRoomPickerOpen(false)
   248→      setCheckInDialogOpen(false)
   249→
   250→      // Find the room number for key card dialog
   251→      const room = availableRooms.find((r) => r.id === variables.roomId)
   252→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   253→      openKeyCardDialog(roomNumber)
   254→    },
   255→  })
   256→
   257→  const checkInMutation = useMutation({
   258→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   259→      const payload: Record<string, unknown> = { status: 'checked_in' }
   260→      if (specialRequests !== undefined) {
   261→        payload.specialRequests = specialRequests
   262→      }
   263→      return apiFetch(`/api/reservations/${reservationId}`, {
   264→        method: 'PATCH',
   265→        headers: { 'Content-Type': 'application/json' },
   266→        body: JSON.stringify(payload),
   267→      })
   268→    },
   269→    onSuccess: (_data, variables) => {
   270→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   271→      setCheckInDialogOpen(false)
   272→
   273→      const roomNumber = selectedArrival?.room?.number || ''
   274→      openKeyCardDialog(roomNumber)
   275→    },
   276→  })
   277→
   278→  const walkInMutation = useMutation({
   279→    mutationFn: async (form: WalkInForm) => {
   280→      // 1. Create guest
   281→      const guestData = await apiFetch('/api/guests', {
   282→        method: 'POST',
   283→        headers: { 'Content-Type': 'application/json' },
   284→        body: JSON.stringify({
   285→          firstName: form.firstName,
   286→          lastName: form.lastName,
   287→          phone: form.phone || null,
   288→          email: form.email || null,
   289→          nationality: form.nationality || null,
   290→          idType: form.idType || null,
   291→          idNumber: form.idNumber || null,
   292→        }),
   293→      })
   294→      const guestId = guestData.guest.id
   295→
   296→      // 2. Find a vacant room
   297→      const vacantRoom = availableRooms[0]
   298→      if (!vacantRoom) throw new Error('No available rooms')
   299→
   300→      // 3. Create reservation as checked_in (walk-in)
   301→      const checkOutDate = form.checkOutDate || (() => {
   302→        const d = new Date()
   303→        d.setDate(d.getDate() + 1)
   304→        return d
   305→      })()
   306→
   307→      const resData = await apiFetch('/api/reservations', {
   308→        method: 'POST',
   309→        headers: { 'Content-Type': 'application/json' },
   310→        body: JSON.stringify({
   311→          guestId,
   312→          roomId: vacantRoom.id,
   313→          adults: form.adults,
   314→          children: form.children,
   315→          checkIn: new Date().toISOString(),
   316→          checkOut: checkOutDate.toISOString(),
   317→          roomRate: form.roomRate,
   318→          source: 'walk_in',
   319→          reservationType: 'walk_in',
   320→          guaranteed: false,
   321→        }),
   322→      })
   323→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   324→    },
   325→    onSuccess: (result) => {
   326→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   327→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   328→      queryClient.invalidateQueries({ queryKey: ['guests'] })
   329→      setWalkInDialogOpen(false)
   330→      setWalkInForm(initialWalkInForm)
   331→
   332→      // Open key card dialog
   333→      openKeyCardDialog(result.roomNumber)
   334→    },
   335→    onError: (error) => {
   336→      toast.error('Walk-in check-in failed', {
   337→        description: error instanceof Error ? error.message : 'Unknown error',
   338→      })
   339→    },
   340→  })
   341→
   342→  // ─── Handlers ──────────────────────────────────────────────────────
   343→
   344→  const handleAssignRoom = (arrival: Arrival) => {
   345→    setSelectedArrival(arrival)
   346→    setRoomPickerOpen(true)
   347→  }
   348→
   349→  const handleQuickCheckIn = (arrival: Arrival) => {
   350→    if (!arrival.room) {
   351→      setSelectedArrival(arrival)
   352→      setRoomPickerOpen(true)
   353→      return
   354→    }
   355→    setSelectedArrival(arrival)
   356→    resetCheckInForm()
   357→    setCheckInDialogOpen(true)
   358→  }
   359→
   360→  const handleOpenWalkIn = () => {
   361→    setWalkInForm(initialWalkInForm)
   362→    setWalkInDialogOpen(true)
   363→  }
   364→
   365→  const buildSpecialRequests = (arrival: Arrival): string => {
   366→    const parts: string[] = []
   367→
   368→    // Preserve existing requests
   369→    if (arrival.specialRequests) {
   370→      parts.push(arrival.specialRequests)
   371→    }
   372→
   373→    // Early check-in note
   374→    if (earlyCheckIn) {
   375→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   376→    }
   377→
   378→    // Check-out time override
   379→    if (checkOutTime !== settings.defaultCheckOut) {
   380→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   381→    }
   382→
   383→    // Room preference
   384→    if (roomPreference) {
   385→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   386→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   387→    }
   388→
   389→    // Pillow type
   390→    if (pillowType) {
   391→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   392→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   393→    }
   394→
   395→    // Wake-up call
   396→    if (wakeupCall) {
   397→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   398→    }
   399→
   400→    return parts.join(' | ')
   401→  }
   402→
   403→  const confirmCheckIn = () => {
   404→    if (!selectedArrival) return
   405→
   406→    const updatedRequests = buildSpecialRequests(selectedArrival)
   407→
   408→    if (selectedRoomId) {
   409→      assignRoomMutation.mutate({
   410→        reservationId: selectedArrival.id,
   411→        roomId: selectedRoomId,
   412→        specialRequests: updatedRequests,
   413→      })
   414→    } else if (selectedArrival.room) {
   415→      checkInMutation.mutate({
   416→        reservationId: selectedArrival.id,
   417→        specialRequests: updatedRequests,
   418→      })
   419→    }
   420→  }
   421→
   422→  const openKeyCardDialog = (roomNumber: string) => {
   423→    setKeyCardRoomNumber(roomNumber)
   424→    setKeyCardIssued(false)
   425→    setKeyCardProcessed(false)
   426→    setKeyCardDialogOpen(true)
   427→  }
   428→
   429→  const handleKeyCardConfirm = () => {
   430→    setKeyCardProcessed(true)
   431→    setKeyCardDialogOpen(false)
   432→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   433→      duration: 5000,
   434→      icon: <Mail className="size-4" />,
   435→    })
   436→    setSelectedArrival(null)
   437→    setSelectedRoomId('')
   438→  }
   439→
   440→  const resetCheckInForm = () => {
   441→    setEarlyCheckIn(false)
   442→    setCheckOutTime(settings.defaultCheckOut)
   443→    setRoomPreference('')
   444→    setPillowType('')
   445→    setWakeupCall('')
   446→  }
   447→
   448→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   449→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   450→    setWalkInForm((prev) => ({
   451→      ...prev,
   452→      roomTypeId,
   453→      roomRate: roomType?.baseRate || 0,
   454→    }))
   455→  }
   456→
   457→  // ─── Render ─────────────────────────────────────────────────────────
   458→
   459→  return (
   460→    <div className="flex flex-col gap-2">
   461→      {/* Header */}
   462→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   463→        <div>
   464→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   465→          <p className="text-xs text-muted-foreground">
   466→            Guest check-ins scheduled for {formatDate(today)}
   467→          </p>
   468→        </div>
   469→        <Button onClick={handleOpenWalkIn} className="gap-2 shrink-0">
   470→          <Footprints className="size-4" />
   471→          Walk-in Check-in
   472→        </Button>
   473→      </div>
   474→
   475→      {/* Arrivals List */}
   476→      <Card className="py-0">
   477→        <CardContent className="p-0">
   478→          {isLoading ? (
   479→            <div className="p-2 space-y-2">
   480→              {Array.from({ length: 4 }).map((_, i) => (
   481→                <Skeleton key={i} className="h-20 w-full rounded-lg" />
   482→              ))}
   483→            </div>
   484→          ) : sortedArrivals.length === 0 ? (
   485→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   486→              <LogIn className="size-8 mb-2 opacity-50" />
   487→              <p className="text-sm">No arrivals scheduled for today</p>
   488→            </div>
   489→          ) : (
   490→            <div className="divide-y">
   491→              {sortedArrivals.map((arrival) => {
   492→                const isUnassigned = !arrival.room
   493→                const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   494→                return (
   495→                  <div
   496→                    key={arrival.id}
   497→                    className={cn(
   498→                      'p-2.5 transition-colors hover:bg-muted/50',
   499→                      isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/50 dark:bg-amber-950/20',
   500→                      isVip && !isUnassigned && 'border-l-4 border-l-amber-400 bg-amber-50/30 dark:bg-amber-950/10',
   501→                      isVip && isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/70 dark:bg-amber-950/30',
   502→                    )}
   503→                  >
   504→                    <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   505→                      {/* Guest Info */}
   506→                      <div className="flex-1 min-w-0">
   507→                        <div className="flex items-center gap-2 flex-wrap">
   508→                          <span className="font-semibold truncate">
   509→                            {arrival.guest ? `${arrival.guest.firstName} ${arrival.guest.lastName}` : 'Unknown Guest'}
   510→                          </span>
   511→                          {isVip && (
   512→                            <Badge className="text-[10px] px-1.5 py-0 gap-0.5 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700">
   513→                              <Star className="size-3 fill-amber-500 text-amber-500" />
   514→                              <Crown className="size-3" />
   515→                              {arrival.guest?.vipLevel?.toUpperCase()}
   516→                            </Badge>
   517→                          )}
   518→                          {isUnassigned && (
   519→                            <Badge variant="outline" className="text-[10px] px-1.5 py-0 border-amber-400 text-amber-600 dark:text-amber-400">
   520→                              <AlertTriangle className="size-3 mr-0.5" />
   521→                              Unassigned
   522→                            </Badge>
   523→                          )}
   524→                        </div>
   525→                        <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground flex-wrap">
   526→                          <span className="font-mono">{arrival.confirmationNo}</span>
   527→                          <span>•</span>
   528→                          <span className="flex items-center gap-1">
   529→                            <BedDouble className="size-3" />
   530→                            {arrival.room ? (
   531→                              <span>Room {arrival.room.number} ({arrival.room.type.name})</span>
   532→                            ) : (
   533→                              <span className="text-amber-600 dark:text-amber-400">No room assigned</span>
   534→                            )}
   535→                          </span>
   536→                          <span>•</span>
   537→                          <span className="flex items-center gap-1">
   538→                            <Clock className="size-3" />
   539→                            {arrival.adults} adults{arrival.children > 0 ? `, ${arrival.children} children` : ''}
   540→                          </span>
   541→                          <span>•</span>
   542→                          <span className="capitalize">{arrival.source?.replace('_', ' ')}</span>
   543→                        </div>
   544→                        {arrival.specialRequests && (
   545→                          <p className="text-xs text-muted-foreground mt-1 italic truncate">
   546→                            📋 {arrival.specialRequests}
   547→                          </p>
   548→                        )}
   549→                      </div>
   550→
