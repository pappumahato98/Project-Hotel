     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,
     9→  Zap, Eye, Clock, Banknote, Mail, X, ArrowRight, BedSingle, FileText, BookOpen,
    10→  MoreHorizontal,
    11→} from 'lucide-react'
    12→
    13→import { Checkbox } from '@/components/ui/checkbox'
    14→import {
    15→  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger, DropdownMenuSeparator,
    16→} from '@/components/ui/dropdown-menu'
    17→import { Button } from '@/components/ui/button'
    18→import { Badge } from '@/components/ui/badge'
    19→import { Card, CardContent } from '@/components/ui/card'
    20→import { Skeleton } from '@/components/ui/skeleton'
    21→import { Input } from '@/components/ui/input'
    22→import { Label } from '@/components/ui/label'

    31→import {
    32→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription,
    33→} from '@/components/ui/dialog'
    34→import { StatusBadge } from '@/components/shared/status-badge'
    35→import { formatDate, formatTime, formatCurrency, getTodayString } from '@/lib/format'

    40→// ─── Types ──────────────────────────────────────────────────────────────
    41→
    42→interface DepartureGuest {
    43→  id: string
    44→  firstName: string
    45→  lastName: string
    46→  vipLevel: string
    47→}
    48→
    49→interface DepartureRoom {
    50→  id: string
    51→  number: string
    52→  floor: number
    53→  type: { name: string; code: string }
    54→}
    55→
    56→interface FolioLineItem {
    57→  id: string
    58→  description: string
    59→  amount: number
    60→  type: 'charge' | 'payment'
    61→  date: string
    62→}
    63→
    64→interface DepartureFolio {
    65→  id: string
    66→  balance: number
    67→  status: string
    68→  items?: FolioLineItem[]
    69→}
    70→
    71→interface Departure {
    72→  id: string
    73→  confirmationNo: string
    74→  status: string
    75→  checkIn: string
    76→  checkOut: string
    77→  roomRate: number
    78→  totalAmount: number
    79→  paidAmount: number
    80→  creditLimit: number
    81→  guest: DepartureGuest
    82→  room: DepartureRoom
    83→  folios: DepartureFolio[]
    84→}
    85→
    86→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    87→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    88→
    89→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    90→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    91→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    92→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    93→}
    94→
    95→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    96→  cash: 'Cash',
    97→  card: 'Credit/Debit Card',
    98→  bank_transfer: 'Bank Transfer',
    99→}
   100→
   101→// ─── Component ──────────────────────────────────────────────────────────
   102→
   103→export function DeparturesView() {
   104→  const queryClient = useQueryClient()
   105→  const { settings } = useSettingsStore()
   106→  const { preferences } = usePreferencesStore()
   107→  const { navigateTo } = useNavigationStore()
   108→  const today = getTodayString()
   109→
   110→  const [selectedRows, setSelectedRows] = useState<Set<string>>(new Set())
   111→
   112→  // Dialog states
   113→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   114→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   115→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   116→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   117→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   118→
   119→  // Selected departure
   120→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   121→
   122→  // Late checkout
   123→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   124→
   125→  // Payment
   126→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   127→  const [paymentAmount, setPaymentAmount] = useState('')
   128→  const [paymentReference, setPaymentReference] = useState('')
   129→
   130→  // Receipt data
   131→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   132→
   133→  // Fetch today's departures (checked_in with checkOut = today)
   134→  const { data, isLoading } = useQuery({
   135→    queryKey: ['departures', today],
   136→    queryFn: async () => {
   137→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   138→      return apiFetch(`/api/reservations?${params.toString()}`)
   139→    },
   140→    refetchInterval: 30000,
   141→  })
   142→
   143→  const departures: Departure[] = data?.reservations || []
   144→
   145→  // Stats
   146→  const totalDepartures = departures.length
   147→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   148→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   149→  const outstandingBalance = departures.reduce((sum, d) => {
   150→    return sum + (d.folios[0]?.balance || 0)
   151→  }, 0)
   152→
   153→  // Zero-balance pending guests (for express & batch checkout)
   154→  const zeroBalancePending = departures.filter(
   155→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   156→  )
   157→
   158→  // ─── Mutations ─────────────────────────────────────────────────────
   159→
   160→  // Checkout mutation
   161→  const checkoutMutation = useMutation({
   162→    mutationFn: async (reservationId: string) => {
   163→      return apiFetch(`/api/reservations/${reservationId}`, {
   164→        method: 'PATCH',
   165→        headers: { 'Content-Type': 'application/json' },
   166→        body: JSON.stringify({ status: 'checked_out' }),
   167→      })
   168→    },
   169→    onSuccess: () => {
   170→      invalidate.afterCheckout(queryClient)
   171→    },
   172→  })
   173→
   174→  // Payment mutation
   175→  const paymentMutation = useMutation({
   176→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   177→      return apiFetch(`/api/folio/${folioId}`, {
   178→        method: 'POST',
   179→        headers: { 'Content-Type': 'application/json' },
   180→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   181→      })
   182→    },
   183→    onSuccess: () => {
   184→      invalidate.afterFolioChange(queryClient)
   185→      setPaymentDialogOpen(false)
   186→      setPaymentAmount('')
   187→      setPaymentReference('')
   188→      toast.success('Payment recorded successfully')
   189→    },
   190→    onError: () => {
   191→      toast.error('Failed to record payment')
   192→    },
   193→  })
   194→
   195→  // Late checkout mutation
   196→  const lateCheckoutMutation = useMutation({
   197→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   198→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   199→        method: 'PATCH',
   200→        headers: { 'Content-Type': 'application/json' },
   201→        body: JSON.stringify({ checkOut: newCheckOut }),
   202→      })
   203→      // Add surcharge to folio
   204→      if (surcharge > 0) {
   205→        await apiFetch(`/api/folio`, {
   206→          method: 'POST',
   207→          headers: { 'Content-Type': 'application/json' },
   208→          body: JSON.stringify({
   209→            reservationId,
   210→            type: 'charge',
   211→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   212→            amount: surcharge,
   213→          }),
   214→        })
   215→      }
   216→      return result
   217→    },
   218→    onSuccess: () => {
   219→      invalidate.afterReservationChange(queryClient)
   220→      invalidate.afterFolioChange(queryClient)
   221→      setLateCheckoutDialogOpen(false)
   222→      toast.success('Late checkout request processed successfully')
   223→    },
   224→    onError: () => {
   225→      toast.error('Failed to process late checkout request')
   226→    },
   227→  })
   228→
   229→  // Batch checkout mutation
   230→  const batchCheckoutMutation = useMutation({
   231→    mutationFn: async (ids: string[]) => {
   232→      const results = []
   233→      for (let i = 0; i < ids.length; i++) {
   234→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   235→          method: 'PATCH',
   236→          headers: { 'Content-Type': 'application/json' },
   237→          body: JSON.stringify({ status: 'checked_out' }),
   238→        })
   239→        results.push(result)
   240→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   241→      }
   242→      return results
   243→    },
   244→    onSuccess: (_data, ids) => {
   245→      invalidate.afterCheckout(queryClient)
   246→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   247→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   248→    },
   249→    onError: () => {
   250→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   251→    },
   252→  })
   253→
   254→  // ─── Handlers ─────────────────────────────────────────────────────
   255→
   256→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   257→
   258→  const handleExpressCheckout = (dep: Departure) => {
   259→    checkoutMutation.mutate(dep.id, {
   260→      onSuccess: () => {
   261→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   262→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   263→      },
   264→    })
   265→  }
   266→
   267→  const handleReviewFolio = (dep: Departure) => {
   268→    setSelectedDeparture(dep)
   269→    setFolioDialogOpen(true)
   270→  }
   271→
   272→  const handleNavigateToFolio = (dep: Departure) => {
   273→    useFolioContextStore.getState().setFolioContext({
   274→      reservationId: dep.id,
   275→      guestId: dep.guest?.id || '',
   276→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   277→      roomNumber: dep.room?.number || '',
   278→      confirmationNo: dep.confirmationNo,
   279→    })
   280→    navigateTo('front-desk', 'folio')
   281→  }
   282→
   283→  const handleViewLedger = (dep: Departure) => {
   284→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   285→      guestId: dep.guest?.id || '',
   286→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   287→    })
   288→    navigateTo('front-desk', 'guest-ledger')
   289→  }
   290→
   291→  const handleFolioSettleCheckout = () => {
   292→    setFolioDialogOpen(false)
   293→    setCheckoutDialogOpen(true)
   294→  }
   295→
   296→  const handleQuickCheckout = (dep: Departure) => {
   297→    setSelectedDeparture(dep)
   298→    setCheckoutDialogOpen(true)
   299→  }
   300→
   301→  const handleRequestLateCheckout = (dep: Departure) => {
   302→    setSelectedDeparture(dep)
   303→    setLateCheckoutOption('14:00')
   304→    setLateCheckoutDialogOpen(true)
   305→  }
   306→
   307→  const handleConfirmLateCheckout = () => {
   308→    if (!selectedDeparture) return
   309→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   310→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   311→
   312→    // Build new checkout datetime with the selected time
   313→    const currentCheckout = new Date(selectedDeparture.checkOut)
   314→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   315→    currentCheckout.setHours(hours, minutes, 0, 0)
   316→
   317→    lateCheckoutMutation.mutate({
   318→      reservationId: selectedDeparture.id,
   319→      newCheckOut: currentCheckout.toISOString(),
   320→      surcharge,
   321→    })
   322→  }
   323→
   324→  const handleOpenPaymentDialog = () => {
   325→    if (!selectedDeparture) return
   326→    const balance = getBalance(selectedDeparture)
   327→    setPaymentAmount(String(balance))
   328→    setPaymentMethod('cash')
   329→    setPaymentReference('')
   330→    setPaymentDialogOpen(true)
   331→  }
   332→
   333→  const handleRecordPayment = () => {
   334→    if (!selectedDeparture) return
   335→    const folioId = selectedDeparture.folios[0]?.id
   336→    if (!folioId) {
   337→      toast.error('No folio found for this reservation')
   338→      return
   339→    }
   340→    const amount = Number(paymentAmount)
   341→    if (isNaN(amount) || amount <= 0) {
   342→      toast.error('Please enter a valid payment amount')
   343→      return
   344→    }
   345→    paymentMutation.mutate({
   346→      folioId,
   347→      amount,
   348→      method: paymentMethod,
   349→      reference: paymentReference,
   350→    })
   351→  }
   352→
   353→  const confirmCheckout = () => {
   354→    if (!selectedDeparture) return
   355→    const roomNumber = selectedDeparture.room.number
   356→    checkoutMutation.mutate(selectedDeparture.id, {
   357→      onSuccess: () => {
   358→        setCheckoutDialogOpen(false)
   359→        setReceiptData(selectedDeparture)
   360→        setReceiptDialogOpen(true)
   361→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   362→      },
   363→    })
   364→  }
   365→
   366→  const handleBatchCheckout = () => {
   367→    const ids = zeroBalancePending.map((d) => d.id)
   368→    if (ids.length === 0) {
   369→      toast.info('No guests with zero balance to batch checkout')
   370→      return
   371→    }
   372→    batchCheckoutMutation.mutate(ids)
   373→  }
   374→
   375→  const handlePrintReceipt = () => {
   376→    toast.success('Receipt sent to printer')
   377→  }
   378→
   379→  const handleEmailReceipt = () => {
   380→    toast.success('Receipt emailed to guest')
   381→  }
   382→
   383→  // ─── Computed values ──────────────────────────────────────────────
   384→
   385→  const lateCheckoutFee = selectedDeparture
   386→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   387→    : 0
   388→
   389→  // ─── Render ────────────────────────────────────────────────────────
   390→
   391→  return (
   392→    <div className="flex flex-col gap-2">
   393→      {/* Header */}
   394→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   395→        <div>
   396→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   397→          <p className="text-xs text-muted-foreground">
   398→            Guest check-outs scheduled for {formatDate(today)}
   399→          </p>
   400→        </div>
   401→        {zeroBalancePending.length > 0 && (
   402→          <Button
   403→            onClick={handleBatchCheckout}
   404→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   405→            disabled={batchCheckoutMutation.isPending}
   406→          >
   407→            <Zap className="size-4 mr-1.5" />
   408→            {batchCheckoutMutation.isPending
   409→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   410→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   411→          </Button>
   412→        )}
   413→      </div>
   414→
   415→      {/* Departures List */}
   416→      <Card className="py-0">
   417→        <CardContent className="p-0">
   418→          <div className="max-h-[600px] overflow-y-auto">
   419→            <Table>
   420→              <TableHeader>
   421→                <TableRow>
   422→                  <TableHead className="w-[40px]">
   423→                    <Checkbox
   424→                      checked={departures.length > 0 && selectedRows.size === departures.filter(d => d.status === 'checked_in').length}
   425→                      onCheckedChange={(checked) => {
   426→                        if (checked) {
   427→                          setSelectedRows(new Set(departures.filter(d => d.status === 'checked_in').map(d => d.id)))
   428→                        } else {
   429→                          setSelectedRows(new Set())
   430→                        }
   431→                      }}
   432→                    />
   433→                  </TableHead>
   434→                  <TableHead className="w-[70px]">Room</TableHead>
   435→                  <TableHead>Guest</TableHead>
   436→                  <TableHead className="w-[100px]">Check-out</TableHead>
   437→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   438→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   439→                  <TableHead className="w-[50px]"></TableHead>
   440→                </TableRow>
   441→              </TableHeader>
   442→              <TableBody>
   443→                {isLoading ? (
   444→                  Array.from({ length: 5 }).map((_, i) => (
   445→                    <TableRow key={i}>
   446→                      {Array.from({ length: 7 }).map((_, j) => (
   447→                        <TableCell key={j}>
   448→                          <Skeleton className="h-4 w-full" />
   449→                        </TableCell>
   450→                      ))}
   451→                    </TableRow>
   452→                  ))
   453→                ) : departures.length === 0 ? (
   454→                  <TableRow>
   455→                    <TableCell colSpan={7} className="h-24 text-center text-muted-foreground">
   456→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   457→                      No departures scheduled for today
   458→                    </TableCell>
   459→                  </TableRow>
   460→                ) : (
   461→                  departures.map((dep) => {
   462→                    const balance = getBalance(dep)
   463→                    const isPending = dep.status === 'checked_in'
   464→                    const isZeroBalance = balance === 0
   465→                    return (
   466→                      <TableRow key={dep.id}>
   467→                        <TableCell>
   468→                          <Checkbox
   469→                            checked={selectedRows.has(dep.id)}
   470→                            onCheckedChange={(checked) => {
   471→                              const next = new Set(selectedRows)
   472→                              if (checked) next.add(dep.id)
   473→                              else next.delete(dep.id)
   474→                              setSelectedRows(next)
   475→                            }}
   476→                            disabled={dep.status === 'checked_out'}
   477→                          />
   478→                        </TableCell>
   479→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   480→                        <TableCell>
   481→                          <div className="flex items-center gap-2">
   482→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   483→                            {dep.guest.vipLevel !== 'none' && (
   484→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   485→                                VIP
   486→                              </Badge>
   487→                            )}
   488→                          </div>
   489→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   490→                        </TableCell>
   491→                        <TableCell className="text-xs">
   492→                          <div>{formatDate(dep.checkOut)}</div>
   493→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   494→                        </TableCell>
   495→                        <TableCell className="text-right">
   496→                          <div className="flex items-center justify-end gap-1.5">
   497→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   498→                              {formatCurrency(balance)}
   499→                            </span>
   500→                            {/* Outstanding Balance Alert Badges */}
   501→                            {isPending && balance > 5000 && (
   502→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   503→                                HIGH
   504→                              </Badge>
   505→                            )}
   506→                            {isPending && balance > 0 && balance <= 5000 && (
   507→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   508→                                DUE
   509→                              </Badge>
   510→                            )}
   511→                          </div>
   512→                        </TableCell>
   513→                        <TableCell className="text-right">
   514→                          {dep.status === 'checked_out' ? (
   515→                            <div className="flex items-center justify-end gap-1">
   516→                              <BedSingle className="size-3 text-yellow-500" />
   517→                              <StatusBadge status="vacant_dirty" />
   518→                            </div>
   519→                          ) : (
   520→                            <StatusBadge status="checked_in" />
   521→                          )}
   522→                        </TableCell>
   523→                        <TableCell className="text-right">
   524→                          {isPending ? (
   525→                            <DropdownMenu>
   526→                              <DropdownMenuTrigger asChild>
   527→                                <Button variant="ghost" size="icon" className="size-8">
   528→                                  <MoreHorizontal className="size-4" />
   529→                                </Button>
   530→                              </DropdownMenuTrigger>
   531→                              <DropdownMenuContent align="end">
   532→                                {isZeroBalance && (
   533→                                  <DropdownMenuItem onClick={() => handleExpressCheckout(dep)}>
   534→                                    <Zap className="size-4 mr-2 text-green-600" /> Express Checkout
   535→                                  </DropdownMenuItem>
   536→                                )}
   537→                                <DropdownMenuItem onClick={() => handleReviewFolio(dep)}>
   538→                                  <Eye className="size-4 mr-2" /> Review Folio
   539→                                </DropdownMenuItem>
   540→                                <DropdownMenuItem onClick={() => handleNavigateToFolio(dep)}>
   541→                                  <FileText className="size-4 mr-2" /> Full Folio
   542→                                </DropdownMenuItem>
   543→                                <DropdownMenuItem onClick={() => handleViewLedger(dep)}>
   544→                                  <BookOpen className="size-4 mr-2" /> View Ledger
   545→                                </DropdownMenuItem>
   546→                                <DropdownMenuSeparator />
   547→                                <DropdownMenuItem onClick={() => handleRequestLateCheckout(dep)}>
   548→                                  <Clock className="size-4 mr-2" /> Late Checkout
   549→                                </DropdownMenuItem>
