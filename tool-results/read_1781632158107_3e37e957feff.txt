   160→// ─── Helpers ────────────────────────────────────────────────────────────
   161→
   162→function guestFullName(g: FolioGuest | null | undefined): string {
   163→  if (!g) return '—'
   164→  return `${g.firstName} ${g.lastName}`
   165→}
   166→
   167→function folioCharges(f: Folio): number {
   168→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   169→}
   170→
   171→function folioPayments(f: Folio): number {
   172→  return f.payments.reduce((s, p) => s + p.amount, 0)
   173→}
   174→
   175→function folioOutstanding(f: Folio): number {
   176→  return folioCharges(f) - folioPayments(f)
   177→}
   178→
   179→function isVoidedTransaction(t: FolioTransaction): boolean {
   180→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   181→}
   182→
   183→function isVoidedPayment(p: FolioPayment): boolean {
   184→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   185→}
   186→
   187→function useDebounce<T>(value: T, delay: number): T {
   188→  const [debounced, setDebounced] = useState(value)
   189→  useEffect(() => {
   190→    const timer = setTimeout(() => setDebounced(value), delay)
   191→    return () => clearTimeout(timer)
   192→  }, [value, delay])
   193→  return debounced
   194→}

  1586→
  1587→          {/* Notes Tab */}
