   168→function sourceLabel(source: string | null): string {
   169→  if (!source) return 'Direct'
   170→  const map: Record<string, string> = {
   171→    direct: 'Direct',
   172→    booking_com: 'Booking.com',
   173→    expedia: 'Expedia',
   174→    airbnb: 'Airbnb',
   175→    agoda: 'Agoda',
   176→    corporate: 'Corporate',
   177→    walk_in: 'Walk-in',
   178→    phone: 'Phone',
   179→    email: 'Email',
   180→    website: 'Website',
   181→  }
   182→  return map[source] || source.replace(/_/g, ' ').replace(/\b\w/g, c => c.toUpperCase())
   183→}

   235→/** 7-day calendar availability strip */
   236→function CalendarStrip({ summary, reservations }: {
   237→  summary: RoomsSummary | undefined
   238→  reservations: ReservationData[] | undefined
   239→}) {
   240→  const days = useMemo(() => {
   241→    const result: Array<{
   242→      date: Date
   243→      dayName: string
   244→      dateStr: string
   245→      arrivalCount: number
   246→      departureCount: number
   247→      occupancyCount: number
   248→      vacancyPct: number
   249→    }> = []
   250→
   251→    for (let i = 0; i < 7; i++) {
   252→      const d = new Date()
   253→      d.setDate(d.getDate() + i)
   254→      d.setHours(0, 0, 0, 0)
   255→
   256→      const nextD = new Date(d)
   257→      nextD.setDate(nextD.getDate() + 1)
   258→
   259→      let arrivalCount = 0
   260→      let departureCount = 0
   261→      let occupancyCount = 0
   262→
   263→      if (reservations) {
   264→        for (const r of reservations) {
   265→          const ci = new Date(r.checkIn)
   266→          ci.setHours(0, 0, 0, 0)
   267→          const co = new Date(r.checkOut)
   268→          co.setHours(0, 0, 0, 0)
   269→          if (r.status === 'checked_in' || r.status === 'confirmed') {
   270→            if (ci.getTime() === d.getTime()) arrivalCount++
   271→            if (co.getTime() === d.getTime()) departureCount++
   272→            if (ci < nextD && co > d) occupancyCount++
   273→          }
   274→        }
   275→      }
   276→
   277→      const totalRooms = summary?.totalRooms || 36
   278→      const vacancyPct = totalRooms > 0 ? ((totalRooms - occupancyCount) / totalRooms) * 100 : 100
   279→
   280→      result.push({
   281→        date: d,
   282→        dayName: DAY_NAMES[d.getDay()],
   283→        dateStr: `${d.getDate()} ${MONTH_NAMES[d.getMonth()]}`,
   284→        arrivalCount,
   285→        departureCount,
   286→        occupancyCount,
   287→        vacancyPct,
   288→      })
   289→    }
   290→    return result
   291→  }, [reservations, summary])
   292→
   293→  const getBarColor = (vacancyPct: number) => {
   294→    if (vacancyPct > 50) return 'bg-emerald-500'
   295→    if (vacancyPct >= 20) return 'bg-yellow-500'
   296→    return 'bg-red-500'
   297→  }
   298→
   299→  return (
   300→    <Card className="p-4">
   301→      <CardHeader className="p-0 pb-3 mb-3">
   302→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   303→          <CalendarDays className="size-4 text-muted-foreground" />
   304→          7-Day Availability
   305→        </CardTitle>
   306→      </CardHeader>
   307→      <div className="flex gap-1.5">
   308→        {days.map((day, idx) => (
   309→          <div key={idx} className="flex-1 flex flex-col items-center gap-1.5">
   310→            <span className="text-[10px] font-medium text-muted-foreground">{day.dayName}</span>
   311→            <span className="text-[11px] font-semibold">{day.dateStr}</span>
   312→            <div className={cn('w-full h-2 rounded-full', getBarColor(day.vacancyPct))} />
   313→            <div className="flex items-center gap-0.5 h-3">
   314→              {day.arrivalCount > 0 && (
   315→                <div className="size-2 rounded-full bg-blue-500" title={day.arrivalCount + ' arrival(s)'} />
   316→              )}
   317→              {day.departureCount > 0 && (
   318→                <div className="size-2 rounded-full bg-orange-500" title={day.departureCount + ' departure(s)'} />
   319→              )}
   320→            </div>
   321→          </div>
   322→        ))}
   323→      </div>
   324→      <div className="flex items-center gap-4 mt-3 pt-2 border-t">
   325→        <div className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   326→          <div className="size-2 rounded-full bg-blue-500" /> Arrivals
   327→        </div>
   328→        <div className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   329→          <div className="size-2 rounded-full bg-orange-500" /> Departures
   330→        </div>
   331→        <div className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   332→          <div className="w-4 h-2 rounded-full bg-emerald-500" /> High
   333→        </div>
   334→        <div className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   335→          <div className="w-4 h-2 rounded-full bg-yellow-500" /> Medium
   336→        </div>
   337→        <div className="flex items-center gap-1.5 text-[10px] text-muted-foreground">
   338→          <div className="w-4 h-2 rounded-full bg-red-500" /> Low
   339→        </div>
   340→      </div>
   341→    </Card>
   342→  )
   343→}
