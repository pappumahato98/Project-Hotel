     3→import { useState } from 'react'
     4→import { useMutation, useQueryClient, useQuery } from '@tanstack/react-query'
     5→import { toast } from 'sonner'
     6→import { apiFetch } from '@/lib/api'
     7→import {
     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,
     9→  Zap, Eye, Clock, Banknote, Mail, X, ArrowRight, BedSingle, FileText, BookOpen,
    10→} from 'lucide-react'
    11→
    12→import { Button } from '@/components/ui/button'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Card, CardContent } from '@/components/ui/card'
    15→import { Skeleton } from '@/components/ui/skeleton'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Separator } from '@/components/ui/separator'
    19→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select'
    23→import {
    24→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    25→} from '@/components/ui/table'
    26→import {
    27→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription,
    28→} from '@/components/ui/dialog'
    29→import { StatusBadge } from '@/components/shared/status-badge'
    30→import { formatDate, formatTime, formatCurrency, getTodayString } from '@/lib/format'
    31→import { cn } from '@/lib/utils'
    32→import { useSettingsStore, usePreferencesStore, useNavigationStore, useFolioContextStore, useGuestLedgerContextStore } from '@/lib/store'

    35→// ─── Types ──────────────────────────────────────────────────────────────
    36→
    37→interface DepartureGuest {
    38→  id: string
    39→  firstName: string
    40→  lastName: string
    41→  vipLevel: string
    42→}
    43→
    44→interface DepartureRoom {
    45→  id: string
    46→  number: string
    47→  floor: number
    48→  type: { name: string; code: string }
    49→}
    50→
    51→interface FolioLineItem {
    52→  id: string
    53→  description: string
    54→  amount: number
    55→  type: 'charge' | 'payment'
    56→  date: string
    57→}
    58→
    59→interface DepartureFolio {
    60→  id: string
    61→  balance: number
    62→  status: string
    63→  items?: FolioLineItem[]
    64→}
    65→
    66→interface Departure {
    67→  id: string
    68→  confirmationNo: string
    69→  status: string
    70→  checkIn: string
    71→  checkOut: string
    72→  roomRate: number
    73→  totalAmount: number
    74→  paidAmount: number
    75→  creditLimit: number
    76→  guest: DepartureGuest
    77→  room: DepartureRoom
    78→  folios: DepartureFolio[]
    79→}
    80→
    81→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    82→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    83→
    84→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    85→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    86→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    87→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    88→}
    89→
    90→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    91→  cash: 'Cash',
    92→  card: 'Credit/Debit Card',
    93→  bank_transfer: 'Bank Transfer',
    94→}
    95→
    96→// ─── Component ──────────────────────────────────────────────────────────
    97→
    98→export function DeparturesView() {
    99→  const queryClient = useQueryClient()
   100→  const { settings } = useSettingsStore()
   101→  const { preferences } = usePreferencesStore()
   102→  const { navigateTo } = useNavigationStore()
   103→  const today = getTodayString()
   104→
   105→  // Dialog states
   106→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   107→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   108→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   109→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   110→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   111→
   112→  // Selected departure
   113→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   114→
   115→  // Late checkout
   116→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   117→
   118→  // Payment
   119→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   120→  const [paymentAmount, setPaymentAmount] = useState('')
   121→  const [paymentReference, setPaymentReference] = useState('')
   122→
   123→  // Receipt data
   124→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   125→
   126→  // Fetch today's departures (checked_in with checkOut = today)
   127→  const { data, isLoading } = useQuery({
   128→    queryKey: ['departures', today],
   129→    queryFn: async () => {
   130→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   131→      return apiFetch(`/api/reservations?${params.toString()}`)
   132→    },
   133→    refetchInterval: 30000,
   134→  })
   135→
   136→  const departures: Departure[] = data?.reservations || []
   137→
   138→  // Stats
   139→  const totalDepartures = departures.length
   140→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   141→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   142→  const outstandingBalance = departures.reduce((sum, d) => {
   143→    return sum + (d.folios[0]?.balance || 0)
   144→  }, 0)
   145→
   146→  // Zero-balance pending guests (for express & batch checkout)
   147→  const zeroBalancePending = departures.filter(
   148→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   149→  )
   150→
   151→  // ─── Mutations ─────────────────────────────────────────────────────
   152→
   153→  // Checkout mutation
   154→  const checkoutMutation = useMutation({
   155→    mutationFn: async (reservationId: string) => {
   156→      return apiFetch(`/api/reservations/${reservationId}`, {
   157→        method: 'PATCH',
   158→        headers: { 'Content-Type': 'application/json' },
   159→        body: JSON.stringify({ status: 'checked_out' }),
   160→      })
   161→    },
   162→    onSuccess: () => {
   163→      invalidate.afterCheckout(queryClient)
   164→    },
   165→  })
   166→
   167→  // Payment mutation
   168→  const paymentMutation = useMutation({
   169→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   170→      return apiFetch(`/api/folio/${folioId}`, {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   174→      })
   175→    },
   176→    onSuccess: () => {
   177→      invalidate.afterFolioChange(queryClient)
   178→      setPaymentDialogOpen(false)
   179→      setPaymentAmount('')
   180→      setPaymentReference('')
   181→      toast.success('Payment recorded successfully')
   182→    },
   183→    onError: () => {
   184→      toast.error('Failed to record payment')
   185→    },
   186→  })
   187→
   188→  // Late checkout mutation
   189→  const lateCheckoutMutation = useMutation({
   190→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   191→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   192→        method: 'PATCH',
   193→        headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({ checkOut: newCheckOut }),
   195→      })
   196→      // Add surcharge to folio
   197→      if (surcharge > 0) {
   198→        await apiFetch(`/api/folio`, {
   199→          method: 'POST',
   200→          headers: { 'Content-Type': 'application/json' },
   201→          body: JSON.stringify({
   202→            reservationId,
   203→            type: 'charge',
   204→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   205→            amount: surcharge,
   206→          }),
   207→        })
   208→      }
   209→      return result
   210→    },
   211→    onSuccess: () => {
   212→      invalidate.afterReservationChange(queryClient)
   213→      invalidate.afterFolioChange(queryClient)
   214→      setLateCheckoutDialogOpen(false)
   215→      toast.success('Late checkout request processed successfully')
   216→    },
   217→    onError: () => {
   218→      toast.error('Failed to process late checkout request')
   219→    },
   220→  })
   221→
   222→  // Batch checkout mutation
   223→  const batchCheckoutMutation = useMutation({
   224→    mutationFn: async (ids: string[]) => {
   225→      const results = []
   226→      for (let i = 0; i < ids.length; i++) {
   227→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   228→          method: 'PATCH',
   229→          headers: { 'Content-Type': 'application/json' },
   230→          body: JSON.stringify({ status: 'checked_out' }),
   231→        })
   232→        results.push(result)
   233→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   234→      }
   235→      return results
   236→    },
   237→    onSuccess: (_data, ids) => {
   238→      invalidate.afterCheckout(queryClient)
   239→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   240→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   241→    },
   242→    onError: () => {
   243→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   244→    },
   245→  })
   246→
   247→  // ─── Handlers ─────────────────────────────────────────────────────
   248→
   249→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   250→
   251→  const handleExpressCheckout = (dep: Departure) => {
   252→    checkoutMutation.mutate(dep.id, {
   253→      onSuccess: () => {
   254→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   255→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   256→      },
   257→    })
   258→  }
   259→
   260→  const handleReviewFolio = (dep: Departure) => {
   261→    setSelectedDeparture(dep)
   262→    setFolioDialogOpen(true)
   263→  }
   264→
   265→  const handleNavigateToFolio = (dep: Departure) => {
   266→    useFolioContextStore.getState().setFolioContext({
   267→      reservationId: dep.id,
   268→      guestId: dep.guest?.id || '',
   269→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   270→      roomNumber: dep.room?.number || '',
   271→      confirmationNo: dep.confirmationNo,
   272→    })
   273→    navigateTo('front-desk', 'folio')
   274→  }
   275→
   276→  const handleViewLedger = (dep: Departure) => {
   277→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   278→      guestId: dep.guest?.id || '',
   279→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   280→    })
   281→    navigateTo('front-desk', 'guest-ledger')
   282→  }
   283→
   284→  const handleFolioSettleCheckout = () => {
   285→    setFolioDialogOpen(false)
   286→    setCheckoutDialogOpen(true)
   287→  }
   288→
   289→  const handleQuickCheckout = (dep: Departure) => {
   290→    setSelectedDeparture(dep)
   291→    setCheckoutDialogOpen(true)
   292→  }
   293→
   294→  const handleRequestLateCheckout = (dep: Departure) => {
   295→    setSelectedDeparture(dep)
   296→    setLateCheckoutOption('14:00')
   297→    setLateCheckoutDialogOpen(true)
   298→  }
   299→
   300→  const handleConfirmLateCheckout = () => {
   301→    if (!selectedDeparture) return
   302→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   303→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   304→
   305→    // Build new checkout datetime with the selected time
   306→    const currentCheckout = new Date(selectedDeparture.checkOut)
   307→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   308→    currentCheckout.setHours(hours, minutes, 0, 0)
   309→
   310→    lateCheckoutMutation.mutate({
   311→      reservationId: selectedDeparture.id,
   312→      newCheckOut: currentCheckout.toISOString(),
   313→      surcharge,
   314→    })
   315→  }
   316→
   317→  const handleOpenPaymentDialog = () => {
   318→    if (!selectedDeparture) return
   319→    const balance = getBalance(selectedDeparture)
   320→    setPaymentAmount(String(balance))
   321→    setPaymentMethod('cash')
   322→    setPaymentReference('')
   323→    setPaymentDialogOpen(true)
   324→  }
   325→
   326→  const handleRecordPayment = () => {
   327→    if (!selectedDeparture) return
   328→    const folioId = selectedDeparture.folios[0]?.id
   329→    if (!folioId) {
   330→      toast.error('No folio found for this reservation')
   331→      return
   332→    }
   333→    const amount = Number(paymentAmount)
   334→    if (isNaN(amount) || amount <= 0) {
   335→      toast.error('Please enter a valid payment amount')
   336→      return
   337→    }
   338→    paymentMutation.mutate({
   339→      folioId,
   340→      amount,
   341→      method: paymentMethod,
   342→      reference: paymentReference,
   343→    })
   344→  }
   345→
   346→  const confirmCheckout = () => {
   347→    if (!selectedDeparture) return
   348→    const roomNumber = selectedDeparture.room.number
   349→    checkoutMutation.mutate(selectedDeparture.id, {
   350→      onSuccess: () => {
   351→        setCheckoutDialogOpen(false)
   352→        setReceiptData(selectedDeparture)
   353→        setReceiptDialogOpen(true)
   354→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   355→      },
   356→    })
   357→  }
   358→
   359→  const handleBatchCheckout = () => {
   360→    const ids = zeroBalancePending.map((d) => d.id)
   361→    if (ids.length === 0) {
   362→      toast.info('No guests with zero balance to batch checkout')
   363→      return
   364→    }
   365→    batchCheckoutMutation.mutate(ids)
   366→  }
   367→
   368→  const handlePrintReceipt = () => {
   369→    toast.success('Receipt sent to printer')
   370→  }
   371→
   372→  const handleEmailReceipt = () => {
   373→    toast.success('Receipt emailed to guest')
   374→  }
   375→
   376→  // ─── Computed values ──────────────────────────────────────────────
   377→
   378→  const lateCheckoutFee = selectedDeparture
   379→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   380→    : 0
   381→
   382→  // ─── Render ────────────────────────────────────────────────────────
   383→
   384→  return (
   385→    <div className="flex flex-col gap-2">
   386→      {/* Header */}
   387→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   388→        <div>
   389→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   390→          <p className="text-xs text-muted-foreground">
   391→            Guest check-outs scheduled for {formatDate(today)}
   392→          </p>
   393→        </div>
   394→        {zeroBalancePending.length > 0 && (
   395→          <Button
   396→            onClick={handleBatchCheckout}
   397→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   398→            disabled={batchCheckoutMutation.isPending}
   399→          >
   400→            <Zap className="size-4 mr-1.5" />
   401→            {batchCheckoutMutation.isPending
   402→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   403→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   404→          </Button>
   405→        )}
   406→      </div>
   407→
   408→      {/* Departures List */}
   409→      <Card className="py-0">
   410→        <CardContent className="p-0">
   411→          <div className="max-h-[600px] overflow-y-auto">
   412→            <Table>
   413→              <TableHeader>
   414→                <TableRow>
   415→                  <TableHead className="w-[70px]">Room</TableHead>
   416→                  <TableHead>Guest</TableHead>
   417→                  <TableHead className="w-[100px]">Check-out</TableHead>
   418→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   419→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   420→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   421→                </TableRow>
   422→              </TableHeader>
   423→              <TableBody>
   424→                {isLoading ? (
   425→                  Array.from({ length: 5 }).map((_, i) => (
   426→                    <TableRow key={i}>
   427→                      {Array.from({ length: 6 }).map((_, j) => (
   428→                        <TableCell key={j}>
   429→                          <Skeleton className="h-4 w-full" />
   430→                        </TableCell>
   431→                      ))}
   432→                    </TableRow>
   433→                  ))
   434→                ) : departures.length === 0 ? (
   435→                  <TableRow>
   436→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   437→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   438→                      No departures scheduled for today
   439→                    </TableCell>
   440→                  </TableRow>
   441→                ) : (
   442→                  departures.map((dep) => {
   443→                    const balance = getBalance(dep)
   444→                    const isPending = dep.status === 'checked_in'
   445→                    const isZeroBalance = balance === 0
   446→                    return (
   447→                      <TableRow key={dep.id}>
   448→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   449→                        <TableCell>
   450→                          <div className="flex items-center gap-2">
   451→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   452→                            {dep.guest.vipLevel !== 'none' && (
   453→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   454→                                VIP
   455→                              </Badge>
   456→                            )}
   457→                          </div>
   458→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   459→                        </TableCell>
   460→                        <TableCell className="text-xs">
   461→                          <div>{formatDate(dep.checkOut)}</div>
   462→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   463→                        </TableCell>
   464→                        <TableCell className="text-right">
   465→                          <div className="flex items-center justify-end gap-1.5">
   466→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   467→                              {formatCurrency(balance)}
   468→                            </span>
   469→                            {/* Outstanding Balance Alert Badges */}
   470→                            {isPending && balance > 5000 && (
   471→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   472→                                HIGH
   473→                              </Badge>
   474→                            )}
   475→                            {isPending && balance > 0 && balance <= 5000 && (
   476→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   477→                                DUE
   478→                              </Badge>
   479→                            )}
   480→                          </div>
   481→                        </TableCell>
   482→                        <TableCell className="text-right">
   483→                          {dep.status === 'checked_out' ? (
   484→                            <div className="flex items-center justify-end gap-1">
   485→                              <BedSingle className="size-3 text-yellow-500" />
   486→                              <StatusBadge status="vacant_dirty" />
   487→                            </div>
   488→                          ) : (
   489→                            <StatusBadge status="checked_in" />
   490→                          )}
   491→                        </TableCell>
   492→                        <TableCell className="text-right">
   493→                          <div className="flex justify-end gap-1 flex-wrap">
   494→                            {isPending && (
   495→                              <>
   496→                                {/* Express Checkout (zero balance) */}
   497→                                {isZeroBalance && (
   498→                                  <Button
   499→                                    size="sm"
   500→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   501→                                    onClick={() => handleExpressCheckout(dep)}
   502→                                    disabled={checkoutMutation.isPending}
   503→                                  >
   504→                                    <Zap className="size-3 mr-0.5" /> Express
   505→                                  </Button>
   506→                                )}
   507→                                {/* Review Folio */}
   508→                                <Button
   509→                                  size="sm"
   510→                                  variant="outline"
   511→                                  className="text-xs h-7"
   512→                                  onClick={() => handleReviewFolio(dep)}
   513→                                >
   514→                                  <Eye className="size-3 mr-0.5" /> Review Folio
   515→                                </Button>
   516→                                <Button
   517→                                  size="sm"
   518→                                  variant="outline"
   519→                                  className="text-xs h-7"
   520→                                  onClick={() => handleNavigateToFolio(dep)}
   521→                                >
   522→                                  <FileText className="size-3 mr-0.5" /> Full Folio
   523→                                </Button>
   524→                                <Button
   525→                                  size="sm"
   526→                                  variant="outline"
   527→                                  className="text-xs h-7"
   528→                                  onClick={() => handleViewLedger(dep)}
   529→                                >
   530→                                  <BookOpen className="size-3 mr-0.5" /> View Ledger
   531→                                </Button>
   532→                                {/* Late Checkout */}
   533→                                <Button
   534→                                  size="sm"
   535→                                  variant="outline"
   536→                                  className="text-xs h-7"
   537→                                  onClick={() => handleRequestLateCheckout(dep)}
   538→                                >
   539→                                  <Clock className="size-3 mr-0.5" /> Late
   540→                                </Button>
   541→                                {/* Regular Checkout */}
   542→                                <Button
   543→                                  size="sm"
   544→                                  className="text-xs h-7"
   545→                                  onClick={() => handleQuickCheckout(dep)}
   546→                                >
   547→                                  <LogOut className="size-3 mr-0.5" /> Checkout
   548→                                </Button>
   549→                              </>
   550→                            )}
   551→                            {dep.status === 'checked_out' && (
   552→                              <Badge variant="outline" className="text-xs">
   553→                                <CheckCircle2 className="size-3 mr-1" />
   554→                                Checked Out
   555→                              </Badge>
   556→                            )}
   557→                          </div>
