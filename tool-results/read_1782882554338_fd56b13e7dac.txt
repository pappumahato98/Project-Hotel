    98→    98→// ─── Helpers ────────────────────────────────────────────────────────────
    99→    99→
   100→   100→function getOutstandingBalance(res: InHouseReservation): number {
   101→   101→  return res.folios[0]?.balance ?? (res.totalAmount - res.paidAmount)
   102→   102→}
   103→   103→
   104→   104→function getLastPayment(res: InHouseReservation): string | null {
   105→   105→  const payments = res.folios[0]?.payments
   106→   106→  if (!payments || payments.length === 0) return null
   107→   107→  const latest = payments.reduce((a, b) =>
   108→   108→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   109→   109→  )
   110→   110→  return new Date(latest.createdAt).toLocaleDateString()
   111→   111→}
   112→   112→
   113→   113→function getVipBadge(vipLevel: string) {
   114→   114→  if (vipLevel === 'none' || !vipLevel) return null
   115→   115→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   116→   116→  return (
   117→   117→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   118→   118→      {level}
   119→   119→    </Badge>
   120→   120→  )
   121→   121→}
