   179→   179→// ─── Debounce hook ────────────────────────────────────────────────────
   180→   180→function useDebounce<T>(value: T, delay: number): T {
   181→   181→  const [debounced, setDebounced] = useState(value)
   182→   182→  useEffect(() => {
   183→   183→    const timer = setTimeout(() => setDebounced(value), delay)
   184→   184→    return () => clearTimeout(timer)
   185→   185→  }, [value, delay])
   186→   186→  return debounced
   187→   187→}

   189→   189→// ─── Date range presets ──────────────────────────────────────────────
   190→   190→function getDatePreset(preset: string): { from: Date; to: Date } {
   191→   191→  const today = new Date()
   192→   192→  today.setHours(0, 0, 0, 0)
   193→   193→  switch (preset) {
   194→   194→    case 'today':
   195→   195→      return { from: today, to: today }
   196→   196→    case 'this-week': {
   197→   197→      const from = new Date(today)
   198→   198→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   199→   199→      const to = new Date(from)
   200→   200→      to.setDate(from.getDate() + 6) // Sunday
   201→   201→      return { from, to }
   202→   202→    }
   203→   203→    case 'this-month': {
   204→   204→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   205→   205→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   206→   206→      return { from, to }
   207→   207→    }
   208→   208→    case 'next-7': {
   209→   209→      const to = new Date(today)
   210→   210→      to.setDate(today.getDate() + 6)
   211→   211→      return { from: today, to }
   212→   212→    }
   213→   213→    case 'next-14': {
   214→   214→      const to = new Date(today)
   215→   215→      to.setDate(today.getDate() + 13)
   216→   216→      return { from: today, to }
   217→   217→    }
   218→   218→    case 'next-30': {
   219→   219→      const to = new Date(today)
   220→   220→      to.setDate(today.getDate() + 29)
   221→   221→      return { from: today, to }
   222→   222→    }
   223→   223→    default:
   224→   224→      return { from: today, to: today }
   225→   225→  }
   226→   226→}

   228→   228→function toDateString(d: Date): string {
   229→   229→  const y = d.getFullYear()
   230→   230→  const m = String(d.getMonth() + 1).padStart(2, '0')
   231→   231→  const day = String(d.getDate()).padStart(2, '0')
   232→   232→  return `${y}-${m}-${day}`
   233→   233→}
   234→   234→
   235→   235→function formatShortDate(dateStr: string): string {
   236→   236→  if (!dateStr) return ''
   237→   237→  const d = new Date(dateStr + 'T00:00:00')
   238→   238→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   239→   239→}
