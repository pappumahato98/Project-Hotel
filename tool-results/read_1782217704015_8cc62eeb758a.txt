   138→   138→function isSameDay(dateStr: string, todayStr: string): boolean {
   139→   139→  const d = new Date(dateStr)
   140→   140→  const t = new Date(todayStr)
   141→   141→  return d.getFullYear() === t.getFullYear() && d.getMonth() === t.getMonth() && d.getDate() === t.getDate()
   142→   142→}
   143→   143→
   144→   144→function isBeforeDay(dateStr: string, todayStr: string): boolean {
   145→   145→  const d = new Date(dateStr)
   146→   146→  d.setHours(0, 0, 0, 0)
   147→   147→  const t = new Date(todayStr)
   148→   148→  t.setHours(0, 0, 0, 0)
   149→   149→  return d.getTime() < t.getTime()
   150→   150→}
   151→   151→
   152→   152→function nightsBetween(start: string | Date, end: string | Date): number {
   153→   153→  const s = new Date(start)
   154→   154→  const e = new Date(end)
   155→   155→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   156→   156→}

   158→   158→function getCreditStatus(res: Departure): { pct: number; status: 'ok' | 'warning' | 'breach'; color: string } {
   159→   159→  const balance = res.folios[0]?.balance || 0
   160→   160→  const limit = res.creditLimit || 15000
   161→   161→  const pct = limit > 0 ? (balance / limit) * 100 : 0
   162→   162→  if (pct >= 100) return { pct, status: 'breach', color: 'text-red-600' }
   163→   163→  if (pct >= 80) return { pct, status: 'warning', color: 'text-amber-600' }
   164→   164→  return { pct, status: 'ok', color: 'text-green-600' }
   165→   165→}
