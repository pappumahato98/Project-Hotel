   169→function SourceBadge({ source }: { source: string | null }) {
   170→  if (!source) return <span className="text-xs text-muted-foreground">—</span>
   171→
   172→  const colors: Record<string, string> = {
   173→    online: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   174→    walk_in: 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-300 border-violet-300 dark:border-violet-700',
   175→    phone: 'bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   176→    email: 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-300 border-cyan-300 dark:border-cyan-700',
   177→    agent: 'bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300 border-orange-300 dark:border-orange-700',
   178→    corporate: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300 border-slate-300 dark:border-slate-600',
   179→    group: 'bg-pink-100 text-pink-700 dark:bg-pink-950 dark:text-pink-300 border-pink-300 dark:border-pink-700',
   180→    booking_com: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   181→    direct: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-300 dark:border-emerald-700',
   182→  }
   183→
   184→  const colorClass = colors[source] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300 border-gray-300 dark:border-gray-600'
   185→
   186→  return (
   187→    <Badge variant="outline" className={cn('text-[10px] px-1.5 py-0', colorClass)}>
   188→      {source.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}
   189→    </Badge>
   190→  )
   191→}

   259→
   260→export function FrontDeskDashboard() {
   261→  const { navigateTo, setActiveSubModule } = useNavigationStore()
   262→
   263→  // ─── Dashboard data query ────────────────────────────────────
   264→  const {
   265→    data,
   266→    isLoading,
   267→    error,
   268→    refetch,
   269→    isFetching,
   270→  } = useQuery<DashboardData>({
   271→    queryKey: ['front-desk-dashboard'],
   272→    queryFn: () => apiFetch('/api/front-desk/dashboard'),
   273→    refetchInterval: 30000,
   274→  })
   275→
   276→  const snapshot = data?.snapshot
   277→  const timeline = data?.timeline || []
   278→  const upcomingArrivals = data?.upcomingArrivals || []
   279→
   280→  // ─── Quick action handler ───────────────────────────────────
   281→  const handleQuickAction = (actionId: string) => {
   282→    switch (actionId) {
   283→      case 'new-reservation':
   284→        setActiveSubModule('reservations')
   285→        toast.success('Navigated to Reservations', {
   286→          description: 'Create a new booking from the Reservations tab.',
   287→        })
   288→        break
   289→      case 'walk-in-checkin':
   290→        setActiveSubModule('arrivals')
   291→        toast.success('Navigated to Arrivals', {
   292→          description: 'Use the Walk-in Check-in button on the Arrivals page.',
   293→        })
   294→        break
   295→      case 'room-transfer':
   296→        toast.info('Room Transfer', {
   297→          description: 'Go to Calendar and drag a reservation to another room.',
   298→        })
   299→        break
   300→      case 'wake-up-call':
   301→        toast.info('Wake-Up Call', {
   302→          description: 'Schedule wake-up calls from the In-House guest details.',
   303→        })
   304→        break
   305→      case 'late-checkout':
   306→        toast.info('Late Checkout Request', {
   307→          description: 'Manage late checkout requests from the In-House view.',
   308→        })
   309→        break
   310→      case 'express-checkout':
   311→        setActiveSubModule('departures')
   312→        toast.success('Navigated to Departures', {
   313→          description: 'Process express checkout from the Departures tab.',
   314→        })
   315→        break
   316→      default:
   317→        break
   318→    }
   319→  }
   320→
   321→  // ─── Determine overbooking alert visibility ──────────────────
   322→  const showOverbookingAlert = useMemo(() => {
   323→    return snapshot ? snapshot.overbookingCount > 0 : false
   324→  }, [snapshot])
   325→
   326→  // ─── Snapshot cards data ────────────────────────────────────
   327→  const snapshotCards = useMemo(() => {
   328→    if (!snapshot) return []
   329→    const cards = [
   330→      {
   331→        label: 'Arrivals Today',
   332→        value: snapshot.arrivals,
   333→        icon: LogIn,
   334→        iconBg: 'bg-emerald-100 dark:bg-emerald-950',
   335→        iconColor: 'text-emerald-600 dark:text-emerald-400',
   336→      },
   337→      {
   338→        label: 'Departures Today',
   339→        value: snapshot.departures,
   340→        icon: LogOut,
   341→        iconBg: 'bg-rose-100 dark:bg-rose-950',
   342→        iconColor: 'text-rose-600 dark:text-rose-400',
   343→      },
   344→      {
   345→        label: 'In-House Guests',
   346→        value: snapshot.inHouse,
   347→        icon: BedDouble,
   348→        iconBg: 'bg-sky-100 dark:bg-sky-950',
   349→        iconColor: 'text-sky-600 dark:text-sky-400',
   350→      },
   351→      {
   352→        label: 'Available Rooms',
   353→        value: snapshot.available,
   354→        icon: Hotel,
   355→        iconBg: 'bg-teal-100 dark:bg-teal-950',
   356→        iconColor: 'text-teal-600 dark:text-teal-400',
   357→      },
   358→      {
   359→        label: 'Occupancy',
   360→        value: `${snapshot.occupancyPct}%`,
   361→        icon: Percent,
   362→        iconBg: 'bg-amber-100 dark:bg-amber-950',
   363→        iconColor: 'text-amber-600 dark:text-amber-400',
   364→        subtext: `${snapshot.inHouse} / ${snapshot.totalRooms} rooms`,
   365→      },
   366→    ]
   367→
   368→    // Only show overbooking alert if there are overbookings
   369→    if (snapshot.overbookingCount > 0) {
   370→      cards.push({
   371→        label: 'Overbooking Alert',
   372→        value: snapshot.overbookingCount,
   373→        icon: AlertTriangle,
   374→        iconBg: 'bg-red-100 dark:bg-red-950',
   375→        iconColor: 'text-red-600 dark:text-red-400',
   376→        subtext: `${snapshot.overbookingCount} room${snapshot.overbookingCount > 1 ? 's' : ''} double-booked`,
   377→      })
   378→    }
   379→
   380→    return cards
   381→  }, [snapshot])
   382→
   383→  // ─── Render ──────────────────────────────────────────────────
   384→  return (
   385→    <div className="flex flex-col gap-2">
   386→      {/* ─── Header ─────────────────────────────────────────── */}
   387→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   388→        <div>
   389→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Front Desk Dashboard</h2>
   390→          <p className="text-xs text-muted-foreground">
   391→            Real-time operational snapshot — {formatDate(new Date())}
   392→          </p>
   393→        </div>
   394→        <Button
   395→          variant="outline"
   396→          size="sm"
   397→          className="gap-1.5 text-xs text-muted-foreground shrink-0"
   398→          onClick={() => refetch()}
   399→          disabled={isFetching}
   400→        >
   401→          <RefreshCw className={cn('size-3.5', isFetching && 'animate-spin')} />
   402→          {isFetching ? 'Refreshing...' : 'Refresh'}
   403→        </Button>
   404→      </div>
   405→
   406→      {/* ─── Today's Snapshot Cards ──────────────────────────── */}
   407→      {isLoading ? (
   408→        <SnapshotSkeleton />
   409→      ) : error ? (
   410→        <Card>
   411→          <CardContent className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   412→            <AlertTriangle className="size-8 mb-2 opacity-40" />
   413→            <p className="text-sm">Failed to load dashboard data</p>
   414→            <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   415→              Try Again
   416→            </Button>
   417→          </CardContent>
   418→        </Card>
   419→      ) : (
   420→        <div className={cn(
   421→          'grid gap-2',
   422→          showOverbookingAlert
   423→            ? 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-6'
   424→            : 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-5',
   425→        )}>
   426→          {snapshotCards.map((card) => (
   427→            <Card key={card.label} className={cn(
   428→              card.label === 'Overbooking Alert' && 'border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20',
   429→            )}>
   430→              <CardContent className="p-2.5 flex items-center gap-3">
   431→                <div className={cn('flex size-10 items-center justify-center rounded-lg shrink-0', card.iconBg)}>
   432→                  <card.icon className={cn('size-5', card.iconColor)} />
   433→                </div>
   434→                <div className="min-w-0">
   435→                  <p className="text-lg font-bold leading-tight">{card.value}</p>
   436→                  <p className="text-xs text-muted-foreground truncate">{card.label}</p>
   437→                  {'subtext' in card && card.subtext && (
   438→                    <p className="text-[10px] text-muted-foreground mt-0.5 truncate">{card.subtext}</p>
   439→                  )}
   440→                </div>
   441→              </CardContent>
   442→            </Card>
   443→          ))}
   444→        </div>
   445→      )}
   446→
   447→      {/* ─── Quick Actions Panel ─────────────────────────────── */}
   448→      <div>
   449→        <h3 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   450→          <LayoutDashboard className="size-4 text-muted-foreground" />
   451→          Quick Actions
   452→        </h3>
   453→        <div className="grid gap-3 grid-cols-2 sm:grid-cols-3 lg:grid-cols-6">
   454→          {QUICK_ACTIONS.map((action) => (
   455→            <Card
   456→              key={action.id}
   457→              className="cursor-pointer transition-all hover:shadow-md hover:border-primary/30 active:scale-[0.98]"
   458→              onClick={() => handleQuickAction(action.id)}
   459→            >
   460→              <CardContent className="p-2.5 flex flex-col items-center text-center gap-2">
   461→                <div className={cn('flex size-10 items-center justify-center rounded-lg', action.bg)}>
   462→                  <action.icon className={cn('size-5', action.color)} />
   463→                </div>
   464→                <div>
   465→                  <p className="text-xs font-semibold leading-tight">{action.title}</p>
   466→                  <p className="text-[10px] text-muted-foreground mt-0.5 leading-tight">{action.description}</p>
   467→                </div>
   468→              </CardContent>
   469→            </Card>
   470→          ))}
   471→        </div>
   472→      </div>
   473→
   474→      <Separator />
   475→
   476→      {/* ─── Bottom Grid: Timeline + Upcoming Arrivals ──────── */}
   477→      <div className="grid gap-4 lg:grid-cols-2">
   478→        {/* ─── Today's Activity Timeline ──────────────────── */}
   479→        <div>
   480→          {isLoading ? (
   481→            <TimelineSkeleton />
   482→          ) : (
   483→            <Card className="h-full">
   484→              <CardHeader className="pb-3">
   485→                <CardTitle className="text-base flex items-center gap-2">
   486→                  <Clock className="size-4 text-muted-foreground" />
   487→                  Today&apos;s Activity
   488→                </CardTitle>
   489→                <CardDescription>Recent front desk operations for today</CardDescription>
   490→              </CardHeader>
   491→              <CardContent>
   492→                {timeline.length === 0 ? (
   493→                  <div className="flex flex-col items-center justify-center py-10 text-muted-foreground">
   494→                    <Clock className="size-8 mb-2 opacity-40" />
   495→                    <p className="text-sm">No activity recorded yet today</p>
   496→                  </div>
   497→                ) : (
   498→                  <div className="max-h-96 overflow-y-auto pr-1 space-y-0">
   499→                    {timeline.map((entry, idx) => (
   500→                      <div key={entry.id} className="flex gap-3 pb-4">
   501→                        {/* Timeline line + icon */}
   502→                        <div className="flex flex-col items-center shrink-0">
   503→                          <TimelineIcon type={entry.type} />
   504→                          {idx < timeline.length - 1 && (
   505→                            <div className="w-px flex-1 bg-border mt-1" />
   506→                          )}
   507→                        </div>
   508→
   509→                        {/* Content */}
   510→                        <div className="flex-1 min-w-0 pb-1">
   511→                          <div className="flex items-center gap-2 flex-wrap">
   512→                            <span className="text-sm font-medium truncate">
   513→                              {entry.guestName}
   514→                            </span>
   515→                            <Badge
   516→                              variant="outline"
   517→                              className={cn(
   518→                                'text-[10px] px-1.5 py-0 shrink-0',
   519→                                entry.type === 'check_in'
   520→                                  ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-300 dark:border-emerald-700'

   526→                              )}
   527→                            >

   529→                            </Badge>
   530→                          </div>
   531→                          <div className="flex items-center gap-2 mt-1 text-xs text-muted-foreground flex-wrap">
   532→                            <span>{formatTime(entry.time)}</span>
   533→                            {entry.roomNumber && (
