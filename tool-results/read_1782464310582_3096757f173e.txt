   261→   261→  const { navigateTo, setActiveSubModule } = useNavigationStore()
   262→   262→
   263→   263→  // ─── Dashboard data query ────────────────────────────────────
   264→   264→  const {
   265→   265→    data,

   279→   279→
   280→   280→  // ─── Quick action handler ───────────────────────────────────
   281→   281→  const handleQuickAction = (actionId: string) => {
   282→   282→    switch (actionId) {
   283→   283→      case 'new-reservation':
   284→   284→        setActiveSubModule('reservations')
   285→   285→        toast.success('Navigated to Reservations', {
   286→   286→          description: 'Create a new booking from the Reservations tab.',
   287→   287→        })

   289→   289→      case 'walk-in-checkin':
   290→   290→        setActiveSubModule('arrivals')
   291→   291→        toast.success('Navigated to Arrivals', {
   292→   292→          description: 'Use the Walk-in Check-in button on the Arrivals page.',
   293→   293→        })
   294→   294→        break
   295→   295→      case 'room-transfer':
   296→   296→        toast.info('Room Transfer', {
   297→   297→          description: 'Go to Calendar and drag a reservation to another room.',
   298→   298→        })
   299→   299→        break
   300→   300→      case 'wake-up-call':
   301→   301→        toast.info('Wake-Up Call', {
   302→   302→          description: 'Schedule wake-up calls from the In-House guest details.',
   303→   303→        })
   304→   304→        break
   305→   305→      case 'late-checkout':
   306→   306→        toast.info('Late Checkout Request', {
   307→   307→          description: 'Manage late checkout requests from the In-House view.',
   308→   308→        })

   313→   313→          description: 'Process express checkout from the Departures tab.',

   323→   323→    return snapshot ? snapshot.overbookingCount > 0 : false
   324→   324→  }, [snapshot])
   325→   325→
   326→   326→  // ─── Snapshot cards data ────────────────────────────────────

   328→   328→    if (!snapshot) return []
   329→   329→    const cards = [
   330→   330→      {

   333→   333→        icon: LogIn,

   335→   335→        iconColor: 'text-emerald-600 dark:text-emerald-400',
   336→   336→      },
   337→   337→      {
   338→   338→        label: 'Departures Today',
   339→   339→        value: snapshot.departures,
   340→   340→        icon: LogOut,
   341→   341→        iconBg: 'bg-rose-100 dark:bg-rose-950',
   342→   342→        iconColor: 'text-rose-600 dark:text-rose-400',
   343→   343→      },
   344→   344→      {
   345→   345→        label: 'In-House Guests',
   346→   346→        value: snapshot.inHouse,
   347→   347→        icon: BedDouble,
   348→   348→        iconBg: 'bg-sky-100 dark:bg-sky-950',
   349→   349→        iconColor: 'text-sky-600 dark:text-sky-400',
   350→   350→      },
   351→   351→      {
   352→   352→        label: 'Available Rooms',
   353→   353→        value: snapshot.available,
   354→   354→        icon: Hotel,
   355→   355→        iconBg: 'bg-teal-100 dark:bg-teal-950',
   356→   356→        iconColor: 'text-teal-600 dark:text-teal-400',
   357→   357→      },
   358→   358→      {
   359→   359→        label: 'Occupancy',

   362→   362→        iconBg: 'bg-amber-100 dark:bg-amber-950',

   368→   368→    // Only show overbooking alert if there are overbookings

   371→   371→        label: 'Overbooking Alert',

   380→   380→    return cards

   383→   383→  // ─── Render ──────────────────────────────────────────────────

   386→   386→      {/* ─── Header ─────────────────────────────────────────── */}

   399→   399→          disabled={isFetching}

   426→   426→          {snapshotCards.map((card) => (
   427→   427→            <Card key={card.label} className={cn(
   428→   428→              card.label === 'Overbooking Alert' && 'border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20',
   429→   429→            )}>
   430→   430→              <CardContent className="p-2.5 flex items-center gap-3">
   431→   431→                <div className={cn('flex size-10 items-center justify-center rounded-lg shrink-0', card.iconBg)}>
   432→   432→                  <card.icon className={cn('size-5', card.iconColor)} />

   434→   434→                <div className="min-w-0">
   435→   435→                  <p className="text-lg font-bold leading-tight">{card.value}</p>

   437→   437→                  {'subtext' in card && card.subtext && (

   441→   441→              </CardContent>
