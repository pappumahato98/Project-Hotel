   254→// ─── Date Helpers ────────────────────────────────────────────────────────
   255→
   256→function addDays(date: Date, days: number): Date {
   257→  const d = new Date(date)
   258→  d.setDate(d.getDate() + days)
   259→  return d
   260→}
   261→
   262→function dateToKey(date: Date): string {
   263→  const y = date.getFullYear()
   264→  const m = String(date.getMonth() + 1).padStart(2, '0')
   265→  const d = String(date.getDate()).padStart(2, '0')
   266→  return `${y}-${m}-${d}`
   267→}
   268→
   269→function toDateOnly(dateStr: string): Date {
   270→  const d = new Date(dateStr)
   271→  d.setHours(0, 0, 0, 0)
   272→  return d
   273→}
   274→
   275→function isSameDay(a: Date, b: Date): boolean {
   276→  return (
   277→    a.getFullYear() === b.getFullYear() &&
   278→    a.getMonth() === b.getMonth() &&
   279→    a.getDate() === b.getDate()
   280→  )
   281→}
   282→
   283→function isWeekend(date: Date): boolean {
   284→  const day = date.getDay()
   285→  return day === 0 || day === 6
   286→}
   287→
   288→function formatDateShort(dateStr: string): string {
   289→  const d = new Date(dateStr)
   290→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   291→}

   293→// Determine the display category for a reservation block
   294→function getReservationCategory(res: CalendarReservation, todayStr: string): string {
   295→  const checkInStr = dateToKey(toDateOnly(res.checkIn))
   296→  const checkOutStr = dateToKey(toDateOnly(res.checkOut))
   297→
   298→  if (res.status === 'tentative') return 'tentative'
   299→  if (checkInStr === todayStr) return 'arrival_today'
   300→  if (checkOutStr === todayStr && res.status === 'checked_in') return 'departure_today'
   301→  if (res.status === 'checked_in') return 'checked_in'
   302→  return 'confirmed'
   303→}

   736→
   737→      // Snapshot the current reservations data for rollback
   738→      const previousReservations = queryClient.getQueryData(['reservations', startDateStr, endDateStr])
   739→
   740→      // Optimistically update the reservations cache
   741→      queryClient.setQueryData(['reservations', startDateStr, endDateStr], (old: Record<string, unknown> | undefined) => {

   745→          ...old,

   767→          }),
   768→        }
   769→      })

   771→      return { previousReservations }
   772→    },
   773→    onSuccess: () => {
   774→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   775→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   776→      queryClient.invalidateQueries({ queryKey: ['calendar'] })
