   177→// ─── Debounce hook ────────────────────────────────────────────────────
   178→function useDebounce<T>(value: T, delay: number): T {
   179→  const [debounced, setDebounced] = useState(value)
   180→  useEffect(() => {
   181→    const timer = setTimeout(() => setDebounced(value), delay)
   182→    return () => clearTimeout(timer)
   183→  }, [value, delay])
   184→  return debounced
   185→}

   187→// ─── Date range presets ──────────────────────────────────────────────
   188→function getDatePreset(preset: string): { from: Date; to: Date } {
   189→  const today = new Date()
   190→  today.setHours(0, 0, 0, 0)
   191→  switch (preset) {
   192→    case 'today':
   193→      return { from: today, to: today }
   194→    case 'this-week': {
   195→      const from = new Date(today)
   196→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   197→      const to = new Date(from)
   198→      to.setDate(from.getDate() + 6) // Sunday
   199→      return { from, to }
   200→    }
   201→    case 'this-month': {
   202→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   203→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   204→      return { from, to }
   205→    }
   206→    case 'next-7': {
   207→      const to = new Date(today)
   208→      to.setDate(today.getDate() + 6)
   209→      return { from: today, to }
   210→    }
   211→    case 'next-14': {
   212→      const to = new Date(today)
   213→      to.setDate(today.getDate() + 13)
   214→      return { from: today, to }
   215→    }
   216→    case 'next-30': {
   217→      const to = new Date(today)
   218→      to.setDate(today.getDate() + 29)
   219→      return { from: today, to }
   220→    }
   221→    default:
   222→      return { from: today, to: today }
   223→  }
   224→}

   226→function toDateString(d: Date): string {
   227→  const y = d.getFullYear()
   228→  const m = String(d.getMonth() + 1).padStart(2, '0')
   229→  const day = String(d.getDate()).padStart(2, '0')
   230→  return `${y}-${m}-${day}`
   231→}
   232→
   233→function formatShortDate(dateStr: string): string {
   234→  if (!dateStr) return ''
   235→  const d = new Date(dateStr + 'T00:00:00')
   236→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   237→}
