   107→function NumberStepper({ label, value, min, max, onChange }: {
   108→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   109→}) {
   110→  return (
   111→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   112→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   113→      <div className="flex items-center gap-0 ml-auto">
   114→        <button
   115→          type="button"
   116→          onClick={() => onChange(Math.max(min, value - 1))}
   117→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   118→        >
   119→          <Minus className="w-3 h-3" />
   120→        </button>
   121→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   122→        <button
   123→          type="button"
   124→          onClick={() => onChange(Math.min(max, value + 1))}
   125→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   126→        >
   127→          <Plus className="w-3 h-3" />
   128→        </button>
   129→      </div>
   130→    </div>
   131→  )
   132→}

   134→function sourceLabel(source: string | null) {
   135→  switch (source) {
   136→    case 'walk_in': return 'Walk-in'
   137→    case 'online': return 'Online'
   138→    case 'phone': return 'Phone'
   139→    case 'travel_agent': return 'Travel Agent'
   140→    case 'corporate': return 'Corporate'
   141→    default: return source || 'Direct'
   142→  }
   143→}

   146→
   147→export function CheckInLookup({ onBack, prefillReservationId }: CheckInLookupProps) {
   148→  const { setActiveSubModule } = useNavigationStore()
   149→  const { setCheckInSession, setPrefillReservationId } = useFrontDeskContextStore()
   150→
   151→  const today = getTodayString()
   152→
   153→  // State ───────────────────────────────────────────────────────────-------------------------------------------------
   154→  const [expressMode, setExpressMode] = useState(false)
   155→  const [isDirectWalkIn, setIsDirectWalkIn] = useState(false)
   156→  const [searchQuery, setSearchQuery] = useState('')
   157→  const [debouncedQuery, setDebouncedQuery] = useState('')
   158→  const [searchedReservationId, setSearchedReservationId] = useState<string | null>(null)
   159→
   160→  // Walk-in fields
   161→  const [walkInDate, setWalkInDate] = useState(today)
   162→  const [walkInCheckOut, setWalkInCheckOut] = useState(() => {
   163→    const d = new Date()
   164→    d.setDate(d.getDate() + 1)
   165→    return toDateOnly(d)
   166→  })
   167→  const [walkInAdults, setWalkInAdults] = useState(2)
   168→  const [walkInChildren, setWalkInChildren] = useState(0)
   169→  const [guestFirstName, setGuestFirstName] = useState('')
   170→  const [guestLastName, setGuestLastName] = useState('')
   171→  const [guestEmail, setGuestEmail] = useState('')
   172→  const [guestPhone, setGuestPhone] = useState('')
   173→  const [guestNationality, setGuestNationality] = useState('')
   174→
   175→  // Clear prefill on unmount to avoid stale prefills
   176→  useEffect(() => {
   177→    return () => { setPrefillReservationId(null) }
   178→  }, [setPrefillReservationId])
   179→
   180→  // Debounced search ────────────────────────────────────────────────---------------------------------------
   181→  useEffect(() => {
   182→    const timer = setTimeout(() => {
   183→      setDebouncedQuery(searchQuery)
   184→    }, 300)
   185→    return () => clearTimeout(timer)
   186→  }, [searchQuery])
   187→
   188→  // Data Queries ────────────────────────────────────────────────────------------------------------------------
   189→
   190→  const activeReservationId = prefillReservationId || searchedReservationId
   191→
   192→  const {
   193→    data: reservationData,
   194→    isLoading: reservationLoading,
   195→    error: reservationError,
   196→  } = useQuery({
   197→    queryKey: ['reservation', activeReservationId],
   198→    queryFn: async () => {
   199→      if (!activeReservationId) return null
   200→      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   201→      return data.reservation
   202→    },
   203→    enabled: !!activeReservationId,
   204→  })
   205→
   206→  // Guest stay history
   207→  const { data: guestStayHistory } = useQuery({
   208→    queryKey: ['guest-stays', reservationData?.guest?.id],
   209→    queryFn: async () => {
   210→      if (!reservationData?.guest?.id) return null
   211→      try {
   212→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   213→      } catch {
   214→        return null
   215→      }
   216→    },
   217→    enabled: !!reservationData?.guest?.id && !!searchedReservationId,
   218→  })
   219→
   220→  // Today's expected arrivals
   221→  const {
   222→    data: arrivalsData,
   223→    isLoading: arrivalsLoading,
   224→  } = useQuery({
   225→    queryKey: ['arrivals', today],
   226→    queryFn: async () => {
   227→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   228→      return data.reservations || []
   229→    },
   230→    enabled: !isDirectWalkIn && !prefillReservationId,
   231→  })
   232→
   233→  // Available rooms for walk-in mode (for live summary)
   234→  const { data: availableRoomsData } = useQuery({
   235→    queryKey: ['rooms-available-count'],
   236→    queryFn: async () => {
   237→      return apiFetch('/api/rooms?status=vacant_clean,inspected') as Promise<{
   238→        rooms: Array<{ id: string; type: { id: string; name: string; code: string } }>
   239→        roomTypes: Array<{ id: string; name: string; code: string; baseOccupancy: number; maxOccupancy: number }>
   240→        summary?: { available: number; totalRooms: number; occupied: number }
   241→      }>
   242→    },
   243→    enabled: isDirectWalkIn,
   244→  })
   245→
   246→  // Compute available rooms by type for walk-in
   247→  const walkInNights = nightsBetween(walkInDate, walkInCheckOut)
   248→  const totalPax = walkInAdults + walkInChildren
   249→  const canContinue = !!(guestFirstName.trim() && guestLastName.trim())
   250→  const availableRoomTypes = useMemo(() => {
   251→    if (!availableRoomsData) return []
   252→    const counts: Record<string, number> = {}
   253→    for (const r of availableRoomsData.rooms) {
   254→      const rtId = r.type.id
   255→      counts[rtId] = (counts[rtId] || 0) + 1
   256→    }
   257→    return availableRoomsData.roomTypes.map(rt => ({
   258→      ...rt,
   259→      availableCount: counts[rt.id] || 0,
   260→    })).filter(rt => rt.availableCount > 0 && rt.maxOccupancy >= totalPax)
   261→  }, [availableRoomsData, totalPax])
   262→
   263→  // Reservation search (debounced)
   264→  const {
   265→    data: searchResults,
   266→    isLoading: searchLoading,
   267→  } = useQuery({
   268→    queryKey: ['reservation-search', debouncedQuery],
   269→    queryFn: async () => {
   270→      if (!debouncedQuery || debouncedQuery.length < 2) return null
   271→      const data = await apiFetch<{ reservations?: ReservationData[] }>(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   272→      return (data?.reservations || []) as ReservationData[]
   273→    },
   274→    enabled: debouncedQuery.length >= 2,
   275→  })
   276→
   277→  // Handlers ────────────────────────────────────────────────────────-----------------------------------------------
   278→
   279→  const navigateToProcess = useCallback((session: CheckInSession) => {
   280→    setCheckInSession(session)
   281→    setActiveSubModule('check-in-process')
   282→  }, [setCheckInSession, setActiveSubModule])
   283→
   284→  const handleUseReservation = useCallback(() => {
   285→    if (!reservationData) return
   286→    navigateToProcess({
   287→      reservationId: reservationData.id,
   288→      reservationData: reservationData as unknown as Record<string, unknown>,
   289→      isDirectWalkIn: false,
   290→      walkInGuest: null,
   291→      walkInDates: null,
   292→      startAtStep: 1,
   293→    })
   294→    toast.success('Reservation confirmed! Proceeding to check-in.')
   295→  }, [reservationData, navigateToProcess])
   296→
   297→  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   298→    navigateToProcess({
   299→      reservationId: res.id,
   300→      reservationData: res as unknown as Record<string, unknown>,
   301→      isDirectWalkIn: false,
   302→      walkInGuest: null,
   303→      walkInDates: null,
   304→      startAtStep: 2,
   305→    })
   306→    toast.success('Express check-in - skipping to Room & Rate')
   307→  }, [navigateToProcess])
   308→
   309→  const handleWalkInContinue = useCallback(() => {
   310→    if (!guestFirstName || !guestLastName) {
   311→      toast.error('Please enter guest first and last name')
   312→      return
   313→    }
   314→    navigateToProcess({
   315→      reservationId: null,
   316→      reservationData: null,
   317→      isDirectWalkIn: true,
   318→      walkInGuest: {
   319→        firstName: guestFirstName,
   320→        lastName: guestLastName,
   321→        email: guestEmail,
   322→        phone: guestPhone,
   323→        nationality: guestNationality,
   324→      },
   325→      walkInDates: {
   326→        checkIn: walkInDate,
   327→        checkOut: walkInCheckOut,
   328→        adults: walkInAdults,
   329→        children: walkInChildren,
   330→      },
   331→      startAtStep: 1,
   332→    })
   333→  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   334→
   335→  // Reservation Card Renderer ───────────────────────────────────────-----------------------------
   336→
   337→  const renderReservationCard = useCallback((res: ReservationData) => {
   338→    const isSelected = searchedReservationId === res.id
   339→    return (
   340→      <Card
   341→        key={res.id}
   342→        className={cn(
   343→          'p-2.5 cursor-pointer transition-all border',
   344→          isSelected
   345→            ? 'border-teal-400 ring-2 ring-teal-400/20 bg-teal-50/50 dark:bg-teal-950/20'
   346→            : 'hover:bg-muted/50'
   347→        )}
   348→        onClick={() => setSearchedReservationId(res.id)}
   349→      >
   350→        <div className="flex items-start justify-between gap-2">
   351→          <div className="space-y-1 min-w-0 flex-1">
   352→            <div className="flex items-center gap-2 flex-wrap">
   353→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   354→                {res.confirmationNo}
   355→              </Badge>
   356→              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   357→                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   358→                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   359→                  VIP
   360→                </Badge>
   361→              )}
   362→              {res.source && (
   363→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   364→                  {sourceLabel(res.source)}
   365→                </Badge>
   366→              )}
   367→            </div>
   368→            <p className="text-sm font-medium truncate">
   369→              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   370→            </p>
   371→            <div className="flex items-center gap-2 text-xs text-muted-foreground flex-wrap">
   372→              {res.room && <span>Room {res.room.number}</span>}
   373→              {res.room && <RoomTypeBedBadge typeName={res.room.type.name} bedConfig={res.room.type.bedConfig} typeCode={res.room.type.code} pax={res.adults + res.children} inline />}
   374→              <span>·</span>
   375→              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   376→              <span>·</span>
   377→              <span className="font-medium">{formatCurrency(res.roomRate)}/night</span>
   378→            </div>
   379→          </div>
   380→          <div className="flex flex-col items-end gap-1.5 shrink-0">
   381→            {isSelected && (
   382→              <Check className="w-4 h-4 text-teal-600" />
   383→            )}
   384→            {expressMode && (
   385→              <Button
   386→                size="sm"
   387→                className="h-7 text-[11px] px-2 bg-emerald-500 hover:bg-emerald-600 text-white"
   388→                onClick={(e) => {
   389→                  e.stopPropagation()
   390→                  setSearchedReservationId(res.id)
   391→                  handleExpressCheckIn(res)
   392→                }}
   393→              >
   394→                <Zap className="w-3 h-3 mr-1" />
   395→                Express
   396→              </Button>
   397→            )}
   398→          </div>
   399→        </div>
   400→      </Card>
   401→    )
   402→  }, [searchedReservationId, expressMode, handleExpressCheckIn])
   403→
   404→  // Render ──────────────────────────────────────────────────────────-----------------------------------------------
   405→  return (
   406→    <div className="flex flex-col h-full bg-background">
   407→      <div className="flex-1 overflow-y-auto">
   408→        <div className={cn('space-y-2 p-3 sm:p-4', isDirectWalkIn ? '' : 'max-w-2xl mx-auto')}>
   409→
   410→          {/* Prefilled reservation mode */}
   411→          {prefillReservationId ? (
   412→            <>
   413→              {/* Header for prefill mode */}
   414→              <div className="shrink-0 -mx-3 sm:-mx-4 -mt-3 sm:-mt-4 mb-2 px-3 sm:px-4 py-2.5 border-b bg-card">
   415→                <div className="flex items-center gap-3">
   416→                  <button
   417→                    type="button"
   418→                    onClick={onBack}
   419→                    className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors"
   420→                  >
   421→                    <ArrowLeft className="w-4 h-4" />
   422→                    <span className="hidden sm:inline">Back</span>
   423→                  </button>
   424→                  <Separator orientation="vertical" className="h-5" />
   425→                  <h1 className="text-base sm:text-lg font-semibold text-foreground">Guest Check-In</h1>
   426→                </div>
   427→              </div>
   428→              {reservationLoading && (
   429→                <div className="space-y-2">
   430→                  <Skeleton className="h-4 w-3/4" />
   431→                  <Skeleton className="h-40 w-full" />
   432→                </div>
   433→              )}
   434→              {reservationError && (
   435→                <Card className="border-red-200 dark:border-red-800 p-2.5">
   436→                  <div className="flex items-start gap-2 text-red-600 dark:text-red-400">
   437→                    <svg className="h-5 w-5 mt-0.5 shrink-0" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>
   438→                    <div>
   439→                      <p className="font-medium text-sm">Reservation Not Found</p>
   440→                      <p className="text-sm text-muted-foreground mt-1">
   441→                        The reservation could not be found. Please search or use direct walk-in.
   442→                      </p>
   443→                    </div>
   444→                  </div>
   445→                  <Button variant="outline" size="sm" className="mt-2" onClick={() => { setPrefillReservationId(null) }}>
   446→                    Use Direct Walk-in Instead
   447→                  </Button>
   448→                </Card>
   449→              )}
   450→              {reservationData && (
   451→                <>
   452→                  <div className="flex items-center gap-2">
   453→                    <Badge variant="outline" className="border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">
   454→                      <Check className="w-3 h-3 mr-1" />
   455→                      Reservation Found
   456→                    </Badge>
   457→                  </div>
   458→                  <Card className="p-2.5">
   459→                    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   460→                      <InfoItem label="Confirmation #" value={reservationData.confirmationNo} />
   461→                      <div className="space-y-0.5">
   462→                        <span className="text-xs text-muted-foreground">Status</span>
   463→                        <p className="text-sm font-medium flex items-center gap-1"><StatusBadge status={reservationData.status} /></p>
   464→                      </div>
   465→                      <InfoItem label="Guest" value={reservationData.guest ? `${reservationData.guest.firstName} ${reservationData.guest.lastName}` : '-'} />
   466→                      <InfoItem label="Room" value={reservationData.room ? <span className="flex items-center gap-1.5">{reservationData.room.number} <RoomTypeBedBadge typeName={reservationData.room.type.name} bedConfig={reservationData.room.type.bedConfig} typeCode={reservationData.room.type.code} pax={reservationData.adults + reservationData.children} inline /></span> : '-'} />
   467→                      <InfoItem label="Stay" value={`${formatDate(reservationData.checkIn)} → ${formatDate(reservationData.checkOut)}`} />
   468→                      <InfoItem label="Rate" value={`${formatCurrency(reservationData.roomRate)}/night`} />
   469→                    </div>
   470→                    {/* Guest stay history */}
   471→                    {guestStayHistory && guestStayHistory.length > 0 && (
   472→                      <div className="mt-2 pt-2 border-t">
   473→                        <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground mb-1">

   479→                              {stay.roomNumber} · {formatDateShort(stay.checkIn)}
   480→                            </Badge>
   481→                          ))}
   482→                        </div>
   483→                      </div>
   484→                    )}
   485→                  </Card>
   486→                  <div className="flex gap-2">
   487→                    <Button onClick={handleUseReservation} className="flex-1">
   488→                      <Check className="w-4 h-4 mr-2" />
   489→                      Go Check-in
   490→                    </Button>
   491→                    {expressMode && (
   492→                      <Button variant="outline" className="flex-1" onClick={() => handleExpressCheckIn(reservationData)}>
   493→                        <Zap className="w-4 h-4 mr-2 text-amber-500" />
   494→                        Express Check-In
   495→                      </Button>
   496→                    )}
   497→                  </div>
   498→                </>
   499→              )}
   500→            </>
   501→          ) : (
