    40→// ─── Types ──────────────────────────────────────────────────────────────
    41→
    42→interface DepartureGuest {
    43→  id: string
    44→  firstName: string
    45→  lastName: string
    46→  vipLevel: string
    47→}
    48→
    49→interface DepartureRoom {
    50→  id: string
    51→  number: string
    52→  floor: number
    53→  type: { name: string; code: string }
    54→}
    55→
    56→interface FolioLineItem {
    57→  id: string
    58→  description: string
    59→  amount: number
    60→  type: 'charge' | 'payment'
    61→  date: string
    62→}
    63→
    64→interface DepartureFolio {
    65→  id: string
    66→  balance: number
    67→  status: string
    68→  items?: FolioLineItem[]
    69→}
    70→
    71→interface Departure {
    72→  id: string
    73→  confirmationNo: string
    74→  status: string
    75→  checkIn: string
    76→  checkOut: string
    77→  roomRate: number
    78→  totalAmount: number
    79→  paidAmount: number
    80→  creditLimit: number
    81→  guest: DepartureGuest
    82→  room: DepartureRoom
    83→  folios: DepartureFolio[]
    84→}
    85→
    86→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    87→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    88→
    89→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    90→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    91→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    92→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    93→}
    94→
    95→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    96→  cash: 'Cash',
    97→  card: 'Credit/Debit Card',
    98→  bank_transfer: 'Bank Transfer',
    99→}

   101→// ─── Component ──────────────────────────────────────────────────────────
   102→
   103→export function DeparturesView() {
   104→  const queryClient = useQueryClient()
   105→  const { settings } = useSettingsStore()
   106→  const { preferences } = usePreferencesStore()
   107→  const { navigateTo } = useNavigationStore()
   108→  const today = getTodayString()
   109→
   110→  const [selectedRows, setSelectedRows] = useState<Set<string>>(new Set())
   111→
   112→  // Dialog states
   113→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   114→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   115→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   116→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   117→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   118→
   119→  // Selected departure
   120→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   121→
   122→  // Late checkout
   123→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   124→
   125→  // Payment
   126→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   127→  const [paymentAmount, setPaymentAmount] = useState('')
   128→  const [paymentReference, setPaymentReference] = useState('')
   129→
   130→  // Receipt data
   131→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   132→
   133→  // ── Fetch today's departures (both checked_in AND checked_out with checkOut = today) ──
   134→  const { data, isLoading } = useQuery({
   135→    queryKey: ['departures', today],
   136→    queryFn: async () => {
   137→      // Fetch both statuses so we show completed check-outs too
   138→      const params = new URLSearchParams({ status: 'checked_in,checked_out', checkOutDate: today })
   139→      return apiFetch(`/api/reservations?${params.toString()}`)
   140→    },
   141→    refetchInterval: 30000,
   142→  })
   143→
   144→  const departures: Departure[] = data?.reservations || []
   145→
   146→  // ── Fallback: if today has zero departures, also fetch upcoming (next 3 days) ──
   147→  const tomorrow = (() => {
   148→    const d = new Date(); d.setDate(d.getDate() + 1)
   149→    const y = d.getFullYear(); const m = String(d.getMonth()+1).padStart(2,'0'); const day = String(d.getDate()).padStart(2,'0')
   150→    return `${y}-${m}-${day}`
   151→  })()
   152→  const threeDaysOut = (() => {
   153→    const d = new Date(); d.setDate(d.getDate() + 4)
   154→    const y = d.getFullYear(); const m = String(d.getMonth()+1).padStart(2,'0'); const day = String(d.getDate()).padStart(2,'0')
   155→    return `${y}-${m}-${day}`
   156→  })()
   157→
   158→  const { data: upcomingData } = useQuery({
   159→    queryKey: ['departures-upcoming', today],
   160→    queryFn: async () => {
   161→      const params = new URLSearchParams({
   162→        status: 'checked_in,confirmed',
   163→        dateFrom: tomorrow,
   164→        dateTo: threeDaysOut,
   165→      })
   166→      return apiFetch(`/api/reservations?${params.toString()}`)
   167→    },
   168→    refetchInterval: 60000,
   169→    enabled: departures.length === 0,
   170→  })
   171→
   172→  const upcomingDepartures: Departure[] = upcomingData?.reservations || []
   173→
   174→  // Use upcoming as fallback when today is empty
   175→  const displayDepartures = departures.length > 0 ? departures : upcomingDepartures
   176→  const isShowingUpcoming = departures.length === 0 && upcomingDepartures.length > 0
   177→
   178→  // Stats (always based on today's data)
   179→  const totalDepartures = departures.length
   180→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   181→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   182→  const outstandingBalance = departures.reduce((sum, d) => {
   183→    return sum + (d.folios[0]?.balance || 0)
   184→  }, 0)
   185→
   186→  // Zero-balance pending guests (for express & batch checkout)
   187→  const zeroBalancePending = departures.filter(
   188→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   189→  )
   190→
   191→  // ─── Mutations ─────────────────────────────────────────────────────
   192→
   193→  // Checkout mutation
   194→  const checkoutMutation = useMutation({
   195→    mutationFn: async (reservationId: string) => {
   196→      return apiFetch(`/api/reservations/${reservationId}`, {
   197→        method: 'PATCH',
   198→        headers: { 'Content-Type': 'application/json' },
   199→        body: JSON.stringify({ status: 'checked_out' }),
   200→      })
   201→    },
   202→    onSuccess: () => {
   203→      invalidate.afterCheckout(queryClient)
   204→    },
   205→  })
   206→
   207→  // Payment mutation
   208→  const paymentMutation = useMutation({
   209→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   210→      return apiFetch(`/api/folio/${folioId}`, {
   211→        method: 'POST',
   212→        headers: { 'Content-Type': 'application/json' },
   213→        body: JSON.stringify({ type: 'payment', paymentMethod: method, amount, reference }),
   214→      })
   215→    },
   216→    onSuccess: () => {
   217→      invalidate.afterFolioChange(queryClient)
   218→      setPaymentDialogOpen(false)
   219→      setPaymentAmount('')
   220→      setPaymentReference('')
   221→      toast.success('Payment recorded successfully')
   222→    },
   223→    onError: () => {
   224→      toast.error('Failed to record payment')
   225→    },
   226→  })
   227→
   228→  // Late checkout mutation
   229→  const lateCheckoutMutation = useMutation({
   230→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   231→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   232→        method: 'PATCH',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ checkOut: newCheckOut }),
   235→      })
   236→      // Add surcharge to folio
   237→      if (surcharge > 0) {
   238→        await apiFetch(`/api/folio`, {
   239→          method: 'POST',
   240→          headers: { 'Content-Type': 'application/json' },
   241→          body: JSON.stringify({
   242→            reservationId,
   243→            type: 'charge',
   244→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   245→            amount: surcharge,
   246→          }),
   247→        })
   248→      }
   249→      return result
   250→    },
   251→    onSuccess: () => {
   252→      invalidate.afterReservationChange(queryClient)
   253→      invalidate.afterFolioChange(queryClient)
   254→      setLateCheckoutDialogOpen(false)
   255→      toast.success('Late checkout request processed successfully')
   256→    },
   257→    onError: () => {
   258→      toast.error('Failed to process late checkout request')
   259→    },
   260→  })
   261→
   262→  // Batch checkout mutation
   263→  const batchCheckoutMutation = useMutation({
   264→    mutationFn: async (ids: string[]) => {
   265→      const results = []
   266→      for (let i = 0; i < ids.length; i++) {
   267→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   268→          method: 'PATCH',
   269→          headers: { 'Content-Type': 'application/json' },
   270→          body: JSON.stringify({ status: 'checked_out' }),
   271→        })
   272→        results.push(result)
   273→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   274→      }
   275→      return results
   276→    },
   277→    onSuccess: (_data, ids) => {
   278→      invalidate.afterCheckout(queryClient)
   279→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   280→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   281→    },
   282→    onError: () => {
   283→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   284→    },
   285→  })
   286→
   287→  // ─── Handlers ─────────────────────────────────────────────────────
   288→
   289→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   290→
   291→  const handleExpressCheckout = (dep: Departure) => {
   292→    checkoutMutation.mutate(dep.id, {
   293→      onSuccess: () => {
   294→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   295→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   296→      },
   297→    })
   298→  }
   299→
   300→  const handleReviewFolio = (dep: Departure) => {
   301→    setSelectedDeparture(dep)
   302→    setFolioDialogOpen(true)
   303→  }
   304→
   305→  const handleNavigateToFolio = (dep: Departure) => {
   306→    useFolioContextStore.getState().setFolioContext({
   307→      reservationId: dep.id,
   308→      guestId: dep.guest?.id || '',
   309→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   310→      roomNumber: dep.room?.number || '',
   311→      confirmationNo: dep.confirmationNo,
   312→    })
   313→    navigateTo('front-desk', 'folio')
   314→  }
   315→
   316→  const handleViewLedger = (dep: Departure) => {
   317→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   318→      guestId: dep.guest?.id || '',
   319→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   320→    })
   321→    navigateTo('front-desk', 'guest-ledger')
   322→  }
   323→
   324→  const handleFolioSettleCheckout = () => {
   325→    setFolioDialogOpen(false)
   326→    setCheckoutDialogOpen(true)
   327→  }
   328→
   329→  const handleQuickCheckout = (dep: Departure) => {
   330→    setSelectedDeparture(dep)
   331→    setCheckoutDialogOpen(true)
   332→  }
   333→
   334→  const handleRequestLateCheckout = (dep: Departure) => {
   335→    setSelectedDeparture(dep)
   336→    setLateCheckoutOption('14:00')
   337→    setLateCheckoutDialogOpen(true)
   338→  }
   339→
   340→  const handleConfirmLateCheckout = () => {
   341→    if (!selectedDeparture) return
   342→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   343→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   344→
   345→    // Build new checkout datetime with the selected time
   346→    const currentCheckout = new Date(selectedDeparture.checkOut)
   347→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   348→    currentCheckout.setHours(hours, minutes, 0, 0)
   349→
   350→    lateCheckoutMutation.mutate({
   351→      reservationId: selectedDeparture.id,
   352→      newCheckOut: currentCheckout.toISOString(),
   353→      surcharge,
   354→    })
   355→  }
   356→
   357→  const handleOpenPaymentDialog = () => {
   358→    if (!selectedDeparture) return
   359→    const balance = getBalance(selectedDeparture)
   360→    setPaymentAmount(String(balance))
   361→    setPaymentMethod('cash')
   362→    setPaymentReference('')
   363→    setPaymentDialogOpen(true)
   364→  }
   365→
   366→  const handleRecordPayment = () => {
   367→    if (!selectedDeparture) return
   368→    const folioId = selectedDeparture.folios[0]?.id
   369→    if (!folioId) {
   370→      toast.error('No folio found for this reservation')
   371→      return
   372→    }
   373→    const amount = Number(paymentAmount)
   374→    if (isNaN(amount) || amount <= 0) {
   375→      toast.error('Please enter a valid payment amount')
   376→      return
   377→    }
   378→    paymentMutation.mutate({
   379→      folioId,
   380→      amount,
   381→      method: paymentMethod,
   382→      reference: paymentReference,
   383→    })
   384→  }
   385→
   386→  const confirmCheckout = () => {
   387→    if (!selectedDeparture) return
   388→    const roomNumber = selectedDeparture.room.number
   389→    checkoutMutation.mutate(selectedDeparture.id, {
   390→      onSuccess: () => {
   391→        setCheckoutDialogOpen(false)
   392→        setReceiptData(selectedDeparture)
   393→        setReceiptDialogOpen(true)
   394→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   395→      },
   396→    })
   397→  }
   398→
   399→  const handleBatchCheckout = () => {
   400→    const ids = zeroBalancePending.map((d) => d.id)
   401→    if (ids.length === 0) {
   402→      toast.info('No guests with zero balance to batch checkout')
   403→      return
   404→    }
   405→    batchCheckoutMutation.mutate(ids)
   406→  }
   407→
   408→  const handlePrintReceipt = () => {
   409→    toast.success('Receipt sent to printer')
   410→  }
   411→
   412→  const handleEmailReceipt = () => {
   413→    toast.success('Receipt emailed to guest')
   414→  }
   415→
   416→  // ─── Computed values ──────────────────────────────────────────────
   417→
   418→  const lateCheckoutFee = selectedDeparture
   419→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   420→    : 0
   421→
   422→  // ─── Render ────────────────────────────────────────────────────────
   423→
   424→  return (
   425→    <div className="flex flex-col gap-2">
   426→      {/* Header */}
   427→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   428→        <div>
   429→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">
   430→            {isShowingUpcoming ? 'Upcoming Departures' : "Today's Departures"}
   431→          </h2>
   432→          <p className="text-xs text-muted-foreground">
   433→            {isShowingUpcoming
   434→              ? `No departures today (${formatDate(today)}). Showing upcoming check-outs.`
   435→              : `Guest check-outs scheduled for ${formatDate(today)}`}
   436→          </p>
   437→        </div>
   438→        {zeroBalancePending.length > 0 && (
   439→          <Button
   440→            onClick={handleBatchCheckout}
   441→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   442→            disabled={batchCheckoutMutation.isPending}
   443→          >
   444→            <Zap className="size-4 mr-1.5" />
   445→            {batchCheckoutMutation.isPending
   446→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   447→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   448→          </Button>
   449→        )}
   450→      </div>
   451→
   452→      {/* Departures List */}
   453→      <Card className="py-0">
   454→        <CardContent className="p-0">
   455→          <div className="max-h-[600px] overflow-y-auto">
   456→            <Table>
   457→              <TableHeader>
   458→                <TableRow>
   459→                  <TableHead className="w-[40px]">
   460→                    <Checkbox
   461→                      checked={displayDepartures.length > 0 && selectedRows.size === displayDepartures.filter(d => d.status === 'checked_in').length && !isShowingUpcoming}
   462→                      onCheckedChange={(checked) => {
   463→                        if (checked) {
   464→                          setSelectedRows(new Set(displayDepartures.filter(d => d.status === 'checked_in').map(d => d.id)))
   465→                        } else {
   466→                          setSelectedRows(new Set())
   467→                        }
   468→                      }}
   469→                    />
   470→                  </TableHead>
   471→                  <TableHead className="w-[70px]">Room</TableHead>
   472→                  <TableHead>Guest</TableHead>
   473→                  <TableHead className="w-[100px]">Check-out</TableHead>
   474→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   475→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   476→                  <TableHead className="w-[50px]"></TableHead>
   477→                </TableRow>
   478→              </TableHeader>
   479→              <TableBody>
   480→                {isLoading ? (
   481→                  Array.from({ length: 5 }).map((_, i) => (
   482→                    <TableRow key={i}>
   483→                      {Array.from({ length: 7 }).map((_, j) => (
   484→                        <TableCell key={j}>
   485→                          <Skeleton className="h-4 w-full" />
   486→                        </TableCell>
   487→                      ))}
   488→                    </TableRow>
   489→                  ))
   490→                ) : displayDepartures.length === 0 ? (
   491→                  <TableRow>
   492→                    <TableCell colSpan={7} className="h-24 text-center text-muted-foreground">
   493→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   494→                      No departures scheduled
   495→                    </TableCell>
   496→                  </TableRow>
   497→                ) : (
   498→                  displayDepartures.map((dep) => {
   499→                    const balance = getBalance(dep)
   500→                    const isPending = dep.status === 'checked_in'
   501→                    const isZeroBalance = balance === 0
   502→                    return (
   503→                      <TableRow key={dep.id}>
   504→                        <TableCell>
   505→                          <Checkbox
   506→                            checked={selectedRows.has(dep.id)}
   507→                            onCheckedChange={(checked) => {
   508→                              const next = new Set(selectedRows)
   509→                              if (checked) next.add(dep.id)
   510→                              else next.delete(dep.id)
   511→                              setSelectedRows(next)
   512→                            }}
   513→                            disabled={dep.status === 'checked_out'}
   514→                          />
   515→                        </TableCell>
   516→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   517→                        <TableCell>
   518→                          <div className="flex items-center gap-2">
   519→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   520→                            {dep.guest.vipLevel !== 'none' && (
   521→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   522→                                VIP
   523→                              </Badge>
   524→                            )}
   525→                          </div>
   526→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   527→                        </TableCell>
   528→                        <TableCell className="text-xs">
   529→                          <div>{formatDate(dep.checkOut)}</div>
   530→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   531→                        </TableCell>
   532→                        <TableCell className="text-right">
   533→                          <div className="flex items-center justify-end gap-1.5">
   534→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   535→                              {formatCurrency(balance)}
   536→                            </span>
   537→                            {/* Outstanding Balance Alert Badges */}
   538→                            {isPending && balance > 5000 && (
   539→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   540→                                HIGH
   541→                              </Badge>
   542→                            )}
