   104→   104→function nightsBetween(start: string | Date, end: string | Date): number {
   105→   105→  const s = new Date(start)
   106→   106→  const e = new Date(end)
   107→   107→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   108→   108→}

   111→   111→  return toDateOnly(date)
