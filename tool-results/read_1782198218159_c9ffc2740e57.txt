   160→   160→  const queryClient = useQueryClient()

   162→   162→  const { navigateTo } = useNavigationStore()

   170→   170→

   172→   172→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   173→   173→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   174→   174→
   175→   175→  // Check-in form states
   176→   176→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   177→   177→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   178→   178→  const [roomPreference, setRoomPreference] = useState('')
   179→   179→  const [pillowType, setPillowType] = useState('')

   182→   182→  // Key card states

   184→   184→  const [keyCardIssued, setKeyCardIssued] = useState(false)

   192→   192→
   193→   193→  const { data, isLoading } = useQuery({
   194→   194→    queryKey: ['arrivals', today],
   195→   195→    queryFn: async () => {

   197→   197→      return apiFetch(`/api/reservations?${params.toString()}`)

   200→   200→  })
   201→   201→
   202→   202→  const { data: roomsData } = useQuery({
   203→   203→    queryKey: ['rooms', 'vacant'],

   208→   208→  const allRooms: AvailableRoom[] = roomsData?.rooms || []

   219→   219→      if (aVip && !bVip) return -1
   220→   220→      if (!aVip && bVip) return 1
   221→   221→      return 0
   222→   222→    })

   236→   236→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }

   238→   238→        payload.specialRequests = specialRequests
   239→   239→      }

   241→   241→        method: 'PATCH',
   242→   242→        headers: { 'Content-Type': 'application/json' },
   243→   243→        body: JSON.stringify(payload),
   244→   244→      })
   245→   245→    },
   246→   246→    onSuccess: (_data, variables) => {
   247→   247→      invalidate.afterReservationChange(queryClient)
   248→   248→      setRoomPickerOpen(false)
   249→   249→      setCheckInDialogOpen(false)
   250→   250→
   251→   251→      // Find the room number for key card dialog
   252→   252→      const room = availableRooms.find((r) => r.id === variables.roomId)
   253→   253→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   254→   254→      openKeyCardDialog(roomNumber)
   255→   255→    },
   256→   256→  })
   257→   257→
   258→   258→  const checkInMutation = useMutation({
   259→   259→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   260→   260→      const payload: Record<string, unknown> = { status: 'checked_in' }
   261→   261→      if (specialRequests !== undefined) {
   262→   262→        payload.specialRequests = specialRequests
   263→   263→      }
   264→   264→      return apiFetch(`/api/reservations/${reservationId}`, {
   265→   265→        method: 'PATCH',
   266→   266→        headers: { 'Content-Type': 'application/json' },
   267→   267→        body: JSON.stringify(payload),
   268→   268→      })
   269→   269→    },
   270→   270→    onSuccess: (_data, variables) => {
   271→   271→      invalidate.afterCheckIn(queryClient)
   272→   272→      setCheckInDialogOpen(false)
   273→   273→
   274→   274→      const roomNumber = selectedArrival?.room?.number || ''
   275→   275→      openKeyCardDialog(roomNumber)
   276→   276→    },
   277→   277→  })
   278→   278→
   279→   279→  const walkInMutation = useMutation({
   280→   280→    mutationFn: async (form: WalkInForm) => {
   281→   281→      // 1. Create guest
   282→   282→      const guestData = await apiFetch('/api/guests', {
   283→   283→        method: 'POST',
   284→   284→        headers: { 'Content-Type': 'application/json' },

   286→   286→          firstName: form.firstName,
   287→   287→          lastName: form.lastName,
   288→   288→          phone: form.phone || null,
   289→   289→          email: form.email || null,
   290→   290→          nationality: form.nationality || null,
   291→   291→          idType: form.idType || null,
   292→   292→          idNumber: form.idNumber || null,
   293→   293→        }),
   294→   294→      })
   295→   295→      const guestId = guestData.guest.id
   296→   296→
   297→   297→      // 2. Find a vacant room
   298→   298→      const vacantRoom = availableRooms[0]
   299→   299→      if (!vacantRoom) throw new Error('No available rooms')
   300→   300→
   301→   301→      // 3. Create reservation as checked_in (walk-in)
   302→   302→      const checkOutDate = form.checkOutDate || (() => {
   303→   303→        const d = new Date()
   304→   304→        d.setDate(d.getDate() + 1)
   305→   305→        return d
   306→   306→      })()
   307→   307→
   308→   308→      const resData = await apiFetch('/api/reservations', {
   309→   309→        method: 'POST',
   310→   310→        headers: { 'Content-Type': 'application/json' },
   311→   311→        body: JSON.stringify({
   312→   312→          guestId,
   313→   313→          roomId: vacantRoom.id,
   314→   314→          adults: form.adults,
   315→   315→          children: form.children,
   316→   316→          checkIn: new Date().toISOString(),
   317→   317→          checkOut: checkOutDate.toISOString(),
   318→   318→          roomRate: form.roomRate,
   319→   319→          source: 'walk_in',
   320→   320→          reservationType: 'walk_in',
   321→   321→          guaranteed: false,
   322→   322→        }),
   323→   323→      })
   324→   324→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   325→   325→    },
   326→   326→    onSuccess: (result) => {
   327→   327→      invalidate.afterCheckIn(queryClient)
   328→   328→      setWalkInDialogOpen(false)
   329→   329→      setWalkInForm(initialWalkInForm)
   330→   330→
   331→   331→      // Open key card dialog
   332→   332→      openKeyCardDialog(result.roomNumber)
   333→   333→    },
   334→   334→    onError: (error) => {
   335→   335→      toast.error('Walk-in check-in failed', {
   336→   336→        description: error instanceof Error ? error.message : 'Unknown error',
   337→   337→      })
   338→   338→    },
   339→   339→  })
   340→   340→
   341→   341→  // ─── Handlers ──────────────────────────────────────────────────────
   342→   342→

   385→   385→    // Early check-in note

   389→   389→
   390→   390→    // Check-out time override
   391→   391→    if (checkOutTime !== settings.defaultCheckOut) {
   392→   392→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   393→   393→    }
   394→   394→
   395→   395→    // Room preference
   396→   396→    if (roomPreference) {
   397→   397→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   398→   398→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   399→   399→    }
   400→   400→
   401→   401→    // Pillow type
   402→   402→    if (pillowType) {
   403→   403→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType

   405→   405→    }
   406→   406→
   407→   407→    // Wake-up call
   408→   408→    if (wakeupCall) {
   409→   409→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   410→   410→    }
   411→   411→
   412→   412→    return parts.join(' | ')
   413→   413→  }
   414→   414→
   415→   415→  const confirmCheckIn = () => {
   416→   416→    if (!selectedArrival) return
   417→   417→
   418→   418→    const updatedRequests = buildSpecialRequests(selectedArrival)
   419→   419→
   420→   420→    if (selectedRoomId) {
   421→   421→      assignRoomMutation.mutate({
   422→   422→        reservationId: selectedArrival.id,
   423→   423→        roomId: selectedRoomId,
   424→   424→        specialRequests: updatedRequests,
   425→   425→      })
   426→   426→    } else if (selectedArrival.room) {
   427→   427→      checkInMutation.mutate({
   428→   428→        reservationId: selectedArrival.id,
   429→   429→        specialRequests: updatedRequests,
   430→   430→      })
   431→   431→    }
   432→   432→  }
   433→   433→
   434→   434→  const openKeyCardDialog = (roomNumber: string) => {

   437→   437→    setKeyCardProcessed(false)
   438→   438→    setKeyCardDialogOpen(true)
   439→   439→  }
   440→   440→

   446→   446→      icon: <Mail className="size-4" />,

   450→   450→  }
   451→   451→
   452→   452→  const resetCheckInForm = () => {

   457→   457→    setWakeupCall('')
   458→   458→  }
   459→   459→
   460→   460→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   461→   461→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   462→   462→    setWalkInForm((prev) => ({
   463→   463→      ...prev,
   464→   464→      roomTypeId,
   465→   465→      roomRate: roomType?.baseRate || 0,
   466→   466→    }))
   467→   467→  }
   468→   468→
   469→   469→  // ─── Render ─────────────────────────────────────────────────────────
   470→   470→
   471→   471→  return (
   472→   472→    <div className="flex flex-col gap-2">
   473→   473→      {/* Header */}
