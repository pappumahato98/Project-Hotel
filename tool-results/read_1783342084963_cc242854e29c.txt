   141→   141→// ── Photo Capture Component ──────────────────────────────────
   142→   142→function PhotoCaptureSection({
   143→   143→  photos,
   144→   144→  onCapture,
   145→   145→  onRemove,
   146→   146→}: {
   147→   147→  photos: string[]
   148→   148→  onCapture: (dataUrl: string) => void
   149→   149→  onRemove: (index: number) => void
   150→   150→}) {
   151→   151→  const videoRef = useRef<HTMLVideoElement>(null)
   152→   152→  const streamRef = useRef<MediaStream | null>(null)
   153→   153→  const [cameraOpen, setCameraOpen] = useState(false)
   154→   154→  const [cameraError, setCameraError] = useState<string | null>(null)
   155→   155→
   156→   156→  const startCamera = useCallback(async () => {
   157→   157→    try {
   158→   158→      setCameraError(null)
   159→   159→      const stream = await navigator.mediaDevices.getUserMedia({
   160→   160→        video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
   161→   161→      })
   162→   162→      streamRef.current = stream
   163→   163→      if (videoRef.current) {
   164→   164→        videoRef.current.srcObject = stream
   165→   165→      }
   166→   166→      setCameraOpen(true)
   167→   167→    } catch (err) {
   168→   168→      console.error('Camera error:', err)
   169→   169→      setCameraError('Camera access denied or unavailable')
   170→   170→    }
   171→   171→  }, [])
   172→   172→
   173→   173→  const stopCamera = useCallback(() => {
   174→   174→    if (streamRef.current) {
   175→   175→      streamRef.current.getTracks().forEach((t) => t.stop())
   176→   176→      streamRef.current = null
   177→   177→    }
   178→   178→    setCameraOpen(false)
   179→   179→  }, [])
   180→   180→
   181→   181→  const capturePhoto = useCallback(() => {
   182→   182→    const video = videoRef.current
   183→   183→    if (!video) return
   184→   184→    const canvas = document.createElement('canvas')
   185→   185→    canvas.width = video.videoWidth
   186→   186→    canvas.height = video.videoHeight
   187→   187→    const ctx = canvas.getContext('2d')
   188→   188→    if (!ctx) return
   189→   189→    ctx.drawImage(video, 0, 0)
   190→   190→    const dataUrl = canvas.toDataURL('image/jpeg', 0.85)
   191→   191→    onCapture(dataUrl)
   192→   192→    stopCamera()
   193→   193→  }, [onCapture, stopCamera])
   194→   194→
   195→   195→  return (
   196→   196→    <div className="space-y-2">
   197→   197→      <p className="text-sm font-semibold">Photo Evidence</p>
   198→   198→
   199→   199→      {/* Camera Preview */}
   200→   200→      {cameraOpen && (
   201→   201→        <div className="relative rounded-lg border overflow-hidden bg-black">
   202→   202→          <video
   203→   203→            ref={videoRef}
   204→   204→            autoPlay
   205→   205→            playsInline
   206→   206→            muted
   207→   207→            className="w-full h-48 object-cover"
   208→   208→          />
   209→   209→          <div className="absolute bottom-2 left-1/2 -translate-x-1/2 flex gap-2">
   210→   210→            <Button
   211→   211→              size="sm"
   212→   212→              className="gap-1.5 bg-white text-black hover:bg-gray-200"
   213→   213→              onClick={capturePhoto}
   214→   214→            >
   215→   215→              <Camera className="h-4 w-4" /> Capture
   216→   216→            </Button>
   217→   217→            <Button
   218→   218→              size="sm"
   219→   219→              variant="outline"
   220→   220→              className="gap-1.5 bg-white/90 text-black hover:bg-gray-200 border-gray-300"
   221→   221→              onClick={stopCamera}
   222→   222→            >
   223→   223→              <X className="h-4 w-4" /> Cancel
   224→   224→            </Button>
   225→   225→          </div>
   226→   226→        </div>
   227→   227→      )}
   228→   228→
   229→   229→      {cameraError && (
   230→   230→        <p className="text-xs text-red-500 flex items-center gap-1">
   231→   231→          <AlertCircle className="h-3 w-3" /> {cameraError}
   232→   232→        </p>
   233→   233→      )}
   234→   234→
   235→   235→      {/* Photo Grid */}
   236→   236→      {photos.length > 0 && (
   237→   237→        <div className="grid grid-cols-3 gap-2">
   238→   238→          {photos.map((photo, idx) => (
   239→   239→            <div key={idx} className="relative group rounded-md border overflow-hidden aspect-square">
   240→   240→              <img src={photo} alt={`Evidence ${idx + 1}`} className="w-full h-full object-cover" />
   241→   241→              <button
   242→   242→                onClick={() => onRemove(idx)}
   243→   243→                className="absolute top-1 right-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity"
   244→   244→              >
   245→   245→                <Trash2 className="h-3 w-3" />
   246→   246→              </button>
   247→   247→            </div>
   248→   248→          ))}
   249→   249→        </div>
   250→   250→      )}
   251→   251→
   252→   252→      {/* Open Camera Button */}
   253→   253→      {!cameraOpen && (
   254→   254→        <Button variant="outline" className="w-full gap-2" onClick={startCamera}>
   255→   255→          <Camera className="h-4 w-4" />
   256→   256→          {photos.length > 0 ? `Add More Photo (${photos.length} captured)` : 'Capture Photo Evidence'}
   257→   257→        </Button>
   258→   258→      )}
   259→   259→    </div>
   260→   260→  )
   261→   261→}

   315→   315→// ── Inspection Dialog ────────────────────────────────────────
   316→   316→function InspectionDialog({
   317→   317→  task,
   318→   318→  open,
   319→   319→  onOpenChange,
   320→   320→}: {
   321→   321→  task: HkTask | null
   322→   322→  open: boolean
   323→   323→  onOpenChange: (open: boolean) => void
   324→   324→}) {
   325→   325→  const queryClient = useQueryClient()
   326→   326→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   327→   327→  const [photos, setPhotos] = useState<string[]>([])
   328→   328→  const [rejectReason, setRejectReason] = useState('')
   329→   329→  const [showRejectForm, setShowRejectForm] = useState(false)
   330→   330→  const [showAuditTrail, setShowAuditTrail] = useState(false)
   331→   331→
   332→   332→  const approveMutation = useMutation({
   333→   333→    mutationFn: async () => {
   334→   334→      if (!task) return
   335→   335→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   336→   336→        if (v) acc[k] = true
   337→   337→        return acc
   338→   338→      }, {} as Record<string, boolean>)
   339→   339→      return apiFetch('/api/housekeeping', {
   340→   340→        method: 'POST',
   341→   341→        headers: { 'Content-Type': 'application/json' },
   342→   342→        body: JSON.stringify({
   343→   343→          action: 'approve-inspection',
   344→   344→          id: task.id,
   345→   345→          performedBy: 'Inspector',
   346→   346→          checklist,
   347→   347→          photos: photos.length > 0 ? photos : undefined,
   348→   348→        }),
   349→   349→      })
   350→   350→    },
   351→   351→    onSuccess: () => {
   352→   352→      invalidate.afterRoomStatusChange(queryClient)
   353→   353→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   354→   354→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   355→   355→      toast.success(`Room ${task?.room.number} inspection passed`)
   356→   356→      resetAndClose()
   357→   357→    },
   358→   358→    onError: (err: Error) => {
   359→   359→      toast.error(err.message || 'Failed to approve inspection')
   360→   360→    },
   361→   361→  })
   362→   362→
   363→   363→  const rejectMutation = useMutation({
   364→   364→    mutationFn: async () => {
   365→   365→      if (!task) return
   366→   366→      if (!rejectReason.trim()) {
   367→   367→        toast.error('Please provide a reason for rejection')
   368→   368→        throw new Error('Reason required')
   369→   369→      }
   370→   370→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   371→   371→        if (v) acc[k] = true
   372→   372→        return acc
   373→   373→      }, {} as Record<string, boolean>)
   374→   374→      return apiFetch('/api/housekeeping', {
   375→   375→        method: 'POST',
   376→   376→        headers: { 'Content-Type': 'application/json' },
   377→   377→        body: JSON.stringify({
   378→   378→          action: 'reject-inspection',
   379→   379→          id: task.id,
   380→   380→          performedBy: 'Inspector',
   381→   381→          reason: rejectReason.trim(),
   382→   382→          checklist,
   383→   383→          photos: photos.length > 0 ? photos : undefined,
   384→   384→        }),
   385→   385→      })
   386→   386→    },
   387→   387→    onSuccess: () => {
   388→   388→      invalidate.afterRoomStatusChange(queryClient)
   389→   389→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   390→   390→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   391→   391→      toast.info(`Room ${task?.room.number} rejected & reassigned for cleaning`)
   392→   392→      resetAndClose()
   393→   393→    },
   394→   394→    onError: (err: Error) => {
   395→   395→      if (err.message !== 'Reason required') {
   396→   396→        toast.error(err.message || 'Failed to reject inspection')
   397→   397→      }
   398→   398→    },
   399→   399→  })
   400→   400→
   401→   401→  const resetAndClose = () => {
   402→   402→    setCheckedItems({})
   403→   403→    setPhotos([])
   404→   404→    setRejectReason('')
   405→   405→    setShowRejectForm(false)
   406→   406→    setShowAuditTrail(false)
   407→   407→    onOpenChange(false)
   408→   408→  }
   409→   409→
   410→   410→  if (!task) return null
   411→   411→
   412→   412→  const toggleCheck = (id: string) => {
   413→   413→    setCheckedItems((prev) => ({ ...prev, [id]: !prev[id] }))
   414→   414→  }
   415→   415→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   416→   416→  const allChecked = checkedCount === INSPECTION_ITEMS.length
   417→   417→  const passCount = INSPECTION_ITEMS.length
   418→   418→  const passPercent = Math.round((checkedCount / passCount) * 100)
   419→   419→
   420→   420→  const handleCapturePhoto = (dataUrl: string) => {
   421→   421→    setPhotos((prev) => [...prev, dataUrl])
   422→   422→    toast.success('Photo captured')
   423→   423→  }
   424→   424→
   425→   425→  const handleRemovePhoto = (index: number) => {
   426→   426→    setPhotos((prev) => prev.filter((_, i) => i !== index))
   427→   427→  }
   428→   428→
   429→   429→  const isSubmitting = approveMutation.isPending || rejectMutation.isPending
   430→   430→
   431→   431→  return (
   432→   432→    <Dialog open={open} onOpenChange={(v) => { if (!v) resetAndClose(); else onOpenChange(true) }}>
   433→   433→      <DialogContent className="sm:max-w-lg max-w-[95vw] max-h-[90vh] flex flex-col p-0 gap-0">
   434→   434→        {/* Fixed Header */}
   435→   435→        <div className="shrink-0 px-6 pt-6 pb-0">
   436→   436→          <DialogHeader>
   437→   437→            <DialogTitle className="flex items-center gap-2">
   438→   438→              <ShieldCheck className="h-5 w-5 text-purple-600" />
   439→   439→              Inspect Room {task.room.number}
   440→   440→            </DialogTitle>
   441→   441→            <DialogDescription className="mt-1">
   442→   442→              Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   443→   443→              {task.assignedTo ? ` · Cleaned by ${task.assignedTo}` : ''}
   444→   444→            </DialogDescription>
   445→   445→          </DialogHeader>
   446→   446→        </div>
   447→   447→
   448→   448→        {/* Scrollable Body */}
   449→   449→        <ScrollArea className="flex-1 px-6 py-4 max-h-[60vh]">
   450→   450→          <div className="space-y-4 pr-2">
   451→   451→            {/* Progress */}
   452→   452→            <div className="rounded-lg border p-3">
   453→   453→              <div className="flex items-center justify-between mb-2">
   454→   454→                <p className="text-sm font-medium">Inspection Progress</p>
   455→   455→                <span className="text-sm text-muted-foreground">{checkedCount}/{passCount} items</span>
   456→   456→              </div>
   457→   457→              <div className="h-2 rounded-full bg-muted overflow-hidden">
   458→   458→                <div
   459→   459→                  className={cn(
   460→   460→                    'h-full rounded-full transition-all',

   587→   587→        </div>

   589→   589→    </Dialog>
   590→   590→  )

   594→   594→export function InspectionView() {
   595→   595→  const [selectedTask, setSelectedTask] = useState<HkTask | null>(null)
   596→   596→
