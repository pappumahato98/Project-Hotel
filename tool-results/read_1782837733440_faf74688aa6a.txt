   180→   180→// ─── Debounce hook ────────────────────────────────────────────────────
   181→   181→function useDebounce<T>(value: T, delay: number): T {
   182→   182→  const [debounced, setDebounced] = useState(value)
   183→   183→  useEffect(() => {
   184→   184→    const timer = setTimeout(() => setDebounced(value), delay)
   185→   185→    return () => clearTimeout(timer)
   186→   186→  }, [value, delay])
   187→   187→  return debounced
   188→   188→}

   190→   190→// ─── Date range presets ──────────────────────────────────────────────
   191→   191→function getDatePreset(preset: string): { from: Date; to: Date } {
   192→   192→  const today = new Date()
   193→   193→  today.setHours(0, 0, 0, 0)
   194→   194→  switch (preset) {
   195→   195→    case 'today':
   196→   196→      return { from: today, to: today }
   197→   197→    case 'this-week': {
   198→   198→      const from = new Date(today)
   199→   199→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   200→   200→      const to = new Date(from)
   201→   201→      to.setDate(from.getDate() + 6) // Sunday
   202→   202→      return { from, to }
   203→   203→    }
   204→   204→    case 'this-month': {
   205→   205→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   206→   206→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   207→   207→      return { from, to }
   208→   208→    }
   209→   209→    case 'next-7': {
   210→   210→      const to = new Date(today)
   211→   211→      to.setDate(today.getDate() + 6)
   212→   212→      return { from: today, to }
   213→   213→    }
   214→   214→    case 'next-14': {
   215→   215→      const to = new Date(today)
   216→   216→      to.setDate(today.getDate() + 13)
   217→   217→      return { from: today, to }
   218→   218→    }
   219→   219→    case 'next-30': {
   220→   220→      const to = new Date(today)
   221→   221→      to.setDate(today.getDate() + 29)
   222→   222→      return { from: today, to }
   223→   223→    }
   224→   224→    default:
   225→   225→      return { from: today, to: today }
   226→   226→  }
   227→   227→}

   229→   229→function toDateString(d: Date): string {
   230→   230→  const y = d.getFullYear()
   231→   231→  const m = String(d.getMonth() + 1).padStart(2, '0')
   232→   232→  const day = String(d.getDate()).padStart(2, '0')
   233→   233→  return `${y}-${m}-${day}`
   234→   234→}

   236→   236→function formatShortDate(dateStr: string): string {
   237→   237→  if (!dateStr) return ''
   238→   238→  const d = new Date(dateStr + 'T00:00:00')
   239→   239→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   240→   240→}
