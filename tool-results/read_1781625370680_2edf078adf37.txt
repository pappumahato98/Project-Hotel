   101→// ─── Helper ──────────────────────────────────────────────────────────────
   102→
   103→function nightsBetween(start: string | Date, end: string | Date): number {
   104→  const s = new Date(start)
   105→  const e = new Date(end)
   106→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   107→}
   108→
   109→function formatDateValue(date: Date): string {
   110→  return toDateOnly(date)
   111→}

   548→  function getCreditStatus(reservation: InHouseReservation) {
   549→    const balance = reservation.folios[0]?.balance || 0
   550→    const pct = (balance / reservation.creditLimit) * 100
   551→    if (pct >= 100) return { status: 'breach', pct, color: 'text-red-600' }
   552→    if (pct >= 80) return { status: 'warning', pct, color: 'text-amber-600' }
   553→    return { status: 'ok', pct, color: 'text-green-600' }
   554→  }
