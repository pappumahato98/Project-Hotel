   151→
   152→interface CheckInViewProps {
   153→  reservationId?: string
   154→  onComplete?: () => void

   156→
   157→export function CheckInView({ reservationId, onComplete }: CheckInViewProps) {
   158→  const queryClient = useQueryClient()
   159→  const { settings } = useSettingsStore()
   160→  const today = getTodayString()
   161→
   162→  // ─── Step state ────────────────────────────────────────────────
   163→  const [currentStep, setCurrentStep] = useState(0)
   164→
   165→  // ─── Selection states ────────────────────────────────────────
   166→  const [selectedReservationId, setSelectedReservationId] = useState<string>('')
   167→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   168→  const [roomSearchFilter, setRoomSearchFilter] = useState('')
   169→
   170→  // ─── Registration card form ────────────────────────────────────
   171→  const [regTitle, setRegTitle] = useState('')
   172→  const [regFirstName, setRegFirstName] = useState('')
   173→  const [regLastName, setRegLastName] = useState('')
   174→  const [regContactNo, setRegContactNo] = useState('')
   175→  const [regEmail, setRegEmail] = useState('')
   176→  const [regAddress, setRegAddress] = useState('')
   177→  const [regNationality, setRegNationality] = useState('')
   178→  const [regIdType, setRegIdType] = useState('passport')
   179→  const [regIdNumber, setRegIdNumber] = useState('')
   180→  const [regCity, setRegCity] = useState('')
   181→  const [regCountry, setRegCountry] = useState('')
   182→
   183→  // ─── Stay preferences ─────────────────────────────────────────
   184→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   185→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   186→  const [pillowType, setPillowType] = useState('')
   187→  const [roomPreference, setRoomPreference] = useState('')
   188→  const [wakeupCall, setWakeupCall] = useState('')
   189→
   190→  // ─── Billing ──────────────────────────────────────────────────
   191→  const [depositAmount, setDepositAmount] = useState('')
   192→  const [paymentMethod, setPaymentMethod] = useState('')
   193→  const [billingNotes, setBillingNotes] = useState('')
   194→
   195→  // ─── Key card dialog ───────────────────────────────────────────
   196→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   197→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   198→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   199→  const [keyCardGuestName, setKeyCardGuestName] = useState('')
   200→  const [keyCardCheckOut, setKeyCardCheckOut] = useState('')
   201→
   202→  // ─── Queries ─────────────────────────────────────────────────
   203→
   204→  // Fetch specific reservation by ID (if prop provided)
   205→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   206→    queryKey: ['reservation', reservationId],
   207→    queryFn: async () => {
   208→      const res = await fetch(`/api/reservations/${reservationId}`)
   209→      if (!res.ok) throw new Error('Failed to fetch reservation')
   210→      return res.json()
   211→    },
   212→    enabled: !!reservationId,
   213→  })
   214→
   215→  // Fetch today's confirmed arrivals
   216→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   217→    queryKey: ['arrivals', today],
   218→    queryFn: async () => {
   219→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   220→      const res = await fetch(`/api/reservations?${params.toString()}`)
   221→      if (!res.ok) throw new Error('Failed to fetch arrivals')
   222→      return res.json()
   223→    },
   224→    enabled: !reservationId,
   225→  })
   226→
   227→  // Fetch all rooms with types
   228→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   229→    queryKey: ['rooms', 'all'],
   230→    queryFn: async () => {
   231→      const res = await fetch('/api/rooms')
   232→      if (!res.ok) throw new Error('Failed to fetch rooms')
   233→      return res.json()
   234→    },
   235→  })
   236→
   237→  // ─── Derived data ──────────────────────────────────────────────
   238→
   239→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   240→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   241→
   242→  // The reservation we're working with
   243→  const activeReservation: ReservationInfo | null = reservationId
   244→    ? specificReservation
   245→    : arrivals.find((a) => a.id === selectedReservationId) || null
   246→
   247→  // Room type data from API (not hardcoded)
   248→  const roomTypes: RoomTypeData[] = useMemo(() => {
   249→    if (!roomsData?.roomTypes) return []
   250→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   251→      id: rt.id,
   252→      name: rt.name,
   253→      code: rt.code,
   254→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   255→    }))
   256→  }, [roomsData])
   257→
   258→  // Available rooms (vacant_clean or inspected)
   259→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   260→  const availableRooms = allRooms.filter(
   261→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   262→  )
   263→
   264→  // Filter available rooms by search
   265→  const filteredAvailableRooms = useMemo(() => {
   266→    let filtered = availableRooms
   267→    if (roomSearchFilter.trim()) {
   268→      const q = roomSearchFilter.toLowerCase()
   269→      filtered = filtered.filter(
   270→        (r: RoomInfo) =>
   271→          r.number.toLowerCase().includes(q) ||
   272→          r.type.name.toLowerCase().includes(q) ||
   273→          String(r.floor).includes(q) ||
   274→          (r.wing && r.wing.toLowerCase().includes(q))
   275→      )
   276→    }
   277→    return filtered
   278→  }, [availableRooms, roomSearchFilter])
   279→
   280→  // Group rooms by floor
   281→  const roomsByFloor = useMemo(() => {
   282→    const grouped = new Map<number, RoomInfo[]>()
   283→    for (const room of filteredAvailableRooms) {
   284→      const floor = room.floor
   285→      if (!grouped.has(floor)) grouped.set(floor, [])
   286→      grouped.get(floor)!.push(room)
   287→    }
   288→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   289→  }, [filteredAvailableRooms])
   290→
   291→  // Get room type name by ID
   292→  const getRoomTypeName = (typeId: string) => {
   293→    const rt = roomTypes.find((t) => t.id === typeId)
   294→    return rt?.name || 'Unknown'
   295→  }
   296→
   297→  // Get room rate by type ID
   298→  const getRoomTypeRate = (typeId: string) => {
   299→    const rt = roomTypes.find((t) => t.id === typeId)
   300→    return rt?.baseRate || 0
   301→  }
   302→
   303→  // ─── Initialize form when reservation loads ───────────────────
   304→
   305→  React.useEffect(() => {
   306→    if (specificReservation && reservationId) {
   307→      setSelectedReservationId(specificReservation.id)
   308→      if (specificReservation.room?.id) {
   309→        setSelectedRoomId(specificReservation.room.id)
   310→      }
   311→      setCurrentStep(specificReservation.guest ? 1 : 0)
   312→    }
   313→  }, [specificReservation, reservationId])
   314→
   315→  // Pre-fill registration fields when guest is selected
   316→  React.useEffect(() => {
   317→    if (activeReservation?.guest) {
   318→      const g = activeReservation.guest
   319→      setRegTitle('')
   320→      setRegFirstName(g.firstName || '')
   321→      setRegLastName(g.lastName || '')
   322→      setRegContactNo(g.phone || '')
   323→      setRegEmail(g.email || '')
   324→      setRegNationality(g.nationality || '')
   325→      setRegIdType(g.idType || 'passport')
   326→      setRegIdNumber(g.idNumber || '')
   327→      setRegAddress(g.address || '')
   328→      setRegCity(g.city || '')
   329→      setRegCountry(g.country || '')
   330→    }
   331→  }, [activeReservation?.guest])
   332→
   333→  // Auto-set deposit when room is selected
   334→  React.useEffect(() => {
   335→    if (activeReservation?.roomRate && !depositAmount) {
   336→      setDepositAmount(String(activeReservation.roomRate))
   337→    }
   338→  }, [activeReservation?.roomRate, depositAmount])
   339→
   340→  // ─── Mutations ────────────────────────────────────────────────
   341→
   342→  const checkInMutation = useMutation({
   343→    mutationFn: async ({
   344→      reservationId: resId,
   345→      roomId,
   346→      specialRequests,
   347→    }: {
   348→      reservationId: string
   349→      roomId: string
   350→      specialRequests: string
   351→    }) => {
   352→      const res = await fetch(`/api/reservations/${resId}`, {
   353→        method: 'PATCH',
   354→        headers: { 'Content-Type': 'application/json' },
   355→        body: JSON.stringify({
   356→          roomId,
   357→          status: 'checked_in',
   358→          specialRequests,
   359→        }),
   360→      })
   361→      if (!res.ok) throw new Error('Failed to check in guest')
   362→      return res.json()
   363→    },
   364→    onSuccess: () => {
   365→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   366→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   367→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   368→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   369→      toast.success('Guest checked in successfully!')
   370→    },
   371→    onError: (error) => {
   372→      toast.error('Check-in failed', {
   373→        description: error instanceof Error ? error.message : 'Unknown error',
   374→      })
   375→    },
   376→  })
   377→
   378→  // ─── Special Requests Builder ─────────────────────────────────
   379→
   380→  const buildSpecialRequests = (): string => {
   381→    const parts: string[] = []
   382→
   383→    if (activeReservation?.specialRequests) {
   384→      parts.push(activeReservation.specialRequests)
   385→    }
   386→    if (earlyCheckIn) {
   387→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   388→    }
   389→    if (checkOutTime !== settings.defaultCheckOut) {
   390→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   391→    }
   392→    if (roomPreference) {
   393→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   394→      parts.push(`[PREFERENCE] Room: ${label}`)
   395→    }
   396→    if (pillowType) {
   397→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   398→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   399→    }
   400→    if (wakeupCall) {
   401→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   402→    }
   403→    if (billingNotes) {
   404→      parts.push(`[BILLING] ${billingNotes}`)
   405→    }
   406→    if (paymentMethod) {
   407→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   408→    }
   409→    if (depositAmount && parseFloat(depositAmount) > 0) {
   410→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   411→    }
   412→
   413→    return parts.join(' | ')
   414→  }
   415→
   416→  // ─── Step Navigation ──────────────────────────────────────────
   417→
   418→  const canGoNext = (): boolean => {
   419→    switch (currentStep) {
   420→      case 0: return !!activeReservation
   421→      case 1: return true // Guest verification is info-only
   422→      case 2: return !!selectedRoomId
   423→      case 3: return true // Preferences are optional
   424→      case 4: return true // Billing is optional
   425→      case 5: return keyCardIssued
   426→      default: return false
   427→    }
   428→  }
   429→
   430→  const handleNext = () => {
   431→    if (currentStep === 5) {
   432→      onComplete?.()
   433→      return
   434→    }
   435→
   436→    // On step 4 → 5, execute check-in
   437→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   438→      const requests = buildSpecialRequests()
   439→      checkInMutation.mutate(
   440→        {
   441→          reservationId: activeReservation.id,
   442→          roomId: selectedRoomId,
   443→          specialRequests: requests,
   444→        },
   445→        {
   446→          onSuccess: () => {
   447→            // Find the selected room number for key card
   448→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   449→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   450→            const guestName = activeReservation.guest
   451→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   452→              : 'Guest'
   453→
   454→            setKeyCardRoomNumber(roomNumber)
   455→            setKeyCardGuestName(guestName)
   456→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   457→            setKeyCardIssued(false)
   458→            setKeyCardDialogOpen(true)
   459→            setCurrentStep(5)
   460→          },
   461→        }
   462→      )
   463→      return
   464→    }
   465→
   466→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   467→  }
   468→
   469→  const handleBack = () => {
   470→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   471→  }
   472→
   473→  // ─── Render: Step Indicator ────────────────────────────────────
   474→
   475→  const renderStepIndicator = () => (
   476→    <div className="mb-4">
   477→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   478→        {STEP_LABELS.map((label, idx) => (
   479→          <React.Fragment key={idx}>
   480→            <div className="flex flex-col items-center gap-1 min-w-0">
   481→              <div
   482→                className={cn(
   483→                  'flex size-7 shrink-0 items-center justify-center rounded-full border-2 text-[10px] font-bold transition-colors',
   484→                  idx < currentStep && 'border-green-500 bg-green-500 text-white',
   485→                  idx === currentStep && 'border-primary bg-primary text-primary-foreground',
   486→                  idx > currentStep && 'border-muted-foreground/30 bg-background text-muted-foreground'
   487→                )}
   488→              >
   489→                {idx < currentStep ? <Check className="size-4" /> : idx + 1}
   490→              </div>
   491→              <span
   492→                className={cn(
   493→                  'text-[10px] sm:text-xs font-medium whitespace-nowrap',
   494→                  idx === currentStep ? 'text-primary' : 'text-muted-foreground'
   495→                )}
   496→              >
   497→                {label}
   498→              </span>
   499→            </div>
   500→            {idx < STEP_LABELS.length - 1 && (
   501→              <div
   502→                className={cn(
   503→                  'flex-1 h-0.5 mx-1 min-w-[12px] max-w-[40px] transition-colors',
   504→                  idx < currentStep ? 'bg-green-500' : 'bg-muted-foreground/20'
   505→                )}
   506→              />
   507→            )}
   508→          </React.Fragment>
   509→        ))}
   510→      </div>
   511→    </div>
   512→  )
   513→
   514→  // ─── Render: Step 0 - Select Arrival ─────────────────────────
   515→
   516→  const renderSelectArrival = () => {
   517→    if (loadingArrivals) {
   518→      return (
   519→        <div className="space-y-3 p-4">
   520→          {Array.from({ length: 4 }).map((_, i) => (
   521→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   522→          ))}
   523→        </div>
   524→      )
   525→    }
   526→
   527→    if (arrivals.length === 0) {
   528→      return (
   529→        <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   530→          <UserCheck className="size-12 mb-3 opacity-30" />
   531→          <p className="text-sm font-medium">No arrivals scheduled for today</p>
   532→          <p className="text-xs mt-1">Check-in date: {formatDate(today)}</p>
   533→        </div>
   534→      )
   535→    }
   536→
   537→    return (
   538→      <div className="space-y-2">
   539→        <div className="flex items-center gap-2 mb-3">
   540→          <Search className="size-4 text-muted-foreground" />
   541→          <p className="text-sm text-muted-foreground">
   542→            {arrivals.length} confirmed arrival{arrivals.length !== 1 ? 's' : ''} for today
   543→          </p>
   544→        </div>
   545→        <div className="max-h-96 overflow-y-auto space-y-2">
   546→          {arrivals.map((arrival) => {
   547→            const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   548→            const isUnassigned = !arrival.room
   549→            const isSelected = arrival.id === selectedReservationId
   550→
   551→            return (
   552→              <button
   553→                key={arrival.id}
   554→                onClick={() => {
   555→                  setSelectedReservationId(arrival.id)
   556→                  // Pre-fill room if already assigned
   557→                  if (arrival.room?.id) {
   558→                    setSelectedRoomId(arrival.room.id)
   559→                  }
   560→                  setCurrentStep(1)
   561→                }}
