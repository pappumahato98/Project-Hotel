   162→// ─── Component ──────────────────────────────────────────────────────────
   163→
   164→export function ArrivalsView() {
   165→  const queryClient = useQueryClient()
   166→  const { settings } = useSettingsStore()
   167→  const { navigateTo } = useNavigationStore()
   168→  const today = getTodayString()
   169→
   170→  // Dialog states
   171→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   172→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   173→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   174→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   175→
   176→  // Selection states
   177→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   178→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   179→
   180→  // Check-in form states
   181→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   182→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   183→  const [roomPreference, setRoomPreference] = useState('')
   184→  const [pillowType, setPillowType] = useState('')
   185→  const [wakeupCall, setWakeupCall] = useState('')
   186→
   187→  // Key card states
   188→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   189→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   190→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   191→
   192→  // Mobile detail view state
   193→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   194→
   195→  // Walk-in form state
   196→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   197→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   198→
   199→  // Listen for walk-in dialog trigger from dashboard / other modules
   200→  useEffect(() => {
   201→    const unsub = useFrontDeskContextStore.subscribe((state) => {
   202→      if (state.triggerWalkInDialog) {
   203→        state.setTriggerWalkInDialog(false)
   204→        setWalkInForm(initialWalkInForm)
   205→        setWalkInDialogOpen(true)
   206→      }
   207→    })
   208→    return unsub
   209→  }, [])
   210→
   211→  // ─── Queries ────────────────────────────────────────────────────────
   212→
   213→  // Fetch room types from API for walk-in dialog
   214→  const { data: roomTypesData } = useQuery({
   215→    queryKey: ['rooms', 'types'],
   216→    queryFn: () => apiFetch('/api/room-types'),
   217→  })
   218→  const apiRoomTypes: { id: string; name: string; code: string; baseRate: number; bedConfig?: string }[] = roomTypesData?.roomTypes || []
   219→
   220→  const { data, isLoading } = useQuery({
   221→    queryKey: ['arrivals', today],
   222→    queryFn: async () => {
   223→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   224→      return apiFetch(`/api/reservations?${params.toString()}`)
   225→    },
   226→    refetchInterval: 30000,
   227→  })
   228→
   229→  const { data: roomsData } = useQuery({
   230→    queryKey: ['rooms', 'vacant'],
   231→    queryFn: () => apiFetch('/api/rooms'),
   232→  })
   233→
   234→  const arrivals: Arrival[] = data?.reservations || []
   235→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   236→  const availableRooms = allRooms.filter(
   237→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   238→  )
   239→
   240→  // Filter available rooms by selected room type for walk-in
   241→  const walkInFilteredRooms = useMemo(() => {
   242→    if (!walkInForm.roomTypeId) return availableRooms
   243→    const rt = apiRoomTypes.find((t) => t.id === walkInForm.roomTypeId)
   244→    if (!rt) return availableRooms
   245→    return availableRooms.filter((r) => r.type.code === rt.code)
   246→  }, [availableRooms, walkInForm.roomTypeId, apiRoomTypes])
   247→
   248→  // Estimated total for walk-in
   249→  const walkInEstimatedTotal = useMemo(() => {
   250→    if (!walkInForm.roomRate || !walkInForm.checkOutDate) return 0
   251→    const nights = Math.max(1, Math.ceil(
   252→      (walkInForm.checkOutDate.getTime() - new Date().setHours(0, 0, 0, 0)) / (1000 * 60 * 60 * 24)
   253→    ))
   254→    const taxRate = (settings.taxRate as number) || 13
   255→    const subtotal = walkInForm.roomRate * nights
   256→    return subtotal + subtotal * (taxRate / 100)
   257→  }, [walkInForm.roomRate, walkInForm.checkOutDate, settings.taxRate])
   258→
   259→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   260→
   261→  const sortedArrivals = useMemo(() => {
   262→    return [...arrivals].sort((a, b) => {
   263→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   264→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   265→      if (aVip && !bVip) return -1
   266→      if (!aVip && bVip) return 1
   267→      return 0
   268→    })
   269→  }, [arrivals])
   270→
   271→  // ─── Stats ──────────────────────────────────────────────────────────
   272→
   273→  const totalArrivals = arrivals.length
   274→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   275→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   276→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   277→
   278→  // ─── Mutations ──────────────────────────────────────────────────────
   279→
   280→  const assignRoomMutation = useMutation({
   281→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   282→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   283→      if (specialRequests !== undefined) {
   284→        payload.specialRequests = specialRequests
   285→      }
   286→      return apiFetch(`/api/reservations/${reservationId}`, {
   287→        method: 'PATCH',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify(payload),
   290→      })
   291→    },
   292→    onSuccess: (_data, variables) => {
   293→      invalidate.afterReservationChange(queryClient)
   294→      setRoomPickerOpen(false)
   295→      setCheckInDialogOpen(false)
   296→
   297→      // Find the room number for key card dialog
   298→      const room = availableRooms.find((r) => r.id === variables.roomId)
   299→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   300→      openKeyCardDialog(roomNumber)
   301→    },
   302→  })
   303→
   304→  const checkInMutation = useMutation({
   305→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   306→      const payload: Record<string, unknown> = { status: 'checked_in' }
   307→      if (specialRequests !== undefined) {
   308→        payload.specialRequests = specialRequests
   309→      }
   310→      return apiFetch(`/api/reservations/${reservationId}`, {
   311→        method: 'PATCH',
   312→        headers: { 'Content-Type': 'application/json' },
   313→        body: JSON.stringify(payload),
   314→      })
   315→    },
   316→    onSuccess: (_data, variables) => {
   317→      invalidate.afterCheckIn(queryClient)
   318→      setCheckInDialogOpen(false)
   319→
   320→      const roomNumber = selectedArrival?.room?.number || ''
   321→      openKeyCardDialog(roomNumber)
   322→    },
   323→  })
   324→
   325→  const walkInMutation = useMutation({
   326→    mutationFn: async (form: WalkInForm) => {
   327→      // 1. Create guest
   328→      const guestData = await apiFetch('/api/guests', {
   329→        method: 'POST',
   330→        headers: { 'Content-Type': 'application/json' },
   331→        body: JSON.stringify({
   332→          firstName: form.firstName,
   333→          lastName: form.lastName,
   334→          phone: form.phone || null,
   335→          email: form.email || null,
   336→          nationality: form.nationality || null,
   337→          idType: form.idType || null,
   338→          idNumber: form.idNumber || null,
   339→        }),
   340→      })
   341→      const guestId = guestData.guest.id
   342→
   343→      // 2. Resolve room — use selected room, or auto-pick first filtered room
   344→      let room
   345→      if (form.roomId) {
   346→        room = availableRooms.find((r) => r.id === form.roomId)
   347→      } else {
   348→        room = walkInFilteredRooms[0]
   349→      }
   350→      if (!room) throw new Error('No available rooms matching the selected room type')
   351→
   352→      // 3. Create reservation as checked_in (walk-in)
   353→      const checkOutDate = form.checkOutDate || (() => {
   354→        const d = new Date()
   355→        d.setDate(d.getDate() + 1)
   356→        return d
   357→      })()
   358→
   359→      const resData = await apiFetch('/api/reservations', {
   360→        method: 'POST',
   361→        headers: { 'Content-Type': 'application/json' },
   362→        body: JSON.stringify({
   363→          guestId,
   364→          roomId: room.id,
   365→          adults: form.adults,
   366→          children: form.children,
   367→          checkIn: new Date().toISOString(),
   368→          checkOut: checkOutDate.toISOString(),
   369→          roomRate: form.roomRate,
   370→          specialRequests: form.specialRequests || null,
   371→          source: 'walk_in',
   372→          reservationType: 'walk_in',
   373→          guaranteed: false,
   374→        }),
   375→      })
   376→      return { roomNumber: room.number, reservation: resData.reservation }
   377→    },
   378→    onSuccess: (result) => {
   379→      invalidate.afterCheckIn(queryClient)
   380→      setWalkInDialogOpen(false)
   381→      setWalkInForm(initialWalkInForm)
   382→
   383→      // Open key card dialog
   384→      openKeyCardDialog(result.roomNumber)
   385→    },
   386→    onError: (error) => {
   387→      toast.error('Walk-in check-in failed', {
   388→        description: error instanceof Error ? error.message : 'Unknown error',
   389→      })
   390→    },
   391→  })
   392→
   393→  // ─── Handlers ──────────────────────────────────────────────────────
   394→
   395→  const handleAssignRoom = (arrival: Arrival) => {
   396→    setSelectedArrival(arrival)
   397→    setRoomPickerOpen(true)
   398→  }
   399→
   400→  const handleQuickCheckIn = (arrival: Arrival) => {
   401→    if (!arrival.room) {
   402→      setSelectedArrival(arrival)
   403→      setRoomPickerOpen(true)
   404→      return
   405→    }
   406→    setSelectedArrival(arrival)
   407→    resetCheckInForm()
   408→    setCheckInDialogOpen(true)
   409→  }
   410→
   411→  const handleOpenWalkIn = () => {
   412→    setWalkInForm(initialWalkInForm)
   413→    setWalkInDialogOpen(true)
   414→  }
   415→
   416→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   417→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   418→    navigateTo('front-desk', 'check-in')
   419→  }
   420→
   421→  const handleViewLedger = (arrival: Arrival) => {
   422→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   423→      guestId: arrival.guest?.id || '',
   424→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   425→    })
   426→    navigateTo('front-desk', 'guest-ledger')
   427→  }
   428→
   429→  const buildSpecialRequests = (arrival: Arrival): string => {
   430→    const parts: string[] = []
   431→
   432→    // Preserve existing requests
   433→    if (arrival.specialRequests) {
   434→      parts.push(arrival.specialRequests)
   435→    }
   436→
   437→    // Early check-in note
   438→    if (earlyCheckIn) {
   439→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   440→    }
   441→
   442→    // Check-out time override
   443→    if (checkOutTime !== settings.defaultCheckOut) {
   444→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   445→    }
   446→
   447→    // Room preference
   448→    if (roomPreference) {
   449→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   450→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   451→    }
   452→
   453→    // Pillow type
   454→    if (pillowType) {
   455→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   456→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   457→    }
   458→
   459→    // Wake-up call
   460→    if (wakeupCall) {
   461→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   462→    }
   463→
   464→    return parts.join(' | ')
   465→  }
   466→
   467→  const confirmCheckIn = () => {
   468→    if (!selectedArrival) return
   469→
   470→    const updatedRequests = buildSpecialRequests(selectedArrival)
   471→
   472→    if (selectedRoomId) {
   473→      assignRoomMutation.mutate({
   474→        reservationId: selectedArrival.id,
   475→        roomId: selectedRoomId,
   476→        specialRequests: updatedRequests,
   477→      })
   478→    } else if (selectedArrival.room) {
   479→      checkInMutation.mutate({
   480→        reservationId: selectedArrival.id,
   481→        specialRequests: updatedRequests,
   482→      })
   483→    }
   484→  }
   485→
   486→  const openKeyCardDialog = (roomNumber: string) => {
   487→    setKeyCardRoomNumber(roomNumber)
   488→    setKeyCardIssued(false)
   489→    setKeyCardProcessed(false)
   490→    setKeyCardDialogOpen(true)
   491→  }
   492→
   493→  const handleKeyCardConfirm = () => {
   494→    setKeyCardProcessed(true)
   495→    setKeyCardDialogOpen(false)
   496→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   497→      duration: 5000,
   498→      icon: <Mail className="size-4" />,
   499→    })
   500→    setSelectedArrival(null)
   501→    setSelectedRoomId('')
   502→  }
   503→
   504→  const resetCheckInForm = () => {
   505→    setEarlyCheckIn(false)
   506→    setCheckOutTime(settings.defaultCheckOut)
   507→    setRoomPreference('')
   508→    setPillowType('')
   509→    setWakeupCall('')
   510→  }
   511→
   512→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   513→    const roomType = apiRoomTypes.find((rt) => rt.id === roomTypeId)
   514→    setWalkInForm((prev) => ({
   515→      ...prev,
   516→      roomTypeId,
   517→      roomId: '',
   518→      roomRate: roomType?.baseRate || 0,
   519→    }))
   520→  }
   521→
   522→  const handleSelectArrival = (arrival: Arrival) => {
   523→    setSelectedArrival(arrival)
   524→    setSelectedRoomId('')
   525→    setMobileShowDetail(true)
   526→  }
   527→
   528→  const handleBackToList = () => {
   529→    setMobileShowDetail(false)
   530→  }
   531→
   532→  // ─── Render ─────────────────────────────────────────────────────────
   533→
   534→  return (
   535→    <div className="flex flex-col gap-3">
   536→      {/* Header */}
   537→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 shrink-0">
   538→        <div>
   539→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   540→          <p className="text-xs text-muted-foreground">
   541→            Guest check-ins scheduled for {formatDate(today)}
   542→          </p>
   543→        </div>
   544→        <div className="flex items-center gap-2 shrink-0">
   545→          <Badge variant="secondary" className="text-[10px]">
   546→            {totalArrivals} expected
   547→          </Badge>
   548→          {vipArrivals > 0 && (
   549→            <Badge variant="outline" className="text-[10px] border-amber-300 text-amber-700 dark:text-amber-400">
   550→              <Crown className="size-3 mr-0.5" />
   551→              {vipArrivals} VIP
   552→            </Badge>
   553→          )}
   554→          {unassignedArrivals > 0 && (
   555→            <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-600 dark:text-amber-400">
   556→              <AlertTriangle className="size-3 mr-0.5" />
   557→              {unassignedArrivals} unassigned
   558→            </Badge>
   559→          )}
   560→          <Button onClick={handleOpenWalkIn} size="sm" className="gap-1.5">
