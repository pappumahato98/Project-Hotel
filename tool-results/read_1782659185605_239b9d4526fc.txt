    42→  id: string
    43→  confirmationNo: string
    44→  checkIn: string
    45→  checkOut: string
    46→  roomRate: number
    47→  status: string
    48→  creditLimit: number
    49→  room: { number: string } | null

    51→
    52→interface SettlementGuest {
    53→  id: string
    54→  firstName: string
    55→  lastName: string

    58→
    59→interface SettlementTransaction {
    60→  id: string
    61→  transactionType: string
    62→  description: string
    63→  amount: number
    64→  taxAmount: number
    65→  totalAmount: number
    66→  quantity: number
    67→  createdAt: string
    68→}
    69→
    70→interface SettlementPayment {
    71→  id: string
    72→  paymentMethod: string
    73→  amount: number
    74→  reference: string | null
    75→  cardType: string | null
    76→  receivedBy: string | null
    77→  createdAt: string
    78→}
    79→
    80→interface SettlementFolio {
    81→  id: string
    82→  folioType: string
    83→  status: string
    84→  balance: number
    85→  createdAt: string
    86→  updatedAt: string
    87→  reservation: SettlementReservation
    88→  guest: SettlementGuest
    89→  transactions: SettlementTransaction[]
    90→  payments: SettlementPayment[]
    91→}
    92→
    93→interface FolioListResponse {
    94→  folios: SettlementFolio[]
    95→  stats: {
    96→    openFolios: number
    97→    totalOutstanding: number
    98→    todayCharges: number
    99→    todayPayments: number
   100→  } | null
   101→  settings: {
   102→    taxRate: number
   103→    serviceCharge: number
   104→  }
   105→}
   106→
   107→type StatusFilter = 'all' | 'open' | 'ready'
   108→type SortField = 'guestName' | 'roomNumber' | 'balance' | 'transactionCount' | 'updatedAt'
   109→type SortDir = 'asc' | 'desc'

   121→// ─── Helpers ────────────────────────────────────────────────────────────
   122→
   123→function guestDisplayName(guest: SettlementGuest): string {
   124→  return `${guest.firstName} ${guest.lastName}`.trim()
   125→}
   126→
   127→function roomDisplay(reservation: SettlementReservation): string {
   128→  return reservation.room?.number ?? '—'
   129→}
   130→
   131→function computeTotals(folio: SettlementFolio) {
   132→  const totalCharges = folio.transactions.reduce((s, t) => s + t.totalAmount, 0)
   133→  const totalPayments = folio.payments.reduce((s, p) => s + p.amount, 0)
   134→  return { totalCharges, totalPayments, balance: totalCharges - totalPayments }
   135→}
   136→
   137→function isReadyForSettlement(folio: SettlementFolio): boolean {
   138→  // A folio is "ready" if it has an open status and its reservation is checked_in (departure pending)
   139→  return folio.status === 'open' && folio.reservation.status === 'checked_in'
   140→}

   143→
   144→export function SettlementView() {
   145→  const queryClient = useQueryClient()
   146→
   147→  // ── State ──
   148→  const [search, setSearch] = useState('')
   149→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all')
   150→  const [sortField, setSortField] = useState<SortField>('balance')
   151→  const [sortDir, setSortDir] = useState<SortDir>('desc')
   152→
   153→  // Dialog state
   154→  const [selectedFolio, setSelectedFolio] = useState<SettlementFolio | null>(null)
   155→  const [dialogOpen, setDialogOpen] = useState(false)
   156→  const [confirmOpen, setConfirmOpen] = useState(false)
   157→
   158→  // Payment form state
   159→  const [paymentMethod, setPaymentMethod] = useState('cash')
   160→  const [paymentAmount, setPaymentAmount] = useState('')
   161→  const [paymentReference, setPaymentReference] = useState('')
   162→  const [cardType, setCardType] = useState('')
   163→
   164→  // ── Query ──
   165→  const { data, isLoading, isError, error } = useQuery<FolioListResponse>({
   166→    queryKey: qk.folios(),
   167→    queryFn: () => apiFetch<FolioListResponse>('/api/folio'),
   168→  })
   169→
   170→  // ── Mutations ──
   171→  const recordPaymentMutation = useMutation({
   172→    mutationFn: ({ folioId, paymentMethod, amount, reference, cardType }: {
   173→      folioId: string
   174→      paymentMethod: string
   175→      amount: number
   176→      reference?: string
   177→      cardType?: string
   178→    }) => apiFetch('/api/folio/' + folioId, {
   179→      method: 'POST',
   180→      headers: { 'Content-Type': 'application/json' },
   181→      body: JSON.stringify({
   182→        type: 'payment',
   183→        paymentMethod,
   184→        amount,
   185→        reference: reference || null,
   186→        cardType: cardType || null,
   187→        receivedBy: 'System',
   188→      }),
   189→    }),
   190→    onSuccess: () => {
   191→      invalidate.afterFolioChange(queryClient)
   192→      toast.success('Payment recorded successfully')
   193→    },
   194→    onError: (err: Error) => {
   195→      toast.error(err.message || 'Failed to record payment')
   196→    },
   197→  })
   198→
   199→  const settleMutation = useMutation({
   200→    mutationFn: (folioId: string) => apiFetch('/api/folio/' + folioId, {
   201→      method: 'PATCH',
   202→      headers: { 'Content-Type': 'application/json' },
   203→      body: JSON.stringify({ status: 'closed' }),
   204→    }),
   205→    onSuccess: () => {
   206→      invalidate.afterFolioChange(queryClient)
   207→      invalidate.afterCheckout(queryClient)
   208→      toast.success('Folio settled and closed successfully')
   209→      closeDialog()
   210→    },
   211→    onError: (err: Error) => {
   212→      toast.error(err.message || 'Failed to settle folio')
   213→    },
   214→  })
   215→
   216→  // ── Derived data ──
   217→  const openFolios = useMemo(() => {
   218→    if (!data) return []
   219→    return data.folios.filter(
   220→      f => f.status === 'open' || (f.status === 'locked')
   221→    )
   222→  }, [data])
   223→
   224→  const filteredFolios = useMemo(() => {
   225→    let list = openFolios
   226→
   227→    // Status filter
   228→    if (statusFilter === 'open') {
   229→      list = list.filter(f => !isReadyForSettlement(f))
   230→    } else if (statusFilter === 'ready') {
   231→      list = list.filter(f => isReadyForSettlement(f))
   232→    }
   233→
   234→    // Search
   235→    if (search.trim()) {
   236→      const q = search.toLowerCase().trim()
   237→      list = list.filter(f => {
   238→        const name = guestDisplayName(f.guest).toLowerCase()
   239→        const room = roomDisplay(f.reservation).toLowerCase()
   240→        const conf = f.reservation.confirmationNo.toLowerCase()
   241→        return name.includes(q) || room.includes(q) || conf.includes(q)
   242→      })
   243→    }
   244→
   245→    // Sort
   246→    list = [...list].sort((a, b) => {
   247→      let cmp = 0
   248→      switch (sortField) {
   249→        case 'guestName': {
   250→          const an = guestDisplayName(a.guest)
   251→          const bn = guestDisplayName(b.guest)
   252→          cmp = an.localeCompare(bn)
   253→          break
   254→        }
   255→        case 'roomNumber': {
   256→          cmp = roomDisplay(a.reservation).localeCompare(roomDisplay(b.reservation), undefined, { numeric: true })
   257→          break
   258→        }
   259→        case 'balance':
   260→          cmp = computeTotals(a).balance - computeTotals(b).balance
   261→          break
   262→        case 'transactionCount':
   263→          cmp = a.transactions.length - b.transactions.length
   264→          break
   265→        case 'updatedAt':
   266→          cmp = new Date(a.updatedAt).getTime() - new Date(b.updatedAt).getTime()
   267→          break
   268→      }
   269→      return sortDir === 'asc' ? cmp : -cmp
   270→    })
   271→
   272→    return list
   273→  }, [openFolios, statusFilter, search, sortField, sortDir])
   274→
   275→  const totals = useMemo(() => {
   276→    const totalBalance = filteredFolios.reduce((s, f) => s + computeTotals(f).balance, 0)
   277→    const readyCount = filteredFolios.filter(f => isReadyForSettlement(f)).length
   278→    return { totalBalance, readyCount, count: filteredFolios.length }
   279→  }, [filteredFolios])
   280→
   281→  // ── Handlers ──
   282→  const handleSort = useCallback((field: SortField) => {
   283→    if (sortField === field) {
   284→      setSortDir(d => d === 'asc' ? 'desc' : 'asc')
   285→    } else {
   286→      setSortField(field)
   287→      setSortDir('desc')
   288→    }
   289→  }, [sortField])
   290→
   291→  const openSettleDialog = useCallback((folio: SettlementFolio) => {
   292→    setSelectedFolio(folio)
   293→    const computed = computeTotals(folio)
   294→    setPaymentAmount(String(Math.max(0, computed.balance)))
   295→    setPaymentMethod('cash')
   296→    setPaymentReference('')
   297→    setCardType('')
   298→    setDialogOpen(true)
   299→  }, [])
   300→
   301→  const closeDialog = useCallback(() => {
   302→    setDialogOpen(false)
   303→    setConfirmOpen(false)
   304→    setSelectedFolio(null)
   305→  }, [])
   306→
   307→  const handleRecordPayment = useCallback(async () => {
   308→    if (!selectedFolio) return
   309→    const amount = parseFloat(paymentAmount)
   310→    if (isNaN(amount) || amount <= 0) {
   311→      toast.error('Please enter a valid payment amount')
   312→      return
   313→    }
   314→    await recordPaymentMutation.mutateAsync({
   315→      folioId: selectedFolio.id,
   316→      paymentMethod,
   317→      amount,
   318→      reference: paymentReference || undefined,
   319→      cardType: paymentMethod === 'card' ? cardType : undefined,
   320→    })
   321→    // Refetch the folio detail to get updated balance
   322→    queryClient.invalidateQueries({ queryKey: qk.folioDetail(selectedFolio.id) as unknown[] })
   323→    // Refresh the list
   324→    queryClient.invalidateQueries({ queryKey: qk.folios() as unknown[] })
   325→  }, [selectedFolio, paymentAmount, paymentMethod, paymentReference, cardType, recordPaymentMutation, queryClient])
   326→
   327→  const handleSettle = useCallback(async () => {
   328→    if (!selectedFolio) return
   329→    await settleMutation.mutateAsync(selectedFolio.id)
   330→  }, [selectedFolio, settleMutation])
   331→
   332→  const SortableHeader = ({ field, children }: { field: SortField; children: React.ReactNode }) => (
   333→    <Button
   334→      variant="ghost"
   335→      size="sm"
   336→      className="-ml-3 h-8 text-xs font-semibold text-muted-foreground hover:text-foreground"
   337→      onClick={() => handleSort(field)}
   338→    >
   339→      {children}
   340→      <ArrowUpDown className="ml-1 h-3 w-3" />
   341→    </Button>
   342→  )
   343→
   344→  // ── Render ──
   345→  return (
   346→    <div className="space-y-4">
   347→      {/* ── Header ── */}
   348→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   349→        <div>
   350→          <h1 className="text-xl font-semibold tracking-tight flex items-center gap-2">
   351→            <Calculator className="h-5 w-5 text-emerald-600" />
   352→            Settlement
   353→          </h1>
   354→          <p className="text-sm text-muted-foreground mt-0.5">
   355→            Close out open folios and process final payments at checkout
   356→          </p>
   357→        </div>
   358→      </div>
   359→
   360→      {/* ── Summary Cards ── */}
   361→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   362→        <Card className="p-4">
   363→          <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Open Folios</div>
   364→          <div className="text-2xl font-bold mt-1">{totals.count}</div>
   365→        </Card>
   366→        <Card className="p-4">
   367→          <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Ready to Settle</div>
   368→          <div className="text-2xl font-bold mt-1 text-emerald-600">{totals.readyCount}</div>
   369→        </Card>
   370→        <Card className="p-4">
   371→          <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Total Outstanding</div>
   372→          <div className="text-2xl font-bold mt-1">{formatCurrency(totals.totalBalance)}</div>
   373→        </Card>
   374→        <Card className="p-4">
   375→          <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider">Today&apos;s Revenue</div>
   376→          <div className="text-2xl font-bold mt-1">
   377→            {data?.stats ? formatCurrency(data.stats.todayPayments) : '—'}
   378→          </div>
   379→        </Card>
   380→      </div>
   381→
   382→      {/* ── Filter Bar ── */}
   383→      <div className="flex flex-col sm:flex-row gap-3">
   384→        <div className="relative flex-1 max-w-sm">
   385→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   386→          <Input
   387→            placeholder="Search guest name, room, or confirmation #..."
   388→            value={search}
   389→            onChange={e => setSearch(e.target.value)}
   390→            className="pl-9 h-9"
   391→          />
   392→          {search && (
   393→            <button
   394→              onClick={() => setSearch('')}
   395→              className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   396→            >
   397→              <X className="h-3.5 w-3.5" />
   398→            </button>
   399→          )}
   400→        </div>
   401→        <Select value={statusFilter} onValueChange={v => setStatusFilter(v as StatusFilter)}>
   402→          <SelectTrigger className="w-full sm:w-40 h-9">
   403→            <SelectValue />
   404→          </SelectTrigger>
   405→          <SelectContent>
   406→            <SelectItem value="all">All Statuses</SelectItem>
   407→            <SelectItem value="ready">Ready to Settle</SelectItem>
   408→            <SelectItem value="open">Open</SelectItem>
   409→          </SelectContent>
   410→        </Select>
   411→      </div>
   412→
   413→      {/* ── Table ── */}
   414→      {isLoading && (
   415→        <div className="space-y-3">
   416→          <Skeleton className="h-10 w-full" />
   417→          {Array.from({ length: 5 }).map((_, i) => (
   418→            <Skeleton key={i} className="h-14 w-full" />
   419→          ))}
   420→        </div>
   421→      )}
   422→
   423→      {isError && (
   424→        <Card className="p-6 text-center border-red-200 bg-red-50 dark:border-red-900 dark:bg-red-950">
   425→          <AlertTriangle className="h-8 w-8 text-red-500 mx-auto mb-2" />
   426→          <p className="text-sm text-red-700 dark:text-red-300 font-medium">Failed to load folio data</p>
   427→          <p className="text-xs text-red-500 dark:text-red-400 mt-1">{(error as Error)?.message}</p>
   428→        </Card>
   429→      )}
   430→
   431→      {!isLoading && !isError && (
   432→        <div className="rounded-lg border bg-card overflow-hidden">
   433→          <div className="overflow-x-auto">
   434→            <Table>
   435→              <TableHeader>
   436→                <TableRow className="bg-muted/50">
   437→                  <TableHead className="text-xs font-semibold">
   438→                    <SortableHeader field="guestName">Guest</SortableHeader>
   439→                  </TableHead>
   440→                  <TableHead className="text-xs font-semibold">
   441→                    <SortableHeader field="roomNumber">Room</SortableHeader>
   442→                  </TableHead>
   443→                  <TableHead className="text-xs font-semibold hidden md:table-cell">Confirmation #</TableHead>
   444→                  <TableHead className="text-xs font-semibold hidden lg:table-cell">Check-Out</TableHead>
   445→                  <TableHead className="text-xs font-semibold text-right">
   446→                    <SortableHeader field="balance">Balance</SortableHeader>
   447→                  </TableHead>
   448→                  <TableHead className="text-xs font-semibold text-center">
   449→                    <SortableHeader field="transactionCount">Txns</SortableHeader>
   450→                  </TableHead>
   451→                  <TableHead className="text-xs font-semibold text-center">Status</TableHead>
   452→                  <TableHead className="text-xs font-semibold text-right">Action</TableHead>
   453→                </TableRow>
   454→              </TableHeader>
   455→              <TableBody>
   456→                {filteredFolios.length === 0 ? (
   457→                  <TableRow>
   458→                    <TableCell colSpan={8} className="h-32 text-center">
   459→                      <div className="flex flex-col items-center gap-1.5 text-muted-foreground">
   460→                        <Receipt className="h-8 w-8 opacity-40" />
   461→                        <p className="text-sm font-medium">No folios to settle</p>
   462→                        <p className="text-xs">
   463→                          {search || statusFilter !== 'all'
   464→                            ? 'Try adjusting your filters'
   465→                            : 'All open folios have been settled'}
   466→                        </p>
   467→                      </div>
   468→                    </TableCell>
   469→                  </TableRow>
   470→                ) : (
   471→                  filteredFolios.map(folio => {
   472→                    const computed = computeTotals(folio)
   473→                    const ready = isReadyForSettlement(folio)
   474→                    return (
   475→                      <TableRow key={folio.id} className="group">
   476→                        {/* Guest */}
   477→                        <TableCell>
   478→                          <div className="flex flex-col">
   479→                            <span className="font-medium text-sm">
   480→                              {guestDisplayName(folio.guest)}
   481→                            </span>
   482→                            <span className="text-xs text-muted-foreground capitalize">
   483→                              {folio.folioType} folio
   484→                            </span>
   485→                          </div>
   486→                        </TableCell>
   487→
   488→                        {/* Room */}
   489→                        <TableCell>
   490→                          <span className="font-mono text-sm font-medium">
   491→                            {roomDisplay(folio.reservation)}
   492→                          </span>
   493→                        </TableCell>
   494→
   495→                        {/* Confirmation */}
   496→                        <TableCell className="hidden md:table-cell">
   497→                          <span className="font-mono text-xs text-muted-foreground">
   498→                            {folio.reservation.confirmationNo}
   499→                          </span>
   500→                        </TableCell>
   501→
   502→                        {/* Check-out */}
   503→                        <TableCell className="hidden lg:table-cell">
   504→                          <span className="text-xs text-muted-foreground">
   505→                            {formatDate(folio.reservation.checkOut)}
   506→                          </span>
   507→                        </TableCell>
   508→
   509→                        {/* Balance */}
   510→                        <TableCell className="text-right">
   511→                          <span className={cn(
   512→                            'font-mono font-semibold text-sm',
   513→                            computed.balance > 0 ? 'text-amber-600' : 'text-emerald-600',
   514→                          )}>
   515→                            {formatCurrency(computed.balance)}
   516→                          </span>
   517→                        </TableCell>
   518→
   519→                        {/* Transaction count */}
   520→                        <TableCell className="text-center">
   521→                          <span className="text-xs text-muted-foreground font-mono">
   522→                            {folio.transactions.length + folio.payments.length}
   523→                          </span>
   524→                        </TableCell>
   525→
   526→                        {/* Status */}
   527→                        <TableCell className="text-center">
   528→                          {ready ? (
   529→                            <Badge className="bg-emerald-100 text-emerald-700 border-emerald-300 dark:bg-emerald-950 dark:text-emerald-300 dark:border-emerald-800 text-xs font-medium">
   530→                              <CheckCircle2 className="h-3 w-3 mr-1" />
   531→                              Ready
   532→                            </Badge>
   533→                          ) : (
   534→                            <StatusBadge status={folio.status} />
   535→                          )}
   536→                        </TableCell>
   537→
   538→                        {/* Action */}
   539→                        <TableCell className="text-right">
   540→                          <Button
   541→                            size="sm"
   542→                            className={cn(
   543→                              'h-7 text-xs',
   544→                              ready
   545→                                ? 'bg-emerald-600 hover:bg-emerald-700 text-white'
   546→                                : 'bg-amber-600 hover:bg-amber-700 text-white',
   547→                            )}
   548→                            onClick={() => openSettleDialog(folio)}
   549→                          >
   550→                            <CreditCard className="h-3 w-3 mr-1" />
   551→                            Settle
   552→                          </Button>
   553→                        </TableCell>
   554→                      </TableRow>
   555→                    )
   556→                  })
   557→                )}
   558→              </TableBody>
   559→            </Table>
   560→          </div>
