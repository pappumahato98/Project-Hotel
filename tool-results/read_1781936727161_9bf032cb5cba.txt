    92→function fetchEmployees(department?: string, status?: string) {
    93→  const params = new URLSearchParams()
    94→  if (department) params.set('department', department)
    95→  if (status) params.set('status', status)
    96→  return apiFetch(`/api/employees?${params.toString()}`)
    97→}
    98→
    99→export function EmployeesView() {
   100→  const queryClient = useQueryClient()
   101→  const [filterDept, setFilterDept] = useState('')
   102→  const [filterStatus, setFilterStatus] = useState('')
   103→  const [searchQuery, setSearchQuery] = useState('')
   104→  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null)
   105→
   106→  // Add/Edit dialog state
   107→  const [showFormDialog, setShowFormDialog] = useState(false)
   108→  const [editingEmployee, setEditingEmployee] = useState<Employee | null>(null)
   109→  const [form, setForm] = useState<EmployeeFormData>(emptyForm)
   110→
   111→  // Delete dialog state
   112→  const [deleteTarget, setDeleteTarget] = useState<Employee | null>(null)
   113→
   114→  const { data, isLoading } = useQuery({
   115→    queryKey: ['employees', filterDept, filterStatus],
   116→    queryFn: () => fetchEmployees(filterDept || undefined, filterStatus || undefined),
   117→  })
   118→
   119→  const departments = data?.departmentBreakdown ? Object.keys(data.departmentBreakdown) : []
   120→
   121→  const filteredEmployees = data?.employees?.filter((emp: Employee) => {
   122→    if (!searchQuery) return true
   123→    const q = searchQuery.toLowerCase()
   124→    return (
   125→      emp.firstName.toLowerCase().includes(q) ||
   126→      emp.lastName.toLowerCase().includes(q) ||
   127→      emp.department.toLowerCase().includes(q) ||
   128→      emp.position.toLowerCase().includes(q) ||
   129→      emp.email?.toLowerCase().includes(q)
   130→    )
   131→  })
   132→
   133→  // Stats
   134→  const activeCount = data?.employees?.filter((e: Employee) => e.status === 'active').length ?? 0
   135→  const onLeaveCount = data?.employees?.filter((e: Employee) => e.status === 'on_leave').length ?? 0
   136→  const totalSalary = data?.employees?.reduce((sum: number, e: Employee) => sum + (e.salary ?? 0), 0) ?? 0
   137→
   138→  // Create mutation
   139→  const createMutation = useMutation({
   140→    mutationFn: async (formData: EmployeeFormData) => {
   141→      return apiFetch('/api/employees', {
   142→        method: 'POST',
   143→        headers: { 'Content-Type': 'application/json' },
   144→        body: JSON.stringify({
   145→          ...formData,
   146→          salary: formData.salary ? Number(formData.salary) : null,
   147→          hireDate: formData.hireDate || null,
   148→        }),
   149→      })
   150→    },
   151→    onSuccess: () => {
   152→      queryClient.invalidateQueries({ queryKey: ['employees'] })
   153→      toast.success('Employee added successfully')
   154→      closeFormDialog()
   155→    },
   156→    onError: () => {
   157→      toast.error('Failed to add employee')
   158→    },
   159→  })
   160→
   161→  // Update mutation
   162→  const updateMutation = useMutation({
   163→    mutationFn: async ({ id, formData }: { id: string; formData: EmployeeFormData }) => {
   164→      return apiFetch(`/api/employees/${id}`, {
   165→        method: 'PATCH',
   166→        headers: { 'Content-Type': 'application/json' },
   167→        body: JSON.stringify({
   168→          ...formData,
   169→          salary: formData.salary ? Number(formData.salary) : null,
   170→          hireDate: formData.hireDate || null,
   171→        }),
   172→      })
   173→    },
   174→    onSuccess: () => {
   175→      queryClient.invalidateQueries({ queryKey: ['employees'] })
   176→      toast.success('Employee updated successfully')
   177→      closeFormDialog()
   178→    },
   179→    onError: () => {
   180→      toast.error('Failed to update employee')
   181→    },
   182→  })
   183→
   184→  // Delete mutation
   185→  const deleteMutation = useMutation({
   186→    mutationFn: async (id: string) => {
   187→      return apiFetch(`/api/employees/${id}`, { method: 'DELETE' })
   188→    },
   189→    onSuccess: () => {
   190→      queryClient.invalidateQueries({ queryKey: ['employees'] })
   191→      toast.success('Employee deleted successfully')
   192→      setDeleteTarget(null)
   193→      if (selectedEmployee?.id === deleteTarget?.id) setSelectedEmployee(null)
   194→    },
   195→    onError: () => {
   196→      toast.error('Failed to delete employee')
   197→    },
   198→  })
   199→
   200→  function openCreateDialog() {
   201→    setEditingEmployee(null)
   202→    setForm(emptyForm)
   203→    setShowFormDialog(true)
   204→  }
   205→
   206→  function openEditDialog(emp: Employee) {
   207→    setEditingEmployee(emp)
   208→    setForm({
   209→      firstName: emp.firstName,
   210→      lastName: emp.lastName,
   211→      email: emp.email ?? '',
   212→      phone: emp.phone ?? '',
   213→      department: emp.department,
   214→      position: emp.position,
   215→      role: emp.role,
   216→      hireDate: emp.hireDate ? emp.hireDate.split('T')[0] : '',
   217→      salary: emp.salary ? String(emp.salary) : '',
   218→      status: emp.status,
   219→    })
   220→    setShowFormDialog(true)
   221→  }
   222→
   223→  function closeFormDialog() {
   224→    setShowFormDialog(false)
   225→    setEditingEmployee(null)
   226→    setForm(emptyForm)
   227→  }
   228→
   229→  function handleSubmit(e: React.FormEvent) {
   230→    e.preventDefault()
   231→    if (!form.firstName || !form.lastName) {
   232→      toast.error('First name and last name are required')
   233→      return
   234→    }
   235→    if (editingEmployee) {
   236→      updateMutation.mutate({ id: editingEmployee.id, formData: form })
   237→    } else {
   238→      createMutation.mutate(form)
   239→    }
   240→  }
   241→
   242→  function handleDelete() {
   243→    if (deleteTarget) {
   244→      deleteMutation.mutate(deleteTarget.id)
   245→    }
   246→  }
   247→
   248→  const isSubmitting = createMutation.isPending || updateMutation.isPending
   249→
   250→  return (
   251→    <div className="flex flex-1 flex-col gap-2 p-6 overflow-y-auto">
   252→      {/* Header */}
   253→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   254→        <div>
   255→          <h1 className="text-sm font-semibold text-gray-700 dark:text-gray-200 tracking-tight">Staff Directory</h1>
   256→          <p className="text-xs text-muted-foreground">Manage and view all hotel staff members</p>
   257→        </div>
   258→        <div className="flex items-center gap-2">
   259→          <Badge variant="outline" className="text-[11px]">
   260→            {data?.total ?? 0} employees
   261→          </Badge>
   262→          <Button onClick={openCreateDialog} className="gap-2">
   263→            <Plus className="h-4 w-4" />
   264→            Add Employee
   265→          </Button>
   266→        </div>
   267→      </div>
   268→
   269→      {/* Summary Cards */}
   270→      <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-4">
   271→        <Card className="p-2.5">
   272→          <div className="flex items-center gap-2">
   273→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-950">
   274→              <Users className="h-5 w-5 text-emerald-600" />
   275→            </div>
   276→            <div>
   277→              <p className="text-xs text-muted-foreground">Total Staff</p>
   278→              <p className="text-lg font-bold">{data?.total ?? 0}</p>
   279→            </div>
   280→          </div>
   281→        </Card>
   282→        <Card className="p-2.5">
   283→          <div className="flex items-center gap-2">
   284→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-green-100 dark:bg-green-950">
   285→              <UserCheck className="h-5 w-5 text-green-600" />
   286→            </div>
   287→            <div>
   288→              <p className="text-xs text-muted-foreground">Active</p>
   289→              <p className="text-lg font-bold">{activeCount}</p>
   290→            </div>
   291→          </div>
   292→        </Card>
   293→        <Card className="p-2.5">
   294→          <div className="flex items-center gap-2">
   295→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-950">
   296→              <UserX className="h-5 w-5 text-amber-600" />
   297→            </div>
   298→            <div>
   299→              <p className="text-xs text-muted-foreground">On Leave</p>
   300→              <p className="text-lg font-bold">{onLeaveCount}</p>
   301→            </div>
   302→          </div>
   303→        </Card>
   304→        <Card className="p-2.5">
   305→          <div className="flex items-center gap-2">
   306→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-950">
   307→              <DollarSign className="h-5 w-5 text-teal-600" />
   308→            </div>
   309→            <div>
   310→              <p className="text-xs text-muted-foreground">Avg Salary</p>
   311→              <p className="text-lg font-bold">{activeCount > 0 ? formatCurrency(Math.round(totalSalary / (data?.employees?.length ?? 1))) : '—'}</p>
   312→            </div>
   313→          </div>
   314→        </Card>
   315→      </div>
   316→
   317→      {/* Department Breakdown */}
   318→      {departments.length > 0 && (
   319→        <div className="grid gap-2 sm:grid-cols-3 lg:grid-cols-5">
   320→          {departments.map((dept: string) => (
   321→            <Card key={dept} className="p-4 hover:bg-muted/50 transition-colors cursor-pointer" onClick={() => setFilterDept(filterDept === dept ? '' : dept)}>
   322→              <div className="flex items-center justify-between">
   323→                <p className="text-xs text-muted-foreground truncate">{dept}</p>
   324→                <Building2 className="h-4 w-4 text-muted-foreground shrink-0" />
   325→              </div>
   326→              <p className="mt-1 text-lg font-bold">{data?.departmentBreakdown?.[dept] ?? 0}</p>
   327→            </Card>
   328→          ))}
   329→        </div>
   330→      )}
   331→
   332→      {/* Filters */}
   333→      <div className="flex flex-wrap items-center gap-2">
   334→        <div className="relative flex-1 min-w-[200px] max-w-sm">
   335→          <Search className="absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   336→          <Input
   337→            placeholder="Search employees..."
   338→            value={searchQuery}
   339→            onChange={(e) => setSearchQuery(e.target.value)}
   340→            className="h-7 text-xs pl-9"
   341→          />
   342→        </div>
   343→        <select
   344→          value={filterDept}
   345→          onChange={(e) => setFilterDept(e.target.value)}
   346→          className="h-7 rounded-md border bg-background px-3 text-xs"
   347→        >
   348→          <option value="">All Departments</option>
   349→          {departments.map((d: string) => (
   350→            <option key={d} value={d}>{d}</option>
   351→          ))}
   352→        </select>
   353→        <select
   354→          value={filterStatus}
   355→          onChange={(e) => setFilterStatus(e.target.value)}
   356→          className="h-7 rounded-md border bg-background px-3 text-xs"
   357→        >
   358→          <option value="">All Status</option>
   359→          <option value="active">Active</option>
   360→          <option value="on_leave">On Leave</option>
   361→          <option value="resigned">Resigned</option>
   362→          <option value="terminated">Terminated</option>
   363→        </select>
   364→        {(filterDept || filterStatus || searchQuery) && (
   365→          <button
   366→            type="button"
   367→            onClick={() => { setFilterDept(''); setFilterStatus(''); setSearchQuery('') }}
   368→            className="inline-flex items-center justify-center size-7 rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors shrink-0"
   369→            title="Clear all filters"
   370→          >
   371→            <X className="size-3.5" strokeWidth={2.5} />
   372→          </button>
   373→        )}
   374→      </div>
   375→
   376→      {/* Employee Table */}
   377→      <Card className="py-0">
   378→        <CardContent className="p-0">
   379→          <ScrollArea className="max-h-[500px]">
   380→            <Table>
   381→              <TableHeader>
   382→                <TableRow>
   383→                  <TableHead>Name</TableHead>
   384→                  <TableHead>Department</TableHead>
   385→                  <TableHead>Position</TableHead>
   386→                  <TableHead>Role</TableHead>
   387→                  <TableHead>Status</TableHead>
   388→                  <TableHead className="hidden md:table-cell">Phone</TableHead>
   389→                  <TableHead className="hidden lg:table-cell">Hire Date</TableHead>
   390→                  <TableHead className="hidden lg:table-cell">Salary</TableHead>
   391→                  <TableHead className="w-[100px]">Actions</TableHead>
   392→                </TableRow>
   393→              </TableHeader>
   394→              <TableBody>
   395→                {isLoading ? (
   396→                  Array.from({ length: 8 }).map((_, i) => (
   397→                    <TableRow key={i}>
   398→                      {Array.from({ length: 9 }).map((_, j) => (
   399→                        <TableCell key={j}>
   400→                          <Skeleton className="h-4 w-[100px]" />
   401→                        </TableCell>
   402→                      ))}
   403→                    </TableRow>
   404→                  ))
   405→                ) : filteredEmployees?.length === 0 ? (
   406→                  <TableRow>
   407→                    <TableCell colSpan={9} className="h-24 text-center text-muted-foreground">
   408→                      No employees found.
   409→                    </TableCell>
   410→                  </TableRow>
   411→                ) : (
   412→                  filteredEmployees?.map((emp: Employee) => (
   413→                    <TableRow
   414→                      key={emp.id}
   415→                      className="cursor-pointer hover:bg-muted/50"
   416→                      onClick={() => setSelectedEmployee(emp)}
   417→                    >
   418→                      <TableCell className="font-medium">
   419→                        {emp.firstName} {emp.lastName}
   420→                      </TableCell>
   421→                      <TableCell>{emp.department}</TableCell>
   422→                      <TableCell>{emp.position}</TableCell>
   423→                      <TableCell>
   424→                        <span className="capitalize text-xs">{emp.role.replace('_', ' ')}</span>
   425→                      </TableCell>
   426→                      <TableCell>
   427→                        <StatusBadge status={emp.status} />
   428→                      </TableCell>
   429→                      <TableCell className="hidden md:table-cell">
   430→                        {emp.phone ?? '—'}
   431→                      </TableCell>
   432→                      <TableCell className="hidden lg:table-cell">
   433→                        {emp.hireDate ? formatDateShort(emp.hireDate) : '—'}
   434→                      </TableCell>
   435→                      <TableCell className="hidden lg:table-cell">
   436→                        {emp.salary ? formatCurrency(emp.salary) : '—'}
   437→                      </TableCell>
   438→                      <TableCell>
   439→                        <div className="flex items-center gap-1" onClick={(e) => e.stopPropagation()}>
   440→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => openEditDialog(emp)}>
   441→                            <Pencil className="h-3.5 w-3.5" />
   442→                          </Button>
   443→                          <Button variant="ghost" size="icon" className="h-7 w-7 text-destructive hover:text-destructive" onClick={() => setDeleteTarget(emp)}>
   444→                            <Trash2 className="h-3.5 w-3.5" />
   445→                          </Button>
   446→                        </div>
   447→                      </TableCell>
   448→                    </TableRow>
   449→                  ))
   450→                )}
   451→              </TableBody>
   452→            </Table>
   453→          </ScrollArea>
   454→        </CardContent>
   455→      </Card>
   456→
   457→      {/* Employee Detail Dialog */}
   458→      <Dialog open={!!selectedEmployee} onOpenChange={() => setSelectedEmployee(null)}>
   459→        <DialogContent className="sm:max-w-md">
   460→          <DialogHeader>
   461→            <DialogTitle>
   462→              {selectedEmployee?.firstName} {selectedEmployee?.lastName}
   463→            </DialogTitle>
   464→          </DialogHeader>
   465→          {selectedEmployee && (
   466→            <div className="space-y-4">
   467→              <div className="flex items-center gap-4">
   468→                <div className="flex h-16 w-16 items-center justify-center rounded-full bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 text-xl font-bold">
   469→                  {selectedEmployee.firstName[0]}{selectedEmployee.lastName[0]}
   470→                </div>
   471→                <div>
   472→                  <p className="text-lg font-semibold">{selectedEmployee.firstName} {selectedEmployee.lastName}</p>
   473→                  <p className="text-sm text-muted-foreground">{selectedEmployee.position}</p>
   474→                  <StatusBadge status={selectedEmployee.status} />
   475→                </div>
   476→              </div>
   477→              <div className="space-y-2 text-sm">
   478→                <div className="flex items-center gap-2">
   479→                  <Building2 className="h-4 w-4 text-muted-foreground" />
   480→                  <span className="text-muted-foreground">Department:</span>
   481→                  <span className="font-medium">{selectedEmployee.department}</span>
   482→                </div>
   483→                <div className="flex items-center gap-2">
   484→                  <Users className="h-4 w-4 text-muted-foreground" />
   485→                  <span className="text-muted-foreground">Role:</span>
   486→                  <span className="capitalize font-medium">{selectedEmployee.role.replace('_', ' ')}</span>
   487→                </div>
   488→                {selectedEmployee.phone && (
   489→                  <div className="flex items-center gap-2">
   490→                    <Search className="h-4 w-4 text-muted-foreground" />
   491→                    <span className="text-muted-foreground">Phone:</span>
   492→                    <span className="font-medium">{selectedEmployee.phone}</span>
   493→                  </div>
   494→                )}
   495→                {selectedEmployee.email && (
   496→                  <div className="flex items-center gap-2">
   497→                    <Search className="h-4 w-4 text-muted-foreground" />
   498→                    <span className="text-muted-foreground">Email:</span>
   499→                    <span className="font-medium">{selectedEmployee.email}</span>
   500→                  </div>
   501→                )}
   502→                {selectedEmployee.hireDate && (
   503→                  <div className="flex items-center gap-2">
   504→                    <Building2 className="h-4 w-4 text-muted-foreground" />
   505→                    <span className="text-muted-foreground">Hire Date:</span>
   506→                    <span className="font-medium">{formatDateShort(selectedEmployee.hireDate)}</span>
   507→                  </div>

   512→                    <span className="text-muted-foreground">Salary:</span>
   513→                    <span className="font-medium">{formatCurrency(selectedEmployee.salary)}</span>
   514→                  </div>

   516→              </div>
   517→              <div className="flex gap-2 pt-2">
   518→                <Button variant="outline" className="flex-1 gap-2" onClick={() => { openEditDialog(selectedEmployee); setSelectedEmployee(null) }}>
   519→                  <Pencil className="h-4 w-4" /> Edit
   520→                </Button>
   521→                <Button variant="outline" className="flex-1 gap-2 text-destructive hover:text-destructive" onClick={() => { setDeleteTarget(selectedEmployee); setSelectedEmployee(null) }}>
   522→                  <Trash2 className="h-4 w-4" /> Delete
   523→                </Button>
   524→              </div>
   525→            </div>
   526→          )}
   527→        </DialogContent>
   528→      </Dialog>
   529→
   530→      {/* Create / Edit Dialog */}
   531→      <Dialog open={showFormDialog} onOpenChange={(open) => { if (!open) closeFormDialog() }}>
   532→        <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   533→          <DialogHeader>
   534→            <DialogTitle>{editingEmployee ? 'Edit Employee' : 'Add New Employee'}</DialogTitle>
   535→            <DialogDescription>
   536→              {editingEmployee ? 'Update employee details below.' : 'Fill in the details to add a new employee.'}
