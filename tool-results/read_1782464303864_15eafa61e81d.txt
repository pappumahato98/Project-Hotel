   194→function generatePreviewConfirmation(): string {
   195→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
   196→  let result = 'MRD-'
   197→  for (let i = 0; i < 6; i++) {
   198→    result += chars.charAt(Math.floor(Math.random() * chars.length))
   199→  }
   200→  return result
   201→}

   203→function formatDayOfWeek(dateStr: string): string {
   204→  const d = new Date(dateStr)
   205→  if (isNaN(d.getTime())) return ''
   206→  return d.toLocaleDateString('en-US', { weekday: 'short' })
   207→}
   208→
   209→function formatDateWithDay(dateStr: string): string {
   210→  if (!dateStr) return ''
   211→  const d = new Date(dateStr)
   212→  const day = formatDayOfWeek(dateStr)
   213→  const formatted = d.toLocaleDateString('en-US', { day: '2-digit', month: 'short' })
   214→  return `${day} ${formatted}`
   215→}

   217→function useDebounce<T>(value: T, delay: number): T {
   218→  const [debounced, setDebounced] = useState(value)
   219→  useEffect(() => {
   220→    const timer = setTimeout(() => setDebounced(value), delay)
   221→    return () => clearTimeout(timer)
   222→  }, [value, delay])
   223→  return debounced
   224→}
