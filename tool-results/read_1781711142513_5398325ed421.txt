   212→   212→    },

   252→   252→
   253→   253→  // ── Derived data ────────────────────────────────────────────────────
   254→   254→  const stats = listData?.stats
   255→   255→  const pendingSummary = pendingData?.summary
   256→   256→  const postings = listData?.postings || []
   257→   257→  const pendingReservations = pendingData?.pendingReservations || []
   258→   258→  const pagination = listData?.pagination
   259→   259→
   260→   260→  const isWorking = voidMutation.isPending || postMutation.isPending || bulkPostMutation.isPending
