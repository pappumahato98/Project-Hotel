   121→   121→
   122→   122→  // Late checkout

   124→   124→

   127→   127→  const [paymentAmount, setPaymentAmount] = useState('')
   128→   128→  const [paymentReference, setPaymentReference] = useState('')
   129→   129→
   130→   130→  // Receipt data
   131→   131→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   132→   132→
   133→   133→  // ── Fetch today's departures (both checked_in AND checked_out with checkOut = today) ──
   134→   134→  const { data, isLoading } = useQuery({
   135→   135→    queryKey: ['departures', today],
   136→   136→    queryFn: async () => {
   137→   137→      // Fetch both statuses so we show completed check-outs too
   138→   138→      const params = new URLSearchParams({ status: 'checked_in,checked_out', checkOutDate: today })
   139→   139→      return apiFetch(`/api/reservations?${params.toString()}`)

   141→   141→    refetchInterval: 30000,
   142→   142→  })

   145→   145→
   146→   146→  // ── Fallback: if today has zero departures, also fetch upcoming (next 3 days) ──
   147→   147→  const tomorrow = (() => {
   148→   148→    const d = new Date(); d.setDate(d.getDate() + 1)
   149→   149→    const y = d.getFullYear(); const m = String(d.getMonth()+1).padStart(2,'0'); const day = String(d.getDate()).padStart(2,'0')
   150→   150→    return `${y}-${m}-${day}`
   151→   151→  })()
   152→   152→  const threeDaysOut = (() => {
   153→   153→    const d = new Date(); d.setDate(d.getDate() + 4)
   154→   154→    const y = d.getFullYear(); const m = String(d.getMonth()+1).padStart(2,'0'); const day = String(d.getDate()).padStart(2,'0')
   155→   155→    return `${y}-${m}-${day}`
   156→   156→  })()

   159→   159→    queryKey: ['departures-upcoming', today],
   160→   160→    queryFn: async () => {
   161→   161→      const params = new URLSearchParams({
   162→   162→        status: 'checked_in,confirmed',
   163→   163→        dateFrom: tomorrow,
   164→   164→        dateTo: threeDaysOut,
   165→   165→      })
   166→   166→      return apiFetch(`/api/reservations?${params.toString()}`)
   167→   167→    },
   168→   168→    refetchInterval: 60000,
   169→   169→    enabled: departures.length === 0,
   170→   170→  })

   172→   172→  const upcomingDepartures: Departure[] = upcomingData?.reservations || []

   188→   188→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   189→   189→  )
   190→   190→

   195→   195→    mutationFn: async (reservationId: string) => {
   196→   196→      return apiFetch(`/api/reservations/${reservationId}`, {
   197→   197→        method: 'PATCH',
   198→   198→        headers: { 'Content-Type': 'application/json' },
   199→   199→        body: JSON.stringify({ status: 'checked_out' }),
   200→   200→      })
   201→   201→    },
   202→   202→    onSuccess: () => {
   203→   203→      invalidate.afterCheckout(queryClient)
   204→   204→    },
   205→   205→  })
   206→   206→
   207→   207→  // Payment mutation
   208→   208→  const paymentMutation = useMutation({
   209→   209→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   210→   210→      return apiFetch(`/api/folio/${folioId}`, {
   211→   211→        method: 'POST',
   212→   212→        headers: { 'Content-Type': 'application/json' },
   213→   213→        body: JSON.stringify({ type: 'payment', paymentMethod: method, amount, reference }),
   214→   214→      })
   215→   215→    },
   216→   216→    onSuccess: () => {
   217→   217→      invalidate.afterFolioChange(queryClient)
   218→   218→      setPaymentDialogOpen(false)
   219→   219→      setPaymentAmount('')
   220→   220→      setPaymentReference('')
   221→   221→      toast.success('Payment recorded successfully')
   222→   222→    },
   223→   223→    onError: () => {
   224→   224→      toast.error('Failed to record payment')
   225→   225→    },
   226→   226→  })
   227→   227→
   228→   228→  // Late checkout mutation
   229→   229→  const lateCheckoutMutation = useMutation({
   230→   230→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   231→   231→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   232→   232→        method: 'PATCH',
   233→   233→        headers: { 'Content-Type': 'application/json' },
   234→   234→        body: JSON.stringify({ checkOut: newCheckOut }),
   235→   235→      })

   237→   237→      if (surcharge > 0) {
   238→   238→        await apiFetch(`/api/folio`, {
   239→   239→          method: 'POST',
   240→   240→          headers: { 'Content-Type': 'application/json' },
   241→   241→          body: JSON.stringify({
   242→   242→            reservationId,
   243→   243→            type: 'charge',
   244→   244→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   245→   245→            amount: surcharge,
   246→   246→          }),
   247→   247→        })
   248→   248→      }
   249→   249→      return result
   250→   250→    },
   251→   251→    onSuccess: () => {
   252→   252→      invalidate.afterReservationChange(queryClient)
   253→   253→      invalidate.afterFolioChange(queryClient)
   254→   254→      setLateCheckoutDialogOpen(false)
   255→   255→      toast.success('Late checkout request processed successfully')
   256→   256→    },
   257→   257→    onError: () => {
   258→   258→      toast.error('Failed to process late checkout request')
   259→   259→    },
   260→   260→  })
   261→   261→
   262→   262→  // Batch checkout mutation
   263→   263→  const batchCheckoutMutation = useMutation({
   264→   264→    mutationFn: async (ids: string[]) => {
   265→   265→      const results = []
   266→   266→      for (let i = 0; i < ids.length; i++) {
   267→   267→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   268→   268→          method: 'PATCH',
   269→   269→          headers: { 'Content-Type': 'application/json' },
   270→   270→          body: JSON.stringify({ status: 'checked_out' }),
   271→   271→        })
   272→   272→        results.push(result)
   273→   273→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   274→   274→      }
   275→   275→      return results
   276→   276→    },
   277→   277→    onSuccess: (_data, ids) => {
   278→   278→      invalidate.afterCheckout(queryClient)
   279→   279→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   280→   280→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   281→   281→    },
   282→   282→    onError: () => {
   283→   283→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   284→   284→    },
   285→   285→  })
   286→   286→

   288→   288→

   290→   290→

   293→   293→      onSuccess: () => {
   294→   294→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   295→   295→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   296→   296→      },
   297→   297→    })
   298→   298→  }
   299→   299→

   302→   302→    setFolioDialogOpen(true)

   314→   314→  }
   315→   315→
   316→   316→  const handleViewLedger = (dep: Departure) => {
   317→   317→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   318→   318→      guestId: dep.guest?.id || '',
   319→   319→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   320→   320→    })
   321→   321→    navigateTo('front-desk', 'guest-ledger')
   322→   322→  }
   323→   323→
   324→   324→  const handleFolioSettleCheckout = () => {
   325→   325→    setFolioDialogOpen(false)
   326→   326→    setCheckoutDialogOpen(true)
   327→   327→  }
   328→   328→
   329→   329→  const handleQuickCheckout = (dep: Departure) => {

   331→   331→    setCheckoutDialogOpen(true)

   336→   336→    setLateCheckoutOption('14:00')

   380→   380→      amount,
   381→   381→      method: paymentMethod,
   382→   382→      reference: paymentReference,
   383→   383→    })

   406→   406→  }
   407→   407→
   408→   408→  const handlePrintReceipt = () => {
   409→   409→    toast.success('Receipt sent to printer')
   410→   410→  }
   411→   411→
   412→   412→  const handleEmailReceipt = () => {
   413→   413→    toast.success('Receipt emailed to guest')
   414→   414→  }
   415→   415→
   416→   416→  // ─── Computed values ──────────────────────────────────────────────
   417→   417→
   418→   418→  const lateCheckoutFee = selectedDeparture
   419→   419→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   420→   420→    : 0
   421→   421→
   422→   422→  // ─── Render ────────────────────────────────────────────────────────
   423→   423→
   424→   424→  return (
   425→   425→    <div className="flex flex-col gap-2">
   426→   426→      {/* Header */}
   427→   427→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   428→   428→        <div>
   429→   429→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">

   432→   432→          <p className="text-xs text-muted-foreground">

   443→   443→          >
   444→   444→            <Zap className="size-4 mr-1.5" />

   446→   446→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   447→   447→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   448→   448→          </Button>
   449→   449→        )}
   450→   450→      </div>
   451→   451→
   452→   452→      {/* Departures List */}
   453→   453→      <Card className="py-0">
   454→   454→        <CardContent className="p-0">
   455→   455→          <div className="max-h-[600px] overflow-y-auto">
   456→   456→            <Table>
   457→   457→              <TableHeader>
   458→   458→                <TableRow>
