   172→function generatePreviewConfirmation(): string {
   173→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
   174→  let result = 'MRD-'
   175→  for (let i = 0; i < 6; i++) {
   176→    result += chars.charAt(Math.floor(Math.random() * chars.length))
   177→  }
   178→  return result
   179→}
