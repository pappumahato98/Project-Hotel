     1→'use client'
     2→
     3→import { toast } from 'sonner'
     4→import { useState } from 'react'
     5→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     6→import { apiFetch } from '@/lib/api'
     7→import { Card, CardContent } from '@/components/ui/card'
     8→import { Badge } from '@/components/ui/badge'
     9→import { Button } from '@/components/ui/button'
    10→import { Input } from '@/components/ui/input'
    11→import { Label } from '@/components/ui/label'
    12→import { Textarea } from '@/components/ui/textarea'
    13→import {
    14→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    15→} from '@/components/ui/table'
    16→import {
    17→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    18→} from '@/components/ui/select'
    19→import {
    20→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    21→} from '@/components/ui/dialog'
    22→import {
    23→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    24→} from '@/components/ui/alert-dialog'
    25→import { Skeleton } from '@/components/ui/skeleton'
    26→import { ScrollArea } from '@/components/ui/scroll-area'
    27→import { Separator } from '@/components/ui/separator'
    28→import {
    29→  ClipboardList, AlertTriangle, CheckCircle, Plus, Eye, Pencil,
    30→  ThumbsUp, ThumbsDown, PackageCheck, Trash2, Search, X, MinusCircle, PlusCircle,
    31→} from 'lucide-react'
    32→import { cn } from '@/lib/utils'
    33→
    34→// ── Types ────────────────────────────────────────────────────
    35→interface RequisitionItem {
    36→  name: string
    37→  quantity: string
    38→  unit: string
    39→}
    40→
    41→interface Requisition {
    42→  id: string
    43→  requestDate: string
    44→  department: string
    45→  requestor: string
    46→  items: RequisitionItem[]
    47→  status: string
    48→  priority: string
    49→  totalItems: number
    50→  notes?: string
    51→  approvedBy?: string
    52→  approvedAt?: string
    53→}
    54→
    55→const DEPARTMENTS = [
    56→  'Kitchen', 'Housekeeping', 'Front Desk', 'Engineering', 'F&B', 'Spa', 'Laundry',
    57→]
    58→
    59→const PRIORITIES = ['High', 'Normal', 'Low']
    60→
    61→const EMPTY_REQ_ITEM = { name: '', quantity: '', unit: 'pcs' }
    62→
    63→const EMPTY_REQ_FORM = {
    64→  department: 'Kitchen',
    65→  requestor: '',
    66→  priority: 'Normal',
    67→  notes: '',
    68→  items: [{ ...EMPTY_REQ_ITEM }],
    69→}
    70→
    71→type ReqForm = typeof EMPTY_REQ_FORM
    72→
    73→// ── API helpers ──────────────────────────────────────────────
    74→function fetchRequisitions() {
    75→  return apiFetch('/api/requisitions')
    76→}
    77→
    78→// ── Component ────────────────────────────────────────────────
    79→export function RequisitionsView() {
    80→  const queryClient = useQueryClient()
    81→  const [searchQuery, setSearchQuery] = useState('')
    82→  const [statusFilter, setStatusFilter] = useState('all')
    83→
    84→  // Dialog states
    85→  const [createOpen, setCreateOpen] = useState(false)
    86→  const [editOpen, setEditOpen] = useState(false)
    87→  const [viewOpen, setViewOpen] = useState(false)
    88→  const [deleteOpen, setDeleteOpen] = useState(false)
    89→  const [selectedReq, setSelectedReq] = useState<Requisition | null>(null)
    90→  const [form, setForm] = useState<ReqForm>(EMPTY_REQ_FORM)
    91→
    92→  // ── Query ──────────────────────────────────────────────────
    93→  const { data, isLoading } = useQuery({
    94→    queryKey: ['requisitions'],
    95→    queryFn: fetchRequisitions,
    96→  })
    97→
    98→  const filteredReqs = data?.requisitions?.filter((req: Requisition) => {
    99→    if (statusFilter !== 'all' && req.status !== statusFilter) return false
   100→    if (!searchQuery) return true
   101→    const q = searchQuery.toLowerCase()
   102→    return (
   103→      req.requestor.toLowerCase().includes(q) ||
   104→      req.department.toLowerCase().includes(q) ||
   105→      req.id.toLowerCase().includes(q)
   106→    )
   107→  })
   108→
   109→  // ── Mutations ───────────────────────────────────────────────
   110→  const createMutation = useMutation({
   111→    mutationFn: async (body: ReqForm) => {
   112→      return apiFetch('/api/requisitions', {
   113→        method: 'POST',
   114→        headers: { 'Content-Type': 'application/json' },
   115→        body: JSON.stringify({
   116→          department: body.department,
   117→          requestor: body.requestor,
   118→          priority: body.priority.toLowerCase(),
   119→          notes: body.notes,
   120→          items: body.items.filter((i) => i.name && i.quantity),
   121→          totalItems: body.items.filter((i) => i.name && i.quantity).length,
   122→        }),
   123→      })
   124→    },
   125→    onSuccess: () => {
   126→      queryClient.invalidateQueries({ queryKey: ['requisitions'] })
   127→      toast.success('Requisition created successfully')
   128→      setCreateOpen(false)
   129→      setForm(EMPTY_REQ_FORM)
   130→    },
   131→    onError: () => toast.error('Failed to create requisition'),
   132→  })
   133→
   134→  const updateMutation = useMutation({
   135→    mutationFn: async ({ id, ...body }: { id: string; [key: string]: unknown }) => {
   136→      return apiFetch('/api/requisitions', {
   137→        method: 'PATCH',
   138→        headers: { 'Content-Type': 'application/json' },
   139→        body: JSON.stringify({ id, ...body }),
   140→      })
   141→    },
   142→    onSuccess: () => {
   143→      queryClient.invalidateQueries({ queryKey: ['requisitions'] })
   144→      setEditOpen(false)
   145→      setSelectedReq(null)
   146→      setForm(EMPTY_REQ_FORM)
   147→    },
   148→    onError: () => toast.error('Failed to update requisition'),
   149→  })
   150→
   151→  const statusMutation = useMutation({
   152→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   153→      return apiFetch('/api/requisitions', {
   154→        method: 'PATCH',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({ id, status }),
   157→      })
   158→    },
   159→    onSuccess: (_data, variables) => {
   160→      queryClient.invalidateQueries({ queryKey: ['requisitions'] })
   161→      toast.success(`Requisition ${variables.status === 'approved' ? 'approved' : variables.status === 'rejected' ? 'rejected' : 'marked as received'}`)
   162→      setSelectedReq(null)
   163→    },
   164→    onError: () => toast.error('Failed to update requisition status'),
   165→  })
   166→
   167→  const deleteMutation = useMutation({
   168→    mutationFn: async (id: string) => {
   169→      return apiFetch('/api/requisitions', {
   170→        method: 'PATCH',
   171→        headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({ id, status: 'rejected', notes: 'Deleted by user' }),
   173→      })
   174→    },
   175→    onSuccess: () => {
   176→      queryClient.invalidateQueries({ queryKey: ['requisitions'] })
   177→      toast.success('Requisition deleted')
   178→      setDeleteOpen(false)
   179→      setSelectedReq(null)
   180→    },
   181→    onError: () => toast.error('Failed to delete requisition'),
   182→  })
   183→
   184→  // ── Helpers ─────────────────────────────────────────────────
   185→  const openCreate = () => {
   186→    setForm(EMPTY_REQ_FORM)
   187→    setCreateOpen(true)
   188→  }
   189→
   190→  const openEdit = (req: Requisition) => {
   191→    setSelectedReq(req)
   192→    setForm({
   193→      department: req.department,
   194→      requestor: req.requestor,
   195→      priority: req.priority.charAt(0).toUpperCase() + req.priority.slice(1),
   196→      notes: req.notes || '',
   197→      items: req.items.length > 0 ? req.items : [{ ...EMPTY_REQ_ITEM }],
   198→    })
   199→    setEditOpen(true)
   200→  }
   201→
   202→  const openView = (req: Requisition) => {
   203→    setSelectedReq(req)
   204→    setViewOpen(true)
   205→  }
   206→
   207→  const openDelete = (req: Requisition) => {
   208→    setSelectedReq(req)
   209→    setDeleteOpen(true)
   210→  }
   211→
   212→  const updateFormItems = (newItems: RequisitionItem[]) => {
   213→    setForm((f) => ({ ...f, items: newItems }))
   214→  }
   215→
   216→  // ── Colors ────────────────────────────────────────────────────
   217→  const statusColors: Record<string, string> = {
   218→    pending: 'border-yellow-300 bg-yellow-50 text-yellow-700 dark:border-yellow-800 dark:bg-yellow-950 dark:text-yellow-300',
   219→    approved: 'border-green-300 bg-green-50 text-green-700 dark:border-green-800 dark:bg-green-950 dark:text-green-300',
   220→    rejected: 'border-red-300 bg-red-50 text-red-700 dark:border-red-800 dark:bg-red-950 dark:text-red-300',
   221→    received: 'border-blue-300 bg-blue-50 text-blue-700 dark:border-blue-800 dark:bg-blue-950 dark:text-blue-300',
   222→  }
   223→
   224→  const priorityColors: Record<string, string> = {
   225→    high: 'border-red-300 bg-red-50 text-red-700 dark:border-red-800 dark:bg-red-950 dark:text-red-300',
   226→    normal: 'border-gray-300 bg-gray-50 text-gray-600 dark:border-gray-700 dark:bg-gray-900 dark:text-gray-400',
   227→    low: 'border-blue-300 bg-blue-50 text-blue-700 dark:border-blue-800 dark:bg-blue-950 dark:text-blue-300',
   228→  }
   229→
   230→  return (
   231→    <div className="space-y-2">
   232→      {/* Header */}
   233→      <div className="flex items-center justify-between">
   234→        <div>
   235→          <h1 className="text-sm font-semibold text-gray-700 dark:text-gray-200 tracking-tight">Requisitions</h1>
   236→          <p className="text-xs text-muted-foreground">Purchase requisitions and supply requests</p>
   237→        </div>
   238→        <div className="flex items-center gap-2">
   239→          <Badge variant="outline" className="text-[11px]">
   240→            {data?.total ?? 0} requisitions
   241→          </Badge>
   242→          <Button className="gap-1" onClick={openCreate}>
   243→            <Plus className="h-4 w-4" />
   244→            New Requisition
   245→          </Button>
   246→        </div>
   247→      </div>
   248→
   249→      {/* Summary */}
   250→      <div className="grid gap-2 sm:grid-cols-3">
   251→        <Card className="p-2.5">
   252→          <div className="flex items-center gap-2">
   253→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-yellow-100 dark:bg-yellow-950">
   254→              <AlertTriangle className="h-5 w-5 text-yellow-600" />
   255→            </div>
   256→            <div>
   257→              <p className="text-xs text-muted-foreground">Pending</p>
   258→              <p className="text-lg font-bold">{data?.summary?.pending ?? '—'}</p>
   259→            </div>
   260→          </div>
   261→        </Card>
   262→        <Card className="p-2.5">
   263→          <div className="flex items-center gap-2">
   264→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-green-100 dark:bg-green-950">
   265→              <CheckCircle className="h-5 w-5 text-green-600" />
   266→            </div>
   267→            <div>
   268→              <p className="text-xs text-muted-foreground">Approved</p>
   269→              <p className="text-lg font-bold">{data?.summary?.approved ?? '—'}</p>
   270→            </div>
   271→          </div>
   272→        </Card>
   273→        <Card className="p-2.5">
   274→          <div className="flex items-center gap-2">
   275→            <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-blue-100 dark:bg-blue-950">
   276→              <ClipboardList className="h-5 w-5 text-blue-600" />
   277→            </div>
   278→            <div>
   279→              <p className="text-xs text-muted-foreground">Received</p>
   280→              <p className="text-lg font-bold">{data?.summary?.received ?? '—'}</p>
   281→            </div>
   282→          </div>
   283→        </Card>
   284→      </div>
   285→
   286→      {/* Filters */}
   287→      <div className="flex flex-wrap items-center gap-2">
   288→        <div className="relative flex-1 min-w-[200px] max-w-sm">
   289→          <Search className="absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   290→          <Input
   291→            placeholder="Search by requestor or department..."
   292→            value={searchQuery}
   293→            onChange={(e) => setSearchQuery(e.target.value)}
   294→            className={cn('h-7 text-xs pl-9', searchQuery && 'pr-7')}
   295→          />
   296→          {searchQuery && (
   297→            <button
   298→              type="button"
   299→              className="absolute right-1.5 top-1/2 -translate-y-1/2 size-5 inline-flex items-center justify-center rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors"
   300→              onClick={() => setSearchQuery('')}
   301→            >
   302→              <X className="size-3" strokeWidth={2.5} />
   303→            </button>
   304→          )}
   305→        </div>
   306→        <div className="relative">
   307→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   308→            <SelectTrigger className={cn('w-[130px] h-7 text-xs', statusFilter !== 'all' && 'pr-8')}>
   309→              <SelectValue placeholder="Status" />
   310→            </SelectTrigger>
   311→            <SelectContent>
   312→              <SelectItem value="all">All Status</SelectItem>
   313→              <SelectItem value="pending">Pending</SelectItem>
   314→              <SelectItem value="approved">Approved</SelectItem>
   315→              <SelectItem value="rejected">Rejected</SelectItem>
   316→              <SelectItem value="received">Received</SelectItem>
   317→            </SelectContent>
   318→          </Select>
   319→          {statusFilter !== 'all' && (
   320→            <button
   321→              type="button"
   322→              className="absolute right-1 top-1/2 -translate-y-1/2 size-5 inline-flex items-center justify-center rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors z-10"
   323→              onClick={(e) => { e.stopPropagation(); setStatusFilter('all') }}
   324→            >
   325→              <X className="size-3" strokeWidth={2.5} />
   326→            </button>
   327→          )}
   328→        </div>
   329→        {(searchQuery || statusFilter !== 'all') && (
   330→          <button
   331→            type="button"
   332→            onClick={() => { setSearchQuery(''); setStatusFilter('all') }}
   333→            className="inline-flex items-center justify-center size-7 rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors shrink-0"
   334→            title="Clear all filters"
   335→          >
   336→            <X className="size-3.5" strokeWidth={2.5} />
   337→          </button>
   338→        )}
   339→      </div>
   340→
   341→      {/* Requisitions Table */}
   342→      <Card className="py-0">
   343→        <CardContent className="p-0">
   344→          <ScrollArea className="max-h-[500px]">
   345→            <Table>
   346→              <TableHeader>
   347→                <TableRow>
   348→                  <TableHead>ID</TableHead>
   349→                  <TableHead>Date</TableHead>
   350→                  <TableHead>Department</TableHead>
   351→                  <TableHead>Requestor</TableHead>
   352→                  <TableHead>Items</TableHead>
   353→                  <TableHead>Priority</TableHead>
   354→                  <TableHead>Status</TableHead>
   355→                  <TableHead className="w-[140px]">Actions</TableHead>
   356→                </TableRow>
   357→              </TableHeader>
   358→              <TableBody>
   359→                {isLoading ? (
   360→                  Array.from({ length: 6 }).map((_, i) => (
   361→                    <TableRow key={i}>
   362→                      {Array.from({ length: 8 }).map((_, j) => (
   363→                        <TableCell key={j}>
   364→                          <Skeleton className="h-4 w-[80px]" />
   365→                        </TableCell>
   366→                      ))}
   367→                    </TableRow>
   368→                  ))
   369→                ) : filteredReqs?.length === 0 ? (
   370→                  <TableRow>
   371→                    <TableCell colSpan={8} className="h-24 text-center text-muted-foreground">
   372→                      No requisitions found.
   373→                    </TableCell>
   374→                  </TableRow>
   375→                ) : (
   376→                  filteredReqs?.map((req: Requisition) => (
   377→                    <TableRow key={req.id}>
   378→                      <TableCell className="font-mono text-xs">{req.id.slice(-8)}</TableCell>
   379→                      <TableCell className="text-xs">
   380→                        {new Date(req.requestDate).toLocaleDateString('en-GB', { day: '2-digit', month: 'short' })}
   381→                      </TableCell>
   382→                      <TableCell className="text-xs">{req.department}</TableCell>
   383→                      <TableCell className="text-xs">{req.requestor}</TableCell>
   384→                      <TableCell>
   385→                        <div className="space-y-0.5">
   386→                          {req.items.slice(0, 2).map((item, idx) => (
   387→                            <p key={idx} className="text-xs text-muted-foreground">
   388→                              {item.name} ({item.quantity} {item.unit})
   389→                            </p>
   390→                          ))}
   391→                          {req.items.length > 2 && (
   392→                            <p className="text-xs text-muted-foreground">+{req.items.length - 2} more</p>
   393→                          )}
   394→                        </div>
   395→                      </TableCell>
   396→                      <TableCell>
   397→                        <Badge variant="outline" className={priorityColors[req.priority] ?? ''}>
   398→                          <span className="capitalize">{req.priority}</span>
   399→                        </Badge>
   400→                      </TableCell>
   401→                      <TableCell>
   402→                        <Badge variant="outline" className={statusColors[req.status] ?? ''}>
   403→                          <span className="capitalize">{req.status}</span>
   404→                        </Badge>
   405→                      </TableCell>
   406→                      <TableCell>
   407→                        <div className="flex items-center gap-1">
   408→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => openView(req)} title="View">
   409→                            <Eye className="h-3.5 w-3.5" />
   410→                          </Button>
   411→                          {req.status === 'pending' && (
   412→                            <>
   413→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => openEdit(req)} title="Edit">
   414→                                <Pencil className="h-3.5 w-3.5" />
   415→                              </Button>
   416→                              <Button variant="ghost" size="icon" className="h-7 w-7 text-green-600 hover:text-green-700" onClick={() => statusMutation.mutate({ id: req.id, status: 'approved' })} title="Approve">
   417→                                <ThumbsUp className="h-3.5 w-3.5" />
   418→                              </Button>
   419→                              <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500 hover:text-red-600" onClick={() => statusMutation.mutate({ id: req.id, status: 'rejected' })} title="Reject">
   420→                                <ThumbsDown className="h-3.5 w-3.5" />
   421→                              </Button>
   422→                              <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500 hover:text-red-600" onClick={() => openDelete(req)} title="Delete">
   423→                                <Trash2 className="h-3.5 w-3.5" />
   424→                              </Button>
   425→                            </>
   426→                          )}
   427→                          {req.status === 'approved' && (
   428→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-blue-600 hover:text-blue-700" onClick={() => statusMutation.mutate({ id: req.id, status: 'received' })} title="Mark Received">
   429→                              <PackageCheck className="h-3.5 w-3.5" />
   430→                            </Button>
   431→                          )}
   432→                        </div>
   433→                      </TableCell>
   434→                    </TableRow>
   435→                  ))
   436→                )}
   437→              </TableBody>
   438→            </Table>
   439→          </ScrollArea>
   440→        </CardContent>
   441→      </Card>
   442→
   443→      {/* ── Create Requisition Dialog ────────────────────────── */}
   444→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   445→        <DialogContent className="sm:max-w-lg max-h-[85vh] overflow-y-auto">
   446→          <DialogHeader>
   447→            <DialogTitle>New Requisition</DialogTitle>
   448→            <DialogDescription>Create a new supply requisition.</DialogDescription>
   449→          </DialogHeader>
   450→          <div className="space-y-4">
   451→            <div className="grid gap-4 sm:grid-cols-2">
   452→              <div className="space-y-2">
   453→                <Label>Department *</Label>
   454→                <Select value={form.department} onValueChange={(v) => setForm((f) => ({ ...f, department: v }))}>
   455→                  <SelectTrigger><SelectValue /></SelectTrigger>
   456→                  <SelectContent>
   457→                    {DEPARTMENTS.map((d) => (
   458→                      <SelectItem key={d} value={d}>{d}</SelectItem>
   459→                    ))}
   460→                  </SelectContent>
   461→                </Select>
   462→              </div>
   463→              <div className="space-y-2">
   464→                <Label>Requestor *</Label>
   465→                <Input placeholder="e.g., Chef Raj" value={form.requestor} onChange={(e) => setForm((f) => ({ ...f, requestor: e.target.value }))} />
   466→              </div>
   467→              <div className="space-y-2">
   468→                <Label>Priority *</Label>
   469→                <Select value={form.priority} onValueChange={(v) => setForm((f) => ({ ...f, priority: v }))}>
   470→                  <SelectTrigger><SelectValue /></SelectTrigger>
   471→                  <SelectContent>
   472→                    {PRIORITIES.map((p) => (
   473→                      <SelectItem key={p} value={p}>{p}</SelectItem>
   474→                    ))}
   475→                  </SelectContent>
   476→                </Select>
   477→              </div>
   478→            </div>
   479→
   480→            {/* Items list */}
   481→            <div className="space-y-3">
   482→              <div className="flex items-center justify-between">
   483→                <Label className="text-sm font-medium">Items *</Label>
   484→                <Button variant="outline" size="sm" className="h-7 text-xs gap-1" onClick={() => updateFormItems([...form.items, { ...EMPTY_REQ_ITEM }])}>
   485→                  <PlusCircle className="h-3 w-3" /> Add Item
   486→                </Button>
   487→              </div>
   488→              {form.items.map((item, idx) => (
   489→                <div key={idx} className="flex items-start gap-2">
   490→                  <div className="flex-1 space-y-2">
   491→                    <div className="grid grid-cols-[1fr_70px_60px] gap-2">
   492→                      <Input placeholder="Item name" value={item.name} onChange={(e) => {
   493→                        const newItems = [...form.items]
   494→                        newItems[idx] = { ...newItems[idx], name: e.target.value }
   495→                        updateFormItems(newItems)
   496→                      }} />
   497→                      <Input placeholder="Qty" value={item.quantity} onChange={(e) => {
   498→                        const newItems = [...form.items]
   499→                        newItems[idx] = { ...newItems[idx], quantity: e.target.value }
