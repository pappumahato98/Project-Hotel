   525→            toast.error(err.error || 'Failed to update reservation')

   528→          }
   529→        }
   530→      }

   532→      // Check in

   558→    } catch (err) {

   560→      toast.error('An error occurred during check-in')
   561→    } finally {
   562→      setIsSubmitting(false)
   563→    }
   564→  }
