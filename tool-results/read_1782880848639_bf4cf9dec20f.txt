   100→   100→
   101→   101→export function DeparturesView() {
   102→   102→  const queryClient = useQueryClient()
   103→   103→  const { settings } = useSettingsStore()
   104→   104→  const { preferences } = usePreferencesStore()
   105→   105→  const { navigateTo } = useNavigationStore()
   106→   106→  const today = getTodayString()
   107→   107→
   108→   108→  // Dialog states
   109→   109→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   110→   110→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   111→   111→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   112→   112→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   113→   113→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   114→   114→
   115→   115→  // Selected departure
   116→   116→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   117→   117→
   118→   118→  // Late checkout
   119→   119→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   120→   120→
   121→   121→  // Payment
   122→   122→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   123→   123→  const [paymentAmount, setPaymentAmount] = useState('')
   124→   124→  const [paymentReference, setPaymentReference] = useState('')
   125→   125→
   126→   126→  // Receipt data
   127→   127→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   128→   128→
   129→   129→  // Fetch today's departures (checked_in with checkOut = today)
   130→   130→  const { data, isLoading } = useQuery({
   131→   131→    queryKey: ['departures', today],
   132→   132→    queryFn: async () => {
   133→   133→      // Fetch both: today's departures AND overdue departures
   134→   134→      const [todayData, overdueData] = await Promise.all([
   135→   135→        apiFetch(`/api/reservations?status=checked_in&checkOutDate=${today}`),
   136→   136→        // Fetch overdue: checked_in with checkOut < today
   137→   137→        apiFetch(`/api/reservations?status=checked_in&checkOutBefore=${today}`),
   138→   138→      ])
   139→   139→      // Merge and deduplicate
   140→   140→      const todayReservations = todayData?.reservations || []
   141→   141→      const overdueReservations = overdueData?.reservations || []
   142→   142→      const seen = new Set<string>()
   143→   143→      const merged = [...todayReservations, ...overdueReservations].filter((r: { id: string }) => {
   144→   144→        if (seen.has(r.id)) return false
   145→   145→        seen.add(r.id)
   146→   146→        return true
   147→   147→      })
   148→   148→      return { reservations: merged }
   149→   149→    },
   150→   150→    refetchInterval: 30000,
   151→   151→  })
   152→   152→
   153→   153→  const departures: Departure[] = data?.reservations || []
   154→   154→
   155→   155→  // Stats
   156→   156→  const overdueCount = departures.filter((d) => new Date(d.checkOut) < new Date(today)).length
   157→   157→  const todayCount = departures.length - overdueCount
   158→   158→  const totalDepartures = departures.length
   159→   159→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   160→   160→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   161→   161→  const outstandingBalance = departures.reduce((sum, d) => {
   162→   162→    return sum + (d.folios[0]?.balance || 0)
   163→   163→  }, 0)
   164→   164→
   165→   165→  // Zero-balance pending guests (for express & batch checkout)
   166→   166→  const zeroBalancePending = departures.filter(
   167→   167→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   168→   168→  )
   169→   169→
   170→   170→  // ─── Mutations ─────────────────────────────────────────────────────
   171→   171→
   172→   172→  // Checkout mutation
   173→   173→  const checkoutMutation = useMutation({
   174→   174→    mutationFn: async (reservationId: string) => {
   175→   175→      return apiFetch(`/api/reservations/${reservationId}`, {
   176→   176→        method: 'PATCH',
   177→   177→        headers: { 'Content-Type': 'application/json' },
   178→   178→        body: JSON.stringify({ status: 'checked_out' }),
   179→   179→      })
   180→   180→    },
   181→   181→    onSuccess: () => {
   182→   182→      invalidate.afterCheckout(queryClient)
   183→   183→    },
   184→   184→  })
   185→   185→
   186→   186→  // Payment mutation
   187→   187→  const paymentMutation = useMutation({
   188→   188→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   189→   189→      return apiFetch(`/api/folio/${folioId}`, {
   190→   190→        method: 'POST',
   191→   191→        headers: { 'Content-Type': 'application/json' },
   192→   192→        body: JSON.stringify({ type: 'payment', paymentMethod: method, amount, reference }),
   193→   193→      })
   194→   194→    },
   195→   195→    onSuccess: () => {
   196→   196→      invalidate.afterFolioChange(queryClient)
   197→   197→      setPaymentDialogOpen(false)
   198→   198→      setPaymentAmount('')
   199→   199→      setPaymentReference('')
   200→   200→      toast.success('Payment recorded successfully')
   201→   201→    },
   202→   202→    onError: () => {
   203→   203→      toast.error('Failed to record payment')
   204→   204→    },
   205→   205→  })
   206→   206→
   207→   207→  // Late checkout mutation
   208→   208→  const lateCheckoutMutation = useMutation({
   209→   209→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   210→   210→      // First get the folio ID for this reservation
   211→   211→      const reservation = departures.find((d) => d.id === reservationId)
   212→   212→      const folioId = reservation?.folios[0]?.id
   213→   213→
   214→   214→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   215→   215→        method: 'PATCH',
   216→   216→        headers: { 'Content-Type': 'application/json' },
   217→   217→        body: JSON.stringify({ checkOut: newCheckOut }),
   218→   218→      })
   219→   219→      // Add surcharge to folio
   220→   220→      if (surcharge > 0 && folioId) {
   221→   221→        await apiFetch(`/api/folio/${folioId}`, {
   222→   222→          method: 'POST',
   223→   223→          headers: { 'Content-Type': 'application/json' },
   224→   224→          body: JSON.stringify({
   225→   225→            type: 'charge',
   226→   226→            transactionType: 'late_checkout',
   227→   227→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   228→   228→            amount: surcharge,
   229→   229→          }),
   230→   230→        })
   231→   231→      }
   232→   232→      return result
   233→   233→    },
   234→   234→    onSuccess: () => {
   235→   235→      invalidate.afterReservationChange(queryClient)
   236→   236→      invalidate.afterFolioChange(queryClient)
   237→   237→      setLateCheckoutDialogOpen(false)
   238→   238→      toast.success('Late checkout request processed successfully')
   239→   239→    },
   240→   240→    onError: () => {
   241→   241→      toast.error('Failed to process late checkout request')
   242→   242→    },
   243→   243→  })
   244→   244→
   245→   245→  // Batch checkout mutation
   246→   246→  const batchCheckoutMutation = useMutation({
   247→   247→    mutationFn: async (ids: string[]) => {
   248→   248→      const results = []
   249→   249→      for (let i = 0; i < ids.length; i++) {
   250→   250→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   251→   251→          method: 'PATCH',
   252→   252→          headers: { 'Content-Type': 'application/json' },
   253→   253→          body: JSON.stringify({ status: 'checked_out' }),
   254→   254→        })
   255→   255→        results.push(result)
   256→   256→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   257→   257→      }
   258→   258→      return results
   259→   259→    },
   260→   260→    onSuccess: (_data, ids) => {
   261→   261→      invalidate.afterCheckout(queryClient)
   262→   262→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   263→   263→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   264→   264→    },
   265→   265→    onError: () => {
   266→   266→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   267→   267→    },
   268→   268→  })
   269→   269→
   270→   270→  // ─── Handlers ─────────────────────────────────────────────────────
   271→   271→
   272→   272→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   273→   273→
   274→   274→  const handleExpressCheckout = (dep: Departure) => {
   275→   275→    checkoutMutation.mutate(dep.id, {
   276→   276→      onSuccess: () => {
   277→   277→        toast.success(`Express checkout complete for Room ${dep.room?.number ?? 'N/A'}`)
   278→   278→        toast.info(`Room ${dep.room?.number ?? 'N/A'} marked for housekeeping - Vacant Dirty`)
   279→   279→      },
   280→   280→    })
   281→   281→  }
   282→   282→
   283→   283→  const handleReviewFolio = (dep: Departure) => {
   284→   284→    setSelectedDeparture(dep)
   285→   285→    setFolioDialogOpen(true)
   286→   286→  }
   287→   287→
   288→   288→  const handleNavigateToFolio = (dep: Departure) => {
   289→   289→    useFolioContextStore.getState().setFolioContext({
   290→   290→      reservationId: dep.id,
   291→   291→      guestId: dep.guest?.id || '',
   292→   292→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   293→   293→      roomNumber: dep.room?.number || '',
   294→   294→      confirmationNo: dep.confirmationNo,
   295→   295→    })
   296→   296→    navigateTo('front-desk', 'folio')
   297→   297→  }
   298→   298→
   299→   299→  const handleViewLedger = (dep: Departure) => {
   300→   300→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   301→   301→      guestId: dep.guest?.id || '',
   302→   302→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   303→   303→    })
   304→   304→    navigateTo('front-desk', 'guest-ledger')
   305→   305→  }
   306→   306→
   307→   307→  const handleFolioSettleCheckout = () => {
   308→   308→    setFolioDialogOpen(false)
   309→   309→    setCheckoutDialogOpen(true)
   310→   310→  }
   311→   311→
   312→   312→  const handleQuickCheckout = (dep: Departure) => {
   313→   313→    setSelectedDeparture(dep)
   314→   314→    setCheckoutDialogOpen(true)
   315→   315→  }
   316→   316→
   317→   317→  const handleRequestLateCheckout = (dep: Departure) => {
   318→   318→    setSelectedDeparture(dep)
   319→   319→    setLateCheckoutOption('14:00')
   320→   320→    setLateCheckoutDialogOpen(true)
   321→   321→  }
   322→   322→
   323→   323→  const handleConfirmLateCheckout = () => {
   324→   324→    if (!selectedDeparture) return
   325→   325→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   326→   326→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   327→   327→
   328→   328→    // Build new checkout datetime with the selected time

   330→   330→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   331→   331→    currentCheckout.setHours(hours, minutes, 0, 0)
   332→   332→
   333→   333→    lateCheckoutMutation.mutate({
   334→   334→      reservationId: selectedDeparture.id,
   335→   335→      newCheckOut: currentCheckout.toISOString(),
   336→   336→      surcharge,
   337→   337→    })
   338→   338→  }
   339→   339→
   340→   340→  const handleOpenPaymentDialog = () => {
   341→   341→    if (!selectedDeparture) return
   342→   342→    const balance = getBalance(selectedDeparture)
   343→   343→    setPaymentAmount(String(balance))
   344→   344→    setPaymentMethod('cash')
   345→   345→    setPaymentReference('')
   346→   346→    setPaymentDialogOpen(true)
   347→   347→  }
   348→   348→
   349→   349→  const handleRecordPayment = () => {

   351→   351→    const folioId = selectedDeparture.folios[0]?.id
   352→   352→    if (!folioId) {
   353→   353→      toast.error('No folio found for this reservation')

   355→   355→    }

   358→   358→      toast.error('Please enter a valid payment amount')
   359→   359→      return
   360→   360→    }
   361→   361→    paymentMutation.mutate({
   362→   362→      folioId,
   363→   363→      amount,
   364→   364→      method: paymentMethod,
   365→   365→      reference: paymentReference,
   366→   366→    })
   367→   367→  }

   372→   372→    checkoutMutation.mutate(selectedDeparture.id, {
   373→   373→      onSuccess: () => {

   375→   375→        setReceiptData(selectedDeparture)
   376→   376→        setReceiptDialogOpen(true)
   377→   377→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   378→   378→      },
   379→   379→    })
   380→   380→  }
   381→   381→
   382→   382→  const handleBatchCheckout = () => {
   383→   383→    const ids = zeroBalancePending.map((d) => d.id)
   384→   384→    if (ids.length === 0) {
   385→   385→      toast.info('No guests with zero balance to batch checkout')
   386→   386→      return
   387→   387→    }
   388→   388→    batchCheckoutMutation.mutate(ids)
   389→   389→  }
   390→   390→
   391→   391→  const handlePrintReceipt = () => {
   392→   392→    toast.success('Receipt sent to printer')
   393→   393→  }
   394→   394→
   395→   395→  const handleEmailReceipt = () => {
   396→   396→    toast.success('Receipt emailed to guest')
   397→   397→  }
   398→   398→
   399→   399→  // ─── Computed values ──────────────────────────────────────────────
   400→   400→
   401→   401→  const lateCheckoutFee = selectedDeparture
   402→   402→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   403→   403→    : 0
   404→   404→
   405→   405→  // ─── Render ────────────────────────────────────────────────────────
   406→   406→
   407→   407→  return (
   408→   408→    <div className="flex flex-col gap-2">
   409→   409→      {/* Header */}
   410→   410→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   411→   411→        <div>
   412→   412→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   413→   413→          <p className="text-xs text-muted-foreground">
   414→   414→            {todayCount} scheduled for {formatDate(today)}{overdueCount > 0 && <span className="text-red-600 font-medium"> · {overdueCount} overdue</span>}
   415→   415→          </p>
   416→   416→        </div>
   417→   417→        {zeroBalancePending.length > 0 && (
   418→   418→          <Button
   419→   419→            onClick={handleBatchCheckout}
   420→   420→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   421→   421→            disabled={batchCheckoutMutation.isPending}
   422→   422→          >
   423→   423→            <Zap className="size-4 mr-1.5" />
   424→   424→            {batchCheckoutMutation.isPending
   425→   425→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   426→   426→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   427→   427→          </Button>
   428→   428→        )}
   429→   429→      </div>
   430→   430→
   431→   431→      {/* Departures List */}
   432→   432→      <Card className="py-0">
   433→   433→        <CardContent className="p-0">
   434→   434→          <div className="max-h-[600px] overflow-y-auto">
   435→   435→            <Table>
   436→   436→              <TableHeader>
   437→   437→                <TableRow>
   438→   438→                  <TableHead className="w-[70px]">Room</TableHead>
   439→   439→                  <TableHead>Guest</TableHead>
   440→   440→                  <TableHead className="w-[100px]">Check-out</TableHead>
   441→   441→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   442→   442→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   443→   443→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   444→   444→                </TableRow>
   445→   445→              </TableHeader>
   446→   446→              <TableBody>
   447→   447→                {isLoading ? (
   448→   448→                  Array.from({ length: 5 }).map((_, i) => (
   449→   449→                    <TableRow key={i}>
   450→   450→                      {Array.from({ length: 6 }).map((_, j) => (
   451→   451→                        <TableCell key={j}>
   452→   452→                          <Skeleton className="h-4 w-full" />
   453→   453→                        </TableCell>
   454→   454→                      ))}
   455→   455→                    </TableRow>
   456→   456→                  ))
   457→   457→                ) : departures.length === 0 ? (
   458→   458→                  <TableRow>
