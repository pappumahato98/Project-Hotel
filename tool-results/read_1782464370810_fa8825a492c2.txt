   398→/** Aging tab content */
   399→function AgingTab({ summary, stays, onNavigateToFolio, onNavigateToReservation }: { summary: LedgerSummary; stays: LedgerStay[]; onNavigateToFolio?: (stay: LedgerStay) => void; onNavigateToReservation?: (stay: LedgerStay) => void }) {
   400→  const totalAging =
   401→    summary.aging.current +
   402→    summary.aging.days30 +
   403→    summary.aging.days60 +
   404→    summary.aging.days90 +
   405→    summary.aging.over90
   406→
   407→  return (
   408→    <div className="space-y-6">
   409→      {/* Horizontal bar chart */}
   410→      <Card className="rounded-xl border bg-card shadow-sm">
   411→        <CardHeader className="pb-3">
   412→          <CardTitle className="text-sm font-semibold">Aging Summary</CardTitle>
   413→        </CardHeader>
   414→        <CardContent className="space-y-4">
   415→          {/* Bar visualization */}
   416→          <div className="flex h-8 rounded-md overflow-hidden bg-muted">
   417→            {totalAging > 0
   418→              ? AGING_COLORS.map(({ key, color }) => {
   419→                  const val = summary.aging[key as keyof typeof summary.aging]
   420→                  const pct = (val / totalAging) * 100
   421→                  if (pct === 0) return null
   422→                  return (
   423→                    <div
   424→                      key={key}
   425→                      className={cn('h-full', color)}
   426→                      style={{ width: `${pct}%` }}
   427→                      title={`${AGING_COLORS.find((c) => c.key === key)?.label}: ${formatCurrency(val)}`}
   428→                    />
   429→                  )
   430→                })
   431→              : <div className="flex-1 flex items-center justify-center text-xs text-muted-foreground">No outstanding balance</div>

   434→
   435→          {/* Legend + values */}
