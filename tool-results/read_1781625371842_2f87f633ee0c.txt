    34→// ─── Types ──────────────────────────────────────────────────────────────
    35→
    36→interface DepartureGuest {
    37→  id: string
    38→  firstName: string
    39→  lastName: string
    40→  vipLevel: string
    41→}
    42→
    43→interface DepartureRoom {
    44→  id: string
    45→  number: string
    46→  floor: number
    47→  type: { name: string; code: string }
    48→}
    49→
    50→interface FolioLineItem {
    51→  id: string
    52→  description: string
    53→  amount: number
    54→  type: 'charge' | 'payment'
    55→  date: string
    56→}
    57→
    58→interface DepartureFolio {
    59→  id: string
    60→  balance: number
    61→  status: string
    62→  items?: FolioLineItem[]
    63→}
    64→
    65→interface Departure {
    66→  id: string
    67→  confirmationNo: string
    68→  status: string
    69→  checkIn: string
    70→  checkOut: string
    71→  roomRate: number
    72→  totalAmount: number
    73→  paidAmount: number
    74→  creditLimit: number
    75→  guest: DepartureGuest
    76→  room: DepartureRoom
    77→  folios: DepartureFolio[]
    78→}
    79→
    80→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    81→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    82→
    83→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    84→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    85→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    86→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    87→}
    88→
    89→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    90→  cash: 'Cash',
    91→  card: 'Credit/Debit Card',
    92→  bank_transfer: 'Bank Transfer',
    93→}
    94→
    95→// ─── Component ──────────────────────────────────────────────────────────
    96→
    97→export function DeparturesView() {
    98→  const queryClient = useQueryClient()
    99→  const { settings } = useSettingsStore()
   100→  const { preferences } = usePreferencesStore()
   101→  const today = getTodayString()
   102→
   103→  // Dialog states
   104→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   105→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   106→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   107→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   108→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   109→
   110→  // Selected departure
   111→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   112→
   113→  // Late checkout
   114→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   115→
   116→  // Payment
   117→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   118→  const [paymentAmount, setPaymentAmount] = useState('')
   119→  const [paymentReference, setPaymentReference] = useState('')
   120→
   121→  // Receipt data
   122→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   123→
   124→  // Fetch today's departures (checked_in with checkOut = today)
   125→  const { data, isLoading } = useQuery({
   126→    queryKey: ['departures', today],
   127→    queryFn: async () => {
   128→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   129→      return apiFetch(`/api/reservations?${params.toString()}`)
   130→    },
   131→    refetchInterval: 30000,
   132→  })
   133→
   134→  const departures: Departure[] = data?.reservations || []
   135→
   136→  // Stats
   137→  const totalDepartures = departures.length
   138→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   139→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   140→  const outstandingBalance = departures.reduce((sum, d) => {
   141→    return sum + (d.folios[0]?.balance || 0)
   142→  }, 0)
   143→
   144→  // Zero-balance pending guests (for express & batch checkout)
   145→  const zeroBalancePending = departures.filter(
   146→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   147→  )
   148→
   149→  // ─── Mutations ─────────────────────────────────────────────────────
   150→
   151→  // Checkout mutation
   152→  const checkoutMutation = useMutation({
   153→    mutationFn: async (reservationId: string) => {
   154→      return apiFetch(`/api/reservations/${reservationId}`, {
   155→        method: 'PATCH',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify({ status: 'checked_out' }),
   158→      })
   159→    },
   160→    onSuccess: () => {
   161→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   162→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   163→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   164→    },
   165→  })
   166→
   167→  // Payment mutation
   168→  const paymentMutation = useMutation({
   169→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   170→      return apiFetch(`/api/folio/${folioId}`, {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   174→      })
   175→    },
   176→    onSuccess: () => {
   177→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   178→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   179→      setPaymentDialogOpen(false)
   180→      setPaymentAmount('')
   181→      setPaymentReference('')
   182→      toast.success('Payment recorded successfully')
   183→    },
   184→    onError: () => {
   185→      toast.error('Failed to record payment')
   186→    },
   187→  })
   188→
   189→  // Late checkout mutation
   190→  const lateCheckoutMutation = useMutation({
   191→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   192→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   193→        method: 'PATCH',
   194→        headers: { 'Content-Type': 'application/json' },
   195→        body: JSON.stringify({ checkOut: newCheckOut }),
   196→      })
   197→      // Add surcharge to folio
   198→      if (surcharge > 0) {
   199→        await apiFetch(`/api/folio`, {
   200→          method: 'POST',
   201→          headers: { 'Content-Type': 'application/json' },
   202→          body: JSON.stringify({
   203→            reservationId,
   204→            type: 'charge',
   205→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   206→            amount: surcharge,
   207→          }),
   208→        })
   209→      }
   210→      return result
   211→    },
   212→    onSuccess: () => {
   213→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   214→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   215→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   216→      setLateCheckoutDialogOpen(false)
   217→      toast.success('Late checkout request processed successfully')
   218→    },
   219→    onError: () => {
   220→      toast.error('Failed to process late checkout request')
   221→    },
   222→  })
   223→
   224→  // Batch checkout mutation
   225→  const batchCheckoutMutation = useMutation({
   226→    mutationFn: async (ids: string[]) => {
   227→      const results = []
   228→      for (let i = 0; i < ids.length; i++) {
   229→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   230→          method: 'PATCH',
   231→          headers: { 'Content-Type': 'application/json' },
   232→          body: JSON.stringify({ status: 'checked_out' }),
   233→        })
   234→        results.push(result)
   235→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   236→      }
   237→      return results
   238→    },
   239→    onSuccess: (_data, ids) => {
   240→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   241→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   242→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   243→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   244→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   245→    },
   246→    onError: () => {
   247→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   248→    },
   249→  })
   250→
   251→  // ─── Handlers ─────────────────────────────────────────────────────
   252→
   253→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   254→
   255→  const handleExpressCheckout = (dep: Departure) => {
   256→    checkoutMutation.mutate(dep.id, {
   257→      onSuccess: () => {
   258→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   259→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   260→      },
   261→    })
   262→  }
   263→
   264→  const handleReviewFolio = (dep: Departure) => {
   265→    setSelectedDeparture(dep)
   266→    setFolioDialogOpen(true)
   267→  }
   268→
   269→  const handleFolioSettleCheckout = () => {
   270→    setFolioDialogOpen(false)
   271→    setCheckoutDialogOpen(true)
   272→  }
   273→
   274→  const handleQuickCheckout = (dep: Departure) => {
   275→    setSelectedDeparture(dep)
   276→    setCheckoutDialogOpen(true)
   277→  }
   278→
   279→  const handleRequestLateCheckout = (dep: Departure) => {
   280→    setSelectedDeparture(dep)
   281→    setLateCheckoutOption('14:00')
   282→    setLateCheckoutDialogOpen(true)
   283→  }
   284→
   285→  const handleConfirmLateCheckout = () => {
   286→    if (!selectedDeparture) return
   287→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   288→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   289→
   290→    // Build new checkout datetime with the selected time
   291→    const currentCheckout = new Date(selectedDeparture.checkOut)
   292→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   293→    currentCheckout.setHours(hours, minutes, 0, 0)
   294→
   295→    lateCheckoutMutation.mutate({
   296→      reservationId: selectedDeparture.id,
   297→      newCheckOut: currentCheckout.toISOString(),
   298→      surcharge,
   299→    })
   300→  }
   301→
   302→  const handleOpenPaymentDialog = () => {
   303→    if (!selectedDeparture) return
   304→    const balance = getBalance(selectedDeparture)
   305→    setPaymentAmount(String(balance))
   306→    setPaymentMethod('cash')
   307→    setPaymentReference('')
   308→    setPaymentDialogOpen(true)
   309→  }
   310→
   311→  const handleRecordPayment = () => {
   312→    if (!selectedDeparture) return
   313→    const folioId = selectedDeparture.folios[0]?.id
   314→    if (!folioId) {
   315→      toast.error('No folio found for this reservation')
   316→      return
   317→    }
   318→    const amount = Number(paymentAmount)
   319→    if (isNaN(amount) || amount <= 0) {
   320→      toast.error('Please enter a valid payment amount')
   321→      return
   322→    }
   323→    paymentMutation.mutate({
   324→      folioId,
   325→      amount,
   326→      method: paymentMethod,
   327→      reference: paymentReference,
   328→    })
   329→  }
   330→
   331→  const confirmCheckout = () => {
   332→    if (!selectedDeparture) return
   333→    const roomNumber = selectedDeparture.room.number
   334→    checkoutMutation.mutate(selectedDeparture.id, {
   335→      onSuccess: () => {
   336→        setCheckoutDialogOpen(false)
   337→        setReceiptData(selectedDeparture)
   338→        setReceiptDialogOpen(true)
   339→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   340→      },
   341→    })
   342→  }
   343→
   344→  const handleBatchCheckout = () => {
   345→    const ids = zeroBalancePending.map((d) => d.id)
   346→    if (ids.length === 0) {
   347→      toast.info('No guests with zero balance to batch checkout')
   348→      return
   349→    }
   350→    batchCheckoutMutation.mutate(ids)
   351→  }
   352→
   353→  const handlePrintReceipt = () => {
   354→    toast.success('Receipt sent to printer')
   355→  }
   356→
   357→  const handleEmailReceipt = () => {
   358→    toast.success('Receipt emailed to guest')
   359→  }
   360→
   361→  // ─── Computed values ──────────────────────────────────────────────
   362→
   363→  const lateCheckoutFee = selectedDeparture
   364→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   365→    : 0
   366→
   367→  // ─── Render ────────────────────────────────────────────────────────
   368→
   369→  return (
   370→    <div className="flex flex-col gap-2">
   371→      {/* Header */}
   372→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   373→        <div>
   374→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   375→          <p className="text-xs text-muted-foreground">
   376→            Guest check-outs scheduled for {formatDate(today)}
   377→          </p>
   378→        </div>
   379→        {zeroBalancePending.length > 0 && (
   380→          <Button
   381→            onClick={handleBatchCheckout}
   382→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   383→            disabled={batchCheckoutMutation.isPending}
   384→          >
   385→            <Zap className="size-4 mr-1.5" />
   386→            {batchCheckoutMutation.isPending
   387→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   388→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   389→          </Button>
   390→        )}
   391→      </div>
   392→
   393→      {/* Departures List */}
   394→      <Card className="py-0">
   395→        <CardContent className="p-0">
   396→          <div className="max-h-[600px] overflow-y-auto">
   397→            <Table>
   398→              <TableHeader>
   399→                <TableRow>
   400→                  <TableHead className="w-[70px]">Room</TableHead>
   401→                  <TableHead>Guest</TableHead>
   402→                  <TableHead className="w-[100px]">Check-out</TableHead>
   403→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   404→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   405→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   406→                </TableRow>
   407→              </TableHeader>
   408→              <TableBody>
   409→                {isLoading ? (
   410→                  Array.from({ length: 5 }).map((_, i) => (
   411→                    <TableRow key={i}>
   412→                      {Array.from({ length: 6 }).map((_, j) => (
   413→                        <TableCell key={j}>
   414→                          <Skeleton className="h-4 w-full" />
   415→                        </TableCell>
   416→                      ))}
   417→                    </TableRow>
   418→                  ))
   419→                ) : departures.length === 0 ? (
   420→                  <TableRow>
   421→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   422→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   423→                      No departures scheduled for today
   424→                    </TableCell>
   425→                  </TableRow>
   426→                ) : (
   427→                  departures.map((dep) => {
   428→                    const balance = getBalance(dep)
   429→                    const isPending = dep.status === 'checked_in'
   430→                    const isZeroBalance = balance === 0
   431→                    return (
   432→                      <TableRow key={dep.id}>
   433→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   434→                        <TableCell>
   435→                          <div className="flex items-center gap-2">
   436→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   437→                            {dep.guest.vipLevel !== 'none' && (
   438→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   439→                                VIP
   440→                              </Badge>
   441→                            )}
   442→                          </div>
   443→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   444→                        </TableCell>
   445→                        <TableCell className="text-xs">
   446→                          <div>{formatDate(dep.checkOut)}</div>
   447→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   448→                        </TableCell>
   449→                        <TableCell className="text-right">
   450→                          <div className="flex items-center justify-end gap-1.5">
   451→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   452→                              {formatCurrency(balance)}
   453→                            </span>
   454→                            {/* Outstanding Balance Alert Badges */}
   455→                            {isPending && balance > 5000 && (
   456→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   457→                                HIGH
   458→                              </Badge>
   459→                            )}
   460→                            {isPending && balance > 0 && balance <= 5000 && (
   461→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   462→                                DUE
   463→                              </Badge>
   464→                            )}
   465→                          </div>
   466→                        </TableCell>
   467→                        <TableCell className="text-right">
   468→                          {dep.status === 'checked_out' ? (
   469→                            <div className="flex items-center justify-end gap-1">
   470→                              <BedSingle className="size-3 text-yellow-500" />
   471→                              <StatusBadge status="vacant_dirty" />
   472→                            </div>
   473→                          ) : (
   474→                            <StatusBadge status="checked_in" />
   475→                          )}
   476→                        </TableCell>
   477→                        <TableCell className="text-right">
   478→                          <div className="flex justify-end gap-1 flex-wrap">
   479→                            {isPending && (
   480→                              <>
   481→                                {/* Express Checkout (zero balance) */}
   482→                                {isZeroBalance && (
   483→                                  <Button
   484→                                    size="sm"
   485→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   486→                                    onClick={() => handleExpressCheckout(dep)}
   487→                                    disabled={checkoutMutation.isPending}
   488→                                  >
   489→                                    <Zap className="size-3 mr-0.5" /> Express
   490→                                  </Button>
   491→                                )}
   492→                                {/* Review Folio */}
   493→                                <Button
   494→                                  size="sm"
   495→                                  variant="outline"
   496→                                  className="text-xs h-7"
   497→                                  onClick={() => handleReviewFolio(dep)}
   498→                                >
   499→                                  <Eye className="size-3 mr-0.5" /> Review Folio
   500→                                </Button>
   501→                                {/* Late Checkout */}
   502→                                <Button
   503→                                  size="sm"
   504→                                  variant="outline"
   505→                                  className="text-xs h-7"
   506→                                  onClick={() => handleRequestLateCheckout(dep)}
   507→                                >
   508→                                  <Clock className="size-3 mr-0.5" /> Late
   509→                                </Button>
   510→                                {/* Regular Checkout */}
   511→                                <Button
   512→                                  size="sm"
   513→                                  className="text-xs h-7"
   514→                                  onClick={() => handleQuickCheckout(dep)}
   515→                                >
   516→                                  <LogOut className="size-3 mr-0.5" /> Checkout
   517→                                </Button>
   518→                              </>
   519→                            )}
   520→                            {dep.status === 'checked_out' && (
   521→                              <Badge variant="outline" className="text-xs">
   522→                                <CheckCircle2 className="size-3 mr-1" />
   523→                                Checked Out
   524→                              </Badge>
   525→                            )}
   526→                          </div>
   527→                        </TableCell>
   528→                      </TableRow>
   529→                    )
   530→                  })
   531→                )}
   532→              </TableBody>
   533→            </Table>
   534→          </div>
   535→        </CardContent>
   536→      </Card>
   537→
   538→      {/* ─── Folio Review Dialog ─────────────────────────────────── */}
   539→      <Dialog open={folioDialogOpen} onOpenChange={setFolioDialogOpen}>
   540→        <DialogContent className="max-w-lg">
   541→          <DialogHeader>
   542→            <DialogTitle>Folio Review</DialogTitle>
   543→            <DialogDescription>
   544→              {selectedDeparture && (
   545→                <>Room {selectedDeparture.room.number} — {selectedDeparture.guest.firstName} {selectedDeparture.guest.lastName}</>
   546→              )}
   547→            </DialogDescription>
   548→          </DialogHeader>
   549→          {selectedDeparture && (
   550→            <div className="space-y-4">
   551→              {/* Folio items table */}
   552→              <div className="rounded-lg border max-h-64 overflow-y-auto">
   553→                <Table>
   554→                  <TableHeader>
   555→                    <TableRow>
