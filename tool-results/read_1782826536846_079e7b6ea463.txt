   503→    }

   507→    }
   508→
   509→    setIsSubmitting(true)
   510→    try {
   511→      let effectiveReservationId = reservationId
   512→
   513→      // ── Walk-in: create guest + reservation first ──
   514→      if (isWalkIn && !effectiveReservationId) {

   522→              firstName: walkInGuest.firstName,

   525→              email: walkInGuest.email || undefined,
   526→            }),
   527→          })

   529→            const err = await guestRes.json()
   530→            toast.error(err.error || 'Failed to create guest')
   531→            setIsSubmitting(false)
   532→            return
   533→          }
   534→          const guestData = await guestRes.json()
   535→          guestId = guestData.guest.id
   536→        }
   537→
   538→        // 2. Create reservation
   539→        const resBody: Record<string, unknown> = {
   540→          guestId,
   541→          roomId: selectedRoom.id,
   542→          roomTypeId: selectedRoom.type.id,
   543→          adults: walkInAdults,
   544→          children: walkInChildren,
   545→          checkIn: checkInDate,

   547→          roomRate,
   548→          source: 'walk_in',
   549→          reservationType: 'walk_in',
   550→          status: 'confirmed',
   551→          specialRequests: walkInSpecialRequests || undefined,
   552→        }
   553→        const resRes = await fetch('/api/reservations', {
   554→          method: 'POST',
   555→          headers: { 'Content-Type': 'application/json' },
   556→          body: JSON.stringify(resBody),
   557→        })
   558→        if (!resRes.ok) {
   559→          const err = await resRes.json()
   560→          toast.error(err.error || 'Failed to create walk-in reservation')
   561→          setIsSubmitting(false)
   562→          return
   563→        }
   564→        const resData = await resRes.json()
   565→        effectiveReservationId = resData.reservation.id
