    34→// ─── Types ──────────────────────────────────────────────────────────────
    35→
    36→interface SettlementGuest {
    37→  id: string
    38→  firstName: string
    39→  lastName: string
    40→  vipLevel: string
    41→}
    42→
    43→interface SettlementRoom {
    44→  id: string
    45→  number: string
    46→  floor: number
    47→  type: { name: string; code: string }
    48→}
    49→
    50→interface SettlementFolio {
    51→  id: string
    52→  balance: number
    53→  status: string
    54→  transactions?: {
    55→    id: string
    56→    description: string
    57→    totalAmount: number
    58→    type: string
    59→    createdAt: string
    60→  }[]
    61→  payments?: {
    62→    id: string
    63→    amount: number
    64→    paymentMethod: string
    65→    reference: string
    66→    createdAt: string
    67→  }[]
    68→}
    69→
    70→interface SettlementDeparture {
    71→  id: string
    72→  confirmationNo: string
    73→  status: string
    74→  checkIn: string
    75→  checkOut: string
    76→  roomRate: number
    77→  totalAmount: number
    78→  paidAmount: number
    79→  creditLimit: number
    80→  guest: SettlementGuest | null
    81→  room: SettlementRoom | null
    82→  folios: SettlementFolio[]
    83→}
    84→
    85→interface OtherFolio {
    86→  id: string
    87→  balance: number
    88→  status: string
    89→  reservation: {
    90→    id: string
    91→    confirmationNo: string
    92→    room: { number: string } | null
    93→    status: string
    94→  } | null
    95→}
    96→
    97→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    98→
    99→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
   100→  cash: 'Cash',
   101→  card: 'Credit/Debit Card',
   102→  bank_transfer: 'Bank Transfer',
   103→}
   104→
   105→type BalanceFilter = 'all' | 'outstanding' | 'settled'
   106→
   107→// ─── Component ──────────────────────────────────────────────────────────
   108→
   109→export function DepartureSettlementView() {
   110→  const queryClient = useQueryClient()
   111→  const { preferences } = usePreferencesStore()
   112→  const { navigateTo } = useNavigationStore()
   113→  const today = getTodayString()
   114→
   115→  // Search & filter
   116→  const [search, setSearch] = useState('')
   117→  const [balanceFilter, setBalanceFilter] = useState<BalanceFilter>('all')
   118→
   119→  // Dialogs
   120→  const [settlementDialogOpen, setSettlementDialogOpen] = useState(false)
   121→  const [checkoutConfirmOpen, setCheckoutConfirmOpen] = useState(false)
   122→
   123→  // Selected
   124→  const [selectedDeparture, setSelectedDeparture] = useState<SettlementDeparture | null>(null)
   125→
   126→  // Payment
   127→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   128→  const [paymentAmount, setPaymentAmount] = useState('')
   129→  const [paymentReference, setPaymentReference] = useState('')
   130→
   131→  // ─── Query ────────────────────────────────────────────────────────────
   132→
   133→  const { data, isLoading } = useQuery({
   134→    queryKey: ['departure-settlement', today],
   135→    queryFn: async () => {
   136→      const params = new URLSearchParams({
   137→        status: 'checked_in',
   138→        checkOutOnOrBefore: today,
   139→      })
   140→      return apiFetch(`/api/reservations?${params.toString()}`)
   141→    },
   142→    refetchInterval: 30000,
   143→  })
   144→
   145→  const departures: SettlementDeparture[] = data?.reservations || []
   146→
   147→  // ─── Other Folios query ──────────────────────────────────────────────
   148→
   149→  const { data: otherFoliosData, isLoading: otherFoliosLoading } = useQuery({
   150→    queryKey: ['settlement-other-folios', selectedDeparture?.guest?.id, selectedDeparture?.id],
   151→    queryFn: async () => {
   152→      if (!selectedDeparture?.guest?.id) return { folios: [] }
   153→      const params = new URLSearchParams({ guestId: selectedDeparture.guest.id })
   154→      return apiFetch<{ folios: OtherFolio[] }>(`/api/folio?${params.toString()}`)
   155→    },
   156→    enabled: settlementDialogOpen && !!selectedDeparture?.guest?.id,
   157→  })
   158→
   159→  const otherFolios = (otherFoliosData?.folios || []).filter(
   160→    (f) => f.reservation?.id !== selectedDeparture?.id,
   161→  )
   162→  const otherBillsTotal = otherFolios.reduce((sum, f) => sum + f.balance, 0)
   163→
   164→  // ─── Helpers ──────────────────────────────────────────────────────────
   165→
   166→  const getBalance = (dep: SettlementDeparture) => dep.folios[0]?.balance || 0
   167→
   168→  const guestName = (dep: SettlementDeparture) =>
   169→    dep.guest
   170→      ? `${dep.guest.firstName || ''} ${dep.guest.lastName || ''}`.trim() || 'Unknown'
   171→      : 'Unknown'
   172→
   173→  const roomNumber = (dep: SettlementDeparture) => dep.room?.number || '—'
   174→
   175→  // ─── Stats ────────────────────────────────────────────────────────────
   176→
   177→  const pendingCount = departures.length
   178→  const withBalanceCount = departures.filter((d) => getBalance(d) > 0).length
   179→  const settledCount = departures.filter((d) => getBalance(d) <= 0).length
   180→  const totalOutstanding = departures.reduce((sum, d) => sum + getBalance(d), 0)
   181→
   182→  // ─── Filtered list ────────────────────────────────────────────────────
   183→
   184→  const filtered = departures.filter((d) => {
   185→    const balance = getBalance(d)
   186→    if (balanceFilter === 'outstanding' && balance <= 0) return false
   187→    if (balanceFilter === 'settled' && balance > 0) return false
   188→    if (search) {
   189→      const s = search.toLowerCase()
   190→      const gName = d.guest
   191→        ? `${d.guest.firstName || ''} ${d.guest.lastName || ''}`.toLowerCase().trim()
   192→        : ''
   193→      const room = (d.room?.number || '').toLowerCase()
   194→      const conf = (d.confirmationNo || '').toLowerCase()
   195→      return gName.includes(s) || room.includes(s) || conf.includes(s)
   196→    }
   197→    return true
   198→  })
   199→
   200→  // sorted list
   201→
   202→  // ─── Mutations ────────────────────────────────────────────────────────
   203→
   204→  const paymentMutation = useMutation({
   205→    mutationFn: async ({ folioId, amount, paymentMethod: method, reference }: { folioId: string; amount: number; paymentMethod: PaymentMethod; reference: string }) => {
   206→      return apiFetch(`/api/folio/${folioId}`, {
   207→        method: 'POST',
   208→        headers: { 'Content-Type': 'application/json' },
   209→        body: JSON.stringify({ type: 'payment', amount, paymentMethod: method, reference }),
   210→      })
   211→    },
   212→    onSuccess: () => {
   213→      invalidate.afterFolioChange(queryClient, selectedDeparture?.guest?.id)
   214→      setPaymentAmount('')
   215→      setPaymentReference('')
   216→      toast.success('Payment recorded successfully')
   217→    },
   218→    onError: () => {
   219→      toast.error('Failed to record payment')
   220→    },
   221→  })
   222→
   223→  const checkoutMutation = useMutation({
   224→    mutationFn: async (reservationId: string) => {
   225→      return apiFetch(`/api/reservations/${reservationId}`, {
   226→        method: 'PATCH',
   227→        headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify({ status: 'checked_out' }),
   229→      })
   230→    },
   231→    onSuccess: () => {
   232→      invalidate.afterCheckout(queryClient)
   233→      setSettlementDialogOpen(false)
   234→      setCheckoutConfirmOpen(false)
   235→      setSelectedDeparture(null)
   236→      toast.success('Checkout complete')
   237→    },
   238→  })
   239→
   240→  // ─── Handlers ─────────────────────────────────────────────────────────
   241→
   242→  const handleOpenSettlement = (dep: SettlementDeparture) => {
   243→    setSelectedDeparture(dep)
   244→    const balance = getBalance(dep)
   245→    setPaymentAmount(String(balance))
   246→    setPaymentMethod('cash')
   247→    setPaymentReference('')
   248→    setSettlementDialogOpen(true)
   249→  }
   250→
   251→  const handleNavigateToFolio = (dep: SettlementDeparture) => {
   252→    useFolioContextStore.getState().setFolioContext({
   253→      reservationId: dep.id,
   254→      guestId: dep.guest?.id || '',
   255→      guestName: guestName(dep),
   256→      roomNumber: roomNumber(dep),
   257→      confirmationNo: dep.confirmationNo,
   258→    })
   259→    navigateTo('front-desk', 'folio')
   260→  }
   261→
   262→  const handleViewLedger = (dep: SettlementDeparture) => {
   263→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   264→      guestId: dep.guest?.id || '',
   265→      guestName: guestName(dep),
   266→    })
   267→    navigateTo('front-desk', 'guest-ledger')
   268→  }
   269→
   270→  const handleRecordPayment = () => {
   271→    if (!selectedDeparture) return
   272→    const folioId = selectedDeparture.folios[0]?.id
   273→    if (!folioId) {
   274→      toast.error('No folio found for this reservation')
   275→      return
   276→    }
   277→    const amount = Number(paymentAmount)
   278→    if (isNaN(amount) || amount <= 0) {
   279→      toast.error('Please enter a valid payment amount')
   280→      return
   281→    }
   282→    paymentMutation.mutate({
   283→      folioId,
   284→      amount,
   285→      paymentMethod,
   286→      reference: paymentReference,
   287→    })
   288→  }
   289→
   290→  const handleForceCheckout = (dep: SettlementDeparture) => {
   291→    setSelectedDeparture(dep)
   292→    setCheckoutConfirmOpen(true)
   293→  }
   294→
   295→  const handleSettleAndCheckout = () => {
   296→    if (!selectedDeparture) return
   297→    const balance = getBalance(selectedDeparture)
   298→    if (balance > 0) {
   299→      toast.error('Please settle the outstanding balance before checkout')
   300→      return
   301→    }
   302→    setSettlementDialogOpen(false)
   303→    setSelectedDeparture(selectedDeparture)
   304→    setCheckoutConfirmOpen(true)
   305→  }
   306→
   307→  const confirmCheckout = () => {
   308→    if (!selectedDeparture) return
   309→    checkoutMutation.mutate(selectedDeparture.id)
   310→  }
   311→
   312→  // ─── Computed ─────────────────────────────────────────────────────────
   313→
   314→  const currentBalance = selectedDeparture ? getBalance(selectedDeparture) : 0
   315→  const grandTotal = currentBalance + otherBillsTotal
   316→
   317→  // Charges & payments from folio transactions
   318→  const folioTransactions = selectedDeparture?.folios[0]?.transactions || []
   319→  const folioPayments = selectedDeparture?.folios[0]?.payments || []
   320→  const totalCharges = folioTransactions.reduce((sum, t) => sum + t.totalAmount, 0) + (selectedDeparture?.totalAmount || 0)
   321→  const totalPaid = folioPayments.reduce((sum, p) => sum + p.amount, 0) + (selectedDeparture?.paidAmount || 0)
   322→
   323→  // ─── Render ───────────────────────────────────────────────────────────
   324→
   325→  return (
   326→    <div className="flex flex-col gap-4">
   327→      {/* Header */}
   328→      <div className="flex items-center gap-3">
   329→        <Button
   330→          variant="outline"
   331→          size="icon"
   332→          className="size-8 shrink-0"
   333→          onClick={() => navigateTo('front-desk', 'departures')}
   334→        >
   335→          <ArrowLeft className="size-4" />
   336→        </Button>
   337→        <div>
   338→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Departure Settlement</h2>
   339→          <p className="text-xs text-muted-foreground">Manage and settle departing guest accounts</p>
   340→        </div>
   341→      </div>
   342→
   343→      {/* Stats Cards */}
   344→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   345→        <Card className="py-0">
   346→          <CardContent className="p-4 flex items-center gap-3">
   347→            <div className="rounded-lg bg-emerald-100 dark:bg-emerald-950 p-2.5 shrink-0">
   348→              <PlaneTakeoff className="size-4 text-emerald-600 dark:text-emerald-400" />
   349→            </div>
   350→            <div>
   351→              <p className="text-xs text-muted-foreground">Pending</p>
   352→              <p className="text-lg font-bold">{pendingCount}</p>
   353→            </div>
   354→          </CardContent>
   355→        </Card>
   356→        <Card className="py-0">
   357→          <CardContent className="p-4 flex items-center gap-3">
   358→            <div className="rounded-lg bg-red-100 dark:bg-red-950 p-2.5 shrink-0">
   359→              <DollarSign className="size-4 text-red-600 dark:text-red-400" />
   360→            </div>
   361→            <div>
   362→              <p className="text-xs text-muted-foreground">With Balance</p>
   363→              <p className="text-lg font-bold text-red-600">{withBalanceCount}</p>
   364→            </div>
   365→          </CardContent>
   366→        </Card>
   367→        <Card className="py-0">
   368→          <CardContent className="p-4 flex items-center gap-3">
   369→            <div className="rounded-lg bg-green-100 dark:bg-green-950 p-2.5 shrink-0">
   370→              <CheckCircle2 className="size-4 text-green-600 dark:text-green-400" />
   371→            </div>
   372→            <div>
   373→              <p className="text-xs text-muted-foreground">Settled</p>
   374→              <p className="text-lg font-bold text-green-600">{settledCount}</p>
   375→            </div>
   376→          </CardContent>
   377→        </Card>
   378→        <Card className="py-0 col-span-2 lg:col-span-1">
   379→          <CardContent className="p-4 flex items-center gap-3">
   380→            <div className="rounded-lg bg-amber-100 dark:bg-amber-950 p-2.5 shrink-0">
   381→              <Receipt className="size-4 text-amber-600 dark:text-amber-400" />
   382→            </div>
   383→            <div>
   384→              <p className="text-xs text-muted-foreground">Total Outstanding</p>
   385→              <p className="text-lg font-bold text-red-600">{formatCurrency(totalOutstanding)}</p>
   386→            </div>
   387→          </CardContent>
   388→        </Card>
   389→      </div>
   390→
   391→      {/* Filters */}
   392→      <div className="flex flex-col sm:flex-row gap-2">
   393→        <div className="relative flex-1">
   394→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   395→          <Input
   396→            placeholder="Search guest, room, or confirmation #..."
   397→            value={search}
   398→            onChange={(e) => setSearch(e.target.value)}
   399→            className="pl-9 h-9 text-sm"
   400→          />
   401→        </div>
   402→        <Select value={balanceFilter} onValueChange={(v) => setBalanceFilter(v as BalanceFilter)}>
   403→          <SelectTrigger className="w-full sm:w-44 h-9 text-sm">
   404→            <SelectValue />
   405→          </SelectTrigger>
   406→          <SelectContent>
   407→            <SelectItem value="all">All Departures</SelectItem>
   408→            <SelectItem value="outstanding">Outstanding Balance</SelectItem>
   409→            <SelectItem value="settled">Settled (Zero)</SelectItem>
   410→          </SelectContent>
   411→        </Select>
   412→      </div>
   413→
   414→      {/* Table */}
   415→      <Card className="py-0">
   416→        <CardContent className="p-0">
   417→          <div className="max-h-[500px] overflow-y-auto">
   418→            <Table>
   419→              <TableHeader>
   420→                <TableRow>
   421→                  <TableHead className="w-[70px]">Room</TableHead>
   422→                  <TableHead>Guest</TableHead>
   423→                  <TableHead className="w-[100px]">Check-out</TableHead>
   424→                  <TableHead className="w-[110px] text-right">Balance</TableHead>
   425→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   426→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   427→                </TableRow>
   428→              </TableHeader>
   429→              <TableBody>
   430→                {isLoading ? (
   431→                  Array.from({ length: 5 }).map((_, i) => (
   432→                    <TableRow key={i}>
   433→                      {Array.from({ length: 6 }).map((_, j) => (
   434→                        <TableCell key={j}>
   435→                          <Skeleton className="h-4 w-full" />
   436→                        </TableCell>
   437→                      ))}
   438→                    </TableRow>
   439→                  ))
   440→                ) : filtered.length === 0 ? (
   441→                  <TableRow>
   442→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   443→                      <PlaneTakeoff className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   444→                      No departures match the current filters
   445→                    </TableCell>
   446→                  </TableRow>
   447→                ) : (
   448→                  filtered.map((dep) => {
   449→                    const balance = getBalance(dep)
   450→                    const isSettled = balance <= 0
   451→                    return (
   452→                      <TableRow key={dep.id}>
   453→                        <TableCell className="font-bold font-mono">{roomNumber(dep)}</TableCell>
   454→                        <TableCell>
   455→                          <div className="flex items-center gap-2">
   456→                            <span className="font-medium">{guestName(dep)}</span>
   457→                            {dep.guest?.vipLevel && dep.guest.vipLevel !== 'none' && (
   458→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   459→                                VIP
   460→                              </Badge>
   461→                            )}
   462→                          </div>
   463→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   464→                        </TableCell>
   465→                        <TableCell className="text-xs">
   466→                          <div>{formatDate(dep.checkOut)}</div>
   467→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   468→                        </TableCell>
   469→                        <TableCell className="text-right">
   470→                          <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   471→                            {formatCurrency(balance)}
   472→                          </span>
   473→                        </TableCell>
   474→                        <TableCell className="text-right">
   475→                          {isSettled ? (
   476→                            <Badge className="bg-green-100 text-green-700 dark:bg-green-950 dark:text-green-300 border-0 text-[10px]">
   477→                              SETTLED
   478→                            </Badge>
   479→                          ) : (
   480→                            <Badge className="bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300 border-0 text-[10px]">
   481→                              OUTSTANDING
   482→                            </Badge>
   483→                          )}
   484→                        </TableCell>
   485→                        <TableCell className="text-right">
   486→                          <div className="flex justify-end gap-1 flex-wrap">
   487→                            <Button
   488→                              size="sm"
   489→                              variant="outline"
   490→                              className="text-xs h-7"
   491→                              onClick={() => handleOpenSettlement(dep)}
   492→                            >
   493→                              <Receipt className="size-3 mr-0.5" /> Settlement
   494→                            </Button>
   495→                            {isSettled && (
   496→                              <Button
   497→                                size="sm"
   498→                                className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   499→                                onClick={() => handleForceCheckout(dep)}
   500→                                disabled={checkoutMutation.isPending}
   501→                              >
   502→                                <LogOut className="size-3 mr-0.5" /> Checkout
   503→                              </Button>
   504→                            )}
   505→                            <Button
   506→                              size="sm"
   507→                              variant="outline"
   508→                              className="text-xs h-7"
   509→                              onClick={() => handleNavigateToFolio(dep)}
   510→                            >
   511→                              <FileText className="size-3 mr-0.5" /> Folio
   512→                            </Button>
   513→                            <Button
   514→                              size="sm"
   515→                              variant="outline"
   516→                              className="text-xs h-7"
   517→                              onClick={() => handleViewLedger(dep)}
   518→                            >
   519→                              <BookOpen className="size-3 mr-0.5" /> Ledger
   520→                            </Button>
   521→                            {!isSettled && (
   522→                              <Button
   523→                                size="sm"
   524→                                variant="outline"
   525→                                className="text-xs h-7 text-orange-600 border-orange-300 hover:bg-orange-50 dark:border-orange-800 dark:hover:bg-orange-950"
   526→                                onClick={() => handleForceCheckout(dep)}
   527→                                disabled={checkoutMutation.isPending}
   528→                              >
   529→                                <AlertTriangle className="size-3 mr-0.5" /> Force CO
   530→                              </Button>
   531→                            )}
   532→                          </div>
   533→                        </TableCell>
   534→                      </TableRow>
   535→                    )
   536→                  })
   537→                )}
   538→              </TableBody>
   539→            </Table>
   540→          </div>
   541→        </CardContent>
   542→      </Card>
   543→
   544→      {/* ─── Settlement Detail Dialog ────────────────────────────── */}
   545→      <Dialog open={settlementDialogOpen} onOpenChange={setSettlementDialogOpen}>
   546→        <DialogContent className="max-w-2xl max-h-[85vh] overflow-y-auto">
   547→          <DialogHeader>
   548→            <DialogTitle>Departure Settlement</DialogTitle>
   549→            <DialogDescription>
   550→              {selectedDeparture && (
