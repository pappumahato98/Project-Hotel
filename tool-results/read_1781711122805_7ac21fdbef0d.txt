   106→// ─── Component ──────────────────────────────────────────────────────────
   107→
   108→export function RoomRatePostingPage() {
   109→  const queryClient = useQueryClient()
   110→
   111→  // State
   112→  const [activeTab, setActiveTab] = useState<TabValue>('pending')
   113→  const [search, setSearch] = useState('')
   114→  const [searchInput, setSearchInput] = useState('')
   115→  const [statusFilter, setStatusFilter] = useState('all')
   116→  const [dateFrom, setDateFrom] = useState('')
   117→  const [dateTo, setDateTo] = useState('')
   118→  const [page, setPage] = useState(1)
   119→  const [expandedRow, setExpandedRow] = useState<string | null>(null)
   120→  const [voidDialogOpen, setVoidDialogOpen] = useState(false)
   121→  const [voidTarget, setVoidTarget] = useState<PostingRow | null>(null)
   122→  const [voidReason, setVoidReason] = useState('')
   123→  const [postConfirmOpen, setPostConfirmOpen] = useState(false)
   124→  const [postTarget, setPostTarget] = useState<PendingReservation | null>(null)
   125→
   126→  // ── Fetch all postings list ──────────────────────────────────────────
   127→  const { data: listData, isLoading: isLoadingList, refetch: refetchList } = useQuery({
   128→    queryKey: ['rate-posting-list', { activeTab, search, statusFilter, dateFrom, dateTo, page }],
   129→    queryFn: async () => {
   130→      const params = new URLSearchParams()
   131→      if (activeTab !== 'pending') {
   132→        params.set('status', activeTab === 'all' ? 'all' : activeTab)
   133→      }
   134→      if (search) params.set('search', search)
   135→      if (dateFrom) params.set('dateFrom', dateFrom)
   136→      if (dateTo) params.set('dateTo', dateTo)
   137→      params.set('page', String(page))
   138→      params.set('limit', '50')
   139→
   140→      return apiFetch<PostingListResponse>(`/api/room-rate-posting/list?${params}`)
   141→    },
   142→    enabled: activeTab !== 'pending',
   143→  })
   144→
   145→  // ── Fetch pending reservations ───────────────────────────────────────
   146→  const { data: pendingData, isLoading: isLoadingPending, refetch: refetchPending } = useQuery({
   147→    queryKey: ['rate-posting-pending'],
   148→    queryFn: () => apiFetch<PendingResponse>('/api/room-rate-posting/pending'),
   149→    enabled: activeTab === 'pending',
   150→  })
   151→
   152→  // ── Void mutation ───────────────────────────────────────────────────
   153→  const voidMutation = useMutation({
   154→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   155→      return apiFetch(`/api/room-rate-posting/${id}`, {
   156→        method: 'PATCH',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ voidReason: reason }),
   159→      })
   160→    },
   161→    onSuccess: () => {
   162→      toast.success('Rate posting voided successfully')
   163→      invalidate.afterFolioChange(queryClient)
   164→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   165→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   166→      setVoidDialogOpen(false)
   167→      setVoidTarget(null)
   168→      setVoidReason('')
   169→    },
   170→    onError: (error) => toast.error(error.message || 'Failed to void posting'),
   171→  })
   172→
   173→  // ── Post single reservation's pending nights ────────────────────────
   174→  const postMutation = useMutation({
   175→    mutationFn: async ({ reservationId, dates }: { reservationId: string; dates: string[] }) => {
   176→      return apiFetch('/api/room-rate-posting', {
   177→        method: 'POST',
   178→        headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({ reservationId, dates }),
   180→      })
   181→    },
   182→    onSuccess: () => {
   183→      toast.success('Pending charges posted successfully')
   184→      invalidate.afterFolioChange(queryClient)
   185→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   186→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   187→      setPostConfirmOpen(false)
   188→      setPostTarget(null)
   189→    },
   190→    onError: (error) => {
   191→      toast.error(error.message || 'Failed to post charges')
   192→      setPostConfirmOpen(false)
   193→      setPostTarget(null)
   194→    },
   195→  })
   196→
   197→  // ── Bulk post all pending ───────────────────────────────────────────
   198→  const bulkPostMutation = useMutation({
   199→    mutationFn: async () => {
   200→      return apiFetch('/api/room-rate-posting/bulk-post', {
   201→        method: 'POST',
   202→        headers: { 'Content-Type': 'application/json' },
   203→        body: JSON.stringify({ postAll: true }),
   204→      })
   205→    },
   206→    onSuccess: (data) => {
   207→      const result = data as { message?: string; summary?: { totalPostedNights: number; successCount: number } }
   208→      toast.success(result?.message || 'Bulk post completed')
   209→      invalidate.afterFolioChange(queryClient)
   210→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   211→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   212→    },
   213→    onError: () => toast.error('Bulk post failed'),
   214→  })
   215→
   216→  // ── Handlers ─────────────────────────────────────────────────────────
   217→  const handleSearch = useCallback(() => {
   218→    setSearch(searchInput)
   219→    setPage(1)
   220→  }, [searchInput])
   221→
   222→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
   223→    if (e.key === 'Enter') handleSearch()
   224→  }, [handleSearch])
   225→
   226→  const clearFilters = useCallback(() => {
   227→    setSearchInput('')
   228→    setSearch('')
   229→    setStatusFilter('all')
   230→    setDateFrom('')
   231→    setDateTo('')
   232→    setPage(1)
   233→  }, [])
   234→
   235→  const handleVoid = useCallback(() => {
   236→    if (!voidTarget || !voidReason.trim()) return
   237→    voidMutation.mutate({ id: voidTarget.id, reason: voidReason.trim() })
   238→  }, [voidTarget, voidReason, voidMutation])
   239→
   240→  const handlePostReservation = useCallback(() => {
   241→    if (!postTarget) return
   242→    postMutation.mutate({ reservationId: postTarget.reservationId, dates: postTarget.pendingNights })
   243→  }, [postTarget, postMutation])
   244→
   245→  const handleBulkPost = useCallback(() => {
   246→    bulkPostMutation.mutate()
   247→  }, [bulkPostMutation])
   248→
   249→  const toggleExpand = useCallback((id: string) => {
   250→    setExpandedRow((prev) => (prev === id ? null : id))
   251→  }, [])
   252→
   253→  // ── Derived data ────────────────────────────────────────────────────
   254→  const stats = listData?.stats
   255→  const pendingSummary = pendingData?.summary
   256→  const postings = listData?.postings || []
   257→  const pendingReservations = pendingData?.pendingReservations || []
   258→  const pagination = listData?.pagination
   259→
   260→  const isWorking = voidMutation.isPending || postMutation.isPending || bulkPostMutation.isPending
   261→
   262→  // ── Render ───────────────────────────────────────────────────────────
   263→  return (
   264→    <div className="space-y-4">
   265→      {/* Header */}
   266→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   267→        <div>
   268→          <h2 className="text-lg font-semibold flex items-center gap-2">
   269→            <DollarSign className="size-5 text-emerald-600" />
   270→            Room Rate Posting
   271→          </h2>
   272→          <p className="text-sm text-muted-foreground mt-0.5">
   273→            Manage daily room charge postings across all reservations
   274→          </p>
   275→        </div>
   276→        <div className="flex items-center gap-2">
   277→          <Button
   278→            variant="outline"
   279→            size="sm"
   280→            onClick={() => {
   281→              refetchList()
   282→              refetchPending()
   283→            }}
   284→            className="gap-1.5"
   285→          >
   286→            <RefreshCw className="size-3.5" />
   287→            Refresh
   288→          </Button>
   289→          {activeTab === 'pending' && pendingReservations.length > 0 && (
   290→            <Button
   291→              size="sm"
   292→              onClick={handleBulkPost}
   293→              disabled={isWorking}
   294→              className="gap-1.5 bg-emerald-600 hover:bg-emerald-700"
   295→            >
   296→              {bulkPostMutation.isPending ? (
   297→                <Loader2 className="size-3.5 animate-spin" />
   298→              ) : (
   299→                <Zap className="size-3.5" />
   300→              )}
   301→              Post All Pending
   302→            </Button>
   303→          )}
   304→        </div>
   305→      </div>
   306→
   307→      {/* Stat Cards */}
   308→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   309→        <StatCard
   310→          icon={<DollarSign className="size-4 text-emerald-600" />}
   311→          label="Today's Posting"
   312→          value={String(stats?.postedToday ?? pendingSummary?.totalReservations ?? 0)}
   313→          sublabel={stats ? `${stats.todayRevenue > 0 ? formatCurrency(stats.todayRevenue) : 'No revenue today'}` : pendingSummary ? `${pendingSummary.totalPendingNights} nights pending` : '—'}
   314→          bgClass="bg-emerald-50 border-emerald-200"
   315→        />
   316→        <StatCard
   317→          icon={<TrendingUp className="size-4 text-blue-600" />}
   318→          label="Total Posted"
   319→          value={String(stats?.totalPosted ?? 0)}
   320→          sublabel={stats ? formatCurrency(stats.totalRevenue) : '—'}
   321→          bgClass="bg-blue-50 border-blue-200"
   322→        />
   323→        <StatCard
   324→          icon={<Users className="size-4 text-orange-600" />}
   325→          label="In-House Guests"
   326→          value={String(stats?.inHouseReservations ?? 0)}
   327→          sublabel="Active reservations"
   328→          bgClass="bg-orange-50 border-orange-200"
   329→        />
   330→        <StatCard
   331→          icon={<AlertCircle className="size-4 text-red-600" />}
   332→          label="Voided"
   333→          value={String(stats?.voidedCount ?? 0)}
   334→          sublabel="Cancelled postings"
   335→          bgClass="bg-red-50 border-red-200"
   336→        />
   337→      </div>
   338→
   339→      {/* Tabs + Filters */}
   340→      <Tabs value={activeTab} onValueChange={(v) => { setActiveTab(v as TabValue); setPage(1) }}>
   341→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   342→          <TabsList>
   343→            <TabsTrigger value="pending" className="gap-1.5">
   344→              <Clock className="size-3.5" />
   345→              Pending
   346→              {pendingSummary && pendingSummary.totalReservations > 0 && (
   347→                <Badge variant="destructive" className="h-4 min-w-4 text-[10px] px-1 ml-1">
   348→                  {pendingSummary.totalReservations}
   349→                </Badge>
   350→              )}
   351→            </TabsTrigger>
   352→            <TabsTrigger value="all" className="gap-1.5">
   353→              <DollarSign className="size-3.5" />
   354→              All
   355→            </TabsTrigger>
   356→            <TabsTrigger value="posted" className="gap-1.5">
   357→              <CheckCircle className="size-3.5" />
   358→              Posted
   359→            </TabsTrigger>
   360→            <TabsTrigger value="voided" className="gap-1.5">
   361→              <Ban className="size-3.5" />
   362→              Voided
   363→            </TabsTrigger>
   364→          </TabsList>
   365→
   366→          {/* Filters (for All/Posted/Voided tabs) */}
   367→          {activeTab !== 'pending' && (
   368→            <div className="flex items-center gap-2 flex-wrap">
   369→              <div className="relative">
   370→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   371→                <Input
   372→                  placeholder="Guest, Room, Conf#..."
   373→                  value={searchInput}
   374→                  onChange={(e) => setSearchInput(e.target.value)}
   375→                  onKeyDown={handleKeyDown}
   376→                  className="h-8 w-[180px] pl-8 text-xs"
   377→                />
   378→              </div>
   379→              <Input
   380→                type="date"
   381→                value={dateFrom}
   382→                onChange={(e) => { setDateFrom(e.target.value); setPage(1) }}
   383→                className="h-8 w-[130px] text-xs"
   384→              />
   385→              <span className="text-xs text-muted-foreground">to</span>
   386→              <Input
   387→                type="date"
   388→                value={dateTo}
   389→                onChange={(e) => { setDateTo(e.target.value); setPage(1) }}
   390→                className="h-8 w-[130px] text-xs"
   391→              />
   392→              <Button variant="ghost" size="sm" onClick={handleSearch} className="h-8 text-xs">
   393→                Search
   394→              </Button>
   395→              {(search || dateFrom || dateTo) && (
   396→                <Button variant="ghost" size="sm" onClick={clearFilters} className="h-8 text-xs text-muted-foreground">
   397→                  Clear
   398→                </Button>
   399→              )}
   400→            </div>
   401→          )}
   402→        </div>
   403→
   404→        {/* ── PENDING TAB ──────────────────────────────────────────── */}
   405→        <TabsContent value="pending" className="mt-4">
   406→          {isLoadingPending ? (
   407→            <Card>
   408→              <CardContent className="p-4 space-y-3">
   409→                {Array.from({ length: 4 }).map((_, i) => (
   410→                  <Skeleton key={i} className="h-14 w-full" />
   411→                ))}
   412→              </CardContent>
   413→            </Card>
   414→          ) : pendingReservations.length === 0 ? (
   415→            <EmptyState
   416→              icon={<CheckCircle className="size-12 text-emerald-400" />}
   417→              title="All Caught Up!"
   418→              description="No pending room charges to post. All in-house guests are up to date."
   419→            />
   420→          ) : (
   421→            <>
   422→              {/* Pending Summary Bar */}
   423→              <Card className="py-2.5 px-4">
   424→                <div className="flex items-center justify-between text-xs">
   425→                  <div className="flex items-center gap-4">
   426→                    <span className="text-muted-foreground">
   427→                      <span className="font-semibold text-foreground">{pendingSummary?.totalReservations}</span> reservations
   428→                    </span>
   429→                    <span className="text-muted-foreground">
   430→                      <span className="font-semibold text-amber-600">{pendingSummary?.totalPendingNights}</span> pending nights
   431→                    </span>
   432→                    <span className="font-semibold tabular-nums">
   433→                      {pendingSummary && formatCurrency(pendingSummary.totalPendingAmount)}
   434→                    </span>
   435→                  </div>
   436→                  <Button
   437→                    size="sm"
   438→                    variant="outline"
   439→                    onClick={handleBulkPost}
   440→                    disabled={isWorking}
   441→                    className="h-7 text-xs gap-1.5"
   442→                  >
   443→                    {bulkPostMutation.isPending ? <Loader2 className="size-3 animate-spin" /> : <Zap className="size-3" />}
   444→                    Post All
   445→                  </Button>
   446→                </div>
   447→              </Card>
   448→
   449→              {/* Pending Table */}
   450→              <Card className="overflow-hidden">
   451→                <div className="max-h-[calc(100vh-380px)] overflow-y-auto">
   452→                  <Table>
   453→                    <TableHeader className="sticky top-0 z-10 bg-muted/95 backdrop-blur-sm">
   454→                      <TableRow>
   455→                        <TableHead className="text-xs w-[40px]"></TableHead>
   456→                        <TableHead className="text-xs">Room</TableHead>
   457→                        <TableHead className="text-xs">Guest</TableHead>
   458→                        <TableHead className="text-xs">Stay</TableHead>
   459→                        <TableHead className="text-xs text-right">Rate/Night</TableHead>
   460→                        <TableHead className="text-xs text-center">Pending</TableHead>
   461→                        <TableHead className="text-xs text-center">Posted</TableHead>
   462→                        <TableHead className="text-xs text-center">Future</TableHead>
   463→                        <TableHead className="text-xs text-right">Pending Amt</TableHead>
   464→                        <TableHead className="text-xs text-right w-[80px]">Action</TableHead>
   465→                      </TableRow>
   466→                    </TableHeader>
   467→                    <TableBody>
   468→                      {pendingReservations.map((res) => (
   469→                        <PendingRow
   470→                          key={res.reservationId}
   471→                          reservation={res}
   472→                          isExpanded={expandedRow === res.reservationId}
   473→                          onToggle={() => toggleExpand(res.reservationId)}
   474→                          onPost={() => { setPostTarget(res); setPostConfirmOpen(true) }}
   475→                          isPosting={postMutation.isPending && postMutation.variables?.reservationId === res.reservationId}
   476→                        />
   477→                      ))}
   478→                    </TableBody>
   479→                  </Table>
   480→                </div>
   481→              </Card>
   482→            </>
   483→          )}
   484→        </TabsContent>
   485→
   486→        {/* ── ALL / POSTED / VOIDED TABS ───────────────────────────── */}
   487→        <TabsContent value={activeTab} className="mt-4">
   488→          {isLoadingList ? (
   489→            <Card>
   490→              <CardContent className="p-4 space-y-3">
   491→                {Array.from({ length: 6 }).map((_, i) => (
   492→                  <Skeleton key={i} className="h-10 w-full" />
   493→                ))}
   494→              </CardContent>
   495→            </Card>
   496→          ) : postings.length === 0 ? (
   497→            <EmptyState
   498→              icon={<DollarSign className="size-12 text-slate-300" />}
   499→              title="No Postings Found"
   500→              description={
   501→                activeTab === 'voided'
   502→                  ? 'No voided postings in the system.'
   503→                  : 'No room rate postings match your filters.'
   504→              }
   505→            />
   506→          ) : (
   507→            <>
   508→              <Card className="overflow-hidden">
   509→                <div className="max-h-[calc(100vh-380px)] overflow-y-auto">
   510→                  <Table>
   511→                    <TableHeader className="sticky top-0 z-10 bg-muted/95 backdrop-blur-sm">
   512→                      <TableRow>
   513→                        <TableHead className="text-xs w-[40px]"></TableHead>
   514→                        <TableHead className="text-xs">Date</TableHead>
   515→                        <TableHead className="text-xs">Room</TableHead>
   516→                        <TableHead className="text-xs">Guest</TableHead>
   517→                        <TableHead className="text-xs">Reservation</TableHead>
   518→                        <TableHead className="text-xs text-right">Room Rate</TableHead>
   519→                        <TableHead className="text-xs text-right">Tax</TableHead>
   520→                        <TableHead className="text-xs text-right">Svc</TableHead>
   521→                        <TableHead className="text-xs text-right">Total</TableHead>
   522→                        <TableHead className="text-xs text-center">Status</TableHead>
   523→                        <TableHead className="text-xs">Posted By</TableHead>
   524→                        <TableHead className="text-xs w-[50px]"></TableHead>
   525→                      </TableRow>
   526→                    </TableHeader>
   527→                    <TableBody>
   528→                      {postings.map((posting) => (
   529→                        <PostingListRow
   530→                          key={posting.id}
   531→                          posting={posting}
   532→                          isExpanded={expandedRow === posting.id}
   533→                          onToggle={() => toggleExpand(posting.id)}
   534→                          onVoid={() => { setVoidTarget(posting); setVoidDialogOpen(true) }}
   535→                        />
   536→                      ))}
   537→                    </TableBody>
   538→                  </Table>
   539→                </div>
   540→              </Card>
   541→
   542→              {/* Pagination */}
   543→              {pagination && pagination.totalPages > 1 && (
   544→                <div className="flex items-center justify-between text-xs">
   545→                  <span className="text-muted-foreground">
   546→                    Showing {((pagination.page - 1) * pagination.limit) + 1}–{Math.min(pagination.page * pagination.limit, pagination.total)} of {pagination.total}
   547→                  </span>
   548→                  <div className="flex items-center gap-1">
   549→                    <Button
   550→                      variant="outline"
   551→                      size="sm"
   552→                      className="h-7 text-xs"
   553→                      disabled={pagination.page <= 1}
   554→                      onClick={() => setPage((p) => p - 1)}
   555→                    >
   556→                      Previous
   557→                    </Button>
   558→                    {Array.from({ length: Math.min(5, pagination.totalPages) }, (_, i) => {
   559→                      let pageNum: number
   560→                      if (pagination.totalPages <= 5) {
   561→                        pageNum = i + 1
   562→                      } else if (pagination.page <= 3) {
   563→                        pageNum = i + 1
   564→                      } else if (pagination.page >= pagination.totalPages - 2) {
   565→                        pageNum = pagination.totalPages - 4 + i
   566→                      } else {
   567→                        pageNum = pagination.page - 2 + i
   568→                      }

   570→                        <Button
   571→                          key={pageNum}
   572→                          variant={pageNum === pagination.page ? 'default' : 'outline'}
   573→                          size="sm"
   574→                          className="h-7 w-7 text-xs p-0"

   576→                        >
   577→                          {pageNum}
   578→                        </Button>
   579→                      )
   580→                    })}
   581→                    <Button
   582→                      variant="outline"
   583→                      size="sm"
   584→                      className="h-7 text-xs"
   585→                      disabled={pagination.page >= pagination.totalPages}
   586→                      onClick={() => setPage((p) => p + 1)}
   587→                    >
   588→                      Next
   589→                    </Button>
   590→                  </div>
   591→                </div>
   592→              )}
   593→            </>
   594→          )}
   595→        </TabsContent>
   596→      </Tabs>
   597→
   598→      {/* ── Void Confirmation Dialog ────────────────────────────────── */}
   599→      <Dialog open={voidDialogOpen} onOpenChange={setVoidDialogOpen}>

   601→          <DialogHeader>
   602→            <DialogTitle className="flex items-center gap-2 text-red-600">
   603→              <Ban className="size-5" />
   604→              Void Rate Posting
   605→            </DialogTitle>
   606→            <DialogDescription>
   607→              This will void the room charge and reverse the folio transaction. This action cannot be undone.
   608→            </DialogDescription>
   609→          </DialogHeader>
   610→          {voidTarget && (
   611→            <div className="space-y-3 py-2">
   612→              <div className="grid grid-cols-2 gap-2 text-xs">
   613→                <div><span className="text-muted-foreground">Date:</span> <span className="font-medium">{formatDate(voidTarget.postingDate)}</span></div>
   614→                <div><span className="text-muted-foreground">Room:</span> <span className="font-medium">{voidTarget.reservation?.room?.number || '—'}</span></div>
   615→                <div><span className="text-muted-foreground">Guest:</span> <span className="font-medium">{voidTarget.reservation?.guest ? `${voidTarget.reservation.guest.firstName} ${voidTarget.reservation.guest.lastName}` : '—'}</span></div>
   616→                <div><span className="text-muted-foreground">Amount:</span> <span className="font-semibold text-red-600">{formatCurrency(voidTarget.totalAmount)}</span></div>
   617→              </div>
   618→              <div>
   619→                <label className="text-xs font-medium text-muted-foreground">Reason for voiding (required)</label>
   620→                <Input
   621→                  value={voidReason}
   622→                  onChange={(e) => setVoidReason(e.target.value)}
   623→                  placeholder="e.g. Incorrect rate, duplicate posting..."
   624→                  className="mt-1 h-8 text-sm"
   625→                />
   626→              </div>
   627→            </div>
   628→          )}
   629→          <DialogFooter className="gap-2">
   630→            <Button variant="outline" onClick={() => setVoidDialogOpen(false)} className="gap-1.5">
   631→              <X className="size-3.5" />
   632→              Cancel
   633→            </Button>
   634→            <Button
   635→              variant="destructive"
   636→              onClick={handleVoid}
   637→              disabled={!voidReason.trim() || voidMutation.isPending}
   638→              className="gap-1.5"
   639→            >
   640→              {voidMutation.isPending ? <Loader2 className="size-3.5 animate-spin" /> : <Ban className="size-3.5" />}
   641→              Void Posting
   642→            </Button>
   643→          </DialogFooter>
   644→        </DialogContent>
   645→      </Dialog>
   646→
   647→      {/* ── Post Confirmation Dialog ────────────────────────────────── */}
   648→      <Dialog open={postConfirmOpen} onOpenChange={setPostConfirmOpen}>
   649→        <DialogContent className="max-w-md">
   650→          <DialogHeader>
   651→            <DialogTitle className="flex items-center gap-2">
   652→              <Zap className="size-5 text-emerald-600" />
   653→              Post Pending Charges
   654→            </DialogTitle>
   655→            <DialogDescription>
   656→              Post all pending room charges for this reservation to the guest folio.
   657→            </DialogDescription>
   658→          </DialogHeader>
   659→          {postTarget && (
   660→            <div className="space-y-2 py-2 text-xs">
   661→              <div className="grid grid-cols-2 gap-2">
   662→                <div><span className="text-muted-foreground">Room:</span> <span className="font-medium">{postTarget.room?.number || '—'}</span></div>
   663→                <div><span className="text-muted-foreground">Guest:</span> <span className="font-medium">{postTarget.guest ? `${postTarget.guest.firstName} ${postTarget.guest.lastName}` : '—'}</span></div>
   664→                <div><span className="text-muted-foreground">Pending Nights:</span> <span className="font-semibold text-amber-600">{postTarget.pendingNights.length}</span></div>
   665→                <div><span className="text-muted-foreground">Amount:</span> <span className="font-semibold">{formatCurrency(postTarget.pendingAmount)}</span></div>
   666→              </div>
   667→              <div className="text-muted-foreground mt-2">
   668→                Nights to post: {postTarget.pendingNights.join(', ')}
   669→              </div>
   670→            </div>
   671→          )}
   672→          <DialogFooter className="gap-2">
   673→            <Button variant="outline" onClick={() => setPostConfirmOpen(false)} className="gap-1.5">
   674→              <X className="size-3.5" />
   675→              Cancel
   676→            </Button>
   677→            <Button
   678→              onClick={handlePostReservation}
   679→              disabled={postMutation.isPending}
   680→              className="gap-1.5 bg-emerald-600 hover:bg-emerald-700"
   681→            >
   682→              {postMutation.isPending ? <Loader2 className="size-3.5 animate-spin" /> : <CheckCircle className="size-3.5" />}
   683→              Post Charges
   684→            </Button>
   685→          </DialogFooter>
   686→        </DialogContent>
   687→      </Dialog>
   688→    </div>
   689→  )
