   106→// ─── Component ──────────────────────────────────────────────────────────
   107→
   108→export function DeparturesView() {
   109→  const queryClient = useQueryClient()
   110→  const { settings } = useSettingsStore()
   111→  const { preferences } = usePreferencesStore()
   112→  const { navigateTo } = useNavigationStore()
   113→  const today = getTodayString()
   114→
   115→  // Dialog states
   116→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   117→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   118→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   119→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   120→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   121→
   122→  // Selected departure
   123→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   124→
   125→  // Late checkout
   126→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   127→
   128→  // Payment
   129→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   130→  const [paymentAmount, setPaymentAmount] = useState('')
   131→  const [paymentReference, setPaymentReference] = useState('')
   132→
   133→  // Receipt data
   134→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   135→
   136→  // Checkbox selection
   137→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   138→  const [compactView, setCompactView] = useState(false)
   139→
   140→  const toggleSelect = (id: string) => {
   141→    setSelectedIds((prev) => {
   142→      const next = new Set(prev)
   143→      if (next.has(id)) next.delete(id)
   144→      else next.add(id)
   145→      return next
   146→    })
   147→  }
   148→
   149→  const handleRefresh = () => {
   150→    queryClient.invalidateQueries({ queryKey: ['departures'] })
   151→  }
   152→
   153→  const handlePrint = () => {
   154→    window.print()
   155→  }
   156→
   157→  // Fetch today's departures (checked_in with checkOut = today)
   158→  const { data, isLoading } = useQuery({
   159→    queryKey: ['departures', today],
   160→    queryFn: async () => {
   161→      // Fetch both: today's departures AND overdue departures
   162→      const [todayData, overdueData] = await Promise.all([
   163→        apiFetch(`/api/reservations?status=checked_in&checkOutDate=${today}`),
   164→        // Fetch overdue: checked_in with checkOut < today
   165→        apiFetch(`/api/reservations?status=checked_in&checkOutBefore=${today}`),
   166→      ])
   167→      // Merge and deduplicate
   168→      const todayReservations = todayData?.reservations || []
   169→      const overdueReservations = overdueData?.reservations || []
   170→      const seen = new Set<string>()
   171→      const merged = [...todayReservations, ...overdueReservations].filter((r: { id: string }) => {
   172→        if (seen.has(r.id)) return false
   173→        seen.add(r.id)
   174→        return true
   175→      })
   176→      return { reservations: merged }
   177→    },
   178→    refetchInterval: 30000,
   179→  })
   180→
   181→  const departures: Departure[] = data?.reservations || []
   182→
   183→  // Stats
   184→  const overdueCount = departures.filter((d) => new Date(d.checkOut) < new Date(today)).length
   185→  const todayCount = departures.length - overdueCount
   186→  const totalDepartures = departures.length
   187→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   188→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   189→  const outstandingBalance = departures.reduce((sum, d) => {
   190→    return sum + (d.folios[0]?.balance || 0)
   191→  }, 0)
   192→
   193→  // Zero-balance pending guests (for express & batch checkout)
   194→  const zeroBalancePending = departures.filter(
   195→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   196→  )
   197→
   198→  // ─── Mutations ─────────────────────────────────────────────────────
   199→
   200→  // Checkout mutation
   201→  const checkoutMutation = useMutation({
   202→    mutationFn: async (reservationId: string) => {
   203→      return apiFetch(`/api/reservations/${reservationId}`, {
   204→        method: 'PATCH',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({ status: 'checked_out' }),
   207→      })
   208→    },
   209→    onSuccess: () => {
   210→      invalidate.afterCheckout(queryClient)
   211→    },
   212→  })
   213→
   214→  // Payment mutation
   215→  const paymentMutation = useMutation({
   216→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   217→      return apiFetch(`/api/folio/${folioId}`, {
   218→        method: 'POST',
   219→        headers: { 'Content-Type': 'application/json' },
   220→        body: JSON.stringify({ type: 'payment', paymentMethod: method, amount, reference }),
   221→      })
   222→    },
   223→    onSuccess: () => {
   224→      invalidate.afterFolioChange(queryClient)
   225→      setPaymentDialogOpen(false)
   226→      setPaymentAmount('')
   227→      setPaymentReference('')
   228→      toast.success('Payment recorded successfully')
   229→    },
   230→    onError: () => {
   231→      toast.error('Failed to record payment')
   232→    },
   233→  })
   234→
   235→  // Late checkout mutation
   236→  const lateCheckoutMutation = useMutation({
   237→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   238→      // First get the folio ID for this reservation
   239→      const reservation = departures.find((d) => d.id === reservationId)
   240→      const folioId = reservation?.folios[0]?.id
   241→
   242→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   243→        method: 'PATCH',
   244→        headers: { 'Content-Type': 'application/json' },
   245→        body: JSON.stringify({ checkOut: newCheckOut }),
   246→      })
   247→      // Add surcharge to folio
   248→      if (surcharge > 0 && folioId) {
   249→        await apiFetch(`/api/folio/${folioId}`, {
   250→          method: 'POST',
   251→          headers: { 'Content-Type': 'application/json' },
   252→          body: JSON.stringify({
   253→            type: 'charge',
   254→            transactionType: 'late_checkout',
   255→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   256→            amount: surcharge,
   257→          }),
   258→        })
   259→      }
   260→      return result
   261→    },
   262→    onSuccess: () => {
   263→      invalidate.afterReservationChange(queryClient)
   264→      invalidate.afterFolioChange(queryClient)
   265→      setLateCheckoutDialogOpen(false)
   266→      toast.success('Late checkout request processed successfully')
   267→    },
   268→    onError: () => {
   269→      toast.error('Failed to process late checkout request')
   270→    },
   271→  })
   272→
   273→  // Batch checkout mutation
   274→  const batchCheckoutMutation = useMutation({
   275→    mutationFn: async (ids: string[]) => {
   276→      const results = []
   277→      for (let i = 0; i < ids.length; i++) {
   278→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   279→          method: 'PATCH',
   280→          headers: { 'Content-Type': 'application/json' },
   281→          body: JSON.stringify({ status: 'checked_out' }),
   282→        })
   283→        results.push(result)
   284→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   285→      }
   286→      return results
   287→    },
   288→    onSuccess: (_data, ids) => {
   289→      invalidate.afterCheckout(queryClient)
   290→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   291→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   292→    },
   293→    onError: () => {
   294→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   295→    },
   296→  })
   297→
   298→  // ─── Handlers ─────────────────────────────────────────────────────
   299→
   300→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   301→
   302→  const toggleSelectAll = () => {
   303→    if (selectedIds.size === departures.length && departures.length > 0) {
   304→      setSelectedIds(new Set())
   305→    } else {
   306→      setSelectedIds(new Set(departures.map((d) => d.id)))
   307→    }
   308→  }
   309→
   310→  const handleExportCSV = () => {
   311→    const headers = ['Room', 'Guest', 'Confirmation', 'Check-Out', 'Balance', 'Status']
   312→    const rows = departures.map((d) => [
   313→      d.room?.number ?? 'Unassigned',
   314→      `${d.guest.firstName} ${d.guest.lastName}`,
   315→      d.confirmationNo,
   316→      d.checkOut,
   317→      String(getBalance(d)),
   318→      d.status,
   319→    ])
   320→    const csv = [headers, ...rows].map((r) => r.map((c) => `"${c}"`).join(',')).join('\n')
   321→    const blob = new Blob([csv], { type: 'text/csv' })
   322→    const url = URL.createObjectURL(blob)
   323→    const a = document.createElement('a')
   324→    a.href = url
   325→    a.download = 'departures.csv'
   326→    a.click()
   327→    URL.revokeObjectURL(url)
   328→  }
   329→
   330→  const handleCheckoutSelected = () => {
   331→    const ids = Array.from(selectedIds)
   332→    if (ids.length === 0) return
   333→    batchCheckoutMutation.mutate(ids)
   334→    setSelectedIds(new Set())
   335→  }
   336→
   337→  const handleExpressCheckout = (dep: Departure) => {
   338→    checkoutMutation.mutate(dep.id, {
   339→      onSuccess: () => {
   340→        toast.success(`Express checkout complete for Room ${dep.room?.number ?? 'N/A'}`)
   341→        toast.info(`Room ${dep.room?.number ?? 'N/A'} marked for housekeeping - Vacant Dirty`)
   342→      },
   343→    })
   344→  }
   345→
   346→  const handleReviewFolio = (dep: Departure) => {
   347→    setSelectedDeparture(dep)
   348→    setFolioDialogOpen(true)
   349→  }
   350→
   351→  const handleNavigateToFolio = (dep: Departure) => {
   352→    useFolioContextStore.getState().setFolioContext({
   353→      reservationId: dep.id,
   354→      guestId: dep.guest?.id || '',
   355→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   356→      roomNumber: dep.room?.number || '',
   357→      confirmationNo: dep.confirmationNo,
   358→    })
   359→    navigateTo('front-desk', 'folio')
   360→  }
   361→
   362→  const handleViewLedger = (dep: Departure) => {
   363→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   364→      guestId: dep.guest?.id || '',
   365→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   366→    })
   367→    navigateTo('front-desk', 'guest-ledger')
   368→  }
   369→
   370→  const handleFolioSettleCheckout = () => {
   371→    setFolioDialogOpen(false)
   372→    setCheckoutDialogOpen(true)
   373→  }
   374→
   375→  const handleQuickCheckout = (dep: Departure) => {
   376→    setSelectedDeparture(dep)
   377→    setCheckoutDialogOpen(true)
   378→  }
   379→
   380→  const handleRequestLateCheckout = (dep: Departure) => {
   381→    setSelectedDeparture(dep)
   382→    setLateCheckoutOption('14:00')
   383→    setLateCheckoutDialogOpen(true)
   384→  }
   385→
   386→  const handleConfirmLateCheckout = () => {
   387→    if (!selectedDeparture) return
   388→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   389→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   390→
   391→    // Build new checkout datetime with the selected time
   392→    const currentCheckout = new Date(selectedDeparture.checkOut)
   393→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   394→    currentCheckout.setHours(hours, minutes, 0, 0)
   395→
   396→    lateCheckoutMutation.mutate({
   397→      reservationId: selectedDeparture.id,
   398→      newCheckOut: currentCheckout.toISOString(),
   399→      surcharge,
   400→    })
   401→  }
   402→
   403→  const handleOpenPaymentDialog = () => {
   404→    if (!selectedDeparture) return
   405→    const balance = getBalance(selectedDeparture)
   406→    setPaymentAmount(String(balance))
   407→    setPaymentMethod('cash')
   408→    setPaymentReference('')
   409→    setPaymentDialogOpen(true)
   410→  }
   411→
   412→  const handleRecordPayment = () => {
   413→    if (!selectedDeparture) return
   414→    const folioId = selectedDeparture.folios[0]?.id
   415→    if (!folioId) {
   416→      toast.error('No folio found for this reservation')
   417→      return
   418→    }
   419→    const amount = Number(paymentAmount)
   420→    if (isNaN(amount) || amount <= 0) {
   421→      toast.error('Please enter a valid payment amount')
   422→      return
   423→    }
   424→    paymentMutation.mutate({
   425→      folioId,
   426→      amount,
   427→      method: paymentMethod,
   428→      reference: paymentReference,
   429→    })
   430→  }
   431→
   432→  const confirmCheckout = () => {
   433→    if (!selectedDeparture) return
   434→    const roomNumber = selectedDeparture.room.number
   435→    checkoutMutation.mutate(selectedDeparture.id, {
   436→      onSuccess: () => {
   437→        setCheckoutDialogOpen(false)
   438→        setReceiptData(selectedDeparture)
   439→        setReceiptDialogOpen(true)
   440→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   441→      },
   442→    })
   443→  }
   444→
   445→  const handleBatchCheckout = () => {
   446→    const ids = zeroBalancePending.map((d) => d.id)
   447→    if (ids.length === 0) {
   448→      toast.info('No guests with zero balance to batch checkout')
   449→      return
   450→    }
   451→    batchCheckoutMutation.mutate(ids)
   452→  }
   453→
   454→  const handlePrintReceipt = () => {
   455→    if (!receiptData) return
   456→    const printWindow = window.open('', '_blank', 'width=400,height=600')
   457→    if (!printWindow) { toast.error('Please allow popups to print receipts'); return }
   458→    const guestName = `${receiptData.guest.firstName} ${receiptData.guest.lastName}`
   459→    const roomNum = receiptData.room?.number ?? 'N/A'
   460→    const roomType = receiptData.room?.type?.name ?? ''
   461→    const balance = getBalance(receiptData)
   462→    printWindow.document.write(`
   463→      <html><head><title>Checkout Receipt</title>
   464→      <style>body{font-family:monospace;max-width:350px;margin:0 auto;padding:20px;font-size:12px}
   465→      .center{text-align:center}.bold{font-weight:bold}.line{border-top:1px dashed #000;margin:8px 0}
   466→      .row{display:flex;justify-content:space-between}.green{color:green}.red{color:red}
   467→      h2{margin:0 0 4px}p.sub{margin:0 0 16px;color:#666;font-size:11px}</style></head>
   468→      <body>
   469→      <div class="center"><h2>MERIDIAN HOTEL</h2><p class="sub">Checkout Receipt</p></div>
   470→      <div class="line"></div>
   471→      <div class="row"><span>Guest:</span><span class="bold">${guestName}</span></div>
   472→      <div class="row"><span>Room:</span><span>${roomNum} ${roomType}</span></div>
   473→      <div class="row"><span>Confirmation:</span><span>${receiptData.confirmationNo}</span></div>
   474→      <div class="row"><span>Check-in:</span><span>${formatDate(receiptData.checkIn)}</span></div>
   475→      <div class="row"><span>Check-out:</span><span>${formatDate(receiptData.checkOut)}</span></div>
   476→      <div class="line"></div>
   477→      <div class="row"><span>Room Total:</span><span>${formatCurrency(receiptData.totalAmount)}</span></div>
   478→      <div class="row"><span>Payments:</span><span class="green">-${formatCurrency(receiptData.paidAmount)}</span></div>
   479→      <div class="line"></div>
   480→      <div class="row"><span class="bold">Balance Due:</span><span class="bold ${balance > 0 ? 'red' : 'green'}">${formatCurrency(balance)}</span></div>
   481→      <div class="line"></div>
   482→      <div class="center" style="margin-top:16px;font-size:10px;color:#999">Thank you for staying with us!</div>
   483→      <script>window.print();window.close();</script>
   484→      </body></html>
   485→    `)
   486→    printWindow.document.close()
   487→    toast.success('Receipt sent to printer')
   488→  }
   489→
   490→  const handleEmailReceipt = () => {
   491→    if (!receiptData) return
   492→    const guestEmail = receiptData.guest.email || 'no email on file'
   493→    // In production, this would call an email API
   494→    toast.success(`Receipt emailed to ${guestEmail}`)
   495→  }
   496→
   497→  // ─── Computed values ──────────────────────────────────────────────
   498→
   499→  const lateCheckoutFee = selectedDeparture
   500→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   501→    : 0
   502→
   503→  // ─── Render ────────────────────────────────────────────────────────
   504→
   505→  return (
   506→    <div className="flex flex-col gap-2">
   507→      {/* Header */}
   508→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   509→        <div>
   510→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   511→          <p className="text-xs text-muted-foreground">
   512→            {todayCount} scheduled for {formatDate(today)}{overdueCount > 0 && <span className="text-red-600 font-medium"> · {overdueCount} overdue</span>}
   513→          </p>
   514→        </div>
   515→        <div className="flex items-center gap-2 shrink-0">
   516→          <DropdownMenu>
   517→            <DropdownMenuTrigger asChild>
   518→              <Button variant="outline" size="icon" className="size-8" aria-label="Actions menu">
   519→                <MoreVertical className="size-4" />
   520→                <span className="sr-only">Actions</span>
   521→              </Button>
   522→            </DropdownMenuTrigger>
   523→            <DropdownMenuContent align="end">
   524→              <DropdownMenuItem onClick={handleRefresh}>
   525→                <RefreshCw className="size-4 mr-2" />
   526→                Refresh Data
   527→              </DropdownMenuItem>
   528→              <DropdownMenuItem onClick={handleExportCSV}>
   529→                <Download className="size-4 mr-2" />
   530→                Export to CSV
