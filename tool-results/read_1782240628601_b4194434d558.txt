   133→function nightsBetween(start: string | Date, end: string | Date): number {
   134→  const s = new Date(start)
   135→  const e = new Date(end)
   136→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   137→}
   138→
   139→function getCreditStatus(res: SettlementDeparture) {
   140→  const balance = res.folios[0]?.balance || 0
   141→  const limit = res.creditLimit || 15000
   142→  const pct = limit > 0 ? (balance / limit) * 100 : 0
   143→  if (pct >= 100) return { pct, status: 'breach' as const, color: 'text-red-600' }
   144→  if (pct >= 80) return { pct, status: 'warning' as const, color: 'text-amber-600' }
   145→  return { pct, status: 'ok' as const, color: 'text-green-600' }
   146→}
