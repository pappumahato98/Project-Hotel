   393→      const newCheckOut = format(addDays(date, stayLength), 'yyyy-MM-dd')
   394→      const newTotal = stayLength * reservation.roomRate
   395→
   396→      updateDatesMutation.mutate({
   397→        id: reservation.id,
   398→        checkIn: newCheckIn,
   399→        checkOut: newCheckOut,
   400→        totalAmount: newTotal,
   401→      })
   402→
   403→      setDraggingId(null)
   404→      setDragOverRoomId(null)
   405→      setDragOverDate(null)
   406→      dragRef.current = null
   407→    },
   408→    [updateDatesMutation],

   472→      toast.success('Reservation created successfully')
   473→    },
   474→    onError: (err: Error) => {
   475→      toast.error(err.message || 'Failed to create reservation')
   476→    },
   477→  })
