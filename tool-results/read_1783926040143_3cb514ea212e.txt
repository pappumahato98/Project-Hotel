   100→   100→
   101→   101→function getOutstandingBalance(res: InHouseReservation): number {
   102→   102→  return res.folios.reduce((sum, f) => sum + (f.balance > 0 ? f.balance : 0), 0)
   103→   103→    || (res.totalAmount - res.paidAmount)
   104→   104→}
   105→   105→
   106→   106→function getLastPayment(res: InHouseReservation): string | null {
   107→   107→  const allPayments = res.folios.flatMap((f) => f.payments || [])
   108→   108→  if (allPayments.length === 0) return null
   109→   109→  const latest = allPayments.reduce((a, b) =>
   110→   110→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   111→   111→  )
   112→   112→  return new Date(latest.createdAt).toLocaleDateString()
   113→   113→}
   114→   114→
   115→   115→function getVipBadge(vipLevel: string) {
   116→   116→  if (vipLevel === 'none' || !vipLevel) return null
   117→   117→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   118→   118→  return (
   119→   119→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   120→   120→      {level}
   121→   121→    </Badge>
   122→   122→  )
   123→   123→}
