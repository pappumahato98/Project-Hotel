   193→function generatePreviewConfirmation(): string {
   194→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
   195→  let result = 'MRD-'
   196→  for (let i = 0; i < 6; i++) {
   197→    result += chars.charAt(Math.floor(Math.random() * chars.length))
   198→  }
   199→  return result
   200→}

   202→function formatDayOfWeek(dateStr: string): string {
   203→  const d = new Date(dateStr)
   204→  return d.toLocaleDateString('en-US', { weekday: 'short' })
   205→}
   206→
   207→function formatDateWithDay(dateStr: string): string {
   208→  if (!dateStr) return ''
   209→  const d = new Date(dateStr)
   210→  const day = formatDayOfWeek(dateStr)
   211→  const formatted = d.toLocaleDateString('en-US', { day: '2-digit', month: 'short' })
   212→  return `${day} ${formatted}`
   213→}

   215→function useDebounce<T>(value: T, delay: number): T {
   216→  const [debounced, setDebounced] = useState(value)
   217→  useEffect(() => {
   218→    const timer = setTimeout(() => setDebounced(value), delay)
   219→    return () => clearTimeout(timer)
   220→  }, [value, delay])
   221→  return debounced
   222→}
