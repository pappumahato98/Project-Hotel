   162→// ─── Helpers ────────────────────────────────────────────────────────────
   163→
   164→function guestFullName(g: FolioGuest | null | undefined): string {
   165→  if (!g) return '—'
   166→  return `${g.firstName} ${g.lastName}`
   167→}
   168→
   169→function folioCharges(f: Folio): number {
   170→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   171→}
   172→
   173→function folioPayments(f: Folio): number {
   174→  return f.payments.reduce((s, p) => s + p.amount, 0)
   175→}
   176→
   177→function folioOutstanding(f: Folio): number {
   178→  return folioCharges(f) - folioPayments(f)
   179→}
   180→
   181→function isVoidedTransaction(t: FolioTransaction): boolean {
   182→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   183→}
   184→
   185→function isVoidedPayment(p: FolioPayment): boolean {
   186→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   187→}
   188→
   189→function useDebounce<T>(value: T, delay: number): T {
   190→  const [debounced, setDebounced] = useState(value)
   191→  useEffect(() => {
   192→    const timer = setTimeout(() => setDebounced(value), delay)
   193→    return () => clearTimeout(timer)
   194→  }, [value, delay])
   195→  return debounced
   196→}
