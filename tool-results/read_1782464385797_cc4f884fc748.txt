     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { apiFetch } from '@/lib/api'
     6→import { format, parseISO } from 'date-fns'

     9→  Search, RefreshCw, ChevronDown, ChevronRight,

    15→} from 'lucide-react'
    16→import { toast } from 'sonner'
    17→import { cn } from '@/lib/utils'
    18→import { formatDate, formatCurrency } from '@/lib/format'
    19→import { invalidate } from '@/lib/queryKeys'
    20→import { useNavigationStore, useFolioContextStore } from '@/lib/store'

    22→import { Button } from '@/components/ui/button'

    40→} from '@/components/ui/dialog'

    44→// ─── Types ──────────────────────────────────────────────────────────────
    45→
    46→interface PendingReservation {
    47→  reservationId: string
    48→  confirmationNo: string
    49→  guest: { id: string; firstName: string; lastName: string } | null
    50→  room: { id: string; number: string; type: { name: string } | null } | null
    51→  checkIn: string
    52→  checkOut: string
    53→  roomRate: number
    54→  folio: { id: string; balance: number; status: string } | null
    55→  pendingNights: string[]
    56→  postedNights: string[]
    57→  futureNights: string[]
    58→  totalNights: number
    59→  pendingAmount: number
    60→}
    61→
    62→interface PendingResponse {
    63→  pendingReservations: PendingReservation[]
    64→  summary: { totalReservations: number; totalPendingNights: number; totalPendingAmount: number }
    65→}
    66→
    67→// ─── Constants ──────────────────────────────────────────────────────────
    68→
    69→type PostingType = 'all' | 'no-room' | 'urgent'
    70→
    71→const POSTING_TYPE_OPTIONS: { value: PostingType; label: string }[] = [
    72→  { value: 'all', label: 'All Pending' },
    73→  { value: 'no-room', label: 'No Room' },
    74→  { value: 'urgent', label: 'Urgent (3+)' },
    75→]
    76→
    77→// ─── Component ──────────────────────────────────────────────────────────
    78→
    79→export function RoomRatePostingPage() {
    80→  const queryClient = useQueryClient()
    81→  const navigateTo = useNavigationStore((s) => s.navigateTo)
    82→  const setFolioContext = useFolioContextStore((s) => s.setFolioContext)
    83→
    84→  // State
    85→  const [search, setSearch] = useState('')
    86→  const [searchInput, setSearchInput] = useState('')
    87→  const [postingType, setPostingType] = useState<PostingType>('all')
    88→  const [dateFilter, setDateFilter] = useState('')
    89→  const [calendarOpen, setCalendarOpen] = useState(false)
    90→  const [expandedRow, setExpandedRow] = useState<string | null>(null)
    91→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
    92→
    93→  // Dialogs
    94→  const [postConfirmOpen, setPostConfirmOpen] = useState(false)
    95→  const [postTarget, setPostTarget] = useState<PendingReservation | null>(null)
    96→  const [bulkPostOpen, setBulkPostOpen] = useState(false)
    97→  const [rateOverride, setRateOverride] = useState('')
    98→  const [showQuickActions, setShowQuickActions] = useState(false)
    99→
   100→  // ── Fetch pending reservations ───────────────────────────────────
   101→  const { data: pendingData, isLoading, refetch } = useQuery({
   102→    queryKey: ['rate-posting-pending'],
   103→    queryFn: () => apiFetch<PendingResponse>('/api/room-rate-posting/pending'),
   104→  })
   105→
   106→  // ── Post single reservation's pending nights ────────────────────
   107→  const postMutation = useMutation({
   108→    mutationFn: async ({ reservationId, dates, customRate }: { reservationId: string; dates: string[]; customRate?: number }) => {
   109→      const body: Record<string, unknown> = { reservationId, dates }
   110→      if (customRate && customRate > 0) body.customRate = customRate
   111→      return apiFetch('/api/room-rate-posting', {
   112→        method: 'POST',
   113→        headers: { 'Content-Type': 'application/json' },
   114→        body: JSON.stringify(body),
   115→      })
   116→    },
   117→    onSuccess: () => {
   118→      toast.success('Pending charges posted successfully')
   119→      invalidate.afterFolioChange(queryClient)
   120→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   121→      setPostConfirmOpen(false)
   122→      setPostTarget(null)
   123→      setRateOverride('')
   124→      setSelectedIds(new Set())
   125→    },
   126→    onError: (error) => {
   127→      toast.error(error.message || 'Failed to post charges')
   128→      setPostConfirmOpen(false)
   129→      setPostTarget(null)
   130→      setRateOverride('')
   131→    },
   132→  })
   133→
   134→  // ── Post selected (batch) ───────────────────────────────────────
   135→  const postSelectedMutation = useMutation({
   136→    mutationFn: async (items: { reservationId: string; dates: string[] }[]) => {
   137→      const results: { reservationId: string; success: boolean; error?: string }[] = []
   138→      for (const item of items) {
   139→        try {
   140→          await apiFetch('/api/room-rate-posting', {
   141→            method: 'POST',
   142→            headers: { 'Content-Type': 'application/json' },
   143→            body: JSON.stringify({ reservationId: item.reservationId, dates: item.dates }),
   144→          })
   145→          results.push({ reservationId: item.reservationId, success: true })
   146→        } catch (e) {
   147→          results.push({ reservationId: item.reservationId, success: false, error: (e as Error).message })
   148→        }
   149→      }
   150→      return results
   151→    },
   152→    onSuccess: (results) => {
   153→      const ok = results.filter((r) => r.success).length
   154→      const fail = results.filter((r) => !r.success).length
   155→      if (fail > 0) toast.warning(`${ok} posted, ${fail} failed`)
   156→      else toast.success(`${ok} reservation(s) posted successfully`)
   157→      invalidate.afterFolioChange(queryClient)
   158→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   159→      setSelectedIds(new Set())
   160→    },
   161→    onError: (error) => toast.error(error.message || 'Batch post failed'),
   162→  })
   163→
   164→  // ── Bulk post all pending ───────────────────────────────────────
   165→  const bulkPostMutation = useMutation({
   166→    mutationFn: async () => {
   167→      return apiFetch('/api/room-rate-posting/bulk-post', {
   168→        method: 'POST',
   169→        headers: { 'Content-Type': 'application/json' },
   170→        body: JSON.stringify({ postAll: true }),
   171→      })
   172→    },
   173→    onSuccess: (data) => {
   174→      const result = data as { message?: string }
   175→      toast.success(result?.message || 'Bulk post completed')
   176→      invalidate.afterFolioChange(queryClient)
   177→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   178→      setBulkPostOpen(false)
   179→    },
   180→    onError: (error) => {
   181→      toast.error(error.message || 'Bulk post failed')
   182→      setBulkPostOpen(false)
   183→    },
   184→  })
   185→
   186→  // ── Derived data ────────────────────────────────────────────────
   187→  const pendingSummary = pendingData?.summary
   188→  const pendingReservations = pendingData?.pendingReservations || []
   189→
   190→  const filteredPendingReservations = useMemo(() => {
   191→    let list = [...pendingReservations]
   192→
   193→    // Posting type filter
   194→    if (postingType === 'no-room') list = list.filter((r) => !r.room)
   195→    else if (postingType === 'urgent') list = list.filter((r) => r.pendingNights.length >= 3)
   196→
   197→    // Search (name, room, conf#)
   198→    if (search) {
   199→      const s = search.toLowerCase()
   200→      list = list.filter((r) =>
   201→        r.guest?.firstName?.toLowerCase().includes(s) ||
   202→        r.guest?.lastName?.toLowerCase().includes(s) ||
   203→        r.confirmationNo?.toLowerCase().includes(s) ||
   204→        r.room?.number?.toLowerCase().includes(s)
   205→      )
   206→    }
   207→
   208→    // Date filter
   209→    if (dateFilter) {
   210→      list = list.filter((r) => r.pendingNights.includes(dateFilter))
   211→    }
   212→
   213→    // Sort by room number ascending
   214→    list.sort((a, b) => (a.room?.number || '').localeCompare(b.room?.number || '', undefined, { numeric: true }))
   215→    return list
   216→  }, [pendingReservations, search, postingType, dateFilter])
   217→
   218→  const pendingStats = useMemo(() => {
   219→    const noRoom = filteredPendingReservations.filter((r) => !r.room).length
   220→    const avgRate = filteredPendingReservations.length > 0
   221→      ? Math.round(filteredPendingReservations.reduce((s, r) => s + r.roomRate, 0) / filteredPendingReservations.length)
   222→      : 0
   223→    const totalFolioBalance = filteredPendingReservations.reduce((s, r) => s + (r.folio?.balance || 0), 0)
   224→    const totalPendingNights = filteredPendingReservations.reduce((s, r) => s + r.pendingNights.length, 0)
   225→    const totalPendingAmount = filteredPendingReservations.reduce((s, r) => s + r.pendingAmount, 0)
   226→    return { noRoom, avgRate, totalFolioBalance, totalPendingNights, totalPendingAmount }
   227→  }, [filteredPendingReservations])
   228→
   229→  const isWorking = postMutation.isPending || bulkPostMutation.isPending || postSelectedMutation.isPending
   230→
   231→  // ── Handlers ─────────────────────────────────────────────────────
   232→  const handleSearch = useCallback(() => setSearch(searchInput), [searchInput])
   233→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => { if (e.key === 'Enter') handleSearch() }, [handleSearch])
   234→
   235→  const clearFilters = useCallback(() => {
   236→    setSearchInput(''); setSearch(''); setPostingType('all'); setDateFilter(''); setCalendarOpen(false)
   237→  }, [])
   238→
   239→  const handlePostReservation = useCallback(() => {
   240→    if (!postTarget) return
   241→    const customRate = rateOverride ? parseFloat(rateOverride) : undefined
   242→    postMutation.mutate({ reservationId: postTarget.reservationId, dates: postTarget.pendingNights, customRate })
   243→  }, [postTarget, rateOverride, postMutation])
   244→
   245→  const handleBulkPost = useCallback(() => bulkPostMutation.mutate(), [bulkPostMutation])
   246→
   247→  const handlePostSelected = useCallback(() => {
   248→    const items = filteredPendingReservations
   249→      .filter((r) => selectedIds.has(r.reservationId))
   250→      .map((r) => ({ reservationId: r.reservationId, dates: r.pendingNights }))
   251→    if (items.length === 0) return
   252→    postSelectedMutation.mutate(items)
   253→  }, [filteredPendingReservations, selectedIds, postSelectedMutation])
   254→
   255→  const toggleExpand = useCallback((id: string) => {
   256→    setExpandedRow((prev) => (prev === id ? null : id))
   257→  }, [])
   258→
   259→  const toggleSelectAll = useCallback(() => {
   260→    if (selectedIds.size === filteredPendingReservations.length) setSelectedIds(new Set())
   261→    else setSelectedIds(new Set(filteredPendingReservations.map((r) => r.reservationId)))
   262→  }, [selectedIds.size, filteredPendingReservations])
   263→
   264→  const toggleSelect = useCallback((id: string) => {
   265→    setSelectedIds((prev) => { const n = new Set(prev); if (n.has(id)) n.delete(id); else n.add(id); return n })
   266→  }, [])
   267→
   268→  // ── Navigation actions ───────────────────────────────────────────
   269→  const handleViewFolio = useCallback((res: PendingReservation) => {
   270→    if (res.guest && res.room && res.folio) {
   271→      setFolioContext({
   272→        reservationId: res.reservationId, guestId: res.guest.id,
   273→        guestName: `${res.guest.firstName} ${res.guest.lastName}`,
   274→        roomNumber: res.room.number, confirmationNo: res.confirmationNo, folioId: res.folio.id,
   275→      })
   276→      navigateTo('front-desk', 'folio')
   277→    } else { toast.error('Cannot open folio — missing guest, room, or folio data') }
   278→  }, [navigateTo, setFolioContext])
   279→
   280→  const handleViewReservation = useCallback((res: PendingReservation) => {
   281→    navigateTo('front-desk', 'reservations')
   282→    toast.info(`Viewing reservation ${res.confirmationNo}`)
   283→  }, [navigateTo])
   284→
   285→  const handleViewGuest = useCallback((res: PendingReservation) => {
   286→    navigateTo('crm', 'profiles')
   287→    toast.info('Viewing guest profile')
   288→  }, [navigateTo])
   289→
   290→  // ── Export CSV ──────────────────────────────────────────────────
   291→  const handleExport = useCallback(() => {
   292→    const rows = filteredPendingReservations.map((r) => ({
   293→      Room: r.room?.number || '', Guest: r.guest ? `${r.guest.firstName} ${r.guest.lastName}` : '',
   294→      'Check-In': formatDate(r.checkIn), 'Check-Out': formatDate(r.checkOut),
   295→      'Rate/Night': r.roomRate, 'Pending Nights': r.pendingNights.length,
   296→      'Posted Nights': r.postedNights.length, 'Future Nights': r.futureNights.length,
   297→      'Pending Amount': r.pendingAmount, 'Folio Balance': r.folio?.balance || 0,
   298→      'Confirmation #': r.confirmationNo,
   299→    }))
   300→    if (rows.length === 0) { toast.info('No data to export'); return }
   301→    const headers = Object.keys(rows[0])
   302→    const csv = [headers.join(','), ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h]}"`).join(','))].join('\n')
   303→    const blob = new Blob([csv], { type: 'text/csv' })
   304→    const url = URL.createObjectURL(blob)
   305→    const a = document.createElement('a'); a.href = url; a.download = `rate-posting-${new Date().toISOString().split('T')[0]}.csv`
   306→    a.click(); URL.revokeObjectURL(url); toast.success(`Exported ${rows.length} records`)
   307→  }, [filteredPendingReservations])
   308→
   309→  const hasActiveFilters = search || postingType !== 'all' || dateFilter
   310→
   311→  // ── Render ───────────────────────────────────────────────────────
   312→  return (
   313→    <TooltipProvider delayDuration={300}>
   314→    <div className="space-y-4">
   315→      {/* ═══════════════ HEADER ═══════════════ */}
   316→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   317→        <div>
   318→          <h2 className="text-lg font-semibold flex items-center gap-2">
   319→            <DollarSign className="size-5 text-emerald-600" />
   320→            Room Rate Posting
   321→          </h2>
   322→          <p className="text-sm text-muted-foreground mt-0.5">
   323→            Manage daily room charge postings for in-house guests
   324→          </p>
   325→        </div>
   326→        <div className="flex items-center gap-2">
   327→          <Button variant="outline" size="sm" onClick={() => refetch()} className="gap-1.5">
   328→            <RefreshCw className="size-3.5" /> Refresh
   329→          </Button>
   330→          <Button variant="outline" size="sm" onClick={handleExport} className="gap-1.5">
   331→            <Download className="size-3.5" /> Export
   332→          </Button>
   333→          <DropdownMenu open={showQuickActions} onOpenChange={setShowQuickActions}>
   334→            <DropdownMenuTrigger asChild>
   335→              <Button size="sm" className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   336→                <Zap className="size-3.5" /> Quick Actions <ChevronDown className="size-3" />
   337→              </Button>
   338→            </DropdownMenuTrigger>
   339→            <DropdownMenuContent align="end" className="w-64">
   340→              <DropdownMenuLabel className="text-xs">Posting Actions</DropdownMenuLabel>
   341→              <DropdownMenuSeparator />
   342→              <DropdownMenuItem
   343→                onClick={() => { setShowQuickActions(false); setBulkPostOpen(true) }}
   344→                disabled={isWorking || pendingReservations.length === 0} className="gap-2.5 py-2"
   345→              >
   346→                <Zap className="size-4 text-emerald-600" />
   347→                <div className="flex-1">
   348→                  <p className="font-medium text-sm">Post All Pending</p>
   349→                  <p className="text-[10px] text-muted-foreground">Post charges for all {pendingSummary?.totalReservations || 0} reservations</p>
   350→                </div>
   351→              </DropdownMenuItem>
   352→              {selectedIds.size > 0 && (
   353→                <DropdownMenuItem
   354→                  onClick={() => { setShowQuickActions(false); handlePostSelected() }}
   355→                  disabled={isWorking} className="gap-2.5 py-2"
   356→                >
   357→                  <ListChecks className="size-4 text-blue-600" />
   358→                  <div className="flex-1">
   359→                    <p className="font-medium text-sm">Post Selected ({selectedIds.size})</p>
   360→                    <p className="text-[10px] text-muted-foreground">Post charges for selected reservations</p>
   361→                  </div>
   362→                </DropdownMenuItem>
   363→              )}
   364→              <DropdownMenuSeparator />
   365→              <DropdownMenuLabel className="text-xs">Navigate To</DropdownMenuLabel>
   366→              <DropdownMenuSeparator />
   367→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'in-house') }} className="gap-2.5 py-2">
   368→                <BedDouble className="size-4 text-teal-600" />
   369→                <div className="flex-1"><p className="font-medium text-sm">In-House Guests</p></div>
   370→              </DropdownMenuItem>
   371→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'folio') }} className="gap-2.5 py-2">
   372→                <Receipt className="size-4 text-violet-600" />
   373→                <div className="flex-1"><p className="font-medium text-sm">Guest Folio</p></div>
   374→              </DropdownMenuItem>
   375→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'calendar') }} className="gap-2.5 py-2">
   376→                <CalendarDays className="size-4 text-sky-600" />
   377→                <div className="flex-1"><p className="font-medium text-sm">Reservation Calendar</p></div>
   378→              </DropdownMenuItem>
   379→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'reports') }} className="gap-2.5 py-2">
   380→                <FileSpreadsheet className="size-4 text-amber-600" />
   381→                <div className="flex-1"><p className="font-medium text-sm">Reports</p></div>
   382→              </DropdownMenuItem>
   383→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('accounting', 'ledger') }} className="gap-2.5 py-2">
   384→                <CreditCard className="size-4 text-purple-600" />
   385→                <div className="flex-1"><p className="font-medium text-sm">Accounting Ledger</p></div>
   386→              </DropdownMenuItem>
   387→            </DropdownMenuContent>
   388→          </DropdownMenu>
   389→        </div>
   390→      </div>
   391→
   392→      {/* ═══════════════ STAT CARDS ═══════════════ */}
   393→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   394→        <StatCard icon={<Clock className="size-4 text-amber-600" />} label="Pending" value={String(pendingSummary?.totalReservations ?? 0)} sublabel={`${pendingSummary?.totalPendingNights ?? 0} nights`} bgClass="bg-amber-50 border-amber-200" />
   395→        <StatCard icon={<Wallet className="size-4 text-emerald-600" />} label="Pending Amount" value={pendingSummary ? formatCurrency(pendingSummary.totalPendingAmount) : 'NPR 0'} sublabel="Total charges" bgClass="bg-emerald-50 border-emerald-200" />
   396→        <StatCard icon={<Users className="size-4 text-blue-600" />} label="In-House" value={String(filteredPendingReservations.length)} sublabel="With pending" bgClass="bg-blue-50 border-blue-200" />
   397→        <StatCard icon={<BarChart3 className="size-4 text-violet-600" />} label="Avg Rate/Night" value={pendingStats.avgRate > 0 ? formatCurrency(pendingStats.avgRate) : 'NPR 0'} sublabel="Of pending" bgClass="bg-violet-50 border-violet-200" />
   398→        <StatCard icon={<Building2 className="size-4 text-sky-600" />} label="Folio Balance" value={pendingStats.totalFolioBalance > 0 ? formatCurrency(pendingStats.totalFolioBalance) : 'NPR 0'} sublabel="Combined balance" bgClass="bg-sky-50 border-sky-200" />
   399→        <StatCard icon={<AlertCircle className="size-4 text-red-600" />} label="Issues" value={String(pendingStats.noRoom)} sublabel={pendingStats.noRoom > 0 ? 'No room assigned' : 'All assigned'} bgClass="bg-red-50 border-red-200" />
   400→      </div>
   401→
   402→      {/* ═══════════════ FILTER BAR ═══════════════ */}
   403→      <div className="flex items-center gap-2 flex-wrap">
   404→        {/* Search box — searches by name, room, conf# */}
   405→        <div className="relative">
   406→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground pointer-events-none" />
   407→          <Input
   408→            placeholder="Search guest, room, conf#..."
   409→            value={searchInput}
   410→            onChange={(e) => setSearchInput(e.target.value)}
   411→            onKeyDown={handleKeyDown}
   412→            className="h-9 w-[330px] pl-8 pr-8 text-sm bg-background border-border/80"
   413→          />
   414→          {search && (
   415→            <button
   416→              onClick={() => { setSearchInput(''); setSearch('') }}
   417→              className="absolute right-2 top-1/2 -translate-y-1/2 p-0.5 rounded-sm text-muted-foreground hover:text-foreground transition-colors"
   418→            >
   419→              <X className="size-3.5" />
   420→            </button>
   421→          )}
   422→        </div>
   423→
   424→        {/* Posting type toggle buttons */}
   425→        <div className="flex items-center rounded-lg border bg-muted/40 p-0.5 gap-0.5">
   426→          {POSTING_TYPE_OPTIONS.map((opt) => (
   427→            <Button
   428→              key={opt.value}
   429→              variant="ghost"
   430→              size="sm"
   431→              onClick={() => setPostingType(opt.value)}
   432→              className={cn(
   433→                'h-8 text-xs px-3 rounded-md transition-all',
   434→                postingType === opt.value
   435→                  ? 'bg-background shadow-sm font-medium text-foreground'
