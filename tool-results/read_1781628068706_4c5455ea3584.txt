   395→/** Aging tab content */
   396→function AgingTab({ summary, stays, onNavigateToFolio, onNavigateToReservation }: { summary: LedgerSummary; stays: LedgerStay[]; onNavigateToFolio?: (stay: LedgerStay) => void; onNavigateToReservation?: (stay: LedgerStay) => void }) {
   397→  const totalAging =
   398→    summary.aging.current +
   399→    summary.aging.days30 +
   400→    summary.aging.days60 +
   401→    summary.aging.days90 +
   402→    summary.aging.over90
   403→
   404→  return (
   405→    <div className="space-y-6">
   406→      {/* Horizontal bar chart */}
   407→      <Card className="rounded-xl border bg-card shadow-sm">

   411→        <CardContent className="space-y-4">
   412→          {/* Bar visualization */}
   413→          <div className="flex h-8 rounded-md overflow-hidden bg-muted">
   414→            {totalAging > 0
   415→              ? AGING_COLORS.map(({ key, color }) => {
   416→                  const val = summary.aging[key as keyof typeof summary.aging]
   417→                  const pct = (val / totalAging) * 100
   418→                  if (pct === 0) return null
   419→                  return (
   420→                    <div
   421→                      key={key}
   422→                      className={cn('h-full', color)}
   423→                      style={{ width: `${pct}%` }}
   424→                      title={`${AGING_COLORS.find((c) => c.key === key)?.label}: ${formatCurrency(val)}`}
   425→                    />
   426→                  )
   427→                })
