   152→// ─── Component ──────────────────────────────────────────────────────────
   153→
   154→interface CheckInViewProps {
   155→  reservationId?: string

   158→
   159→export function CheckInView({ reservationId, onComplete }: CheckInViewProps) {
   160→  const queryClient = useQueryClient()
   161→  const { settings } = useSettingsStore()
   162→  const today = getTodayString()
   163→
   164→  // ─── Step state ────────────────────────────────────────────────
   165→  const [currentStep, setCurrentStep] = useState(0)
   166→
   167→  // ─── Selection states ────────────────────────────────────────
   168→  const [selectedReservationId, setSelectedReservationId] = useState<string>('')
   169→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   170→  const [roomSearchFilter, setRoomSearchFilter] = useState('')
   171→
   172→  // ─── Registration card form ────────────────────────────────────
   173→  const [regTitle, setRegTitle] = useState('')
   174→  const [regFirstName, setRegFirstName] = useState('')
   175→  const [regLastName, setRegLastName] = useState('')
   176→  const [regContactNo, setRegContactNo] = useState('')
   177→  const [regEmail, setRegEmail] = useState('')
   178→  const [regAddress, setRegAddress] = useState('')
   179→  const [regNationality, setRegNationality] = useState('')
   180→  const [regIdType, setRegIdType] = useState('passport')
   181→  const [regIdNumber, setRegIdNumber] = useState('')
   182→  const [regCity, setRegCity] = useState('')
   183→  const [regCountry, setRegCountry] = useState('')
   184→
   185→  // ─── Stay preferences ─────────────────────────────────────────
   186→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   187→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   188→  const [pillowType, setPillowType] = useState('')
   189→  const [roomPreference, setRoomPreference] = useState('')
   190→  const [wakeupCall, setWakeupCall] = useState('')
   191→
   192→  // ─── Billing ──────────────────────────────────────────────────
   193→  const [depositAmount, setDepositAmount] = useState('')
   194→  const [paymentMethod, setPaymentMethod] = useState('')
   195→  const [billingNotes, setBillingNotes] = useState('')
   196→
   197→  // ─── Key card dialog ───────────────────────────────────────────
   198→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   199→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   200→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   201→  const [keyCardGuestName, setKeyCardGuestName] = useState('')
   202→  const [keyCardCheckOut, setKeyCardCheckOut] = useState('')
   203→
   204→  // ─── Queries ─────────────────────────────────────────────────
   205→
   206→  // Fetch specific reservation by ID (if prop provided)
   207→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   208→    queryKey: ['reservation', reservationId],
   209→    queryFn: () => apiFetch(`/api/reservations/${reservationId}`),
   210→    enabled: !!reservationId,
   211→  })
   212→
   213→  // Fetch today's confirmed arrivals
   214→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   215→    queryKey: ['arrivals', today],
   216→    queryFn: async () => {
   217→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   218→      return apiFetch(`/api/reservations?${params.toString()}`)
   219→    },
   220→    enabled: !reservationId,
   221→  })
   222→
   223→  // Fetch all rooms with types
   224→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   225→    queryKey: ['rooms', 'all'],
   226→    queryFn: () => apiFetch('/api/rooms'),
   227→  })
   228→
   229→  // ─── Derived data ──────────────────────────────────────────────
   230→
   231→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   232→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   233→
   234→  // The reservation we're working with
   235→  const activeReservation: ReservationInfo | null = reservationId
   236→    ? specificReservation
   237→    : arrivals.find((a) => a.id === selectedReservationId) || null
   238→
   239→  // Room type data from API (not hardcoded)
   240→  const roomTypes: RoomTypeData[] = useMemo(() => {
   241→    if (!roomsData?.roomTypes) return []
   242→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   243→      id: rt.id,
   244→      name: rt.name,
   245→      code: rt.code,
   246→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   247→    }))
   248→  }, [roomsData])
   249→
   250→  // Available rooms (vacant_clean or inspected)
   251→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   252→  const availableRooms = allRooms.filter(
   253→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   254→  )
   255→
   256→  // Filter available rooms by search
   257→  const filteredAvailableRooms = useMemo(() => {
   258→    let filtered = availableRooms
   259→    if (roomSearchFilter.trim()) {
   260→      const q = roomSearchFilter.toLowerCase()
   261→      filtered = filtered.filter(
   262→        (r: RoomInfo) =>
   263→          r.number.toLowerCase().includes(q) ||
   264→          r.type.name.toLowerCase().includes(q) ||
   265→          String(r.floor).includes(q) ||
   266→          (r.wing && r.wing.toLowerCase().includes(q))
   267→      )
   268→    }
   269→    return filtered
   270→  }, [availableRooms, roomSearchFilter])
   271→
   272→  // Group rooms by floor
   273→  const roomsByFloor = useMemo(() => {
   274→    const grouped = new Map<number, RoomInfo[]>()
   275→    for (const room of filteredAvailableRooms) {
   276→      const floor = room.floor
   277→      if (!grouped.has(floor)) grouped.set(floor, [])
   278→      grouped.get(floor)!.push(room)
   279→    }
   280→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   281→  }, [filteredAvailableRooms])
   282→
   283→  // Get room type name by ID
   284→  const getRoomTypeName = (typeId: string) => {
   285→    const rt = roomTypes.find((t) => t.id === typeId)
   286→    return rt?.name || 'Unknown'
   287→  }
   288→
   289→  // Get room rate by type ID
   290→  const getRoomTypeRate = (typeId: string) => {
   291→    const rt = roomTypes.find((t) => t.id === typeId)
   292→    return rt?.baseRate || 0
   293→  }
   294→
   295→  // ─── Initialize form when reservation loads ───────────────────
   296→
   297→  React.useEffect(() => {
   298→    if (specificReservation && reservationId) {
   299→      setSelectedReservationId(specificReservation.id)
   300→      if (specificReservation.room?.id) {
   301→        setSelectedRoomId(specificReservation.room.id)
   302→      }
   303→      setCurrentStep(specificReservation.guest ? 1 : 0)
   304→    }
   305→  }, [specificReservation, reservationId])
   306→
   307→  // Pre-fill registration fields when guest is selected
   308→  React.useEffect(() => {
   309→    if (activeReservation?.guest) {
   310→      const g = activeReservation.guest
   311→      setRegTitle('')
   312→      setRegFirstName(g.firstName || '')
   313→      setRegLastName(g.lastName || '')
   314→      setRegContactNo(g.phone || '')
   315→      setRegEmail(g.email || '')
   316→      setRegNationality(g.nationality || '')
   317→      setRegIdType(g.idType || 'passport')
   318→      setRegIdNumber(g.idNumber || '')
   319→      setRegAddress(g.address || '')
   320→      setRegCity(g.city || '')
   321→      setRegCountry(g.country || '')
   322→    }
   323→  }, [activeReservation?.guest])
   324→
   325→  // Auto-set deposit when room is selected
   326→  React.useEffect(() => {
   327→    if (activeReservation?.roomRate && !depositAmount) {
   328→      setDepositAmount(String(activeReservation.roomRate))
   329→    }
   330→  }, [activeReservation?.roomRate, depositAmount])
   331→
   332→  // ─── Mutations ────────────────────────────────────────────────
   333→
   334→  const checkInMutation = useMutation({
   335→    mutationFn: async ({
   336→      reservationId: resId,
   337→      roomId,
   338→      specialRequests,
   339→    }: {
   340→      reservationId: string
   341→      roomId: string
   342→      specialRequests: string
   343→    }) => {
   344→      return apiFetch(`/api/reservations/${resId}`, {
   345→        method: 'PATCH',
   346→        headers: { 'Content-Type': 'application/json' },
   347→        body: JSON.stringify({
   348→          roomId,
   349→          status: 'checked_in',
   350→          specialRequests,
   351→        }),
   352→      })
   353→    },
   354→    onSuccess: () => {
   355→      invalidate.afterCheckIn(queryClient)
   356→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   357→      toast.success('Guest checked in successfully!')
   358→    },
   359→    onError: (error) => {
   360→      toast.error('Check-in failed', {
   361→        description: error instanceof Error ? error.message : 'Unknown error',
   362→      })
   363→    },
   364→  })
   365→
   366→  // ─── Special Requests Builder ─────────────────────────────────
   367→
   368→  const buildSpecialRequests = (): string => {
   369→    const parts: string[] = []
   370→
   371→    if (activeReservation?.specialRequests) {
   372→      parts.push(activeReservation.specialRequests)
   373→    }
   374→    if (earlyCheckIn) {
   375→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   376→    }
   377→    if (checkOutTime !== settings.defaultCheckOut) {
   378→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   379→    }
   380→    if (roomPreference) {
   381→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   382→      parts.push(`[PREFERENCE] Room: ${label}`)
   383→    }
   384→    if (pillowType) {
   385→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   386→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   387→    }
   388→    if (wakeupCall) {
   389→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   390→    }
   391→    if (billingNotes) {
   392→      parts.push(`[BILLING] ${billingNotes}`)
   393→    }
   394→    if (paymentMethod) {
   395→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   396→    }
   397→    if (depositAmount && parseFloat(depositAmount) > 0) {
   398→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   399→    }
   400→
   401→    return parts.join(' | ')
   402→  }
   403→
   404→  // ─── Step Navigation ──────────────────────────────────────────
   405→
   406→  const canGoNext = (): boolean => {
   407→    switch (currentStep) {
   408→      case 0: return !!activeReservation
   409→      case 1: return true // Guest verification is info-only
   410→      case 2: return !!selectedRoomId
   411→      case 3: return true // Preferences are optional
   412→      case 4: return true // Billing is optional
   413→      case 5: return keyCardIssued
   414→      default: return false
   415→    }
   416→  }
   417→
   418→  const handleNext = () => {
   419→    if (currentStep === 5) {
   420→      onComplete?.()
   421→      return
   422→    }
   423→
   424→    // On step 4 → 5, execute check-in
   425→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   426→      const requests = buildSpecialRequests()
   427→      checkInMutation.mutate(
   428→        {
   429→          reservationId: activeReservation.id,
   430→          roomId: selectedRoomId,
   431→          specialRequests: requests,
   432→        },
   433→        {
   434→          onSuccess: () => {
   435→            // Find the selected room number for key card
   436→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   437→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   438→            const guestName = activeReservation.guest
   439→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   440→              : 'Guest'
   441→
   442→            setKeyCardRoomNumber(roomNumber)
   443→            setKeyCardGuestName(guestName)
   444→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   445→            setKeyCardIssued(false)
   446→            setKeyCardDialogOpen(true)
   447→            setCurrentStep(5)
   448→          },
   449→        }
   450→      )
   451→      return
   452→    }
   453→
   454→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   455→  }
   456→
   457→  const handleBack = () => {
   458→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   459→  }
   460→
   461→  // ─── Render: Step Indicator ────────────────────────────────────
   462→
   463→  const renderStepIndicator = () => (
   464→    <div className="mb-4">
   465→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   466→        {STEP_LABELS.map((label, idx) => (
   467→          <React.Fragment key={idx}>
   468→            <div className="flex flex-col items-center gap-1 min-w-0">
   469→              <div
   470→                className={cn(
   471→                  'flex size-7 shrink-0 items-center justify-center rounded-full border-2 text-[10px] font-bold transition-colors',
   472→                  idx < currentStep && 'border-green-500 bg-green-500 text-white',
   473→                  idx === currentStep && 'border-primary bg-primary text-primary-foreground',
   474→                  idx > currentStep && 'border-muted-foreground/30 bg-background text-muted-foreground'
   475→                )}
   476→              >
   477→                {idx < currentStep ? <Check className="size-4" /> : idx + 1}
   478→              </div>
   479→              <span
   480→                className={cn(
   481→                  'text-[10px] sm:text-xs font-medium whitespace-nowrap',
   482→                  idx === currentStep ? 'text-primary' : 'text-muted-foreground'
   483→                )}
   484→              >
   485→                {label}
   486→              </span>
   487→            </div>
   488→            {idx < STEP_LABELS.length - 1 && (
   489→              <div
   490→                className={cn(
   491→                  'flex-1 h-0.5 mx-1 min-w-[12px] max-w-[40px] transition-colors',
   492→                  idx < currentStep ? 'bg-green-500' : 'bg-muted-foreground/20'
   493→                )}
   494→              />
   495→            )}
   496→          </React.Fragment>
   497→        ))}
   498→      </div>
   499→    </div>
   500→  )
   501→
   502→  // ─── Render: Step 0 - Select Arrival ─────────────────────────
   503→
   504→  const renderSelectArrival = () => {
   505→    if (loadingArrivals) {
   506→      return (
   507→        <div className="space-y-3 p-4">
   508→          {Array.from({ length: 4 }).map((_, i) => (
   509→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   510→          ))}
   511→        </div>
   512→      )
   513→    }
   514→
   515→    if (arrivals.length === 0) {
   516→      return (
   517→        <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   518→          <UserCheck className="size-12 mb-3 opacity-30" />
   519→          <p className="text-sm font-medium">No arrivals scheduled for today</p>
   520→          <p className="text-xs mt-1">Check-in date: {formatDate(today)}</p>
   521→        </div>
   522→      )
   523→    }
   524→
   525→    return (
   526→      <div className="space-y-2">
   527→        <div className="flex items-center gap-2 mb-3">
   528→          <Search className="size-4 text-muted-foreground" />
   529→          <p className="text-sm text-muted-foreground">
   530→            {arrivals.length} confirmed arrival{arrivals.length !== 1 ? 's' : ''} for today
   531→          </p>
   532→        </div>
   533→        <div className="max-h-96 overflow-y-auto space-y-2">
   534→          {arrivals.map((arrival) => {
   535→            const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   536→            const isUnassigned = !arrival.room
   537→            const isSelected = arrival.id === selectedReservationId
   538→
   539→            return (
   540→              <button
   541→                key={arrival.id}
   542→                onClick={() => {
   543→                  setSelectedReservationId(arrival.id)
   544→                  // Pre-fill room if already assigned
   545→                  if (arrival.room?.id) {
   546→                    setSelectedRoomId(arrival.room.id)
   547→                  }
   548→                  setCurrentStep(1)
   549→                }}
   550→                className={cn(
   551→                  'w-full text-left rounded-lg border p-3 sm:p-4 transition-all',
   552→                  isSelected && 'ring-2 ring-primary',
   553→                  isSelected
   554→                    ? 'border-primary bg-primary/5'
   555→                    : 'border-border hover:border-primary/40 hover:bg-muted/50',
   556→                  isUnassigned && 'border-l-4 border-l-amber-400'
   557→                )}
   558→              >
   559→                <div className="flex items-start gap-3">
   560→                  <Avatar className="size-10 shrink-0">
