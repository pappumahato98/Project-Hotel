    35→// ─── Types ──────────────────────────────────────────────────────────────
    36→
    37→interface SettlementGuest {
    38→  id: string
    39→  firstName: string
    40→  lastName: string
    41→  email: string
    42→  phone: string
    43→  vipLevel: string
    44→}
    45→
    46→interface SettlementRoom {
    47→  id: string
    48→  number: string
    49→  floor: number
    50→  wing: string
    51→  type: { name: string; code: string; bedConfig?: string }
    52→}
    53→
    54→interface SettlementFolio {
    55→  id: string
    56→  balance: number
    57→  status: string
    58→  payments?: { id: string; amount: number; createdAt: string }[]
    59→}
    60→
    61→interface InHouseReservation {
    62→  id: string
    63→  confirmationNo: string
    64→  guest: SettlementGuest
    65→  room: SettlementRoom
    66→  adults: number
    67→  children: number
    68→  checkIn: string
    69→  checkOut: string
    70→  roomRate: number
    71→  totalAmount: number
    72→  paidAmount: number
    73→  folios: SettlementFolio[]
    74→}
    75→
    76→type PaymentMethod = 'cash' | 'card' | 'bank_transfer' | 'digital_wallet'
    77→
    78→type BalanceFilter = 'all' | 'over_10k' | 'over_50k' | 'over_100k'
    79→

    88→  { value: 'all', label: 'All Balances' },
    89→  { value: 'over_10k', label: 'Over 10K NPR' },

    94→// ─── Helpers ────────────────────────────────────────────────────────────
    95→
    96→function getOutstandingBalance(res: InHouseReservation): number {
    97→  return res.folios[0]?.balance ?? (res.totalAmount - res.paidAmount)
    98→}
    99→
   100→function getLastPayment(res: InHouseReservation): string | null {
   101→  const payments = res.folios[0]?.payments
   102→  if (!payments || payments.length === 0) return null
   103→  const latest = payments.reduce((a, b) =>
   104→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   105→  )
   106→  return new Date(latest.createdAt).toLocaleDateString()
   107→}
   108→
   109→function getVipBadge(vipLevel: string) {
   110→  if (vipLevel === 'none' || !vipLevel) return null
   111→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   112→  return (
   113→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   114→      {level}
   115→    </Badge>
   116→  )
   117→}

   121→export function SettlementView() {
   122→  const queryClient = useQueryClient()
   123→  const { navigateTo } = useNavigationStore()
   124→
   125→  // Dialog states
   126→  const [settleDialogOpen, setSettleDialogOpen] = useState(false)
   127→  const [batchDialogOpen, setBatchDialogOpen] = useState(false)
   128→
   129→  // Selected reservation for single settle
   130→  const [selectedReservation, setSelectedReservation] = useState<InHouseReservation | null>(null)
   131→
   132→  // Search & filter
   133→  const [searchQuery, setSearchQuery] = useState('')
   134→  const [balanceFilter, setBalanceFilter] = useState<BalanceFilter>('all')
   135→
   136→  // Payment form (single)
   137→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   138→  const [paymentAmount, setPaymentAmount] = useState('')
   139→  const [paymentReference, setPaymentReference] = useState('')
   140→
   141→  // Batch payment form
   142→  const [batchPaymentMethod, setBatchPaymentMethod] = useState<PaymentMethod>('cash')
   143→
   144→  // ─── Data Fetching ──────────────────────────────────────────────────
   145→
   146→  const { data, isLoading } = useQuery({
   147→    queryKey: ['reservations', 'checked_in', 'settlement'],
   148→    queryFn: async () => {
   149→      return apiFetch<{ reservations: InHouseReservation[]; total: number }>(
   150→        '/api/reservations?status=checked_in'
   151→      )
   152→    },
   153→    refetchInterval: 30000,
   154→  })
   155→
   156→  const reservations: InHouseReservation[] = data?.reservations || []

   158→  // ─── Computed ───────────────────────────────────────────────────────
   159→
   160→  const filteredReservations = useMemo(() => {
   161→    let result = reservations.filter((r) => getOutstandingBalance(r) > 0)
   162→
   163→    // Search filter
   164→    if (searchQuery.trim()) {
   165→      const q = searchQuery.toLowerCase().trim()
   166→      result = result.filter((r) => {
   167→        const guestName = r.guest ? `${r.guest.firstName} ${r.guest.lastName}`.toLowerCase() : ''
   168→        const roomNumber = r.room ? r.room.number.toLowerCase() : ''
   169→        const confirmationNo = r.confirmationNo.toLowerCase()
   170→        return guestName.includes(q) || roomNumber.includes(q) || confirmationNo.includes(q)
   171→      })
   172→    }
   173→
   174→    // Balance filter
   175→    switch (balanceFilter) {
   176→      case 'over_10k':
   177→        result = result.filter((r) => getOutstandingBalance(r) > 10000)
   178→        break
   179→      case 'over_50k':
   180→        result = result.filter((r) => getOutstandingBalance(r) > 50000)
   181→        break
   182→      case 'over_100k':
   183→        result = result.filter((r) => getOutstandingBalance(r) > 100000)
   184→        break
   185→    }
   186→
   187→    return result
   188→  }, [reservations, searchQuery, balanceFilter])
   189→
   190→  const totalOutstanding = filteredReservations.reduce(
   191→    (sum, r) => sum + getOutstandingBalance(r),
   192→    0
   193→  )
   194→  const guestCount = filteredReservations.length
   195→
   196→  // All reservations with outstanding balance (for batch)
   197→  const allOutstanding = reservations.filter((r) => getOutstandingBalance(r) > 0)
   198→  const batchTotalOutstanding = allOutstanding.reduce(
   199→    (sum, r) => sum + getOutstandingBalance(r),
   200→    0
   201→  )
   202→
   203→  // ─── Mutations ─────────────────────────────────────────────────────
   204→
   205→  // Single payment mutation — uses existing API pattern: POST /api/folio/{folioId}
   206→  const paymentMutation = useMutation({
   207→    mutationFn: async ({
   208→      folioId,
   209→      amount,
   210→      method,
   211→      reference,
   212→    }: {
   213→      folioId: string
   214→      amount: number
   215→      method: PaymentMethod
   216→      reference: string
   217→    }) => {
   218→      return apiFetch(`/api/folio/${folioId}/payment`, {
   219→        method: 'POST',
   220→        headers: { 'Content-Type': 'application/json' },
   221→        body: JSON.stringify({ method, amount, reference }),
   222→      })
   223→    },
   224→    onSuccess: () => {
   225→      invalidate.afterFolioChange(queryClient)
   226→      setSettleDialogOpen(false)
   227→      setPaymentAmount('')
   228→      setPaymentReference('')
   229→      toast.success('Payment processed successfully')
   230→    },
   231→    onError: () => {
   232→      toast.error('Failed to process payment. Please try again.')
   233→    },
   234→  })
   235→
   236→  // Batch settlement mutation
   237→  const batchSettleMutation = useMutation({
   238→    mutationFn: async (method: PaymentMethod) => {
   239→      const results: { reservation: InHouseReservation; success: boolean; error?: string }[] = []
   240→      for (let i = 0; i < allOutstanding.length; i++) {
   241→        const res = allOutstanding[i]
   242→        const balance = getOutstandingBalance(res)
   243→        const folioId = res.folios[0]?.id
   244→        if (!folioId) {
   245→          results.push({ reservation: res, success: false, error: 'No folio found' })
   246→          continue
   247→        }
   248→        try {
   249→          await apiFetch(`/api/folio/${folioId}/payment`, {
   250→            method: 'POST',
   251→            headers: { 'Content-Type': 'application/json' },
   252→            body: JSON.stringify({ method, amount: balance, reference: '' }),
   253→          })
   254→          results.push({ reservation: res, success: true })
   255→          toast.success(`Settled ${i + 1} of ${allOutstanding.length}: Room ${res.room.number}`)
   256→        } catch {
   257→          results.push({ reservation: res, success: false, error: 'Payment failed' })
   258→        }
   259→      }
   260→      return results
   261→    },
   262→    onSuccess: (results) => {
   263→      invalidate.afterFolioChange(queryClient)
   264→      setBatchDialogOpen(false)
   265→      const successCount = results.filter((r) => r.success).length
   266→      const failCount = results.filter((r) => !r.success).length
   267→      if (failCount === 0) {
   268→        toast.success(`All ${successCount} accounts settled successfully!`)
   269→      } else {
   270→        toast.warning(`Settled ${successCount} accounts. ${failCount} failed.`)
   271→      }
   272→    },
   273→    onError: () => {
   274→      toast.error('Batch settlement failed. Please try again.')
   275→    },
   276→  })
   277→
   278→  // ─── Handlers ───────────────────────────────────────────────────────
   279→
   280→  const handleSettleClick = (res: InHouseReservation) => {

   285→    setPaymentReference('')
   286→    setSettleDialogOpen(true)
   287→  }
   288→

   298→      toast.error('Please enter a valid payment amount')
   299→      return

   301→    paymentMutation.mutate({
   302→      folioId,
   303→      amount,
   304→      method: paymentMethod,
   305→      reference: paymentReference,
   306→    })
   307→  }
   308→
   309→  const handleSettleAll = () => {

   311→      toast.info('No outstanding balances to settle')
   312→      return
   313→    }
   314→    setBatchPaymentMethod('cash')
   315→    setBatchDialogOpen(true)
   316→  }
   317→
   318→  const handleConfirmBatchSettle = () => {
   319→    batchSettleMutation.mutate(batchPaymentMethod)
   320→  }
   321→
   322→  const handleViewFolio = (res: InHouseReservation) => {
   323→    useFolioContextStore.getState().setFolioContext({
   324→      reservationId: res.id,
   325→      guestId: res.guest.id,
   326→      guestName: `${res.guest.firstName} ${res.guest.lastName}`.trim(),
   327→      roomNumber: res.room.number,

   329→    })
   330→    navigateTo('front-desk', 'folio')
   331→  }
   332→
   333→  // ─── Render ─────────────────────────────────────────────────────────
   334→
   335→  const selectedBalance = selectedReservation
   336→    ? getOutstandingBalance(selectedReservation)

   338→
   339→  return (

   341→      {/* Header */}

   343→        <div>
   344→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">
   345→            Settlement
   346→          </h2>
   347→          <p className="text-xs text-muted-foreground">
   348→            View and settle outstanding guest folio balances

   350→        </div>

   353→            onClick={handleSettleAll}
   354→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"

   356→            <CheckCircle className="size-4 mr-1.5" />
   357→            Settle All ({allOutstanding.length})
   358→          </Button>
   359→        )}
   360→      </div>
   361→
   362→      {/* Summary Stats */}
   363→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   364→        <Card className="py-0">
   365→          <CardContent className="p-4 flex items-center gap-3">
   366→            <div className="flex items-center justify-center size-10 rounded-lg bg-red-100 dark:bg-red-950">
   367→              <Wallet className="size-5 text-red-600 dark:text-red-400" />
   368→            </div>
   369→            <div>
   370→              <p className="text-xs text-muted-foreground font-medium">Total Outstanding</p>
   371→              <p className="text-lg font-bold text-red-600 dark:text-red-400">
   372→                {formatCurrency(totalOutstanding)}
   373→              </p>
   374→            </div>
   375→          </CardContent>
   376→        </Card>
   377→        <Card className="py-0">

   379→            <div className="flex items-center justify-center size-10 rounded-lg bg-amber-100 dark:bg-amber-950">
   380→              <CreditCard className="size-5 text-amber-600 dark:text-amber-400" />
   381→            </div>
   382→            <div>
   383→              <p className="text-xs text-muted-foreground font-medium">Guests with Balance</p>
   384→              <p className="text-lg font-bold">{guestCount}</p>
   385→            </div>
   386→          </CardContent>
   387→        </Card>
   388→        <Card className="py-0">
   389→          <CardContent className="p-4 flex items-center gap-3">
   390→            <div className="flex items-center justify-center size-10 rounded-lg bg-green-100 dark:bg-green-950">
   391→              <Banknote className="size-5 text-green-600 dark:text-green-400" />
   392→            </div>
   393→            <div>
   394→              <p className="text-xs text-muted-foreground font-medium">Avg. Balance</p>
   395→              <p className="text-lg font-bold">
   396→                {guestCount > 0 ? formatCurrency(Math.round(totalOutstanding / guestCount)) : formatCurrency(0)}
   397→              </p>

   399→          </CardContent>

   402→

   405→        <CardContent className="p-3 flex flex-col sm:flex-row gap-3">
   406→          <div className="relative flex-1">

   408→            <Input

   744→          <DialogFooter className="gap-2 sm:gap-0">
