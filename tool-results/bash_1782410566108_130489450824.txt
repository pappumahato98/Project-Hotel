   146	function NumberStepper({ label, value, min, max, onChange }: {
   147	  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   148	}) {
   149	  return (
   150	    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   151	      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   152	      <div className="flex items-center gap-0 ml-auto">
   153	        <button
   154	          type="button"
   155	          onClick={() => onChange(Math.max(min, value - 1))}
   156	          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   157	        >
   158	          <Minus className="w-3 h-3" />
   159	        </button>
   160	        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   161	        <button
   162	          type="button"
   163	          onClick={() => onChange(Math.min(max, value + 1))}
   164	          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   165	        >
   166	          <Plus className="w-3 h-3" />
   167	        </button>
   168	      </div>
   169	    </div>
   170	  )
   171	}

   203	export function CheckInLookup({ onBack, prefillReservationId }: CheckInLookupProps) {
   204	  const { setActiveSubModule } = useNavigationStore()
   205	  const { setCheckInSession, setPrefillReservationId } = useFrontDeskContextStore()
   206	
   207	  const today = getTodayString()
   208	
   209	  // ─── State ──────────────────────────────────────────────────────────────────
   210	  const [activeTab, setActiveTab] = useState<string>('find-reservation')
   211	  const [expressMode, setExpressMode] = useState(false)
   212	  const [searchQuery, setSearchQuery] = useState('')
   213	  const [debouncedQuery, setDebouncedQuery] = useState('')
   214	  const [selectedReservationId, setSelectedReservationId] = useState<string | null>(null)
   215	
   216	  // Walk-in fields
   217	  const [walkInDate, setWalkInDate] = useState(today)
   218	  const [walkInCheckOut, setWalkInCheckOut] = useState(() => {
   219	    const d = new Date()
   220	    d.setDate(d.getDate() + 1)
   221	    return toDateOnly(d)
   222	  })
   223	  const [walkInAdults, setWalkInAdults] = useState(2)
   224	  const [walkInChildren, setWalkInChildren] = useState(0)
   225	  const [guestFirstName, setGuestFirstName] = useState('')
   226	  const [guestLastName, setGuestLastName] = useState('')
   227	  const [guestEmail, setGuestEmail] = useState('')
   228	  const [guestPhone, setGuestPhone] = useState('')
   229	  const [guestNationality, setGuestNationality] = useState('')
   230	
   231	  // ─── Debounced search ──────────────────────────────────────────────────────
   232	  useEffect(() => {
   233	    const timer = setTimeout(() => {
   234	      setDebouncedQuery(searchQuery)
   235	    }, 300)
   236	    return () => clearTimeout(timer)
   237	  }, [searchQuery])
   238	
   239	  // ─── Data Queries ──────────────────────────────────────────────────────────
   240	
   241	  const activeReservationId = prefillReservationId || selectedReservationId
   242	
   243	  const {
   244	    data: reservationData,
   245	    isLoading: reservationLoading,
   246	    error: reservationError,
   247	  } = useQuery({
   248	    queryKey: ['reservation', activeReservationId],
   249	    queryFn: async () => {
   250	      if (!activeReservationId) return null
   251	      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   252	      return data.reservation
   253	    },
   254	    enabled: !!activeReservationId,
   255	  })
   256	
   257	  // Guest stay history
   258	  const { data: guestStayHistory } = useQuery({
   259	    queryKey: ['guest-stays', reservationData?.guest?.id],
   260	    queryFn: async () => {
   261	      if (!reservationData?.guest?.id) return null
   262	      try {
   263	        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   264	      } catch {
   265	        return null
   266	      }
   267	    },
   268	    enabled: !!reservationData?.guest?.id && !!selectedReservationId,
   269	  })
   270	
   271	  // Today's expected arrivals
   272	  const {
   273	    data: arrivalsData,
   274	    isLoading: arrivalsLoading,
   275	  } = useQuery({
   276	    queryKey: ['arrivals', today],
   277	    queryFn: async () => {
   278	      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   279	      return data.reservations || []
   280	    },
   281	    enabled: activeTab === 'find-reservation' && !prefillReservationId,
   282	  })
   283	
   284	  // Reservation search (debounced)
   285	  const {
   286	    data: searchResults,
   287	    isLoading: searchLoading,
   288	  } = useQuery({
   289	    queryKey: ['reservation-search', debouncedQuery],
   290	    queryFn: async () => {
   291	      if (!debouncedQuery || debouncedQuery.length < 2) return null
   292	      const data = await apiFetch(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   293	      return (data.reservations || []) as ReservationData[]
   294	    },
   295	    enabled: debouncedQuery.length >= 2,
   296	  })
   297	
   298	  // Room data for Direct Check-in right panel
   299	  const {
   300	    data: roomsData,
   301	    isLoading: roomsLoading,
   302	  } = useQuery({
   303	    queryKey: ['rooms-summary'],
   304	    queryFn: async () => {
   305	      const data = await apiFetch('/api/rooms') as RoomsApiResponse
   306	      return data
   307	    },
   308	    enabled: activeTab === 'direct-checkin' && !prefillReservationId,
   309	  })
   310	
   311	  // 14-day reservations for calendar summary
   312	  const todayDate = useMemo(() => new Date(), [])
   313	  const dateFrom14 = useMemo(() => {
   314	    const d = new Date(todayDate)
   315	    d.setHours(0, 0, 0, 0)
   316	    return d.toISOString().slice(0, 10)
   317	  }, [todayDate])
   318	  const dateTo14 = useMemo(() => {
   319	    const d = new Date(todayDate)
   320	    d.setDate(d.getDate() + 13)
   321	    d.setHours(23, 59, 59, 999)
   322	    return d.toISOString().slice(0, 10)
   323	  }, [todayDate])
   324	
   325	  interface CalendarReservation {
   326	    id: string
   327	    status: string
   328	    checkIn: string
   329	    checkOut: string
   330	    room?: { number: string } | null
   331	    guest?: { firstName: string; lastName: string } | null
   332	  }
   333	
   334	  const { data: upcomingReservations } = useQuery({
   335	    queryKey: ['upcoming-reservations', dateFrom14, dateTo14],
   336	    queryFn: async () => {
   337	      const data = await apiFetch(`/api/reservations?dateFrom=${dateFrom14}&dateTo=${dateTo14}`) as { reservations: CalendarReservation[] }
   338	      return data.reservations || []
   339	    },
   340	    enabled: activeTab === 'direct-checkin' && !prefillReservationId,
   341	  })
   342	
   343	  // Compute daily calendar summary data
   344	  const calendarSummary = useMemo(() => {
   345	    const totalRooms = roomsData?.summary?.totalRooms || 0
   346	    const oooCount = roomsData?.summary?.outOfOrder || 0
   347	    const sellableRooms = totalRooms - oooCount
   348	
   349	    if (!upcomingReservations || upcomingReservations.length === 0 || totalRooms === 0) {
   350	      // Return empty 14-day grid with just total rooms
   351	      return Array.from({ length: 14 }, (_, i) => {
   352	        const d = new Date(todayDate)
   353	        d.setDate(d.getDate() + i)
   354	        d.setHours(0, 0, 0, 0)
   355	        return {
   356	          date: d,
   357	          dateStr: d.toISOString().slice(0, 10),
   358	          dayName: d.toLocaleDateString('en-US', { weekday: 'short' }),
   359	          dayNum: d.getDate(),
   360	          monthStr: d.toLocaleDateString('en-US', { month: 'short' }),
   361	          occupied: 0,
   362	          reserved: 0,
   363	          vacant: sellableRooms,
   364	          totalRooms,
   365	          sellableRooms,
   366	          isToday: i === 0,
   367	          isWeekend: d.getDay() === 0 || d.getDay() === 6,
   368	        }
   369	      })
   370	    }
   371	
   372	    return Array.from({ length: 14 }, (_, i) => {
   373	      const d = new Date(todayDate)
   374	      d.setDate(d.getDate() + i)
   375	      d.setHours(0, 0, 0, 0)
   376	      const nextDay = new Date(d)
   377	      nextDay.setDate(nextDay.getDate() + 1)
   378	
   379	      const dayStart = d.getTime()
   380	      const dayEnd = nextDay.getTime()
   381	
   382	      let occupied = 0
   383	      let reserved = 0
   384	
   385	      for (const res of upcomingReservations) {
   386	        const ci = new Date(res.checkIn).getTime()
   387	        const co = new Date(res.checkOut).getTime()
   388	
   389	        // Check if reservation overlaps this day
   390	        if (ci < dayEnd && co > dayStart) {
   391	          if (res.status === 'checked_in') {
   392	            occupied++
   393	          } else if (res.status === 'confirmed' || res.status === 'tentative') {
   394	            reserved++
   395	          }
   396	        }
   397	      }
   398	
   399	      const vacant = Math.max(0, sellableRooms - occupied - reserved)
   400	
   401	      return {
   402	        date: d,
   403	        dateStr: d.toISOString().slice(0, 10),
   404	        dayName: d.toLocaleDateString('en-US', { weekday: 'short' }),
   405	        dayNum: d.getDate(),
   406	        monthStr: d.toLocaleDateString('en-US', { month: 'short' }),
   407	        occupied,
   408	        reserved,
   409	        vacant,
   410	        totalRooms,
   411	        sellableRooms,
   412	        isToday: i === 0,
   413	        isWeekend: d.getDay() === 0 || d.getDay() === 6,
   414	      }
   415	    })
   416	  }, [upcomingReservations, roomsData, todayDate])
   417	
   418	  // ─── Handlers ──────────────────────────────────────────────────────────────
   419	
   420	  const navigateToProcess = useCallback((session: CheckInSession) => {
   421	    setCheckInSession(session)
   422	    setActiveSubModule('check-in-process')
   423	  }, [setCheckInSession, setActiveSubModule])
   424	
   425	  const handleUseReservation = useCallback(() => {
   426	    if (!reservationData) return
   427	    navigateToProcess({
   428	      reservationId: reservationData.id,
   429	      reservationData: reservationData as unknown as Record<string, unknown>,
   430	      isDirectWalkIn: false,
   431	      walkInGuest: null,
   432	      walkInDates: null,
   433	      startAtStep: 1,
   434	    })
   435	    toast.success('Reservation confirmed! Proceeding to check-in.')
   436	  }, [reservationData, navigateToProcess])
   437	
   438	  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   439	    navigateToProcess({
   440	      reservationId: res.id,
   441	      reservationData: res as unknown as Record<string, unknown>,
   442	      isDirectWalkIn: false,
   443	      walkInGuest: null,
   444	      walkInDates: null,
   445	      startAtStep: 2,
   446	    })
   447	    toast.success('Express check-in - skipping to Room & Rate')
   448	  }, [navigateToProcess])
   449	
   450	  const handleWalkInContinue = useCallback(() => {
   451	    if (!guestFirstName || !guestLastName) {
   452	      toast.error('Please enter guest first and last name')
   453	      return
   454	    }
   455	    navigateToProcess({
   456	      reservationId: null,
   457	      reservationData: null,
   458	      isDirectWalkIn: true,
   459	      walkInGuest: {
   460	        firstName: guestFirstName,
   461	        lastName: guestLastName,
   462	        email: guestEmail,
   463	        phone: guestPhone,
   464	        nationality: guestNationality,
   465	      },
   466	      walkInDates: {
   467	        checkIn: walkInDate,
   468	        checkOut: walkInCheckOut,
   469	        adults: walkInAdults,
   470	        children: walkInChildren,
   471	      },
   472	      startAtStep: 1,
   473	    })
   474	  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   475	
   476	  // ─── Reservation Card Renderer ─────────────────────────────────────────────
   477	
   478	  const renderReservationCard = useCallback((res: ReservationData) => {
   479	    const isSelected = selectedReservationId === res.id
   480	    return (
   481	      <Card
   482	        key={res.id}
   483	        className={cn(
   484	          'p-3 cursor-pointer transition-all border',
   485	          isSelected
   486	            ? 'border-teal-400 ring-2 ring-teal-400/20 bg-teal-50/50 dark:bg-teal-950/20'
   487	            : 'hover:bg-muted/50'
   488	        )}
   489	        onClick={() => setSelectedReservationId(res.id)}
   490	      >
   491	        <div className="flex items-start justify-between gap-2">
   492	          <div className="space-y-1.5 min-w-0 flex-1">
   493	            <div className="flex items-center gap-2 flex-wrap">
   494	              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   495	                {res.confirmationNo}
   496	              </Badge>
   497	              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   498	                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   499	                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   500	                  VIP
   501	                </Badge>
   502	              )}
   503	              {res.source && (
   504	                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   505	                  {sourceLabel(res.source)}
   506	                </Badge>
   507	              )}
   508	            </div>
   509	            <p className="text-sm font-medium truncate">
   510	              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   511	            </p>
   512	            <div className="flex items-center gap-2 text-xs text-muted-foreground flex-wrap">
   513	              {res.room && <span>Room {res.room.number}</span>}
   514	              <span>·</span>
   515	              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   516	              <span>·</span>
   517	              <span className="font-medium">{formatCurrency(res.roomRate)}/night</span>
   518	            </div>
   519	          </div>
   520	          <div className="flex flex-col items-end gap-1.5 shrink-0">
   521	            {isSelected && (
   522	              <Check className="w-4 h-4 text-teal-600" />
   523	            )}
   524	            {expressMode && (
   525	              <Button
   526	                size="sm"
   527	                className="h-7 text-[11px] px-2 bg-emerald-500 hover:bg-emerald-600 text-white"
   528	                onClick={(e) => {
   529	                  e.stopPropagation()
   530	                  setSelectedReservationId(res.id)
   531	                  handleExpressCheckIn(res)
   532	                }}
   533	              >
   534	                <Zap className="w-3 h-3 mr-1" />
   535	                Express
   536	              </Button>
   537	            )}
   538	          </div>
   539	        </div>
   540	      </Card>
   541	    )
   542	  }, [selectedReservationId, expressMode, handleExpressCheckIn])
   543	
   544	  // ─── Reservation Detail Panel (right side of Tab 1) ────────────────────────
   545	
   546	  const renderReservationDetail = () => {
   547	    if (!selectedReservationId) {
   548	      return (
   549	        <div className="flex flex-col items-center justify-center h-full text-center px-6">
   550	          <div className="w-16 h-16 rounded-full bg-muted flex items-center justify-center mb-4">
   551	            <Eye className="w-7 h-7 text-muted-foreground" />
   552	          </div>
   553	          <p className="text-sm font-medium text-muted-foreground">
   554	            Select a reservation to view details
   555	          </p>
   556	          <p className="text-xs text-muted-foreground/70 mt-1">
   557	            Click on any reservation card from the list
   558	          </p>
   559	        </div>
   560	      )
   561	    }
   562	
   563	    if (reservationLoading) {
   564	      return (
