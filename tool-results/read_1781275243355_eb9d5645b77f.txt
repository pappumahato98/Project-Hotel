   191→   191→
   192→   192→function useReportQuery<T>(type: ReportType, date?: string) {
   193→   193→  return useQuery<T>({
   194→   194→    queryKey: ['front-desk-reports', type, date],
   195→   195→    queryFn: async () => {
   196→   196→      const params = new URLSearchParams({ type })
   197→   197→      if (date) params.set('date', date)
   198→   198→      const res = await fetch(`/api/front-desk/reports?${params.toString()}`)
   199→   199→      if (!res.ok) throw new Error(`Failed to fetch ${type} report`)
   200→   200→      return res.json()
   201→   201→    },
   202→   202→    refetchInterval: 30000,
   203→   203→  })
   204→   204→}

   862→   862→// 6. Occupancy Report (7-day trend)
   863→   863→function OccupancyReport() {

   868→   868→

   871→   871→
   872→   872→  if (days.length === 0) {

   874→   874→  }
   875→   875→

   877→   877→    <div className="space-y-3">

   903→   903→              />
   904→   904→              <StatCard
   905→   905→                label="Peak Occupancy"
   906→   906→                value={`${Math.max(...days.map(d => d.total > 0 ? Math.round((d.occupied / d.total) * 100) : 0))}%`}
   907→   907→                icon={TrendingUp}
