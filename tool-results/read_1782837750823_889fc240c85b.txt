   268→   268→
   269→   269→export function FrontDeskDashboard() {
   270→   270→  const { navigateTo, setActiveSubModule } = useNavigationStore()
   271→   271→  const { setTriggerWalkInDialog } = useFrontDeskContextStore()
   272→   272→
   273→   273→  const handleArrivalClick = (arrival: UpcomingArrival) => {
   274→   274→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   275→   275→    navigateTo('front-desk', 'check-in')
   276→   276→  }
   277→   277→
   278→   278→  // ─── Dashboard data query ────────────────────────────────────
   279→   279→  const {
   280→   280→    data,
   281→   281→    isLoading,
   282→   282→    error,
   283→   283→    refetch,
   284→   284→    isFetching,
   285→   285→  } = useQuery<DashboardData>({
   286→   286→    queryKey: ['front-desk-dashboard'],
   287→   287→    queryFn: () => apiFetch('/api/front-desk/dashboard'),
   288→   288→    refetchInterval: 30000,
   289→   289→  })
   290→   290→
   291→   291→  const snapshot = data?.snapshot
   292→   292→  const timeline = data?.timeline || []
   293→   293→  const upcomingArrivals = data?.upcomingArrivals || []
   294→   294→
   295→   295→  // ─── Quick action handler ───────────────────────────────────
   296→   296→  const handleQuickAction = (actionId: string) => {
   297→   297→    switch (actionId) {
   298→   298→      case 'new-reservation':
   299→   299→        setActiveSubModule('reservations')
   300→   300→        toast.success('Navigated to Reservations', {
   301→   301→          description: 'Create a new booking from the Reservations tab.',
   302→   302→        })
   303→   303→        break
   304→   304→      case 'walk-in-checkin':
   305→   305→        setActiveSubModule('arrivals')
   306→   306→        setTriggerWalkInDialog(true)
   307→   307→        toast.success('Walk-In Check-In', {
   308→   308→          description: 'Register a walk-in guest directly.',
   309→   309→        })
   310→   310→        break
   311→   311→      case 'room-transfer':
   312→   312→        setActiveSubModule('in-house')
   313→   313→        toast.success('Navigated to In-House Guests', {
   314→   314→          description: 'Select a guest and use Room Transfer from their details.',
   315→   315→        })
   316→   316→        break
   317→   317→      case 'wake-up-call':
   318→   318→        setActiveSubModule('wake-up-calls')
   319→   319→        toast.success('Navigated to Wake-up Calls', {
   320→   320→          description: 'Schedule and manage guest wake-up calls.',
   321→   321→        })
   322→   322→        break
   323→   323→      case 'late-checkout':
   324→   324→        setActiveSubModule('departures')
   325→   325→        toast.success('Navigated to Departures', {
   326→   326→          description: 'Manage late checkout requests from the Departures tab.',
   327→   327→        })
   328→   328→        break
   329→   329→      case 'express-checkout':
   330→   330→        setActiveSubModule('departures')
   331→   331→        toast.success('Navigated to Departures', {
   332→   332→          description: 'Process express checkout from the Departures tab.',
   333→   333→        })
   334→   334→        break
   335→   335→      case 'settlement':
   336→   336→        setActiveSubModule('settlement')
   337→   337→        toast.success('Navigated to Settlement', {
   338→   338→          description: 'Settle outstanding guest folio balances.',
   339→   339→        })
   340→   340→        break
   341→   341→      default:
   342→   342→        break
   343→   343→    }
   344→   344→  }
   345→   345→
   346→   346→  // ─── Determine overbooking alert visibility ──────────────────
   347→   347→  const showOverbookingAlert = useMemo(() => {
   348→   348→    return snapshot ? snapshot.overbookingCount > 0 : false
   349→   349→  }, [snapshot])
   350→   350→
   351→   351→  // ─── Snapshot cards data ────────────────────────────────────
   352→   352→  const snapshotCards = useMemo(() => {
   353→   353→    if (!snapshot) return []
   354→   354→    const cards = [
   355→   355→      {
   356→   356→        label: 'Arrivals Today',
   357→   357→        value: snapshot.arrivals,
   358→   358→        icon: LogIn,
   359→   359→        iconBg: 'bg-emerald-100 dark:bg-emerald-950',
   360→   360→        iconColor: 'text-emerald-600 dark:text-emerald-400',
   361→   361→      },
   362→   362→      {
   363→   363→        label: 'Departures Today',
   364→   364→        value: snapshot.departures,
   365→   365→        icon: LogOut,
   366→   366→        iconBg: 'bg-rose-100 dark:bg-rose-950',
   367→   367→        iconColor: 'text-rose-600 dark:text-rose-400',
   368→   368→      },
   369→   369→      {
   370→   370→        label: 'In-House Guests',
   371→   371→        value: snapshot.inHouse,
   372→   372→        icon: BedDouble,
   373→   373→        iconBg: 'bg-sky-100 dark:bg-sky-950',
   374→   374→        iconColor: 'text-sky-600 dark:text-sky-400',
   375→   375→      },
   376→   376→      {
   377→   377→        label: 'Available Rooms',
   378→   378→        value: snapshot.available,
   379→   379→        icon: Hotel,
   380→   380→        iconBg: 'bg-teal-100 dark:bg-teal-950',
   381→   381→        iconColor: 'text-teal-600 dark:text-teal-400',
   382→   382→      },
   383→   383→      {
   384→   384→        label: 'Occupancy',
   385→   385→        value: `${snapshot.occupancyPct}%`,
   386→   386→        icon: Percent,
   387→   387→        iconBg: 'bg-amber-100 dark:bg-amber-950',
   388→   388→        iconColor: 'text-amber-600 dark:text-amber-400',
   389→   389→        subtext: `${snapshot.inHouse} / ${snapshot.totalRooms} rooms`,
   390→   390→      },
   391→   391→    ]
   392→   392→
   393→   393→    // Only show overbooking alert if there are overbookings
   394→   394→    if (snapshot.overbookingCount > 0) {
   395→   395→      cards.push({
   396→   396→        label: 'Overbooking Alert',
   397→   397→        value: snapshot.overbookingCount,
   398→   398→        icon: AlertTriangle,
   399→   399→        iconBg: 'bg-red-100 dark:bg-red-950',
   400→   400→        iconColor: 'text-red-600 dark:text-red-400',
   401→   401→        subtext: `${snapshot.overbookingCount} room${snapshot.overbookingCount > 1 ? 's' : ''} double-booked`,
   402→   402→      })
   403→   403→    }
   404→   404→
   405→   405→    return cards
   406→   406→  }, [snapshot])
   407→   407→
   408→   408→  // ─── Render ──────────────────────────────────────────────────
   409→   409→  return (
   410→   410→    <div className="flex flex-col gap-2">
   411→   411→      {/* ─── Header ─────────────────────────────────────────── */}
   412→   412→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   413→   413→        <div>
   414→   414→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Front Desk Dashboard</h2>
   415→   415→          <p className="text-xs text-muted-foreground">
   416→   416→            Real-time operational snapshot — {formatDate(new Date())}
   417→   417→          </p>
   418→   418→        </div>
   419→   419→        <Button
   420→   420→          variant="outline"
   421→   421→          size="sm"
   422→   422→          className="gap-1.5 text-xs text-muted-foreground shrink-0"
   423→   423→          onClick={() => refetch()}
   424→   424→          disabled={isFetching}
   425→   425→        >
   426→   426→          <RefreshCw className={cn('size-3.5', isFetching && 'animate-spin')} />
   427→   427→          {isFetching ? 'Refreshing...' : 'Refresh'}
   428→   428→        </Button>
   429→   429→      </div>
   430→   430→
   431→   431→      {/* ─── Today's Snapshot Cards ──────────────────────────── */}
   432→   432→      {isLoading ? (
   433→   433→        <SnapshotSkeleton />
   434→   434→      ) : error ? (
   435→   435→        <Card>
   436→   436→          <CardContent className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   437→   437→            <AlertTriangle className="size-8 mb-2 opacity-40" />
   438→   438→            <p className="text-sm">Failed to load dashboard data</p>
   439→   439→            <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   440→   440→              Try Again
   441→   441→            </Button>
   442→   442→          </CardContent>
   443→   443→        </Card>
   444→   444→      ) : (
   445→   445→        <div className={cn(
   446→   446→          'grid gap-2',
   447→   447→          showOverbookingAlert
   448→   448→            ? 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-6'
   449→   449→            : 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-5',
   450→   450→        )}>
   451→   451→          {snapshotCards.map((card) => (
   452→   452→            <Card key={card.label} className={cn(
   453→   453→              card.label === 'Overbooking Alert' && 'border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20',
   454→   454→            )}>
   455→   455→              <CardContent className="p-2.5 flex items-center gap-3">
   456→   456→                <div className={cn('flex size-10 items-center justify-center rounded-lg shrink-0', card.iconBg)}>
   457→   457→                  <card.icon className={cn('size-5', card.iconColor)} />
