   106→function NumberStepper({ label, value, min, max, onChange }: {
   107→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   108→}) {
   109→  return (
   110→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   111→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   112→      <div className="flex items-center gap-0 ml-auto">
   113→        <button
   114→          type="button"
   115→          onClick={() => onChange(Math.max(min, value - 1))}
   116→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   117→        >
   118→          <Minus className="w-3 h-3" />
   119→        </button>
   120→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   121→        <button
   122→          type="button"
   123→          onClick={() => onChange(Math.min(max, value + 1))}
   124→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   125→        >
   126→          <Plus className="w-3 h-3" />
   127→        </button>
   128→      </div>
   129→    </div>
   130→  )
   131→}

   133→function sourceLabel(source: string | null) {
   134→  switch (source) {
   135→    case 'walk_in': return 'Walk-in'
   136→    case 'online': return 'Online'
   137→    case 'phone': return 'Phone'
   138→    case 'travel_agent': return 'Travel Agent'
   139→    case 'corporate': return 'Corporate'
   140→    default: return source || 'Direct'
   141→  }
   142→}

   146→export function CheckInLookup({ onBack, prefillReservationId }: CheckInLookupProps) {
   147→  const { setActiveSubModule } = useNavigationStore()
   148→  const { setCheckInSession, setPrefillReservationId } = useFrontDeskContextStore()
   149→
   150→  const today = getTodayString()
   151→
   152→  // State ───────────────────────────────────────────────────────────-------------------------------------------------
   153→  const [expressMode, setExpressMode] = useState(false)
   154→  const [isDirectWalkIn, setIsDirectWalkIn] = useState(false)
   155→  const [searchQuery, setSearchQuery] = useState('')
   156→  const [debouncedQuery, setDebouncedQuery] = useState('')
   157→  const [searchedReservationId, setSearchedReservationId] = useState<string | null>(null)
   158→
   159→  // Walk-in fields
   160→  const [walkInDate, setWalkInDate] = useState(today)
   161→  const [walkInCheckOut, setWalkInCheckOut] = useState(() => {
   162→    const d = new Date()
   163→    d.setDate(d.getDate() + 1)
   164→    return toDateOnly(d)
   165→  })
   166→  const [walkInAdults, setWalkInAdults] = useState(2)
   167→  const [walkInChildren, setWalkInChildren] = useState(0)
   168→  const [guestFirstName, setGuestFirstName] = useState('')
   169→  const [guestLastName, setGuestLastName] = useState('')
   170→  const [guestEmail, setGuestEmail] = useState('')
   171→  const [guestPhone, setGuestPhone] = useState('')
   172→  const [guestNationality, setGuestNationality] = useState('')
   173→
   174→  // Clear prefill on unmount to avoid stale prefills
   175→  useEffect(() => {
   176→    return () => { setPrefillReservationId(null) }
   177→  }, [setPrefillReservationId])
   178→
   179→  // Debounced search ────────────────────────────────────────────────---------------------------------------
   180→  useEffect(() => {
   181→    const timer = setTimeout(() => {
   182→      setDebouncedQuery(searchQuery)
   183→    }, 300)
   184→    return () => clearTimeout(timer)
   185→  }, [searchQuery])
   186→
   187→  // Data Queries ────────────────────────────────────────────────────------------------------------------------
   188→
   189→  const activeReservationId = prefillReservationId || searchedReservationId
   190→
   191→  const {
   192→    data: reservationData,
   193→    isLoading: reservationLoading,
   194→    error: reservationError,
   195→  } = useQuery({
   196→    queryKey: ['reservation', activeReservationId],
   197→    queryFn: async () => {
   198→      if (!activeReservationId) return null
   199→      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   200→      return data.reservation
   201→    },
   202→    enabled: !!activeReservationId,
   203→  })
   204→
   205→  // Guest stay history
   206→  const { data: guestStayHistory } = useQuery({
   207→    queryKey: ['guest-stays', reservationData?.guest?.id],
   208→    queryFn: async () => {
   209→      if (!reservationData?.guest?.id) return null
   210→      try {
   211→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   212→      } catch {
   213→        return null
   214→      }
   215→    },
   216→    enabled: !!reservationData?.guest?.id && !!searchedReservationId,
   217→  })
   218→
   219→  // Today's expected arrivals
   220→  const {
   221→    data: arrivalsData,
   222→    isLoading: arrivalsLoading,
   223→  } = useQuery({
   224→    queryKey: ['arrivals', today],
   225→    queryFn: async () => {
   226→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   227→      return data.reservations || []
   228→    },
   229→    enabled: !isDirectWalkIn && !prefillReservationId,
   230→  })
   231→
   232→  // Reservation search (debounced)
   233→  const {
   234→    data: searchResults,
   235→    isLoading: searchLoading,
   236→  } = useQuery({
   237→    queryKey: ['reservation-search', debouncedQuery],
   238→    queryFn: async () => {
   239→      if (!debouncedQuery || debouncedQuery.length < 2) return null
   240→      const data = await apiFetch(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   241→      return (data.reservations || []) as ReservationData[]
   242→    },
   243→    enabled: debouncedQuery.length >= 2,
   244→  })
   245→
   246→  // Handlers ────────────────────────────────────────────────────────-----------------------------------------------
   247→
   248→  const navigateToProcess = useCallback((session: CheckInSession) => {
   249→    setCheckInSession(session)
   250→    setActiveSubModule('check-in-process')
   251→  }, [setCheckInSession, setActiveSubModule])
   252→
   253→  const handleUseReservation = useCallback(() => {
   254→    if (!reservationData) return
   255→    navigateToProcess({
   256→      reservationId: reservationData.id,
   257→      reservationData: reservationData as unknown as Record<string, unknown>,
   258→      isDirectWalkIn: false,
   259→      walkInGuest: null,
   260→      walkInDates: null,
   261→      startAtStep: 1,
   262→    })
   263→    toast.success('Reservation confirmed! Proceeding to check-in.')
   264→  }, [reservationData, navigateToProcess])
   265→
   266→  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   267→    navigateToProcess({
   268→      reservationId: res.id,
   269→      reservationData: res as unknown as Record<string, unknown>,
   270→      isDirectWalkIn: false,
   271→      walkInGuest: null,
   272→      walkInDates: null,
   273→      startAtStep: 2,
   274→    })
   275→    toast.success('Express check-in - skipping to Room & Rate')
   276→  }, [navigateToProcess])
   277→
   278→  const handleWalkInContinue = useCallback(() => {
   279→    if (!guestFirstName || !guestLastName) {
   280→      toast.error('Please enter guest first and last name')
   281→      return
   282→    }
   283→    navigateToProcess({
   284→      reservationId: null,
   285→      reservationData: null,
   286→      isDirectWalkIn: true,
   287→      walkInGuest: {
   288→        firstName: guestFirstName,
   289→        lastName: guestLastName,
   290→        email: guestEmail,
   291→        phone: guestPhone,
   292→        nationality: guestNationality,
   293→      },
   294→      walkInDates: {
   295→        checkIn: walkInDate,
   296→        checkOut: walkInCheckOut,
   297→        adults: walkInAdults,
   298→        children: walkInChildren,
   299→      },
   300→      startAtStep: 1,
   301→    })
   302→  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   303→
   304→  // Reservation Card Renderer ───────────────────────────────────────-----------------------------
   305→
   306→  const renderReservationCard = useCallback((res: ReservationData) => {
   307→    const isSelected = searchedReservationId === res.id
   308→    return (
   309→      <Card
   310→        key={res.id}
   311→        className={cn(
   312→          'p-2.5 cursor-pointer transition-all border',
   313→          isSelected
   314→            ? 'border-teal-400 ring-2 ring-teal-400/20 bg-teal-50/50 dark:bg-teal-950/20'
   315→            : 'hover:bg-muted/50'
   316→        )}
   317→        onClick={() => setSearchedReservationId(res.id)}
   318→      >
   319→        <div className="flex items-start justify-between gap-2">
   320→          <div className="space-y-1 min-w-0 flex-1">
   321→            <div className="flex items-center gap-2 flex-wrap">
   322→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   323→                {res.confirmationNo}
   324→              </Badge>
   325→              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   326→                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   327→                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   328→                  VIP
   329→                </Badge>
   330→              )}
   331→              {res.source && (
   332→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   333→                  {sourceLabel(res.source)}
   334→                </Badge>
   335→              )}
   336→            </div>
   337→            <p className="text-sm font-medium truncate">
   338→              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   339→            </p>
   340→            <div className="flex items-center gap-2 text-xs text-muted-foreground flex-wrap">
   341→              {res.room && <span>Room {res.room.number}</span>}
   342→              <span>·</span>
   343→              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   344→              <span>·</span>
   345→              <span className="font-medium">{formatCurrency(res.roomRate)}/night</span>
   346→            </div>
   347→          </div>
   348→          <div className="flex flex-col items-end gap-1.5 shrink-0">
   349→            {isSelected && (
   350→              <Check className="w-4 h-4 text-teal-600" />
   351→            )}
   352→            {expressMode && (
   353→              <Button
   354→                size="sm"
   355→                className="h-7 text-[11px] px-2 bg-emerald-500 hover:bg-emerald-600 text-white"
   356→                onClick={(e) => {
   357→                  e.stopPropagation()
   358→                  setSearchedReservationId(res.id)
   359→                  handleExpressCheckIn(res)
   360→                }}
   361→              >
   362→                <Zap className="w-3 h-3 mr-1" />
   363→                Express
   364→              </Button>
   365→            )}
   366→          </div>
   367→        </div>
   368→      </Card>
   369→    )
   370→  }, [searchedReservationId, expressMode, handleExpressCheckIn])
   371→
   372→  // Render ──────────────────────────────────────────────────────────-----------------------------------------------
   373→  return (
   374→    <div className="flex flex-col h-full bg-background">
   375→      <div className="flex-1 overflow-y-auto">
   376→        <div className="max-w-2xl mx-auto space-y-2 p-3 sm:p-4">
   377→
   378→          {/* Prefilled reservation mode */}
   379→          {prefillReservationId ? (
   380→            <>
   381→              {/* Header for prefill mode */}
   382→              <div className="shrink-0 -mx-3 sm:-mx-4 -mt-3 sm:-mt-4 mb-2 px-3 sm:px-4 py-2.5 border-b bg-card">
   383→                <div className="flex items-center gap-3">
   384→                  <button
   385→                    type="button"
   386→                    onClick={onBack}
   387→                    className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors"
   388→                  >
   389→                    <ArrowLeft className="w-4 h-4" />
   390→                    <span className="hidden sm:inline">Back</span>
   391→                  </button>
   392→                  <Separator orientation="vertical" className="h-5" />
   393→                  <h1 className="text-base sm:text-lg font-semibold text-foreground">Guest Check-In</h1>
   394→                </div>
   395→              </div>
   396→              {reservationLoading && (
   397→                <div className="space-y-2">
   398→                  <Skeleton className="h-4 w-3/4" />
   399→                  <Skeleton className="h-40 w-full" />
   400→                </div>
   401→              )}
   402→              {reservationError && (
   403→                <Card className="border-red-200 dark:border-red-800 p-2.5">
   404→                  <div className="flex items-start gap-2 text-red-600 dark:text-red-400">
   405→                    <svg className="h-5 w-5 mt-0.5 shrink-0" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>
   406→                    <div>
   407→                      <p className="font-medium text-sm">Reservation Not Found</p>
   408→                      <p className="text-sm text-muted-foreground mt-1">
   409→                        The reservation could not be found. Please search or use direct walk-in.
   410→                      </p>
   411→                    </div>
   412→                  </div>
   413→                  <Button variant="outline" size="sm" className="mt-2" onClick={() => { setPrefillReservationId(null) }}>
   414→                    Use Direct Walk-in Instead
   415→                  </Button>
   416→                </Card>

   418→              {reservationData && (
   419→                <>
   420→                  <div className="flex items-center gap-2">
   421→                    <Badge variant="outline" className="border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">
   422→                      <Check className="w-3 h-3 mr-1" />
   423→                      Reservation Found
   424→                    </Badge>

   429→                      <div className="space-y-0.5">
   430→                        <span className="text-xs text-muted-foreground">Status</span>

   435→                      <InfoItem label="Stay" value={`${formatDate(reservationData.checkIn)} → ${formatDate(reservationData.checkOut)}`} />

   439→                    {guestStayHistory && guestStayHistory.length > 0 && (

   453→                  </Card>
   454→                  <div className="flex gap-2">
   455→                    <Button onClick={handleUseReservation} className="flex-1">
   456→                      <Check className="w-4 h-4 mr-2" />
   457→                      Go Check-in
   458→                    </Button>
   459→                    {expressMode && (

   473→                <div className="flex items-center gap-2">
   474→                  <button
