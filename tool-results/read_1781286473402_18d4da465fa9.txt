   193→// ─── Debounce hook ────────────────────────────────────────────────────
   194→function useDebounce<T>(value: T, delay: number): T {
   195→  const [debounced, setDebounced] = useState(value)
   196→  useEffect(() => {
   197→    const timer = setTimeout(() => setDebounced(value), delay)
   198→    return () => clearTimeout(timer)
   199→  }, [value, delay])
   200→  return debounced
   201→}

   203→// ─── Date range presets ──────────────────────────────────────────────
   204→function getDatePreset(preset: string): { from: Date; to: Date } {
   205→  const today = new Date()
   206→  today.setHours(0, 0, 0, 0)
   207→  switch (preset) {
   208→    case 'today':
   209→      return { from: today, to: today }
   210→    case 'this-week': {
   211→      const from = new Date(today)
   212→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   213→      const to = new Date(from)
   214→      to.setDate(from.getDate() + 6) // Sunday
   215→      return { from, to }
   216→    }
   217→    case 'this-month': {
   218→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   219→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   220→      return { from, to }
   221→    }
   222→    case 'next-7': {
   223→      const to = new Date(today)
   224→      to.setDate(today.getDate() + 6)
   225→      return { from: today, to }
   226→    }
   227→    case 'next-14': {
   228→      const to = new Date(today)
   229→      to.setDate(today.getDate() + 13)
   230→      return { from: today, to }
   231→    }
   232→    case 'next-30': {
   233→      const to = new Date(today)
   234→      to.setDate(today.getDate() + 29)
   235→      return { from: today, to }
   236→    }
   237→    default:
   238→      return { from: today, to: today }
   239→  }
   240→}

   242→function toDateString(d: Date): string {
   243→  const y = d.getFullYear()
   244→  const m = String(d.getMonth() + 1).padStart(2, '0')
   245→  const day = String(d.getDate()).padStart(2, '0')
   246→  return `${y}-${m}-${day}`
   247→}
   248→
   249→function formatShortDate(dateStr: string): string {
   250→  if (!dateStr) return ''
   251→  const d = new Date(dateStr + 'T00:00:00')
   252→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   253→}
