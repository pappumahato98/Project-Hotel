   101→// ─── Helper ──────────────────────────────────────────────────────────────
   102→
   103→function nightsBetween(start: string | Date, end: string | Date): number {
   104→  const s = new Date(start)
   105→  const e = new Date(end)
   106→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   107→}
   108→
   109→function formatDateValue(date: Date): string {
   110→  return toDateOnly(date)
   111→}
