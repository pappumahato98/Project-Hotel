   175→function PriorityBadge({ priority }: { priority: Priority }) {
   176→  const styles: Record<Priority, string> = {
   177→    High: 'bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300 border-red-300 dark:border-red-700',
   178→    Normal: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700',
   179→    Low: 'bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   180→  }
   181→  return (
   182→    <Badge variant="outline" className={cn('text-[10px] px-1.5 py-0', styles[priority])}>
   183→      {priority === 'High' && <AlertTriangle className="size-3 mr-0.5" />}
   184→      {priority}
   185→    </Badge>
   186→  )
   187→}

   215→

   218→  const [assignDialogOpen, setAssignDialogOpen] = useState(false)

   228→  const [formPriority, setFormPriority] = useState<Priority>('Normal')

   230→  const [formNotes, setFormNotes] = useState('')

   232→  // ─── Computed values ──────────────────────────────────────────
   233→

   235→    let filtered = waitlist
   236→    if (searchQuery.trim()) {
   237→      const q = searchQuery.toLowerCase()
   238→      filtered = filtered.filter(
   239→        (e) =>
   240→          e.guestName.toLowerCase().includes(q) ||
   241→          e.contactPhone.includes(q) ||
   242→          e.roomPreference.toLowerCase().includes(q),
   243→      )
   244→    }
   245→    if (priorityFilter !== 'all') {
   246→      filtered = filtered.filter((e) => e.priority === priorityFilter)
   247→    }
   248→    if (statusFilter !== 'all') {
   249→      filtered = filtered.filter((e) => e.status === statusFilter)
   250→    }
   251→    return filtered
   252→  }, [waitlist, searchQuery, priorityFilter, statusFilter])
   253→
   254→  const totalWaitlisted = waitlist.filter((e) => e.status === 'Waiting').length
   255→  const avgWaitDays = useMemo(() => {
   256→    const waiting = waitlist.filter((e) => e.status === 'Waiting')
   257→    if (waiting.length === 0) return 0
   258→    const totalDays = waiting.reduce((sum, e) => {
   259→      const added = new Date(e.addedDate)
   260→      const now = new Date()
   261→      return sum + Math.max(1, Math.ceil((now.getTime() - added.getTime()) / (1000 * 60 * 60 * 24)))
   262→    }, 0)
   263→    return (totalDays / waiting.length).toFixed(1)
   264→  }, [waitlist])
   265→  const highPriorityCount = waitlist.filter((e) => e.priority === 'High' && e.status === 'Waiting').length
   266→  const assignedToday = waitlist.filter((e) => e.status === 'Assigned').length
   267→
   268→  // ─── Handlers ────────────────────────────────────────────────
   269→
   270→  function handleAddToWaitlist() {
   271→    if (!formGuestName.trim() || !formCheckIn || !formCheckOut) {
   272→      toast.error('Please fill in guest name and dates')
   273→      return
   274→    }
   275→    const newEntry: WaitlistEntry = {
   276→      id: `wl-${Date.now()}`,
   277→      position: waitlist.length + 1,
   278→      guestName: formGuestName.trim(),
   279→      roomPreference: formRoomPref || 'No preference',
   280→      checkInDate: formCheckIn,
   281→      checkOutDate: formCheckOut,
   282→      guestsCount: parseInt(formGuestsCount) || 1,
   283→      priority: formPriority,
   284→      status: 'Waiting',
   285→      contactPhone: formPhone || 'N/A',
   286→      notes: formNotes || '',
   287→      addedDate: new Date().toISOString().split('T')[0],
   288→    }
   289→    setWaitlist((prev) => [...prev, newEntry])
   290→    setAddDialogOpen(false)
   291→    resetAddForm()
   292→    toast.success(`${newEntry.guestName} added to waitlist`, {
   293→      description: `Position #${newEntry.position}`,
   294→    })
   295→  }
   296→
   297→  function resetAddForm() {
   298→    setFormGuestName('')
   299→    setFormRoomPref('')
   300→    setFormCheckIn('')
   301→    setFormCheckOut('')
   302→    setFormGuestsCount('1')
   303→    setFormPriority('Normal')
   304→    setFormPhone('')
   305→    setFormNotes('')
   306→  }
   307→
   308→  function handleAssignRoom() {
   309→    if (!selectedEntry || !selectedRoomId) return
   310→    const room = AVAILABLE_ROOMS.find((r) => r.id === selectedRoomId)
   311→    setWaitlist((prev) =>
   312→      prev.map((e) => (e.id === selectedEntry.id ? { ...e, status: 'Assigned' as WaitlistStatus } : e)),
   313→    )
   314→    setAssignDialogOpen(false)
   315→    setSelectedRoomId('')
   316→    setSelectedEntry(null)
   317→    toast.success(`Room ${room?.number || selectedRoomId} assigned to ${selectedEntry.guestName}`, {
   318→      description: `${selectedEntry.roomPreference} → Room ${room?.number || ''}`,
   319→    })
   320→  }
   321→
   322→  function handleRemoveFromWaitlist(entry: WaitlistEntry) {
   323→    setWaitlist((prev) =>
   324→      prev.map((e) => (e.id === entry.id ? { ...e, status: 'Cancelled' as WaitlistStatus } : e)),
   325→    )
   326→    toast.info(`${entry.guestName} removed from waitlist`, {
   327→      description: 'Status set to Cancelled',
   328→    })
   329→  }
   330→
   331→  function handleCallGuest(entry: WaitlistEntry) {
   332→    toast.success(`Calling ${entry.guestName}...`, {
   333→      description: entry.contactPhone,
   334→      duration: 4000,
   335→    })
   336→  }
   337→
   338→  function openAssignDialog(entry: WaitlistEntry) {
   339→    setSelectedEntry(entry)
   340→    setSelectedRoomId('')
   341→    setAssignDialogOpen(true)
   342→  }
   343→
   344→  // ─── Render ──────────────────────────────────────────────────
   345→  return (

   347→      {/* Header */}
   348→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   349→        <div>
   350→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Waitlist Management</h2>
   351→          <p className="text-xs text-muted-foreground">
   352→            Manage guests waiting for room availability
   353→          </p>
   354→        </div>
   355→        <Button onClick={() => { resetAddForm(); setAddDialogOpen(true) }} className="gap-2 shrink-0">
   356→          <UserPlus className="size-3.5" />
   357→          Add to Waitlist
   358→        </Button>
   359→      </div>
   360→
   361→      {/* Filters */}
   362→      <Card className="py-0">
   363→        <CardContent className="p-2">
   364→          <div className="flex flex-col sm:flex-row items-start sm:items-center gap-2">
   365→            <div className="flex items-center gap-2 text-xs font-medium text-muted-foreground">
   366→              <Filter className="size-3.5" />
   367→              Filters
   368→            </div>
   369→            <div className="flex flex-col sm:flex-row items-start sm:items-center gap-2 flex-1">
   370→              <div className="relative flex-1 max-w-xs">
   371→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3 text-muted-foreground" />
   372→                <Input
   373→                  placeholder="Search by guest name..."
   374→                  value={searchQuery}
   375→                  onChange={(e) => setSearchQuery(e.target.value)}
   376→                  className={cn('pl-8 h-7 text-xs', searchQuery && 'pr-7')}
   377→                />
   378→                {searchQuery && (
   379→                  <button
   380→                    type="button"
   381→                    className="absolute right-1.5 top-1/2 -translate-y-1/2 size-5 inline-flex items-center justify-center rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors"
   382→                    onClick={() => setSearchQuery('')}
   383→                  >
   384→                    <X className="size-3" strokeWidth={2.5} />
   385→                  </button>
   386→                )}
   387→              </div>
   388→              <div className="relative">
   389→                <Select value={priorityFilter} onValueChange={setPriorityFilter}>
   390→                  <SelectTrigger className={cn('w-[110px] h-7 data-[size=default]:h-7 text-xs', priorityFilter !== 'all' && 'pr-8')}>
   391→                    <SelectValue placeholder="Priority" />
   392→                  </SelectTrigger>
   393→                  <SelectContent>
   394→                    <SelectItem value="all">All Priority</SelectItem>
   395→                    <SelectItem value="High">High</SelectItem>
   396→                    <SelectItem value="Normal">Normal</SelectItem>
   397→                    <SelectItem value="Low">Low</SelectItem>
   398→                  </SelectContent>
   399→                </Select>
   400→                {priorityFilter !== 'all' && (
   401→                  <button
   402→                    type="button"
   403→                    className="absolute right-1 top-1/2 -translate-y-1/2 size-5 inline-flex items-center justify-center rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors z-10"
   404→                    onClick={(e) => { e.stopPropagation(); setPriorityFilter('all') }}
   405→                  >
   406→                    <X className="size-3" strokeWidth={2.5} />
   407→                  </button>
   408→                )}
   409→              </div>

   411→                <Select value={statusFilter} onValueChange={setStatusFilter}>

   416→                    <SelectItem value="all">All Status</SelectItem>

   419→                    <SelectItem value="Cancelled">Cancelled</SelectItem>
   420→                  </SelectContent>
   421→                </Select>
   422→                {statusFilter !== 'all' && (
   423→                  <button
   424→                    type="button"

   427→                  >
   428→                    <X className="size-3" strokeWidth={2.5} />
   429→                  </button>

   431→              </div>
   432→            </div>
   433→            {(searchQuery || priorityFilter !== 'all' || statusFilter !== 'all') && (
   434→              <button
   435→                type="button"
   436→                onClick={() => { setSearchQuery(''); setPriorityFilter('all'); setStatusFilter('all') }}
   437→                className="inline-flex items-center justify-center size-7 rounded-full bg-red-100 dark:bg-red-950/50 text-red-600 dark:text-red-400 hover:bg-red-200 dark:hover:bg-red-900/50 transition-colors shrink-0"
   438→                title="Clear all filters"
   439→              >
   440→                <X className="size-3.5" strokeWidth={2.5} />
   441→              </button>

   443→            <p className="text-[11px] text-muted-foreground">
   444→              Showing {filteredWaitlist.length} of {waitlist.length}

   446→          </div>
   447→        </CardContent>
   448→      </Card>
   449→
   450→      {/* Waitlist Table */}
   451→      <Card className="py-0">
   452→        <CardContent className="p-0 overflow-auto max-h-[65vh]">
   453→          <Table>
   454→            <TableHeader>

   461→                <TableHead className="w-[50px] hidden sm:table-cell">Pax</TableHead>

   484→                  >
   485→                    <TableCell className="font-mono text-xs text-muted-foreground">

   495→                    </TableCell>
   496→                    <TableCell className="hidden md:table-cell">
