   177→   177→// ─── Debounce hook ────────────────────────────────────────────────────
   178→   178→function useDebounce<T>(value: T, delay: number): T {
   179→   179→  const [debounced, setDebounced] = useState(value)
   180→   180→  useEffect(() => {
   181→   181→    const timer = setTimeout(() => setDebounced(value), delay)
   182→   182→    return () => clearTimeout(timer)
   183→   183→  }, [value, delay])
   184→   184→  return debounced
   185→   185→}

   187→   187→// ─── Date range presets ──────────────────────────────────────────────
   188→   188→function getDatePreset(preset: string): { from: Date; to: Date } {
   189→   189→  const today = new Date()
   190→   190→  today.setHours(0, 0, 0, 0)
   191→   191→  switch (preset) {
   192→   192→    case 'today':
   193→   193→      return { from: today, to: today }
   194→   194→    case 'this-week': {
   195→   195→      const from = new Date(today)
   196→   196→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   197→   197→      const to = new Date(from)
   198→   198→      to.setDate(from.getDate() + 6) // Sunday
   199→   199→      return { from, to }
   200→   200→    }
   201→   201→    case 'this-month': {
   202→   202→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   203→   203→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   204→   204→      return { from, to }
   205→   205→    }
   206→   206→    case 'next-7': {
   207→   207→      const to = new Date(today)
   208→   208→      to.setDate(today.getDate() + 6)
   209→   209→      return { from: today, to }
   210→   210→    }
   211→   211→    case 'next-14': {
   212→   212→      const to = new Date(today)
   213→   213→      to.setDate(today.getDate() + 13)
   214→   214→      return { from: today, to }
   215→   215→    }
   216→   216→    case 'next-30': {
   217→   217→      const to = new Date(today)
   218→   218→      to.setDate(today.getDate() + 29)
   219→   219→      return { from: today, to }
   220→   220→    }
   221→   221→    default:
   222→   222→      return { from: today, to: today }
   223→   223→  }
   224→   224→}

   226→   226→function toDateString(d: Date): string {
   227→   227→  const y = d.getFullYear()
   228→   228→  const m = String(d.getMonth() + 1).padStart(2, '0')
   229→   229→  const day = String(d.getDate()).padStart(2, '0')
   230→   230→  return `${y}-${m}-${day}`
   231→   231→}
   232→   232→
   233→   233→function formatShortDate(dateStr: string): string {
   234→   234→  if (!dateStr) return ''
   235→   235→  const d = new Date(dateStr + 'T00:00:00')
   236→   236→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   237→   237→}
