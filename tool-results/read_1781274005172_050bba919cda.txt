   102→   102→function nightsBetween(start: string | Date, end: string | Date): number {
   103→   103→  const s = new Date(start)
   104→   104→  const e = new Date(end)
   105→   105→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   106→   106→}

   109→   109→  return date.toISOString().split('T')[0]
   110→   110→}
