   276→   276→function InlineRowActions({
   277→   277→  row,
   278→   278→  onStatusChange,
   279→   279→  onViewDetail,
   280→   280→  isForceMutated,
   281→   281→}: {
   282→   282→  row: RoomRow
   283→   283→  onStatusChange: (roomId: string, taskId: string | null, newStatus: string) => void
   284→   284→  onViewDetail: (roomId: string) => void
   285→   285→  isForceMutated: boolean
   286→   286→}) {
   287→   287→  const s = row.hkDisplayStatus
   288→   288→  const canClean = ['dirty', 'change_over', 'pending', 'assigned'].includes(s)
   289→   289→  const canMarkCleaned = ['cleaning', 'in_progress'].includes(s)
   290→   290→  const canInspect = s === 'cleaned'
   291→   291→  const canReset = ['failed', 'inspected'].includes(s)
   292→   292→  const canFail = ['cleaning', 'in_progress', 'cleaned'].includes(s)
   293→   293→  const canDirty = ['clean', 'inspected'].includes(s)
   294→   294→  const hasActions = canClean || canMarkCleaned || canInspect || canReset || canFail || canDirty
   295→   295→
   296→   296→  return (
   297→   297→    <div className="flex items-center gap-0.5 flex-nowrap" onPointerDown={(e) => e.stopPropagation()}>
   298→   298→      {/* Occupied indicator */}
   299→   299→      {row.reservationStatus === 'occupied' && (
   300→   300→        <div className="flex items-center gap-0.5 text-amber-500" title={`Occupied by ${row.guestName || 'Guest'}`}>
   301→   301→          <User className="h-3 w-3" />
   302→   302→        </div>
   303→   303→      )}
   304→   304→      {/* View Details — always visible as separate button */}
   305→   305→      <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground"
   306→   306→        onClick={() => {
   307→   307→          if (row.reservationStatus === 'occupied') {
   308→   308→            toast.info(`Room ${row.roomNumber} — Occupied By "${row.guestName || 'Guest'}"`, { duration: 3000 })
   309→   309→          }
   310→   310→          onViewDetail(row.roomId)
   311→   311→        }} title="View Details">
   312→   312→        <Eye className="h-3.5 w-3.5" />
   313→   313→      </Button>
   314→   314→
   315→   315→      {/* Three-dot menu for status actions */}
   316→   316→      <DropdownMenu>
   317→   317→        <DropdownMenuTrigger asChild>
   318→   318→          <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground">
   319→   319→            <MoreHorizontal className="h-3.5 w-3.5" />
   320→   320→          </Button>
   321→   321→        </DropdownMenuTrigger>
   322→   322→        <DropdownMenuContent align="end" className="w-48">
   323→   323→          {/* Start Cleaning — dirty, change_over, pending, assigned */}
   324→   324→          {canClean && (
   325→   325→            <DropdownMenuItem
   326→   326→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'in_progress') }}
   327→   327→              disabled={isForceMutated}
   328→   328→              className="gap-2 text-xs text-sky-600 focus:text-sky-600 focus:bg-sky-50 dark:text-sky-400 dark:focus:bg-sky-950/40"
   329→   329→            >
   330→   330→              <PlayCircle className="h-3.5 w-3.5" /> Start Cleaning
   331→   331→            </DropdownMenuItem>
   332→   332→          )}
   333→   333→
   334→   334→          {/* Mark Cleaned — cleaning, in_progress */}
   335→   335→          {canMarkCleaned && (
   336→   336→            <DropdownMenuItem
   337→   337→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'cleaned') }}
   338→   338→              disabled={isForceMutated}
   339→   339→              className="gap-2 text-xs text-emerald-600 focus:text-emerald-600 focus:bg-emerald-50 dark:text-emerald-400 dark:focus:bg-emerald-950/40"
   340→   340→            >
   341→   341→              <CheckCircle className="h-3.5 w-3.5" /> Mark Cleaned
   342→   342→            </DropdownMenuItem>
   343→   343→          )}
   344→   344→
   345→   345→          {/* Mark Inspected — cleaned */}
   346→   346→          {canInspect && (
   347→   347→            <DropdownMenuItem
   348→   348→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'inspected') }}
   349→   349→              disabled={isForceMutated}
   350→   350→              className="gap-2 text-xs text-purple-600 focus:text-purple-600 focus:bg-purple-50 dark:text-purple-400 dark:focus:bg-purple-950/40"
   351→   351→            >
   352→   352→              <ShieldCheck className="h-3.5 w-3.5" /> Mark Inspected
   353→   353→            </DropdownMenuItem>
   354→   354→          )}
   355→   355→
   356→   356→          {/* Mark Dirty — clean, inspected (re-trigger cleaning) */}
   357→   357→          {canDirty && (
   358→   358→            <DropdownMenuItem
   359→   359→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'dirty') }}
   360→   360→              disabled={isForceMutated}
   361→   361→              className="gap-2 text-xs text-orange-600 focus:text-orange-600 focus:bg-orange-50 dark:text-orange-400 dark:focus:bg-orange-950/40"

   366→   366→
   367→   367→          {/* Reset to Pending — failed, inspected */}
   368→   368→          {canReset && (
   369→   369→            <DropdownMenuItem
   370→   370→              onClick={(e) => { e.preventDefault(); onStatusChange(row.roomId, row.hkTaskId, 'pending') }}
   371→   371→              disabled={isForceMutated}
   372→   372→              className="gap-2 text-xs text-amber-600 focus:text-amber-600 focus:bg-amber-50 dark:text-amber-400 dark:focus:bg-amber-950/40"
   373→   373→            >
   374→   374→              <ArrowUpCircle className="h-3.5 w-3.5" /> Reset to Pending
   375→   375→            </DropdownMenuItem>
   376→   376→          )}
   377→   377→
   378→   378→          {/* Rush — always available (not blocked by force-mutated) */}
   379→   379→          <DropdownMenuSeparator />
