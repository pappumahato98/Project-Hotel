   183→   183→function guestFullName(g: FolioGuest | null | undefined): string {
   184→   184→  if (!g) return '—'
   185→   185→  return `${g.firstName} ${g.lastName}`
   186→   186→}
   187→   187→
   188→   188→function folioCharges(f: Folio): number {
   189→   189→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   190→   190→}
   191→   191→
   192→   192→function folioPayments(f: Folio): number {
   193→   193→  return f.payments.reduce((s, p) => s + p.amount, 0)
   194→   194→}
   195→   195→
   196→   196→function folioOutstanding(f: Folio): number {
   197→   197→  return folioCharges(f) - folioPayments(f)
   198→   198→}
   199→   199→
   200→   200→function isVoidedTransaction(t: FolioTransaction): boolean {
   201→   201→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   202→   202→}
   203→   203→
   204→   204→function isVoidedPayment(p: FolioPayment): boolean {
   205→   205→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   206→   206→}

   208→   208→function useDebounce<T>(value: T, delay: number): T {
   209→   209→  const [debounced, setDebounced] = useState(value)
   210→   210→  useEffect(() => {
   211→   211→    const timer = setTimeout(() => setDebounced(value), delay)
   212→   212→    return () => clearTimeout(timer)
   213→   213→  }, [value, delay])
   214→   214→  return debounced
   215→   215→}
