   158→
   159→export function CheckInView({ reservationId, onComplete }: CheckInViewProps) {
   160→  const queryClient = useQueryClient()
   161→  const { settings } = useSettingsStore()
   162→  const today = getTodayString()
   163→
   164→  // ─── Tab & search state ────────────────────────────────────────
   165→  const [activeTab, setActiveTab] = useState<'find' | 'walkin'>('find')
   166→  const [searchQuery, setSearchQuery] = useState('')
   167→
   168→  // ─── Step state ────────────────────────────────────────────────
   169→  const [currentStep, setCurrentStep] = useState(0)
   170→
   171→  // ─── Selection states ────────────────────────────────────────
   172→  const [selectedReservationId, setSelectedReservationId] = useState<string>('')
   173→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   174→  const [roomSearchFilter, setRoomSearchFilter] = useState('')
   175→
   176→  // ─── Registration card form ────────────────────────────────────
   177→  const [regTitle, setRegTitle] = useState('')
   178→  const [regFirstName, setRegFirstName] = useState('')
   179→  const [regLastName, setRegLastName] = useState('')
   180→  const [regContactNo, setRegContactNo] = useState('')
   181→  const [regEmail, setRegEmail] = useState('')
   182→  const [regAddress, setRegAddress] = useState('')
   183→  const [regNationality, setRegNationality] = useState('')
   184→  const [regIdType, setRegIdType] = useState('passport')
   185→  const [regIdNumber, setRegIdNumber] = useState('')
   186→  const [regCity, setRegCity] = useState('')
   187→  const [regCountry, setRegCountry] = useState('')
   188→
   189→  // ─── Stay preferences ─────────────────────────────────────────
   190→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   191→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   192→  const [pillowType, setPillowType] = useState('')
   193→  const [roomPreference, setRoomPreference] = useState('')
   194→  const [wakeupCall, setWakeupCall] = useState('')
   195→
   196→  // ─── Billing ──────────────────────────────────────────────────
   197→  const [depositAmount, setDepositAmount] = useState('')
   198→  const [paymentMethod, setPaymentMethod] = useState('')
   199→  const [billingNotes, setBillingNotes] = useState('')
   200→
   201→  // ─── Key card dialog ───────────────────────────────────────────
   202→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   203→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   204→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   205→  const [keyCardGuestName, setKeyCardGuestName] = useState('')
   206→  const [keyCardCheckOut, setKeyCardCheckOut] = useState('')
   207→
   208→  // ─── Queries ─────────────────────────────────────────────────
   209→
   210→  // Fetch specific reservation by ID (if prop provided)
   211→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   212→    queryKey: ['reservation', reservationId],
   213→    queryFn: () => apiFetch(`/api/reservations/${reservationId}`),
   214→    enabled: !!reservationId,
   215→  })
   216→
   217→  // Fetch today's confirmed arrivals
   218→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   219→    queryKey: ['arrivals', today],
   220→    queryFn: async () => {
   221→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   222→      return apiFetch(`/api/reservations?${params.toString()}`)
   223→    },
   224→    enabled: !reservationId,
   225→  })
   226→
   227→  // Fetch all rooms with types
   228→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   229→    queryKey: ['rooms', 'all'],
   230→    queryFn: () => apiFetch('/api/rooms'),
   231→  })
   232→
   233→  // ─── Derived data ──────────────────────────────────────────────
   234→
   235→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   236→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   237→
   238→  // The reservation we're working with
   239→  const activeReservation: ReservationInfo | null = reservationId
   240→    ? specificReservation
   241→    : arrivals.find((a) => a.id === selectedReservationId) || null
   242→
   243→  // Room type data from API (not hardcoded)
   244→  const roomTypes: RoomTypeData[] = useMemo(() => {
   245→    if (!roomsData?.roomTypes) return []
   246→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   247→      id: rt.id,
   248→      name: rt.name,
   249→      code: rt.code,
   250→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   251→    }))
   252→  }, [roomsData])
   253→
   254→  // Available rooms (vacant_clean or inspected)
   255→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   256→  const availableRooms = allRooms.filter(
   257→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   258→  )
   259→
   260→  // Filter available rooms by search
   261→  const filteredAvailableRooms = useMemo(() => {
   262→    let filtered = availableRooms
   263→    if (roomSearchFilter.trim()) {
   264→      const q = roomSearchFilter.toLowerCase()
   265→      filtered = filtered.filter(
   266→        (r: RoomInfo) =>
   267→          r.number.toLowerCase().includes(q) ||
   268→          r.type.name.toLowerCase().includes(q) ||
   269→          String(r.floor).includes(q) ||
   270→          (r.wing && r.wing.toLowerCase().includes(q))
   271→      )
   272→    }
   273→    return filtered
   274→  }, [availableRooms, roomSearchFilter])
   275→
   276→  // Group rooms by floor
   277→  const roomsByFloor = useMemo(() => {
   278→    const grouped = new Map<number, RoomInfo[]>()
   279→    for (const room of filteredAvailableRooms) {
   280→      const floor = room.floor
   281→      if (!grouped.has(floor)) grouped.set(floor, [])
   282→      grouped.get(floor)!.push(room)
   283→    }
   284→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   285→  }, [filteredAvailableRooms])
   286→
   287→  // Get room type name by ID
   288→  const getRoomTypeName = (typeId: string) => {
   289→    const rt = roomTypes.find((t) => t.id === typeId)
   290→    return rt?.name || 'Unknown'
   291→  }
   292→
   293→  // Get room rate by type ID
   294→  const getRoomTypeRate = (typeId: string) => {
   295→    const rt = roomTypes.find((t) => t.id === typeId)
   296→    return rt?.baseRate || 0
   297→  }
   298→
   299→  // ─── Room status summary ─────────────────────────────────────
   300→  const roomStatusSummary = useMemo(() => {
   301→    let vacant = 0, occupied = 0, reserved = 0
   302→    for (const r of allRooms) {
   303→      if (r.status === 'vacant_clean' || r.status === 'inspected') vacant++
   304→      else if (r.status === 'occupied') occupied++
   305→      else reserved++
   306→    }
   307→    return { vacant, occupied, reserved, total: allRooms.length }
   308→  }, [allRooms])
   309→
   310→  // Filter arrivals by search query
   311→  const filteredArrivals = useMemo(() => {
   312→    if (!searchQuery.trim()) return arrivals
   313→    const q = searchQuery.toLowerCase()
   314→    return arrivals.filter((a) =>
   315→      a.confirmationNo.toLowerCase().includes(q) ||
   316→      (a.guest?.firstName || '').toLowerCase().includes(q) ||
   317→      (a.guest?.lastName || '').toLowerCase().includes(q) ||
   318→      (a.room?.number || '').includes(q)
   319→    )
   320→  }, [arrivals, searchQuery])
   321→
   322→  // ─── Initialize form when reservation loads ───────────────────
   323→
   324→  React.useEffect(() => {
   325→    if (specificReservation && reservationId) {
   326→      setSelectedReservationId(specificReservation.id)
   327→      if (specificReservation.room?.id) {
   328→        setSelectedRoomId(specificReservation.room.id)
   329→      }
   330→      setCurrentStep(specificReservation.guest ? 1 : 0)
   331→    }
   332→  }, [specificReservation, reservationId])
   333→
   334→  // Pre-fill registration fields when guest is selected
   335→  React.useEffect(() => {
   336→    if (activeReservation?.guest) {
   337→      const g = activeReservation.guest
   338→      setRegTitle('')
   339→      setRegFirstName(g.firstName || '')
   340→      setRegLastName(g.lastName || '')
   341→      setRegContactNo(g.phone || '')
   342→      setRegEmail(g.email || '')
   343→      setRegNationality(g.nationality || '')
   344→      setRegIdType(g.idType || 'passport')
   345→      setRegIdNumber(g.idNumber || '')
   346→      setRegAddress(g.address || '')
   347→      setRegCity(g.city || '')
   348→      setRegCountry(g.country || '')
   349→    }
   350→  }, [activeReservation?.guest])
   351→
   352→  // Auto-set deposit when room is selected
   353→  React.useEffect(() => {
   354→    if (activeReservation?.roomRate && !depositAmount) {
   355→      setDepositAmount(String(activeReservation.roomRate))
   356→    }
   357→  }, [activeReservation?.roomRate, depositAmount])
   358→
   359→  // ─── Mutations ────────────────────────────────────────────────
   360→
   361→  const checkInMutation = useMutation({
   362→    mutationFn: async ({
   363→      reservationId: resId,
   364→      roomId,
   365→      specialRequests,
   366→    }: {
   367→      reservationId: string
   368→      roomId: string
   369→      specialRequests: string
   370→    }) => {
   371→      return apiFetch(`/api/reservations/${resId}`, {
   372→        method: 'PATCH',
   373→        headers: { 'Content-Type': 'application/json' },
   374→        body: JSON.stringify({
   375→          roomId,
   376→          status: 'checked_in',
   377→          specialRequests,
   378→        }),
   379→      })
   380→    },
   381→    onSuccess: () => {
   382→      invalidate.afterCheckIn(queryClient)
   383→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   384→      toast.success('Guest checked in successfully!')
   385→    },
   386→    onError: (error) => {
   387→      toast.error('Check-in failed', {
   388→        description: error instanceof Error ? error.message : 'Unknown error',
   389→      })
   390→    },
   391→  })
   392→
   393→  // ─── Special Requests Builder ─────────────────────────────────
   394→
   395→  const buildSpecialRequests = (): string => {
   396→    const parts: string[] = []
   397→
   398→    if (activeReservation?.specialRequests) {
   399→      parts.push(activeReservation.specialRequests)
   400→    }
   401→    if (earlyCheckIn) {
   402→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   403→    }
   404→    if (checkOutTime !== settings.defaultCheckOut) {
   405→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   406→    }
   407→    if (roomPreference) {
   408→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   409→      parts.push(`[PREFERENCE] Room: ${label}`)
   410→    }
   411→    if (pillowType) {
   412→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   413→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   414→    }
   415→    if (wakeupCall) {
   416→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   417→    }
   418→    if (billingNotes) {
   419→      parts.push(`[BILLING] ${billingNotes}`)
   420→    }
   421→    if (paymentMethod) {
   422→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   423→    }
   424→    if (depositAmount && parseFloat(depositAmount) > 0) {
   425→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   426→    }
   427→
   428→    return parts.join(' | ')
   429→  }
   430→
   431→  // ─── Step Navigation ──────────────────────────────────────────
   432→
   433→  const canGoNext = (): boolean => {
   434→    switch (currentStep) {
   435→      case 0: return !!activeReservation
   436→      case 1: return true // Guest verification is info-only
   437→      case 2: return !!selectedRoomId
   438→      case 3: return true // Preferences are optional
   439→      case 4: return true // Billing is optional
   440→      case 5: return keyCardIssued
   441→      default: return false
   442→    }
   443→  }
   444→
   445→  const handleNext = () => {
   446→    if (currentStep === 5) {
   447→      onComplete?.()
   448→      return
   449→    }
   450→
   451→    // On step 4 → 5, execute check-in
   452→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   453→      const requests = buildSpecialRequests()
   454→      checkInMutation.mutate(
   455→        {
   456→          reservationId: activeReservation.id,
   457→          roomId: selectedRoomId,
   458→          specialRequests: requests,
   459→        },
   460→        {
   461→          onSuccess: () => {
   462→            // Find the selected room number for key card
   463→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   464→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   465→            const guestName = activeReservation.guest
   466→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   467→              : 'Guest'
   468→
   469→            setKeyCardRoomNumber(roomNumber)
   470→            setKeyCardGuestName(guestName)
   471→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   472→            setKeyCardIssued(false)
   473→            setKeyCardDialogOpen(true)
   474→            setCurrentStep(5)
   475→          },
   476→        }
   477→      )
   478→      return
   479→    }
   480→
   481→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   482→  }
   483→
   484→  const handleBack = () => {
   485→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   486→  }
   487→
   488→  // ─── Render: Step Indicator ────────────────────────────────────
   489→
   490→  const renderStepIndicator = () => (
   491→    <div className="mb-4">
   492→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   493→        {STEP_LABELS.map((label, idx) => (
   494→          <React.Fragment key={idx}>
   495→            <div className="flex flex-col items-center gap-1 min-w-0">
   496→              <div
   497→                className={cn(
   498→                  'flex size-7 shrink-0 items-center justify-center rounded-full border-2 text-[10px] font-bold transition-colors',
   499→                  idx < currentStep && 'border-green-500 bg-green-500 text-white',
   500→                  idx === currentStep && 'border-primary bg-primary text-primary-foreground',
   501→                  idx > currentStep && 'border-muted-foreground/30 bg-background text-muted-foreground'
   502→                )}
   503→              >
   504→                {idx < currentStep ? <Check className="size-4" /> : idx + 1}
   505→              </div>
   506→              <span
   507→                className={cn(
   508→                  'text-[10px] sm:text-xs font-medium whitespace-nowrap',
   509→                  idx === currentStep ? 'text-primary' : 'text-muted-foreground'
   510→                )}
   511→              >
   512→                {label}
   513→              </span>
   514→            </div>
   515→            {idx < STEP_LABELS.length - 1 && (
   516→              <div
   517→                className={cn(
   518→                  'flex-1 h-0.5 mx-1 min-w-[12px] max-w-[40px] transition-colors',
   519→                  idx < currentStep ? 'bg-green-500' : 'bg-muted-foreground/20'
   520→                )}
   521→              />
   522→            )}
   523→          </React.Fragment>
   524→        ))}
   525→      </div>
   526→    </div>
   527→  )
   528→
   529→  // ─── Render: Step 0 - Select Arrival ─────────────────────────
   530→
   531→  const renderSelectArrival = () => {
   532→    if (loadingArrivals) {
   533→      return (
   534→        <div className="space-y-3 p-4">
   535→          {Array.from({ length: 4 }).map((_, i) => (
   536→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   537→          ))}
   538→        </div>
   539→      )
   540→    }
   541→
   542→    if (arrivals.length === 0) {
   543→      return (
   544→        <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   545→          <UserCheck className="size-12 mb-3 opacity-30" />
   546→          <p className="text-sm font-medium">No arrivals scheduled for today</p>
   547→          <p className="text-xs mt-1">Check-in date: {formatDate(today)}</p>
   548→        </div>
   549→      )
   550→    }
   551→
   552→    return (
   553→      <div className="space-y-2">
   554→        <div className="flex items-center gap-2 mb-3">
   555→          <Search className="size-4 text-muted-foreground" />
   556→          <p className="text-sm text-muted-foreground">
   557→            {arrivals.length} confirmed arrival{arrivals.length !== 1 ? 's' : ''} for today
   558→          </p>
