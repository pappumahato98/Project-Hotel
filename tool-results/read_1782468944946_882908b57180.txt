   157→// ─── Component ──────────────────────────────────────────────────────────
   158→
   159→export function ArrivalsView() {
   160→  const queryClient = useQueryClient()
   161→  const { settings } = useSettingsStore()
   162→  const { navigateTo } = useNavigationStore()
   163→  const today = getTodayString()
   164→
   165→  // Dialog states
   166→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   167→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   168→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   169→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   170→
   171→  // Selection states
   172→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   173→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   174→
   175→  // Check-in form states
   176→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   177→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   178→  const [roomPreference, setRoomPreference] = useState('')
   179→  const [pillowType, setPillowType] = useState('')
   180→  const [wakeupCall, setWakeupCall] = useState('')
   181→
   182→  // Key card states
   183→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   184→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   185→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   186→
   187→  // Walk-in form state
   188→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   189→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   190→
   191→  // ─── Queries ────────────────────────────────────────────────────────
   192→
   193→  const { data, isLoading } = useQuery({
   194→    queryKey: ['arrivals', today],
   195→    queryFn: async () => {
   196→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   197→      return apiFetch(`/api/reservations?${params.toString()}`)
   198→    },
   199→    refetchInterval: 30000,
   200→  })
   201→
   202→  const { data: roomsData } = useQuery({
   203→    queryKey: ['rooms', 'vacant'],
   204→    queryFn: () => apiFetch('/api/rooms'),
   205→  })
   206→
   207→  const arrivals: Arrival[] = data?.reservations || []
   208→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   209→  const availableRooms = allRooms.filter(
   210→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   211→  )
   212→
   213→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   214→
   215→  const sortedArrivals = useMemo(() => {
   216→    return [...arrivals].sort((a, b) => {
   217→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   218→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   219→      if (aVip && !bVip) return -1
   220→      if (!aVip && bVip) return 1
   221→      return 0
   222→    })
   223→  }, [arrivals])
   224→
   225→  // ─── Stats ──────────────────────────────────────────────────────────
   226→
   227→  const totalArrivals = arrivals.length
   228→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   229→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   230→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   231→
   232→  // ─── Mutations ──────────────────────────────────────────────────────
   233→
   234→  const assignRoomMutation = useMutation({
   235→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   236→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   237→      if (specialRequests !== undefined) {
   238→        payload.specialRequests = specialRequests
   239→      }
   240→      return apiFetch(`/api/reservations/${reservationId}`, {
   241→        method: 'PATCH',
   242→        headers: { 'Content-Type': 'application/json' },
   243→        body: JSON.stringify(payload),
   244→      })
   245→    },
   246→    onSuccess: (_data, variables) => {
   247→      invalidate.afterReservationChange(queryClient)
   248→      setRoomPickerOpen(false)
   249→      setCheckInDialogOpen(false)
   250→
   251→      // Find the room number for key card dialog
   252→      const room = availableRooms.find((r) => r.id === variables.roomId)
   253→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   254→      openKeyCardDialog(roomNumber)
   255→    },
   256→  })
   257→
   258→  const checkInMutation = useMutation({
   259→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   260→      const payload: Record<string, unknown> = { status: 'checked_in' }
   261→      if (specialRequests !== undefined) {
   262→        payload.specialRequests = specialRequests
   263→      }
   264→      return apiFetch(`/api/reservations/${reservationId}`, {
   265→        method: 'PATCH',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify(payload),
   268→      })
   269→    },
   270→    onSuccess: (_data, variables) => {
   271→      invalidate.afterCheckIn(queryClient)
   272→      setCheckInDialogOpen(false)
   273→
   274→      const roomNumber = selectedArrival?.room?.number || ''
   275→      openKeyCardDialog(roomNumber)
   276→    },
   277→  })
   278→
   279→  const walkInMutation = useMutation({
   280→    mutationFn: async (form: WalkInForm) => {
   281→      // 1. Create guest
   282→      const guestData = await apiFetch('/api/guests', {
   283→        method: 'POST',
   284→        headers: { 'Content-Type': 'application/json' },
   285→        body: JSON.stringify({
   286→          firstName: form.firstName,
   287→          lastName: form.lastName,
   288→          phone: form.phone || null,
   289→          email: form.email || null,
   290→          nationality: form.nationality || null,
   291→          idType: form.idType || null,
   292→          idNumber: form.idNumber || null,
   293→        }),
   294→      })
   295→      const guestId = guestData.guest.id
   296→
   297→      // 2. Find a vacant room
   298→      const vacantRoom = availableRooms[0]
   299→      if (!vacantRoom) throw new Error('No available rooms')
   300→
   301→      // 3. Create reservation as checked_in (walk-in)
   302→      const checkOutDate = form.checkOutDate || (() => {
   303→        const d = new Date()
   304→        d.setDate(d.getDate() + 1)
   305→        return d
   306→      })()
   307→
   308→      const resData = await apiFetch('/api/reservations', {
   309→        method: 'POST',
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify({
   312→          guestId,
   313→          roomId: vacantRoom.id,
   314→          adults: form.adults,
   315→          children: form.children,
   316→          checkIn: new Date().toISOString(),
   317→          checkOut: checkOutDate.toISOString(),
   318→          roomRate: form.roomRate,
   319→          source: 'walk_in',
   320→          reservationType: 'walk_in',
   321→          guaranteed: false,
   322→        }),
   323→      })
   324→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   325→    },
   326→    onSuccess: (result) => {
   327→      invalidate.afterCheckIn(queryClient)
   328→      setWalkInDialogOpen(false)
   329→      setWalkInForm(initialWalkInForm)
   330→
   331→      // Open key card dialog
   332→      openKeyCardDialog(result.roomNumber)
   333→    },
   334→    onError: (error) => {
   335→      toast.error('Walk-in check-in failed', {
   336→        description: error instanceof Error ? error.message : 'Unknown error',
   337→      })
   338→    },
   339→  })
   340→
   341→  // ─── Handlers ──────────────────────────────────────────────────────
   342→
   343→  const handleAssignRoom = (arrival: Arrival) => {
   344→    setSelectedArrival(arrival)
   345→    setRoomPickerOpen(true)
   346→  }
   347→
   348→  const handleQuickCheckIn = (arrival: Arrival) => {
   349→    if (!arrival.room) {
   350→      setSelectedArrival(arrival)
   351→      setRoomPickerOpen(true)
   352→      return
   353→    }
   354→    setSelectedArrival(arrival)
   355→    resetCheckInForm()
   356→    setCheckInDialogOpen(true)
   357→  }
   358→
   359→  const handleOpenWalkIn = () => {
   360→    setWalkInForm(initialWalkInForm)
   361→    setWalkInDialogOpen(true)
   362→  }
   363→
   364→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   365→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   366→    navigateTo('front-desk', 'check-in')
   367→  }
   368→
   369→  const handleViewLedger = (arrival: Arrival) => {
   370→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   371→      guestId: arrival.guest?.id || '',
   372→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   373→    })
   374→    navigateTo('front-desk', 'guest-ledger')
   375→  }
   376→
   377→  const buildSpecialRequests = (arrival: Arrival): string => {
   378→    const parts: string[] = []
   379→
   380→    // Preserve existing requests
   381→    if (arrival.specialRequests) {
   382→      parts.push(arrival.specialRequests)
   383→    }
   384→
   385→    // Early check-in note
   386→    if (earlyCheckIn) {
   387→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   388→    }
   389→
   390→    // Check-out time override
   391→    if (checkOutTime !== settings.defaultCheckOut) {
   392→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   393→    }
   394→
   395→    // Room preference
   396→    if (roomPreference) {
   397→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   398→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   399→    }
   400→
   401→    // Pillow type
   402→    if (pillowType) {
   403→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   404→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   405→    }
   406→
   407→    // Wake-up call
   408→    if (wakeupCall) {
   409→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   410→    }
   411→
   412→    return parts.join(' | ')
   413→  }
   414→
   415→  const confirmCheckIn = () => {
   416→    if (!selectedArrival) return
   417→
   418→    const updatedRequests = buildSpecialRequests(selectedArrival)
   419→
   420→    if (selectedRoomId) {
   421→      assignRoomMutation.mutate({
   422→        reservationId: selectedArrival.id,
   423→        roomId: selectedRoomId,
   424→        specialRequests: updatedRequests,
   425→      })
   426→    } else if (selectedArrival.room) {
   427→      checkInMutation.mutate({
   428→        reservationId: selectedArrival.id,
   429→        specialRequests: updatedRequests,
   430→      })
   431→    }
   432→  }
   433→
   434→  const openKeyCardDialog = (roomNumber: string) => {
   435→    setKeyCardRoomNumber(roomNumber)
   436→    setKeyCardIssued(false)
   437→    setKeyCardProcessed(false)
   438→    setKeyCardDialogOpen(true)
   439→  }
   440→
   441→  const handleKeyCardConfirm = () => {
   442→    setKeyCardProcessed(true)
   443→    setKeyCardDialogOpen(false)
   444→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   445→      duration: 5000,
   446→      icon: <Mail className="size-4" />,
   447→    })
   448→    setSelectedArrival(null)
   449→    setSelectedRoomId('')
   450→  }
   451→
   452→  const resetCheckInForm = () => {
   453→    setEarlyCheckIn(false)
   454→    setCheckOutTime(settings.defaultCheckOut)
   455→    setRoomPreference('')
   456→    setPillowType('')
   457→    setWakeupCall('')
   458→  }
   459→
   460→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   461→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   462→    setWalkInForm((prev) => ({
   463→      ...prev,
   464→      roomTypeId,
   465→      roomRate: roomType?.baseRate || 0,
   466→    }))
   467→  }
   468→
   469→  // ─── Render ─────────────────────────────────────────────────────────
   470→
   471→  return (
   472→    <div className="flex flex-col gap-2">
   473→      {/* Header */}
   474→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   475→        <div>
   476→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   477→          <p className="text-xs text-muted-foreground">
   478→            Guest check-ins scheduled for {formatDate(today)}
   479→          </p>
   480→        </div>
   481→        <Button onClick={handleOpenWalkIn} className="gap-2 shrink-0">
   482→          <Footprints className="size-4" />
   483→          Walk-in Check-in
   484→        </Button>
   485→      </div>
   486→
   487→      {/* Arrivals List */}
   488→      <Card className="py-0">
   489→        <CardContent className="p-0">
   490→          {isLoading ? (
   491→            <div className="p-2 space-y-2">
   492→              {Array.from({ length: 4 }).map((_, i) => (
   493→                <Skeleton key={i} className="h-20 w-full rounded-lg" />
   494→              ))}
   495→            </div>
   496→          ) : sortedArrivals.length === 0 ? (
   497→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   498→              <LogIn className="size-8 mb-2 opacity-50" />
   499→              <p className="text-sm">No arrivals scheduled for today</p>
   500→            </div>
   501→          ) : (
   502→            <div className="divide-y">
   503→              {sortedArrivals.map((arrival) => {
   504→                const isUnassigned = !arrival.room
   505→                const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   506→                return (
   507→                  <div
   508→                    key={arrival.id}
   509→                    className={cn(
   510→                      'p-2.5 transition-colors hover:bg-muted/50',
   511→                      isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/50 dark:bg-amber-950/20',
   512→                      isVip && !isUnassigned && 'border-l-4 border-l-amber-400 bg-amber-50/30 dark:bg-amber-950/10',
   513→                      isVip && isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/70 dark:bg-amber-950/30',
   514→                    )}
   515→                  >
   516→                    <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   517→                      {/* Guest Info */}
   518→                      <div className="flex-1 min-w-0">
   519→                        <div className="flex items-center gap-2 flex-wrap">
   520→                          <span className="font-semibold truncate">
   521→                            {arrival.guest ? `${arrival.guest.firstName} ${arrival.guest.lastName}` : 'Unknown Guest'}
   522→                          </span>
   523→                          {isVip && (
   524→                            <Badge className="text-[10px] px-1.5 py-0 gap-0.5 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700">
   525→                              <Star className="size-3 fill-amber-500 text-amber-500" />
   526→                              <Crown className="size-3" />
   527→                              {arrival.guest?.vipLevel?.toUpperCase()}
   528→                            </Badge>
   529→                          )}
   530→                          {isUnassigned && (
   531→                            <Badge variant="outline" className="text-[10px] px-1.5 py-0 border-amber-400 text-amber-600 dark:text-amber-400">
   532→                              <AlertTriangle className="size-3 mr-0.5" />
   533→                              Unassigned
   534→                            </Badge>
   535→                          )}
   536→                        </div>
   537→                        <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground flex-wrap">
   538→                          <span className="font-mono">{arrival.confirmationNo}</span>
   539→                          <span>•</span>
   540→                          <span className="flex items-center gap-1">
   541→                            <BedDouble className="size-3" />
   542→                            {arrival.room ? (
   543→                              <span>Room {arrival.room.number} ({arrival.room.type.name})</span>
   544→                            ) : (
   545→                              <span className="text-amber-600 dark:text-amber-400">No room assigned</span>
   546→                            )}
   547→                          </span>
   548→                          <span>•</span>
