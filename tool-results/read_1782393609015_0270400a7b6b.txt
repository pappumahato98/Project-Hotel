   183→// ─── Helpers ────────────────────────────────────────────────────────────
   184→
   185→function guestFullName(g: FolioGuest | null | undefined): string {
   186→  if (!g) return '—'
   187→  return `${g.firstName} ${g.lastName}`
   188→}
   189→
   190→function folioCharges(f: Folio): number {
   191→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   192→}
   193→
   194→function folioPayments(f: Folio): number {
   195→  return f.payments.reduce((s, p) => s + p.amount, 0)
   196→}
   197→
   198→function folioOutstanding(f: Folio): number {
   199→  return folioCharges(f) - folioPayments(f)
   200→}
   201→
   202→function isVoidedTransaction(t: FolioTransaction): boolean {
   203→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   204→}
   205→
   206→function isVoidedPayment(p: FolioPayment): boolean {
   207→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   208→}
   209→
   210→function useDebounce<T>(value: T, delay: number): T {
   211→  const [debounced, setDebounced] = useState(value)
   212→  useEffect(() => {
   213→    const timer = setTimeout(() => setDebounced(value), delay)
   214→    return () => clearTimeout(timer)
   215→  }, [value, delay])
   216→  return debounced
   217→}
