   102→// ─── Helper ──────────────────────────────────────────────────────────────
   103→
   104→function nightsBetween(start: string | Date, end: string | Date): number {
   105→  const s = new Date(start)
   106→  const e = new Date(end)
   107→  return Math.max(1, Math.ceil((e.getTime() - s.getTime()) / (1000 * 60 * 60 * 24)))
   108→}
   109→
   110→function formatDateValue(date: Date): string {
   111→  return toDateOnly(date)
   112→}
