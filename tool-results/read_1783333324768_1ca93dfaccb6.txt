   275→// ── Inline Row Action Buttons ─────────────────────────────────
   276→function InlineRowActions({
   277→  row,
   278→  onStatusChange,
   279→  onViewDetail,
   280→  isForceMutated,
   281→}: {
   282→  row: RoomRow
   283→  onStatusChange: (roomId: string, taskId: string | null, newStatus: string) => void
   284→  onViewDetail: (roomId: string) => void
   285→  isForceMutated: boolean
   286→}) {
   287→  const s = row.hkDisplayStatus
   288→  const canClean = ['dirty', 'change_over', 'pending', 'assigned'].includes(s)
   289→  const canMarkCleaned = ['cleaning', 'in_progress'].includes(s)
   290→  const canInspect = s === 'cleaned'
   291→  const canReset = ['failed', 'inspected'].includes(s)
   292→  const canFail = ['cleaning', 'in_progress', 'cleaned'].includes(s)
   293→  const canDirty = ['clean', 'inspected'].includes(s)
   294→  const hasActions = canClean || canMarkCleaned || canInspect || canReset || canFail || canDirty
   295→
   296→  return (
   297→    <div className="flex items-center gap-0.5 flex-nowrap" onPointerDown={(e) => e.stopPropagation()}>
   298→      {/* Occupied indicator + Lock icon next to Eye */}
   299→      {row.reservationStatus === 'occupied' && (
   300→        <span className="text-amber-500" title={`Occupied by ${row.guestName || 'Guest'}`}>
   301→          <ShieldAlert className="h-3.5 w-3.5" />
   302→        </span>
   303→      )}
   304→      {/* View Details — always visible as separate button */}
   305→      <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground"
   306→        onClick={() => {
   307→          if (row.reservationStatus === 'occupied') {
   308→            toast.info(`Room ${row.roomNumber} — Occupied By "${row.guestName || 'Guest'}"`, { duration: 3000 })
   309→          }
   310→          onViewDetail(row.roomId)
   311→        }} title="View Details">
   312→        <Eye className="h-3.5 w-3.5" />
   313→      </Button>
   314→
   315→      {/* Three-dot menu for status actions */}
   316→      <DropdownMenu modal={false}>
   317→        <DropdownMenuTrigger asChild>
   318→          <Button variant="ghost" size="sm" className="h-8 w-8 sm:h-7 sm:w-7 p-0 text-muted-foreground hover:text-foreground">
   319→            <MoreHorizontal className="h-3.5 w-3.5" />
   320→          </Button>
   321→        </DropdownMenuTrigger>
   322→        <DropdownMenuContent align="start" side="bottom" sideOffset={4} collisionPadding={8} className="w-48">
   323→          {/* Start Cleaning — dirty, change_over, pending, assigned */}
   324→          {canClean && (
   325→            <DropdownMenuItem
   326→              onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'in_progress')}
   327→              disabled={isForceMutated}
   328→              className="gap-2 text-xs text-sky-600 focus:text-sky-600 focus:bg-sky-50 dark:text-sky-400 dark:focus:bg-sky-950/40"
   329→            >
   330→              <PlayCircle className="h-4 w-4" /> Start Cleaning
   331→            </DropdownMenuItem>
   332→          )}
   333→
   334→          {/* Mark Cleaned — cleaning, in_progress */}
   335→          {canMarkCleaned && (
   336→            <DropdownMenuItem
   337→              onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'cleaned')}
   338→              disabled={isForceMutated}
   339→              className="gap-2 text-xs text-emerald-600 focus:text-emerald-600 focus:bg-emerald-50 dark:text-emerald-400 dark:focus:bg-emerald-950/40"
   340→            >
   341→              <CheckCircle className="h-4 w-4" /> Mark Cleaned
   342→            </DropdownMenuItem>
   343→          )}
   344→
   345→          {/* Mark Inspected — cleaned */}
   346→          {canInspect && (
   347→            <DropdownMenuItem
   348→              onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'inspected')}
   349→              disabled={isForceMutated}
   350→              className="gap-2 text-xs text-purple-600 focus:text-purple-600 focus:bg-purple-50 dark:text-purple-400 dark:focus:bg-purple-950/40"
   351→            >
   352→              <ShieldCheck className="h-4 w-4" /> Mark Inspected
   353→            </DropdownMenuItem>
   354→          )}
   355→
   356→          {/* Mark Dirty — clean, inspected (re-trigger cleaning) */}
   357→          {canDirty && (
   358→            <DropdownMenuItem
   359→              onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'dirty')}
   360→              disabled={isForceMutated}
   361→              className="gap-2 text-xs text-orange-600 focus:text-orange-600 focus:bg-orange-50 dark:text-orange-400 dark:focus:bg-orange-950/40"
   362→            >
   363→              <RotateCcw className="h-4 w-4" /> Mark Dirty
   364→            </DropdownMenuItem>
   365→          )}
   366→
   367→          {/* Reset to Pending — failed, inspected */}
   368→          {canReset && (
   369→            <DropdownMenuItem
   370→              onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'pending')}
   371→              disabled={isForceMutated}
   372→              className="gap-2 text-xs text-amber-600 focus:text-amber-600 focus:bg-amber-50 dark:text-amber-400 dark:focus:bg-amber-950/40"
   373→            >
   374→              <ArrowUpCircle className="h-4 w-4" /> Reset to Pending
   375→            </DropdownMenuItem>
   376→          )}
   377→
   378→          {/* Rush — always available (not blocked by force-mutated) */}
   379→          <DropdownMenuSeparator />
   380→          <DropdownMenuItem
   381→            onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'rush')}
   382→            className="gap-2 text-xs text-red-500 focus:text-red-500 focus:bg-red-50 dark:text-red-400 dark:focus:bg-red-950/40"
   383→          >
   384→            <AlertTriangle className="h-4 w-4" /> Set Rush Priority
   385→          </DropdownMenuItem>
   386→
   387→          {/* Mark Failed — cleaning, in_progress, cleaned */}
   388→          {canFail && (
   389→            <>
   390→              <DropdownMenuSeparator />
   391→              <DropdownMenuItem
   392→                onSelect={() => onStatusChange(row.roomId, row.hkTaskId, 'failed')}
   393→                disabled={isForceMutated}
   394→                className="gap-2 text-xs text-red-600 focus:text-red-600 focus:bg-red-50 dark:text-red-400 dark:focus:bg-red-950/40"
   395→              >
   396→                <XCircle className="h-4 w-4" /> Mark Failed
   397→              </DropdownMenuItem>
   398→            </>
   399→          )}
   400→
   401→          {/* No actions hint — only shows if nothing else is available (edge case) */}
   402→          {!hasActions && (
   403→            <div className="px-2 py-1.5 text-xs text-muted-foreground text-center">
   404→              No actions available
   405→            </div>
   406→          )}
   407→        </DropdownMenuContent>
   408→      </DropdownMenu>
   409→
   410→      {/* Locked badge */}
   411→      {isForceMutated && (
   412→        <Badge variant="outline" className="text-[9px] text-red-500 border-red-300 dark:border-red-700 shrink-0">
   413→          <Ban className="h-2.5 w-2.5 mr-0.5" /> Locked
   414→        </Badge>
   415→      )}
   416→    </div>
   417→  )
   418→}
