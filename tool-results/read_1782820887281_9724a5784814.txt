   108→   108→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   109→   109→}) {
   110→   110→  return (

   122→   122→        <button
   123→   123→          type="button"
   124→   124→          onClick={() => onChange(Math.min(max, value + 1))}

   135→   135→  switch (source) {

   138→   138→    case 'phone': return 'Phone'

   141→   141→    default: return source || 'Direct'
   142→   142→  }
