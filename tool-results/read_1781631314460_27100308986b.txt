   193→function generatePreviewConfirmation(): string {
   194→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
   195→  let result = 'MRD-'
   196→  for (let i = 0; i < 6; i++) {
   197→    result += chars.charAt(Math.floor(Math.random() * chars.length))
   198→  }
   199→  return result
   200→}

   202→function formatDayOfWeek(dateStr: string): string {
   203→  const d = new Date(dateStr)
   204→  if (isNaN(d.getTime())) return ''
   205→  return d.toLocaleDateString('en-US', { weekday: 'short' })
   206→}
   207→
   208→function formatDateWithDay(dateStr: string): string {
   209→  if (!dateStr) return ''
   210→  const d = new Date(dateStr)
   211→  const day = formatDayOfWeek(dateStr)
   212→  const formatted = d.toLocaleDateString('en-US', { day: '2-digit', month: 'short' })
   213→  return `${day} ${formatted}`
   214→}

   216→function useDebounce<T>(value: T, delay: number): T {
   217→  const [debounced, setDebounced] = useState(value)
   218→  useEffect(() => {
   219→    const timer = setTimeout(() => setDebounced(value), delay)
   220→    return () => clearTimeout(timer)
   221→  }, [value, delay])
   222→  return debounced
   223→}
