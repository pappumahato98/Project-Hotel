   156→// ─── Component ──────────────────────────────────────────────────────────
   157→
   158→export function ArrivalsView() {
   159→  const queryClient = useQueryClient()
   160→  const { settings } = useSettingsStore()
   161→  const { navigateTo } = useNavigationStore()
   162→  const today = getTodayString()
   163→
   164→  // Dialog states
   165→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   166→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   167→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   168→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   169→
   170→  // Selection states
   171→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   172→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   173→
   174→  // Check-in form states
   175→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   176→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   177→  const [roomPreference, setRoomPreference] = useState('')
   178→  const [pillowType, setPillowType] = useState('')
   179→  const [wakeupCall, setWakeupCall] = useState('')
   180→
   181→  // Key card states
   182→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   183→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   184→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   185→
   186→  // Walk-in form state
   187→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   188→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   189→
   190→  // ─── Queries ────────────────────────────────────────────────────────
   191→
   192→  const { data, isLoading } = useQuery({
   193→    queryKey: ['arrivals', today],
   194→    queryFn: async () => {
   195→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   196→      return apiFetch(`/api/reservations?${params.toString()}`)
   197→    },
   198→    refetchInterval: 30000,
   199→  })
   200→
   201→  const { data: roomsData } = useQuery({
   202→    queryKey: ['rooms', 'vacant'],
   203→    queryFn: () => apiFetch('/api/rooms'),
   204→  })
   205→
   206→  const arrivals: Arrival[] = data?.reservations || []
   207→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   208→  const availableRooms = allRooms.filter(
   209→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   210→  )
   211→
   212→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   213→
   214→  const sortedArrivals = useMemo(() => {
   215→    return [...arrivals].sort((a, b) => {
   216→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   217→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   218→      if (aVip && !bVip) return -1
   219→      if (!aVip && bVip) return 1
   220→      return 0
   221→    })
   222→  }, [arrivals])
   223→
   224→  // ─── Stats ──────────────────────────────────────────────────────────
   225→
   226→  const totalArrivals = arrivals.length
   227→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   228→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   229→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   230→
   231→  // ─── Mutations ──────────────────────────────────────────────────────
   232→
   233→  const assignRoomMutation = useMutation({
   234→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   235→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   236→      if (specialRequests !== undefined) {
   237→        payload.specialRequests = specialRequests
   238→      }
   239→      return apiFetch(`/api/reservations/${reservationId}`, {
   240→        method: 'PATCH',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify(payload),
   243→      })
   244→    },
   245→    onSuccess: (_data, variables) => {
   246→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   247→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   248→      setRoomPickerOpen(false)
   249→      setCheckInDialogOpen(false)
   250→
   251→      // Find the room number for key card dialog
   252→      const room = availableRooms.find((r) => r.id === variables.roomId)
   253→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   254→      openKeyCardDialog(roomNumber)
   255→    },
   256→  })
   257→
   258→  const checkInMutation = useMutation({
   259→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   260→      const payload: Record<string, unknown> = { status: 'checked_in' }
   261→      if (specialRequests !== undefined) {
   262→        payload.specialRequests = specialRequests
   263→      }
   264→      return apiFetch(`/api/reservations/${reservationId}`, {
   265→        method: 'PATCH',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify(payload),
   268→      })
   269→    },
   270→    onSuccess: (_data, variables) => {
   271→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   272→      setCheckInDialogOpen(false)
   273→
   274→      const roomNumber = selectedArrival?.room?.number || ''
   275→      openKeyCardDialog(roomNumber)
   276→    },
   277→  })
   278→
   279→  const walkInMutation = useMutation({
   280→    mutationFn: async (form: WalkInForm) => {
   281→      // 1. Create guest
   282→      const guestData = await apiFetch('/api/guests', {
   283→        method: 'POST',
   284→        headers: { 'Content-Type': 'application/json' },
   285→        body: JSON.stringify({
   286→          firstName: form.firstName,
   287→          lastName: form.lastName,
   288→          phone: form.phone || null,
   289→          email: form.email || null,
   290→          nationality: form.nationality || null,
   291→          idType: form.idType || null,
   292→          idNumber: form.idNumber || null,
   293→        }),
   294→      })
   295→      const guestId = guestData.guest.id
   296→
   297→      // 2. Find a vacant room
   298→      const vacantRoom = availableRooms[0]
   299→      if (!vacantRoom) throw new Error('No available rooms')
   300→
   301→      // 3. Create reservation as checked_in (walk-in)
   302→      const checkOutDate = form.checkOutDate || (() => {
   303→        const d = new Date()
   304→        d.setDate(d.getDate() + 1)
   305→        return d
   306→      })()
   307→
   308→      const resData = await apiFetch('/api/reservations', {
   309→        method: 'POST',
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify({
   312→          guestId,
   313→          roomId: vacantRoom.id,
   314→          adults: form.adults,
   315→          children: form.children,
   316→          checkIn: new Date().toISOString(),
   317→          checkOut: checkOutDate.toISOString(),
   318→          roomRate: form.roomRate,
   319→          source: 'walk_in',
   320→          reservationType: 'walk_in',
   321→          guaranteed: false,
   322→        }),
   323→      })
   324→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   325→    },
   326→    onSuccess: (result) => {
   327→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   328→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   329→      queryClient.invalidateQueries({ queryKey: ['guests'] })
   330→      setWalkInDialogOpen(false)
   331→      setWalkInForm(initialWalkInForm)
   332→
   333→      // Open key card dialog
   334→      openKeyCardDialog(result.roomNumber)
   335→    },
   336→    onError: (error) => {
   337→      toast.error('Walk-in check-in failed', {
   338→        description: error instanceof Error ? error.message : 'Unknown error',
   339→      })
   340→    },
   341→  })
   342→
   343→  // ─── Handlers ──────────────────────────────────────────────────────
   344→
   345→  const handleAssignRoom = (arrival: Arrival) => {
   346→    setSelectedArrival(arrival)
   347→    setRoomPickerOpen(true)
   348→  }
   349→
   350→  const handleQuickCheckIn = (arrival: Arrival) => {
   351→    if (!arrival.room) {
   352→      setSelectedArrival(arrival)
   353→      setRoomPickerOpen(true)
   354→      return
   355→    }
   356→    setSelectedArrival(arrival)
   357→    resetCheckInForm()
   358→    setCheckInDialogOpen(true)
   359→  }
   360→
   361→  const handleOpenWalkIn = () => {
   362→    setWalkInForm(initialWalkInForm)
   363→    setWalkInDialogOpen(true)
   364→  }
   365→
   366→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   367→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   368→    navigateTo('front-desk', 'check-in')
   369→  }
   370→
   371→  const handleViewLedger = (arrival: Arrival) => {
   372→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   373→      guestId: arrival.guest?.id || '',
   374→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   375→    })
   376→    navigateTo('front-desk', 'guest-ledger')
   377→  }
   378→
   379→  const buildSpecialRequests = (arrival: Arrival): string => {
   380→    const parts: string[] = []
   381→
   382→    // Preserve existing requests
   383→    if (arrival.specialRequests) {
   384→      parts.push(arrival.specialRequests)
   385→    }
   386→
   387→    // Early check-in note
   388→    if (earlyCheckIn) {
   389→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   390→    }
   391→
   392→    // Check-out time override
   393→    if (checkOutTime !== settings.defaultCheckOut) {
   394→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   395→    }
   396→
   397→    // Room preference
   398→    if (roomPreference) {
   399→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   400→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   401→    }
   402→
   403→    // Pillow type
   404→    if (pillowType) {
   405→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   406→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   407→    }
   408→
   409→    // Wake-up call
   410→    if (wakeupCall) {
   411→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   412→    }
   413→
   414→    return parts.join(' | ')
   415→  }
   416→
   417→  const confirmCheckIn = () => {
   418→    if (!selectedArrival) return
   419→
   420→    const updatedRequests = buildSpecialRequests(selectedArrival)
   421→
   422→    if (selectedRoomId) {
   423→      assignRoomMutation.mutate({
   424→        reservationId: selectedArrival.id,
   425→        roomId: selectedRoomId,
   426→        specialRequests: updatedRequests,
   427→      })
   428→    } else if (selectedArrival.room) {
   429→      checkInMutation.mutate({
   430→        reservationId: selectedArrival.id,
   431→        specialRequests: updatedRequests,
   432→      })
   433→    }
   434→  }
   435→
   436→  const openKeyCardDialog = (roomNumber: string) => {
   437→    setKeyCardRoomNumber(roomNumber)
   438→    setKeyCardIssued(false)
   439→    setKeyCardProcessed(false)
   440→    setKeyCardDialogOpen(true)
   441→  }
   442→
   443→  const handleKeyCardConfirm = () => {
   444→    setKeyCardProcessed(true)
   445→    setKeyCardDialogOpen(false)
   446→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   447→      duration: 5000,
   448→      icon: <Mail className="size-4" />,
   449→    })
   450→    setSelectedArrival(null)
   451→    setSelectedRoomId('')
   452→  }
   453→
   454→  const resetCheckInForm = () => {
   455→    setEarlyCheckIn(false)
   456→    setCheckOutTime(settings.defaultCheckOut)
   457→    setRoomPreference('')
   458→    setPillowType('')
   459→    setWakeupCall('')
   460→  }
   461→
   462→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   463→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   464→    setWalkInForm((prev) => ({
   465→      ...prev,
   466→      roomTypeId,
   467→      roomRate: roomType?.baseRate || 0,
   468→    }))
   469→  }
   470→
   471→  // ─── Render ─────────────────────────────────────────────────────────
   472→
   473→  return (
   474→    <div className="flex flex-col gap-2">
   475→      {/* Header */}
   476→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   477→        <div>
   478→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   479→          <p className="text-xs text-muted-foreground">
   480→            Guest check-ins scheduled for {formatDate(today)}
   481→          </p>
   482→        </div>
   483→        <Button onClick={handleOpenWalkIn} className="gap-2 shrink-0">
   484→          <Footprints className="size-4" />
   485→          Walk-in Check-in
   486→        </Button>
   487→      </div>
   488→
   489→      {/* Arrivals List */}
   490→      <Card className="py-0">
   491→        <CardContent className="p-0">
   492→          {isLoading ? (
   493→            <div className="p-2 space-y-2">
   494→              {Array.from({ length: 4 }).map((_, i) => (
   495→                <Skeleton key={i} className="h-20 w-full rounded-lg" />
   496→              ))}
   497→            </div>
   498→          ) : sortedArrivals.length === 0 ? (
   499→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   500→              <LogIn className="size-8 mb-2 opacity-50" />
   501→              <p className="text-sm">No arrivals scheduled for today</p>
   502→            </div>
   503→          ) : (
   504→            <div className="divide-y">
   505→              {sortedArrivals.map((arrival) => {
   506→                const isUnassigned = !arrival.room
   507→                const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   508→                return (
   509→                  <div
   510→                    key={arrival.id}
   511→                    className={cn(
   512→                      'p-2.5 transition-colors hover:bg-muted/50',
   513→                      isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/50 dark:bg-amber-950/20',
   514→                      isVip && !isUnassigned && 'border-l-4 border-l-amber-400 bg-amber-50/30 dark:bg-amber-950/10',
   515→                      isVip && isUnassigned && 'border-l-4 border-l-amber-500 bg-amber-50/70 dark:bg-amber-950/30',
   516→                    )}
   517→                  >
   518→                    <div className="flex flex-col sm:flex-row sm:items-center gap-2">
   519→                      {/* Guest Info */}
   520→                      <div className="flex-1 min-w-0">
   521→                        <div className="flex items-center gap-2 flex-wrap">
   522→                          <span className="font-semibold truncate">
   523→                            {arrival.guest ? `${arrival.guest.firstName} ${arrival.guest.lastName}` : 'Unknown Guest'}
   524→                          </span>
   525→                          {isVip && (
   526→                            <Badge className="text-[10px] px-1.5 py-0 gap-0.5 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-300 dark:border-amber-700">
   527→                              <Star className="size-3 fill-amber-500 text-amber-500" />
   528→                              <Crown className="size-3" />
   529→                              {arrival.guest?.vipLevel?.toUpperCase()}
   530→                            </Badge>
   531→                          )}
   532→                          {isUnassigned && (
   533→                            <Badge variant="outline" className="text-[10px] px-1.5 py-0 border-amber-400 text-amber-600 dark:text-amber-400">
   534→                              <AlertTriangle className="size-3 mr-0.5" />
   535→                              Unassigned
   536→                            </Badge>
   537→                          )}
   538→                        </div>
   539→                        <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground flex-wrap">
   540→                          <span className="font-mono">{arrival.confirmationNo}</span>
   541→                          <span>•</span>
   542→                          <span className="flex items-center gap-1">
   543→                            <BedDouble className="size-3" />
   544→                            {arrival.room ? (
   545→                              <span>Room {arrival.room.number} ({arrival.room.type.name})</span>
   546→                            ) : (
   547→                              <span className="text-amber-600 dark:text-amber-400">No room assigned</span>
