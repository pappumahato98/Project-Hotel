   257→   257→function TaskDetailDialog({ task, open, onOpenChange }: { task: HkTask | null; open: boolean; onOpenChange: (open: boolean) => void }) {
   258→   258→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   259→   259→  const queryClient = useQueryClient()
   260→   260→
   261→   261→  const updateStatusMutation = useMutation({
   262→   262→    mutationFn: (body: { id: string; status: string; inspectedBy?: string }) =>
   263→   263→      apiFetch('/api/housekeeping', {
   264→   264→        method: 'POST',
   265→   265→        headers: { 'Content-Type': 'application/json' },
   266→   266→        body: JSON.stringify({ ...body, action: 'update-task-status' }),
   267→   267→      }),
   268→   268→    onSuccess: () => {
   269→   269→      toast.success('Task status updated')
   270→   270→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   271→   271→      queryClient.invalidateQueries({ queryKey: ['housekeeping-rooms'] })
   272→   272→      onOpenChange(false)
   273→   273→    },
   274→   274→  })
   275→   275→
   276→   276→  if (!task) return null
   277→   277→
   278→   278→  const toggleCheck = (item: string) => {
   279→   279→    setCheckedItems((prev) => ({ ...prev, [item]: !prev[item] }))
   280→   280→  }
   281→   281→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   282→   282→
   283→   283→  return (
   284→   284→    <Dialog open={open} onOpenChange={onOpenChange}>
   285→   285→      <DialogContent className="sm:max-w-lg">
   286→   286→        <DialogHeader>
   287→   287→          <DialogTitle className="flex items-center gap-2">
   288→   288→            Room {task.room.number} — {formatTaskType(task.taskType)}
   289→   289→          </DialogTitle>
   290→   290→          <DialogDescription>
   291→   291→            Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   292→   292→          </DialogDescription>
   293→   293→        </DialogHeader>
   294→   294→
   295→   295→        <div className="space-y-4">
   296→   296→          <div className="grid grid-cols-2 gap-3">
   297→   297→            <div className="rounded-lg border p-3 space-y-1">
   298→   298→              <p className="text-xs text-muted-foreground">Priority</p>
   299→   299→              <Badge variant="outline" className={cn(priorityColor(task.priority))}>
   300→   300→                {task.priority.charAt(0).toUpperCase() + task.priority.slice(1)}
   301→   301→              </Badge>
   302→   302→            </div>
   303→   303→            <div className="rounded-lg border p-3 space-y-1">

   310→   310→            </div>
   311→   311→            <div className="rounded-lg border p-3 space-y-1">
   312→   312→              <p className="text-xs text-muted-foreground">Est. Time</p>

   388→   388→function AttendantView({ tasks }: { tasks: HkTask[] }) {
   389→   389→  const attendants = new Map<string, HkTask[]>()
   390→   390→  tasks.forEach((task) => {
   391→   391→    const name = task.assignedTo || 'Unassigned'
   392→   392→    if (!attendants.has(name)) attendants.set(name, [])
   393→   393→    attendants.get(name)!.push(task)
   394→   394→  })
   395→   395→
   396→   396→  return (
   397→   397→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   398→   398→      {Array.from(attendants.entries()).map(([name, attTasks]) => (
   399→   399→        <Card key={name}>
   400→   400→          <CardContent className="p-3 space-y-2">
