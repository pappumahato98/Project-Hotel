   142→   142→function PhotoCaptureSection({
   143→   143→  photos,
   144→   144→  onCapture,
   145→   145→  onRemove,
   146→   146→}: {
   147→   147→  photos: string[]
   148→   148→  onCapture: (dataUrl: string) => void
   149→   149→  onRemove: (index: number) => void
   150→   150→}) {
   151→   151→  const videoRef = useRef<HTMLVideoElement>(null)
   152→   152→  const streamRef = useRef<MediaStream | null>(null)
   153→   153→  const [cameraOpen, setCameraOpen] = useState(false)
   154→   154→  const [cameraError, setCameraError] = useState<string | null>(null)
   155→   155→
   156→   156→  const startCamera = useCallback(async () => {
   157→   157→    try {
   158→   158→      setCameraError(null)
   159→   159→      const stream = await navigator.mediaDevices.getUserMedia({
   160→   160→        video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
   161→   161→      })
   162→   162→      streamRef.current = stream
   163→   163→      if (videoRef.current) {
   164→   164→        videoRef.current.srcObject = stream
   165→   165→      }
   166→   166→      setCameraOpen(true)
   167→   167→    } catch (_err) {
   168→   168→      setCameraError('Camera access denied or unavailable')
   169→   169→    }
   170→   170→  }, [])
   171→   171→
   172→   172→  const stopCamera = useCallback(() => {
   173→   173→    if (streamRef.current) {
   174→   174→      streamRef.current.getTracks().forEach((t) => t.stop())
   175→   175→      streamRef.current = null
   176→   176→    }
   177→   177→    setCameraOpen(false)
   178→   178→  }, [])
   179→   179→
   180→   180→  const capturePhoto = useCallback(() => {
   181→   181→    const video = videoRef.current
   182→   182→    if (!video) return
   183→   183→    const canvas = document.createElement('canvas')
   184→   184→    canvas.width = video.videoWidth
   185→   185→    canvas.height = video.videoHeight
   186→   186→    const ctx = canvas.getContext('2d')
   187→   187→    if (!ctx) return
   188→   188→    ctx.drawImage(video, 0, 0)
   189→   189→    const dataUrl = canvas.toDataURL('image/jpeg', 0.85)
   190→   190→    onCapture(dataUrl)
   191→   191→    stopCamera()
   192→   192→  }, [onCapture, stopCamera])
   193→   193→
   194→   194→  return (
   195→   195→    <div className="space-y-2">
   196→   196→      <p className="text-sm font-semibold">Photo Evidence</p>
   197→   197→
   198→   198→      {/* Camera Preview */}
   199→   199→      {cameraOpen && (
   200→   200→        <div className="relative rounded-lg border overflow-hidden bg-black max-h-36 sm:max-h-48">
   201→   201→          <video
   202→   202→            ref={videoRef}
   203→   203→            autoPlay
   204→   204→            playsInline
   205→   205→            muted
   206→   206→            className="w-full h-36 sm:h-48 object-cover"
   207→   207→          />
   208→   208→          <div className="absolute bottom-2 left-1/2 -translate-x-1/2 flex gap-2">
   209→   209→            <Button
   210→   210→              size="sm"
   211→   211→              className="gap-1.5 bg-white text-black hover:bg-gray-200"
   212→   212→              onClick={capturePhoto}
   213→   213→            >
   214→   214→              <Camera className="h-4 w-4" /> Capture
   215→   215→            </Button>
   216→   216→            <Button
   217→   217→              size="sm"
   218→   218→              variant="outline"
   219→   219→              className="gap-1.5 bg-white/90 text-black hover:bg-gray-200 border-gray-300"
   220→   220→              onClick={stopCamera}
   221→   221→            >
   222→   222→              <X className="h-4 w-4" /> Cancel
   223→   223→            </Button>
   224→   224→          </div>
   225→   225→        </div>
   226→   226→      )}
   227→   227→
   228→   228→      {cameraError && (
   229→   229→        <p className="text-xs text-red-500 flex items-center gap-1">
   230→   230→          <AlertCircle className="h-3 w-3" /> {cameraError}
   231→   231→        </p>
   232→   232→      )}
   233→   233→
   234→   234→      {/* Photo Grid */}
   235→   235→      {photos.length > 0 && (
   236→   236→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   237→   237→          {photos.map((photo, idx) => (
   238→   238→            <div key={idx} className="relative group rounded-md border overflow-hidden">
   239→   239→              <img src={photo} alt={`Evidence ${idx + 1}`} className="w-full h-auto max-h-24 sm:max-h-28 object-cover" />
   240→   240→              <button
   241→   241→                onClick={() => onRemove(idx)}
   242→   242→                className="absolute top-1 right-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-70 sm:opacity-0 sm:group-hover:opacity-100 transition-opacity"
   243→   243→              >
   244→   244→                <Trash2 className="h-3 w-3" />
   245→   245→              </button>
   246→   246→            </div>
   247→   247→          ))}
   248→   248→        </div>
   249→   249→      )}
   250→   250→
   251→   251→      {/* Open Camera Button */}
   252→   252→      {!cameraOpen && (
   253→   253→        <Button variant="outline" className="w-full gap-2" onClick={startCamera}>
   254→   254→          <Camera className="h-4 w-4" />
   255→   255→          {photos.length > 0 ? `Add More Photo (${photos.length} captured)` : 'Capture Photo Evidence'}
   256→   256→        </Button>
   257→   257→      )}
   258→   258→    </div>
   259→   259→  )
   260→   260→}

   315→   315→function InspectionDialog({
   316→   316→  task,
   317→   317→  open,
   318→   318→  onOpenChange,
   319→   319→}: {
   320→   320→  task: HkTask | null
   321→   321→  open: boolean
   322→   322→  onOpenChange: (open: boolean) => void
   323→   323→}) {
   324→   324→  const queryClient = useQueryClient()
   325→   325→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   326→   326→  const [photos, setPhotos] = useState<string[]>([])
   327→   327→  const [rejectReason, setRejectReason] = useState('')
   328→   328→  const [showRejectForm, setShowRejectForm] = useState(false)
   329→   329→  const [showAuditTrail, setShowAuditTrail] = useState(false)
   330→   330→
   331→   331→  const approveMutation = useMutation({
   332→   332→    mutationFn: async () => {
   333→   333→      if (!task) return
   334→   334→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   335→   335→        if (v) acc[k] = true
   336→   336→        return acc
   337→   337→      }, {} as Record<string, boolean>)
   338→   338→      return apiFetch('/api/housekeeping', {
   339→   339→        method: 'POST',
   340→   340→        headers: { 'Content-Type': 'application/json' },
   341→   341→        body: JSON.stringify({
   342→   342→          action: 'approve-inspection',
   343→   343→          id: task.id,
   344→   344→          performedBy: 'Inspector',
   345→   345→          checklist,
   346→   346→          photos: photos.length > 0 ? photos : undefined,
   347→   347→        }),
   348→   348→      })
   349→   349→    },
   350→   350→    onSuccess: () => {
   351→   351→      invalidate.afterRoomStatusChange(queryClient)
   352→   352→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   353→   353→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   354→   354→      toast.success(`Room ${task?.room.number} inspection passed`)
   355→   355→      resetAndClose()
   356→   356→    },
   357→   357→    onError: (err: Error) => {
   358→   358→      toast.error(err.message || 'Failed to approve inspection')
   359→   359→    },
   360→   360→  })
   361→   361→
   362→   362→  const rejectMutation = useMutation({
   363→   363→    mutationFn: async () => {
   364→   364→      if (!task) return
   365→   365→      if (!rejectReason.trim()) {
   366→   366→        toast.error('Please provide a reason for rejection')
   367→   367→        throw new Error('Reason required')
   368→   368→      }
   369→   369→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   370→   370→        if (v) acc[k] = true
   371→   371→        return acc
   372→   372→      }, {} as Record<string, boolean>)
   373→   373→      return apiFetch('/api/housekeeping', {
   374→   374→        method: 'POST',
   375→   375→        headers: { 'Content-Type': 'application/json' },
   376→   376→        body: JSON.stringify({
   377→   377→          action: 'reject-inspection',
   378→   378→          id: task.id,
   379→   379→          performedBy: 'Inspector',
   380→   380→          reason: rejectReason.trim(),
   381→   381→          checklist,
   382→   382→          photos: photos.length > 0 ? photos : undefined,
   383→   383→        }),
   384→   384→      })
   385→   385→    },
   386→   386→    onSuccess: () => {
   387→   387→      invalidate.afterRoomStatusChange(queryClient)
   388→   388→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   389→   389→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   390→   390→      toast.info(`Room ${task?.room.number} rejected & reassigned for cleaning`)
   391→   391→      resetAndClose()
   392→   392→    },
   393→   393→    onError: (err: Error) => {
   394→   394→      if (err.message !== 'Reason required') {
   395→   395→        toast.error(err.message || 'Failed to reject inspection')
   396→   396→      }
   397→   397→    },
   398→   398→  })
   399→   399→
   400→   400→  const resetAndClose = () => {
   401→   401→    setCheckedItems({})
   402→   402→    setPhotos([])
   403→   403→    setRejectReason('')
   404→   404→    setShowRejectForm(false)
   405→   405→    setShowAuditTrail(false)
   406→   406→    onOpenChange(false)
   407→   407→  }
   408→   408→
   409→   409→  if (!task) return null
   410→   410→
   411→   411→  const toggleCheck = (id: string) => {
   412→   412→    setCheckedItems((prev) => ({ ...prev, [id]: !prev[id] }))
   413→   413→  }
   414→   414→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   415→   415→  const allChecked = checkedCount === INSPECTION_ITEMS.length
   416→   416→  const passCount = INSPECTION_ITEMS.length
   417→   417→  const passPercent = Math.round((checkedCount / passCount) * 100)
   418→   418→
   419→   419→  const handleCapturePhoto = (dataUrl: string) => {
   420→   420→    setPhotos((prev) => [...prev, dataUrl])
   421→   421→    toast.success('Photo captured')
   422→   422→  }
   423→   423→
   424→   424→  const handleRemovePhoto = (index: number) => {
   425→   425→    setPhotos((prev) => prev.filter((_, i) => i !== index))
   426→   426→  }
   427→   427→
   428→   428→  const isSubmitting = approveMutation.isPending || rejectMutation.isPending
   429→   429→
   430→   430→  return (
   431→   431→    <Dialog open={open} onOpenChange={(v) => { if (!v) resetAndClose(); else onOpenChange(true) }}>
   432→   432→      <DialogContent className="w-[calc(100vw-0.5rem)] sm:w-auto sm:max-w-lg max-h-[calc(100dvh-1rem)] sm:max-h-[calc(100vh-2rem)] flex flex-col p-0 gap-0 overflow-hidden rounded-2xl sm:rounded-lg">
   433→   433→        {/* Fixed Header */}
   434→   434→        <div className="shrink-0 px-4 sm:px-6 pt-4 sm:pt-5 pb-2">
   435→   435→          <DialogHeader>
   436→   436→            <DialogTitle className="flex items-center gap-1.5 sm:gap-2 text-sm sm:text-lg">
   437→   437→              <ShieldCheck className="h-5 w-5 text-purple-600 shrink-0" />
   438→   438→              <span>Inspect Room {task.room.number}</span>
   439→   439→            </DialogTitle>
   440→   440→            <DialogDescription className="mt-1.5 text-xs sm:text-sm leading-relaxed">
   441→   441→              <span className="inline-flex flex-wrap gap-x-1.5 gap-y-0.5">
   442→   442→                <span>Floor {task.room.floor}</span>
   443→   443→                {task.room.wing && <span>· {task.room.wing} Wing</span>}
   444→   444→                <span>· {task.room.type.name}</span>
   445→   445→                {task.assignedTo && <span>· by {task.assignedTo}</span>}
   446→   446→              </span>
   447→   447→            </DialogDescription>
   448→   448→          </DialogHeader>
   449→   449→        </div>
   450→   450→
   451→   451→        {/* Scrollable Body */}
   452→   452→        <ScrollArea className="flex-1 min-h-0 px-3 sm:px-6">
   453→   453→          <div className="space-y-3 sm:space-y-4 pb-4 pr-1">
   454→   454→            {/* Progress */}

   597→   597→export function InspectionView() {
   598→   598→  const [selectedTask, setSelectedTask] = useState<HkTask | null>(null)
   599→   599→
   600→   600→  const { data, isLoading } = useQuery<{
   601→   601→    tasks: HkTask[]
   602→   602→    summary: HkSummary
   603→   603→  }>({
   604→   604→    queryKey: ['housekeeping-tasks'],

   606→   606→  })
   607→   607→
   608→   608→  const tasks = data?.tasks || []
   609→   609→  const summary = data?.summary

   652→   652→      {/* Awaiting Inspection */}
