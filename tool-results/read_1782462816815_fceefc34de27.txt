    62→    62→// ─── Quick Search Dialog ────────────────────────────────────────────
    63→    63→function QuickSearchDialog() {
    64→    64→  const { searchOpen, setSearchOpen, navigateTo } = useNavigationStore()
    65→    65→  const [query, setQuery] = React.useState('')
    66→    66→
    67→    67→  const filteredItems = React.useMemo(() => {
    68→    68→    if (!query.trim()) return NAV_ITEMS
    69→    69→    const lower = query.toLowerCase()
    70→    70→    return NAV_ITEMS.filter(
    71→    71→      (item) =>
    72→    72→        item.label.toLowerCase().includes(lower) ||
    73→    73→        item.children?.some((c) => c.label.toLowerCase().includes(lower))
    74→    74→    )
    75→    75→  }, [query])
    76→    76→
    77→    77→  React.useEffect(() => {
    78→    78→    const down = (e: KeyboardEvent) => {
    79→    79→      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
    80→    80→        e.preventDefault()
    81→    81→        setSearchOpen(!searchOpen)
    82→    82→      }
    83→    83→    }
    84→    84→    document.addEventListener('keydown', down)
    85→    85→    return () => document.removeEventListener('keydown', down)
    86→    86→  }, [searchOpen, setSearchOpen])
    87→    87→
    88→    88→  return (
    89→    89→    <CommandDialog open={searchOpen} onOpenChange={setSearchOpen}>
    90→    90→      <CommandInput
    91→    91→        placeholder="Search modules, pages..."
    92→    92→        value={query}
    93→    93→        onValueChange={setQuery}
    94→    94→      />
    95→    95→      <CommandList>
    96→    96→        <CommandEmpty>No results found.</CommandEmpty>
    97→    97→        {filteredItems.map((item) => (
    98→    98→          <React.Fragment key={item.id}>
    99→    99→            <CommandGroup heading={item.label}>
   100→   100→              <CommandItem
   101→   101→                onSelect={() => {
   102→   102→                  navigateTo(item.id)
   103→   103→                  setSearchOpen(false)
   104→   104→                  setQuery('')
   105→   105→                }}
   106→   106→              >
   107→   107→                <item.icon className="mr-2 size-4 text-muted-foreground" />
   108→   108→                <span>{item.label}</span>
   109→   109→              </CommandItem>
   110→   110→              {item.children?.map((child) => (
   111→   111→                <CommandItem
   112→   112→                  key={child.id}
   113→   113→                  onSelect={() => {
   114→   114→                    navigateTo(item.id, child.id)
   115→   115→                    setSearchOpen(false)
   116→   116→                    setQuery('')
   117→   117→                  }}
   118→   118→                  className="pl-8"
   119→   119→                >
   120→   120→                  <span>{child.label}</span>
   121→   121→                </CommandItem>
   122→   122→              ))}
   123→   123→            </CommandGroup>
   124→   124→            <CommandSeparator />
   125→   125→          </React.Fragment>
   126→   126→        ))}
   127→   127→      </CommandList>
   128→   128→    </CommandDialog>
   129→   129→  )
   130→   130→}
