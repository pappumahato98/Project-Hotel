   162→   162→function guestFullName(g: FolioGuest | null | undefined): string {
   163→   163→  if (!g) return '—'
   164→   164→  return `${g.firstName} ${g.lastName}`
   165→   165→}
   166→   166→
   167→   167→function folioCharges(f: Folio): number {
   168→   168→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   169→   169→}
   170→   170→
   171→   171→function folioPayments(f: Folio): number {
   172→   172→  return f.payments.reduce((s, p) => s + p.amount, 0)
   173→   173→}
   174→   174→
   175→   175→function folioOutstanding(f: Folio): number {
   176→   176→  return folioCharges(f) - folioPayments(f)
   177→   177→}
   178→   178→
   179→   179→function isVoidedTransaction(t: FolioTransaction): boolean {
   180→   180→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   181→   181→}
   182→   182→
   183→   183→function isVoidedPayment(p: FolioPayment): boolean {
   184→   184→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   185→   185→}
   186→   186→
   187→   187→function useDebounce<T>(value: T, delay: number): T {
   188→   188→  const [debounced, setDebounced] = useState(value)
   189→   189→  useEffect(() => {
   190→   190→    const timer = setTimeout(() => setDebounced(value), delay)
   191→   191→    return () => clearTimeout(timer)
   192→   192→  }, [value, delay])
   193→   193→  return debounced
   194→   194→}
