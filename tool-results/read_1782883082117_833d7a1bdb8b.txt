   269→   269→function NewOrderDialog({
   270→   270→  open,
   271→   271→  onClose,
   272→   272→  onSubmit,
   273→   273→}: {
   274→   274→  open: boolean
   275→   275→  onClose: () => void
   276→   276→  onSubmit: (order: Omit<RoomServiceOrder, 'id' | 'createdAt' | 'status' | 'total'> & { total: number }) => void
   277→   277→}) {
   278→   278→  const [selectedGuest, setSelectedGuest] = useState('')
   279→   279→  const [orderItems, setOrderItems] = useState<{ name: string; quantity: number; price: number }[]>([])
   280→   280→  const [specialInstructions, setSpecialInstructions] = useState('')
   281→   281→  const [selectedMenuCategory, setSelectedMenuCategory] = useState('all')
   282→   282→
   283→   283→  const selectedGuestData = MOCK_GUESTS.find((g) => g.id === selectedGuest)
   284→   284→
   285→   285→  const menuCategories = ['all', 'Appetizer', 'Main Course', 'Beverage', 'Breakfast', 'Bread', 'Salad', 'Dessert']
   286→   286→  const filteredMenu = MOCK_MENU_ITEMS.filter((m) => selectedMenuCategory === 'all' || m.category === selectedMenuCategory)
   287→   287→
   288→   288→  const handleAddItem = (menuItem: typeof MOCK_MENU_ITEMS[0]) => {
   289→   289→    const existing = orderItems.find((i) => i.name === menuItem.name)
   290→   290→    if (existing) {
   291→   291→      setOrderItems(orderItems.map((i) => i.name === menuItem.name ? { ...i, quantity: i.quantity + 1 } : i))
   292→   292→    } else {
   293→   293→      setOrderItems([...orderItems, { name: menuItem.name, quantity: 1, price: menuItem.price }])
   294→   294→    }
   295→   295→  }
   296→   296→
   297→   297→  const handleRemoveItem = (name: string) => {
   298→   298→    setOrderItems(orderItems.filter((i) => i.name !== name))
   299→   299→  }
   300→   300→
   301→   301→  const handleQtyChange = (name: string, delta: number) => {
   302→   302→    setOrderItems(orderItems.map((i) => {
   303→   303→      if (i.name !== name) return i
   304→   304→      const newQty = Math.max(0, i.quantity + delta)
   305→   305→      return { ...i, quantity: newQty }
   306→   306→    }).filter((i) => i.quantity > 0))
   307→   307→  }
   308→   308→
   309→   309→  const total = orderItems.reduce((s, i) => s + i.price * i.quantity, 0)
   310→   310→
   311→   311→  const handleSubmit = () => {
   312→   312→    if (!selectedGuest || orderItems.length === 0) {
   313→   313→      toast.error('Please select a guest and add at least one item')
   314→   314→      return
   315→   315→    }
   316→   316→    onSubmit({
   317→   317→      roomNumber: selectedGuestData!.room,
   318→   318→      floor: selectedGuestData!.floor,
   319→   319→      guestName: selectedGuestData!.name,
   320→   320→      phone: '+977-98' + Math.floor(10000000 + Math.random() * 90000000).toString(),
   321→   321→      items: orderItems,
   322→   322→      specialInstructions,
   323→   323→      total,
   324→   324→    })
   325→   325→    setSelectedGuest('')
   326→   326→    setOrderItems([])
   327→   327→    setSpecialInstructions('')
   328→   328→    onClose()
   329→   329→  }
   330→   330→
   331→   331→  return (
   332→   332→    <Dialog open={open} onOpenChange={onClose}>
   333→   333→      <DialogContent className="sm:max-w-2xl max-h-[85vh] overflow-y-auto">
   334→   334→        <DialogHeader>
   335→   335→          <DialogTitle className="flex items-center gap-2">
   336→   336→            <BellRing className="h-4 w-4" />
   337→   337→            New Room Service Order
   338→   338→          </DialogTitle>
   339→   339→        </DialogHeader>
   340→   340→        <div className="space-y-4">
   341→   341→          {/* Guest Selection */}
   342→   342→          <div>
   343→   343→            <Label className="text-sm font-medium">Select Guest</Label>
   344→   344→            <Select value={selectedGuest} onValueChange={setSelectedGuest}>
   345→   345→              <SelectTrigger className="mt-1.5">
   346→   346→                <SelectValue placeholder="Choose in-house guest..." />
   347→   347→              </SelectTrigger>
   348→   348→              <SelectContent>
   349→   349→                {MOCK_GUESTS.map((g) => (
   350→   350→                  <SelectItem key={g.id} value={g.id}>
   351→   351→                    {g.name} — Room {g.room} (Floor {g.floor})
   352→   352→                  </SelectItem>
   353→   353→                ))}
   354→   354→              </SelectContent>
   355→   355→            </Select>
   356→   356→          </div>
   357→   357→
   358→   358→          {/* Menu Items */}
   359→   359→          <div>
   360→   360→            <Label className="text-sm font-medium">Add Items from Menu</Label>
   361→   361→            <Select value={selectedMenuCategory} onValueChange={setSelectedMenuCategory}>
   362→   362→              <SelectTrigger className="mt-1.5 mb-2">
   363→   363→                <SelectValue />
   364→   364→              </SelectTrigger>
   365→   365→              <SelectContent>
   366→   366→                {menuCategories.map((cat) => (
