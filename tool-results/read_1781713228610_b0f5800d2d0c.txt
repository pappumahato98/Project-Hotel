    47→
    48→interface PostingRow {
    49→  id: string
    50→  reservationId: string
    51→  folioId: string
    52→  roomId: string
    53→  postingDate: string
    54→  roomRate: number
    55→  taxAmount: number
    56→  serviceCharge: number
    57→  totalAmount: number
    58→  status: string
    59→  postedBy: string | null
    60→  voidedBy: string | null
    61→  voidReason: string | null
    62→  createdAt: string
    63→  reservation: {
    64→    id: string
    65→    confirmationNo: string
    66→    status: string
    67→    checkIn: string
    68→    checkOut: string
    69→    roomRate: number
    70→    guest: { id: string; firstName: string; lastName: string } | null

    72→  } | null
    73→  folio: { id: string; balance: number; status: string } | null
    74→}
    75→
    76→interface PendingReservation {
    77→  reservationId: string
    78→  confirmationNo: string
    79→  guest: { id: string; firstName: string; lastName: string } | null
    80→  room: { id: string; number: string; type: { name: string } | null } | null
    81→  checkIn: string
    82→  checkOut: string
    83→  roomRate: number
    84→  folio: { id: string; balance: number; status: string } | null
    85→  pendingNights: string[]
    86→  postedNights: string[]
    87→  futureNights: string[]
    88→  totalNights: number
    89→  pendingAmount: number
    90→}
    91→
    92→interface PostingListResponse {
    93→  postings: PostingRow[]
    94→  pagination: { page: number; limit: number; total: number; totalPages: number }
    95→  stats: {
    96→    totalPosted: number
    97→    postedToday: number
    98→    totalRevenue: number
    99→    todayRevenue: number
   100→    inHouseReservations: number
   101→    voidedCount: number
   102→  }
   103→}
   104→
   105→interface PendingResponse {
   106→  pendingReservations: PendingReservation[]
   107→  summary: { totalReservations: number; totalPendingNights: number; totalPendingAmount: number }
   108→}

   122→// ─── Component ──────────────────────────────────────────────────────────
   123→
   124→export function RoomRatePostingPage() {
   125→  const queryClient = useQueryClient()
   126→
   127→  // State
   128→  const [activeTab, setActiveTab] = useState<TabValue>('pending')
   129→  const [search, setSearch] = useState('')
   130→  const [searchInput, setSearchInput] = useState('')
   131→  const [roomFilter, setRoomFilter] = useState('')
   132→  const [dateFrom, setDateFrom] = useState('')
   133→  const [dateTo, setDateTo] = useState('')
   134→  const [page, setPage] = useState(1)
   135→  const [sortBy, setSortBy] = useState('room-asc')
   136→  const [expandedRow, setExpandedRow] = useState<string | null>(null)
   137→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   138→
   139→  // Dialogs
   140→  const [voidDialogOpen, setVoidDialogOpen] = useState(false)
   141→  const [voidTarget, setVoidTarget] = useState<PostingRow | null>(null)
   142→  const [voidReason, setVoidReason] = useState('')
   143→  const [postConfirmOpen, setPostConfirmOpen] = useState(false)
   144→  const [postTarget, setPostTarget] = useState<PendingReservation | null>(null)
   145→  const [bulkPostOpen, setBulkPostOpen] = useState(false)
   146→  const [rateOverride, setRateOverride] = useState('')
   147→  const [showQuickActions, setShowQuickActions] = useState(false)
   148→
   149→  const tableRef = useRef<HTMLDivElement>(null)
   150→
   151→  // ── Fetch all postings list ──────────────────────────────────────────
   152→  const { data: listData, isLoading: isLoadingList, refetch: refetchList } = useQuery({
   153→    queryKey: ['rate-posting-list', { activeTab, search, dateFrom, dateTo, page }],
   154→    queryFn: async () => {
   155→      const params = new URLSearchParams()
   156→      if (activeTab !== 'pending') {
   157→        params.set('status', activeTab === 'all' ? 'all' : activeTab)
   158→      }
   159→      if (search) params.set('search', search)
   160→      if (dateFrom) params.set('dateFrom', dateFrom)
   161→      if (dateTo) params.set('dateTo', dateTo)
   162→      params.set('page', String(page))
   163→      params.set('limit', '50')
   164→
   165→      return apiFetch<PostingListResponse>(`/api/room-rate-posting/list?${params}`)
   166→    },
   167→    enabled: activeTab !== 'pending',
   168→  })
   169→
   170→  // ── Fetch pending reservations ───────────────────────────────────────
   171→  const { data: pendingData, isLoading: isLoadingPending, refetch: refetchPending } = useQuery({
   172→    queryKey: ['rate-posting-pending'],
   173→    queryFn: () => apiFetch<PendingResponse>('/api/room-rate-posting/pending'),
   174→    enabled: activeTab === 'pending',
   175→  })
   176→
   177→  // ── Void mutation ───────────────────────────────────────────────────
   178→  const voidMutation = useMutation({
   179→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   180→      return apiFetch(`/api/room-rate-posting/${id}`, {
   181→        method: 'PATCH',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify({ voidReason: reason }),
   184→      })
   185→    },
   186→    onSuccess: () => {
   187→      toast.success('Rate posting voided successfully')
   188→      invalidate.afterFolioChange(queryClient)
   189→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   190→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   191→      setVoidDialogOpen(false)
   192→      setVoidTarget(null)
   193→      setVoidReason('')
   194→    },
   195→    onError: (error) => toast.error(error.message || 'Failed to void posting'),
   196→  })
   197→
   198→  // ── Post single reservation's pending nights ────────────────────────
   199→  const postMutation = useMutation({
   200→    mutationFn: async ({ reservationId, dates, customRate }: { reservationId: string; dates: string[]; customRate?: number }) => {
   201→      const body: Record<string, unknown> = { reservationId, dates }
   202→      if (customRate && customRate > 0) body.customRate = customRate
   203→      return apiFetch('/api/room-rate-posting', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify(body),
   207→      })
   208→    },
   209→    onSuccess: () => {
   210→      toast.success('Pending charges posted successfully')
   211→      invalidate.afterFolioChange(queryClient)
   212→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   213→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   214→      setPostConfirmOpen(false)
   215→      setPostTarget(null)
   216→      setRateOverride('')
   217→      setSelectedIds(new Set())
   218→    },
   219→    onError: (error) => {
   220→      toast.error(error.message || 'Failed to post charges')
   221→      setPostConfirmOpen(false)
   222→      setPostTarget(null)
   223→      setRateOverride('')
   224→    },
   225→  })
   226→
   227→  // ── Post selected (batch) ───────────────────────────────────────────
   228→  const postSelectedMutation = useMutation({
   229→    mutationFn: async (items: { reservationId: string; dates: string[] }[]) => {
   230→      const results = []
   231→      for (const item of items) {
   232→        try {
   233→          await apiFetch('/api/room-rate-posting', {
   234→            method: 'POST',
   235→            headers: { 'Content-Type': 'application/json' },
   236→            body: JSON.stringify({ reservationId: item.reservationId, dates: item.dates }),
   237→          })
   238→          results.push({ reservationId: item.reservationId, success: true })
   239→        } catch (e) {
   240→          results.push({ reservationId: item.reservationId, success: false, error: (e as Error).message })
   241→        }
   242→      }
   243→      return results
   244→    },
   245→    onSuccess: (results) => {
   246→      const ok = results.filter((r) => r.success).length
   247→      const fail = results.filter((r) => !r.success).length
   248→      if (fail > 0) {
   249→        toast.warning(`${ok} posted, ${fail} failed`)
   250→      } else {
   251→        toast.success(`${ok} reservation(s) posted successfully`)
   252→      }
   253→      invalidate.afterFolioChange(queryClient)
   254→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   255→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   256→      setSelectedIds(new Set())
   257→    },
   258→    onError: (error) => toast.error(error.message || 'Batch post failed'),
   259→  })
   260→
   261→  // ── Bulk post all pending ───────────────────────────────────────────
   262→  const bulkPostMutation = useMutation({
   263→    mutationFn: async () => {
   264→      return apiFetch('/api/room-rate-posting/bulk-post', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify({ postAll: true }),
   268→      })
   269→    },
   270→    onSuccess: (data) => {
   271→      const result = data as { message?: string; summary?: { totalPostedNights: number; successCount: number } }
   272→      toast.success(result?.message || 'Bulk post completed')
   273→      invalidate.afterFolioChange(queryClient)
   274→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   275→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   276→      setBulkPostOpen(false)
   277→    },
   278→    onError: (error) => {
   279→      toast.error(error.message || 'Bulk post failed')
   280→      setBulkPostOpen(false)
   281→    },
   282→  })
   283→
   284→  // ── Derived data ────────────────────────────────────────────────────
   285→  const stats = listData?.stats
   286→  const pendingSummary = pendingData?.summary
   287→  const postings = listData?.postings || []
   288→  const pendingReservations = pendingData?.pendingReservations || []
   289→  const pagination = listData?.pagination
   290→
   291→  // Filter + sort pending
   292→  const filteredPendingReservations = useMemo(() => {
   293→    let list = [...pendingReservations]
   294→    if (roomFilter) {
   295→      list = list.filter((r) => r.room?.number?.toLowerCase().includes(roomFilter.toLowerCase()))
   296→    }
   297→    if (search) {
   298→      const s = search.toLowerCase()
   299→      list = list.filter((r) =>
   300→        r.guest?.firstName?.toLowerCase().includes(s) ||
   301→        r.guest?.lastName?.toLowerCase().includes(s) ||
   302→        r.confirmationNo?.toLowerCase().includes(s)
   303→      )
   304→    }
   305→    list.sort((a, b) => {
   306→      switch (sortBy) {
   307→        case 'room-desc': return (b.room?.number || '').localeCompare(a.room?.number || '', undefined, { numeric: true })
   308→        case 'guest-asc': {
   309→          const ga = a.guest ? `${a.guest.firstName} ${a.guest.lastName}` : ''
   310→          const gb = b.guest ? `${b.guest.firstName} ${b.guest.lastName}` : ''
   311→          return ga.localeCompare(gb)
   312→        }
   313→        case 'amount-desc': return b.pendingAmount - a.pendingAmount
   314→        case 'pending-desc': return b.pendingNights.length - a.pendingNights.length
   315→        default: return (a.room?.number || '').localeCompare(b.room?.number || '', undefined, { numeric: true })
   316→      }
   317→    })
   318→    return list
   319→  }, [pendingReservations, roomFilter, search, sortBy])
   320→
   321→  // Calculated stats for pending tab
   322→  const pendingStats = useMemo(() => {
   323→    const noRoom = filteredPendingReservations.filter((r) => !r.room).length
   324→    const avgRate = filteredPendingReservations.length > 0
   325→      ? Math.round(filteredPendingReservations.reduce((s, r) => s + r.roomRate, 0) / filteredPendingReservations.length)
   326→      : 0
   327→    const totalFolioBalance = filteredPendingReservations.reduce((s, r) => s + (r.folio?.balance || 0), 0)
   328→    return { noRoom, avgRate, totalFolioBalance }
   329→  }, [filteredPendingReservations])
   330→
   331→  const isWorking = voidMutation.isPending || postMutation.isPending || bulkPostMutation.isPending || postSelectedMutation.isPending
   332→
   333→  // ── Handlers ─────────────────────────────────────────────────────────
   334→  const handleSearch = useCallback(() => {
   335→    setSearch(searchInput)
   336→    setPage(1)
   337→  }, [searchInput])
   338→
   339→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
   340→    if (e.key === 'Enter') handleSearch()
   341→  }, [handleSearch])
   342→
   343→  const clearFilters = useCallback(() => {
   344→    setSearchInput('')
   345→    setSearch('')
   346→    setRoomFilter('')
   347→    setDateFrom('')
   348→    setDateTo('')
   349→    setPage(1)
   350→  }, [])
   351→
   352→  const handleVoid = useCallback(() => {
   353→    if (!voidTarget || !voidReason.trim()) return
   354→    voidMutation.mutate({ id: voidTarget.id, reason: voidReason.trim() })
   355→  }, [voidTarget, voidReason, voidMutation])
   356→
   357→  const handlePostReservation = useCallback(() => {
   358→    if (!postTarget) return
   359→    const customRate = rateOverride ? parseFloat(rateOverride) : undefined
   360→    postMutation.mutate({ reservationId: postTarget.reservationId, dates: postTarget.pendingNights, customRate })
   361→  }, [postTarget, rateOverride, postMutation])
   362→
   363→  const handleBulkPost = useCallback(() => {
   364→    bulkPostMutation.mutate()
   365→  }, [bulkPostMutation])
   366→
   367→  const handlePostSelected = useCallback(() => {
   368→    const items = filteredPendingReservations
   369→      .filter((r) => selectedIds.has(r.reservationId))
   370→      .map((r) => ({ reservationId: r.reservationId, dates: r.pendingNights }))
   371→    if (items.length === 0) return
   372→    postSelectedMutation.mutate(items)
   373→  }, [filteredPendingReservations, selectedIds, postSelectedMutation])
   374→
   375→  const toggleExpand = useCallback((id: string) => {
   376→    setExpandedRow((prev) => (prev === id ? null : id))
   377→  }, [])
   378→
   379→  const toggleSelectAll = useCallback(() => {
   380→    if (selectedIds.size === filteredPendingReservations.length) {
   381→      setSelectedIds(new Set())
   382→    } else {
   383→      setSelectedIds(new Set(filteredPendingReservations.map((r) => r.reservationId)))
   384→    }
   385→  }, [selectedIds.size, filteredPendingReservations])
   386→
   387→  const toggleSelect = useCallback((id: string) => {
   388→    setSelectedIds((prev) => {
   389→      const next = new Set(prev)
   390→      if (next.has(id)) next.delete(id)
   391→      else next.add(id)
   392→      return next
   393→    })
   394→  }, [])
   395→
   396→  // ── Export CSV ──────────────────────────────────────────────────────
   397→  const handleExport = useCallback(() => {
   398→    const rows = activeTab === 'pending'
   399→      ? filteredPendingReservations.map((r) => ({
   400→          Room: r.room?.number || '',
   401→          Guest: r.guest ? `${r.guest.firstName} ${r.guest.lastName}` : '',
   402→          'Check-In': formatDate(r.checkIn),
   403→          'Check-Out': formatDate(r.checkOut),
   404→          'Rate/Night': r.roomRate,
   405→          'Pending Nights': r.pendingNights.length,
   406→          'Posted Nights': r.postedNights.length,
   407→          'Future Nights': r.futureNights.length,
   408→          'Pending Amount': r.pendingAmount,
   409→        }))
   410→      : postings.map((p) => ({
   411→          Date: formatDate(p.postingDate),
   412→          Room: p.reservation?.room?.number || '',
   413→          Guest: p.reservation?.guest ? `${p.reservation.guest.firstName} ${p.reservation.guest.lastName}` : '',
   414→          'Confirmation #': p.reservation?.confirmationNo || '',
   415→          'Room Rate': p.roomRate,
   416→          Tax: p.taxAmount,
   417→          'Service Charge': p.serviceCharge,
   418→          Total: p.totalAmount,
   419→          Status: p.status,
   420→          'Posted By': p.postedBy || 'System',
   421→        }))
   422→
   423→    if (rows.length === 0) {
   424→      toast.info('No data to export')
   425→      return
   426→    }
   427→
   428→    const headers = Object.keys(rows[0])
   429→    const csv = [
   430→      headers.join(','),
   431→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h]}"`).join(',')),
   432→    ].join('\n')
   433→
   434→    const blob = new Blob([csv], { type: 'text/csv' })
   435→    const url = URL.createObjectURL(blob)
   436→    const a = document.createElement('a')
   437→    a.href = url
   438→    a.download = `rate-posting-${activeTab}-${new Date().toISOString().split('T')[0]}.csv`
   439→    a.click()
   440→    URL.revokeObjectURL(url)
   441→    toast.success(`Exported ${rows.length} records`)
   442→  }, [activeTab, filteredPendingReservations, postings])
   443→
   444→  // ── Render ───────────────────────────────────────────────────────────
   445→  return (
   446→    <TooltipProvider delayDuration={300}>
   447→    <div className="space-y-4">
   448→      {/* ═══════════════ HEADER ═══════════════ */}
   449→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   450→        <div>
   451→          <h2 className="text-lg font-semibold flex items-center gap-2">
   452→            <DollarSign className="size-5 text-emerald-600" />
   453→            Room Rate Posting
   454→          </h2>
   455→          <p className="text-sm text-muted-foreground mt-0.5">
   456→            Manage daily room charge postings across all reservations
   457→          </p>
   458→        </div>
   459→        <div className="flex items-center gap-2">
   460→          <Button
   461→            variant="outline"
   462→            size="sm"
   463→            onClick={() => { refetchList(); refetchPending() }}
   464→            className="gap-1.5"
   465→          >
   466→            <RefreshCw className="size-3.5" />
   467→            Refresh
   468→          </Button>
   469→          <Button
   470→            variant="outline"
   471→            size="sm"
   472→            onClick={handleExport}
   473→            className="gap-1.5"
   474→          >
   475→            <Download className="size-3.5" />
   476→            Export
   477→          </Button>
   478→          <DropdownMenu open={showQuickActions} onOpenChange={setShowQuickActions}>
   479→            <DropdownMenuTrigger asChild>
   480→              <Button size="sm" className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   481→                <Zap className="size-3.5" />
   482→                Quick Actions
   483→                <ChevronDown className="size-3" />
   484→              </Button>
   485→            </DropdownMenuTrigger>
   486→            <DropdownMenuContent align="end" className="w-56">
   487→              <DropdownMenuLabel>Posting Actions</DropdownMenuLabel>
   488→              <DropdownMenuSeparator />
   489→              {activeTab === 'pending' && (
   490→                <>
   491→                  <DropdownMenuItem
   492→                    onClick={() => { setShowQuickActions(false); setBulkPostOpen(true) }}
   493→                    disabled={isWorking || pendingReservations.length === 0}
   494→                    className="gap-2"
   495→                  >
   496→                    <Zap className="size-4 text-emerald-600" />
   497→                    <div>
   498→                      <p className="font-medium">Post All Pending</p>
   499→                      <p className="text-[10px] text-muted-foreground">Post charges for all {pendingSummary?.totalReservations || 0} reservations</p>
   500→                    </div>
   501→                  </DropdownMenuItem>
   502→                  {selectedIds.size > 0 && (
   503→                    <DropdownMenuItem
   504→                      onClick={() => { setShowQuickActions(false); handlePostSelected() }}
   505→                      disabled={isWorking}
   506→                      className="gap-2"
   507→                    >
   508→                      <ListChecks className="size-4 text-blue-600" />
   509→                      <div>
   510→                        <p className="font-medium">Post Selected ({selectedIds.size})</p>
   511→                        <p className="text-[10px] text-muted-foreground">Post charges for {selectedIds.size} selected reservations</p>
   512→                      </div>
   513→                    </DropdownMenuItem>
   514→                  )}
   515→                </>
   516→              )}
   517→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); setActiveTab('posted') }}>
