   190→  const [guestLastName, setGuestLastName] = useState(checkInSession?.walkInGuest?.lastName || '')

   194→  const [guestIdType, setGuestIdType] = useState('')
   195→  const [guestIdNumber, setGuestIdNumber] = useState('')
   196→
   197→  // ─── Room & Rate ───────────────────────────────────────────────

   208→    checkInSession?.walkInDates?.checkOut || sessionReservation?.checkOut?.split('T')[0] || (() => {
   209→      const d = new Date()
   210→      d.setDate(d.getDate() + 1)
   211→      return toDateOnly(d)
   212→    })()
   213→  )
   214→  const [adults, setAdults] = useState(
   215→    checkInSession?.walkInDates?.adults || sessionReservation?.adults || 2
   216→  )
   217→  const [children, setChildren] = useState(
   218→    checkInSession?.walkInDates?.children || sessionReservation?.children || 0
   219→  )
   220→
   221→  // ─── Guest & Stay extras (step 1) ──────────────────────────────
   222→  const [specialRequests, setSpecialRequests] = useState(sessionReservation?.specialRequests || '')
   223→  const [purposeOfVisit, setPurposeOfVisit] = useState('')
   224→  const [guestInfoEditable, setGuestInfoEditable] = useState(false)
   225→  const [guestNotes, setGuestNotes] = useState('')
   226→
   227→  // ─── Documents ────────────────────────────────────────────────
   228→  const [collectDocuments, setCollectDocuments] = useState(false)
   229→  const [documents, setDocuments] = useState<GuestDocument[]>([])
   230→  const [editingDoc, setEditingDoc] = useState<Partial<GuestDocument>>({})
   231→
   232→  // ─── Payment ───────────────────────────────────────────────────
   233→  const [collectAdvance, setCollectAdvance] = useState(false)
   234→  const [advanceAmount, setAdvanceAmount] = useState<string>('')
   235→  const [advanceMethod, setAdvanceMethod] = useState<string>('')
   236→  const [advanceReference, setAdvanceReference] = useState('')
   237→  const [confirmChecked, setConfirmChecked] = useState(false)
   238→
   239→  // ─── Success ──────────────────────────────────────────────────
   240→  const [checkInResult, setCheckInResult] = useState<CheckInResult | null>(null)
   241→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   242→  const [showRatePosting, setShowRatePosting] = useState(false)
   243→
   244→  // ─── Data queries ──────────────────────────────────────────────
   245→
   246→  const {
   247→    data: reservationData,
   248→    isLoading: reservationLoading,
   249→  } = useQuery({
   250→    queryKey: ['reservation', confirmedReservationId],
   251→    queryFn: async () => {
   252→      if (!confirmedReservationId) return null
   253→      const data = await apiFetch(`/api/reservations/${confirmedReservationId}`) as { reservation: ReservationData }

   257→  })
   258→
   259→  // Guest stay history
   260→  const { data: guestStayHistory } = useQuery({
   261→    queryKey: ['guest-stays', reservationData?.guest?.id],
   262→    queryFn: async () => {
   263→      if (!reservationData?.guest?.id) return null
   264→      try {
   265→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string; guestName?: string; status?: string }>

   270→    enabled: !!reservationData?.guest?.id,
   271→  })

   273→  // Rooms list
   274→  const { data: roomsData } = useQuery({
   275→    queryKey: ['rooms-available'],
   276→    queryFn: async () => {
   277→      return apiFetch('/api/rooms?status=vacant_clean,inspected') as Promise<{
   278→        rooms: RoomData[]
   279→        roomTypes: Array<{
   280→          id: string; name: string; code: string; baseOccupancy: number
   281→          maxOccupancy: number; bedConfig: string; areaSqFt: number | null
   282→          view: string | null; amenities: string | null
   283→          ratePlans: RatePlanData[]
   284→        }>
   285→      }>
   286→    },
   287→  })
   288→
   289→  // ─── Check-in mutation ─────────────────────────────────────────
   290→  const checkInMutation = useMutation({
   291→    mutationFn: async (payload: Record<string, unknown>) => {
   292→      return apiFetch('/api/check-in', {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify(payload),
   296→      }) as Promise<CheckInResult>
   297→    },
   298→    onSuccess: (data) => {
   299→      setCheckInResult(data)
   300→      setCurrentStep(4) // success screen marker
   301→      toast.success('Guest checked in successfully!')
   302→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   303→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   304→      queryClient.invalidateQueries({ queryKey: ['rooms-available'] })
   305→      queryClient.invalidateQueries({ queryKey: ['front-desk-dashboard'] })
   306→    },
   307→    onError: (err: Error) => {
   308→      toast.error(err.message || 'Check-in failed')
   309→    },
   310→  })
   311→
   312→  // ─── Derived values ─────────────────────────────────────────────
   313→
   314→  const nights = useMemo(() => nightsBetween(checkInDate, checkOutDate), [checkInDate, checkOutDate])
   315→  const taxRate = settings?.taxRate ?? 13
   316→  const serviceChargeRate = settings?.serviceCharge ?? 10
   317→  const subtotal = roomRate * nights
   318→  const taxAmount = subtotal * (taxRate / 100)
   319→  const serviceChargeAmount = subtotal * (serviceChargeRate / 100)
   320→  const grandTotal = subtotal + taxAmount + serviceChargeAmount
   321→  const advanceNum = parseFloat(advanceAmount) || 0
   322→  const balanceDue = grandTotal - advanceNum
   323→
   324→  const selectedRoom = useMemo(() => {
   325→    if (!roomsData || !selectedRoomId) return null
   326→    return roomsData.rooms.find(r => r.id === selectedRoomId) || null
   327→  }, [roomsData, selectedRoomId])
   328→
   329→  const selectedRoomType = useMemo(() => {
   330→    if (!roomsData || !selectedRoom) return null
   331→    return roomsData.roomTypes.find(rt => rt.id === selectedRoom.type.id) || null
   332→  }, [roomsData, selectedRoom])
   333→
   334→  const ratePlans = useMemo(() => {
   335→    if (!selectedRoomType) return []
   336→    return selectedRoomType.ratePlans || []
   337→  }, [selectedRoomType])
   338→
   339→  const selectedRatePlan = useMemo(() => {
   340→    return ratePlans.find(rp => rp.id === selectedRatePlanId) || null
   341→  }, [ratePlans, selectedRatePlanId])
   342→
   343→  // All available rooms of the selected room's type
   344→  const sameTypeRooms = useMemo(() => {
   345→    if (!roomsData || !selectedRoom) return []
   346→    return roomsData.rooms.filter(
   347→      r => r.type.id === selectedRoom.type.id && (r.status === 'vacant_clean' || r.status === 'inspected')
   348→    )
   349→  }, [roomsData, selectedRoom])
   350→
   351→  // Filtered rooms
   352→  const filteredRooms = useMemo(() => {
   353→    if (!roomsData) return []
   354→    let available = roomsData.rooms.filter(
   355→      r => r.status === 'vacant_clean' || r.status === 'inspected'
   356→    )
   357→    if (roomTypeFilter) {
   358→      available = available.filter(r => r.type.id === roomTypeFilter)
   359→    }
   360→    if (roomFloorFilter) {
   361→      available = available.filter(r => String(r.floor) === roomFloorFilter)
   362→    }
   363→    if (!roomSearchFilter) return available
   364→    const f = roomSearchFilter.toLowerCase()
   365→    return available.filter(r =>
   366→      r.number.toLowerCase().includes(f) ||
   367→      r.type.name.toLowerCase().includes(f) ||
   368→      r.type.code.toLowerCase().includes(f) ||
   369→      (r.wing && r.wing.toLowerCase().includes(f))
   370→    )
   371→  }, [roomsData, roomSearchFilter, roomTypeFilter, roomFloorFilter])
   372→
   373→  // Room counts by type
   374→  const roomCountsByType = useMemo(() => {
   375→    if (!roomsData) return {} as Record<string, number>
   376→    const counts: Record<string, number> = {}
   377→    for (const r of roomsData.rooms) {
   378→      if (r.status === 'vacant_clean' || r.status === 'inspected') {
   379→        counts[r.type.id] = (counts[r.type.id] || 0) + 1
   380→      }
   381→    }
   382→    return counts
   383→  }, [roomsData])
   384→
   385→  // Unique floors
   386→  const uniqueFloors = useMemo(() => {
   387→    if (!roomsData) return []
   388→    const floors = new Set(roomsData.rooms.map(r => r.floor))
   389→    return Array.from(floors).sort((a, b) => a - b)
   390→  }, [roomsData])
   391→
   392→  // Occupancy check
   393→  const totalOccupancy = adults + children
   394→  const maxOccupancy = selectedRoom?.type.maxOccupancy || 0
   395→  const occupancyExceeded = maxOccupancy > 0 && totalOccupancy > maxOccupancy
   396→

   398→  const nightsDisplay = useMemo(() => {

   465→      placeOfIssue: editingDoc.placeOfIssue || '',

   479→    }

   481→      toast.error('Please confirm all details are correct')
   482→      return
   483→    }

   485→    const payload: Record<string, unknown> = {
   486→      source: confirmedReservationId ? 'reservation' : 'direct',
   487→      roomId: selectedRoomId,
   488→      roomTypeId: selectedRoom?.type.id,
   489→      ratePlanId: selectedRatePlanId || undefined,
   490→      adults,
   491→      children,
   492→      roomRate,
   493→      advanceAmount: collectAdvance && advanceNum > 0 ? advanceNum : undefined,
   494→      advanceMethod: collectAdvance && advanceNum > 0 ? (advanceMethod || 'cash') : undefined,
   495→      advanceReference: collectAdvance && advanceNum > 0 ? advanceReference : undefined,
   496→      documents: collectDocuments && documents.length > 0
   497→        ? documents.map(d => ({
   498→            docType: d.docType,
   499→            docNumber: d.docNumber || undefined,
   500→            issueCountry: d.issueCountry || undefined,
   501→            issueDate: d.issueDate || undefined,
   502→            expiryDate: d.expiryDate || undefined,
   503→            placeOfIssue: d.placeOfIssue || undefined,
   504→          }))
   505→        : undefined,
   506→      checkedInBy: `${user?.firstName || ''} ${user?.lastName || ''}`.trim() || 'System',
   507→    }
   508→
   509→    if (confirmedReservationId) {
   510→      payload.reservationId = confirmedReservationId
   511→    } else {
   512→      payload.guest = {
   513→        firstName: guestFirstName,
   514→        lastName: guestLastName,
   515→        email: guestEmail || undefined,
   516→        phone: guestPhone || undefined,
   517→        nationality: guestNationality || undefined,
   518→        idType: guestIdType || undefined,
   519→        idNumber: guestIdNumber || undefined,
   520→      }
   521→      payload.checkIn = checkInDate
   522→      payload.checkOut = checkOutDate
   523→    }
   524→
   525→    checkInMutation.mutate(payload)
   526→  }, [
   527→    confirmedReservationId, selectedRoomId, selectedRoom, selectedRatePlanId,
   528→    adults, children, roomRate, collectAdvance, advanceNum, advanceMethod, advanceReference,
   529→    collectDocuments, documents, user, guestFirstName, guestLastName,
   530→    guestEmail, guestPhone, guestNationality, guestIdType, guestIdNumber,
   531→    checkInDate, checkOutDate, confirmChecked, checkInMutation,
   532→  ])
   533→
   534→  const handleStepClick = useCallback((step: number) => {
   535→    if (step < currentStep) setCurrentStep(step)
   536→  }, [currentStep])
   537→
