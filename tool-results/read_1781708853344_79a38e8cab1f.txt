   206→    checkInSession?.walkInDates?.checkIn || sessionReservation?.checkIn?.split('T')[0] || today

   209→    checkInSession?.walkInDates?.checkOut || sessionReservation?.checkOut?.split('T')[0] || (() => {
   210→      const d = new Date()
   211→      d.setDate(d.getDate() + 1)
   212→      return toDateOnly(d)
   213→    })()
   214→  )
   215→  const [adults, setAdults] = useState(
   216→    checkInSession?.walkInDates?.adults || sessionReservation?.adults || 2
   217→  )
   218→  const [children, setChildren] = useState(
   219→    checkInSession?.walkInDates?.children || sessionReservation?.children || 0
   220→  )
   221→
   222→  // ─── Guest & Stay extras (step 1) ──────────────────────────────
   223→  const [specialRequests, setSpecialRequests] = useState(sessionReservation?.specialRequests || '')
   224→  const [purposeOfVisit, setPurposeOfVisit] = useState('')
   225→  const [guestInfoEditable, setGuestInfoEditable] = useState(false)
   226→  const [guestNotes, setGuestNotes] = useState('')
   227→
   228→  // ─── Documents ────────────────────────────────────────────────
   229→  const [collectDocuments, setCollectDocuments] = useState(false)
   230→  const [documents, setDocuments] = useState<GuestDocument[]>([])
   231→  const [editingDoc, setEditingDoc] = useState<Partial<GuestDocument>>({})
   232→
   233→  // ─── Payment ───────────────────────────────────────────────────
   234→  const [collectAdvance, setCollectAdvance] = useState(false)
   235→  const [advanceAmount, setAdvanceAmount] = useState<string>('')
   236→  const [advanceMethod, setAdvanceMethod] = useState<string>('')
   237→  const [advanceReference, setAdvanceReference] = useState('')

   244→

   247→  const {
   248→    data: reservationData,
   249→    isLoading: reservationLoading,
   250→  } = useQuery({
   251→    queryKey: ['reservation', confirmedReservationId],
   252→    queryFn: async () => {

   254→      const data = await apiFetch(`/api/reservations/${confirmedReservationId}`) as { reservation: ReservationData }

   260→  // Guest stay history

   262→    queryKey: ['guest-stays', reservationData?.guest?.id],
   263→    queryFn: async () => {
   264→      if (!reservationData?.guest?.id) return null
   265→      try {
   266→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string; guestName?: string; status?: string }>

   275→  const { data: roomsData } = useQuery({
   276→    queryKey: ['rooms-available'],
   277→    queryFn: async () => {
   278→      return apiFetch('/api/rooms?status=vacant_clean,inspected') as Promise<{
   279→        rooms: RoomData[]
   280→        roomTypes: Array<{
   281→          id: string; name: string; code: string; baseOccupancy: number
   282→          maxOccupancy: number; bedConfig: string; areaSqFt: number | null
   283→          view: string | null; amenities: string | null
   284→          ratePlans: RatePlanData[]
   285→        }>
   286→      }>
   287→    },
   288→  })
   289→
   290→  // ─── Check-in mutation ─────────────────────────────────────────
   291→  const checkInMutation = useMutation({
   292→    mutationFn: async (payload: Record<string, unknown>) => {
   293→      return apiFetch('/api/check-in', {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify(payload),
   297→      }) as Promise<CheckInResult>

   301→      setCurrentStep(4) // success screen marker

   304→      queryClient.invalidateQueries({ queryKey: ['rooms-available'] })
   305→    },
   306→    onError: (err: Error) => {
   307→      toast.error(err.message || 'Check-in failed')
   308→    },
   309→  })
   310→
   311→  // ─── Derived values ─────────────────────────────────────────────
   312→
   313→  const nights = useMemo(() => nightsBetween(checkInDate, checkOutDate), [checkInDate, checkOutDate])
   314→  const taxRate = settings?.taxRate ?? 13
   315→  const serviceChargeRate = settings?.serviceCharge ?? 10
   316→  const subtotal = roomRate * nights
   317→  const taxAmount = subtotal * (taxRate / 100)
   318→  const serviceChargeAmount = subtotal * (serviceChargeRate / 100)
   319→  const grandTotal = subtotal + taxAmount + serviceChargeAmount
   320→  const advanceNum = parseFloat(advanceAmount) || 0
   321→  const balanceDue = grandTotal - advanceNum
   322→
   323→  const selectedRoom = useMemo(() => {
   324→    if (!roomsData || !selectedRoomId) return null
   325→    return roomsData.rooms.find(r => r.id === selectedRoomId) || null
   326→  }, [roomsData, selectedRoomId])
   327→
   328→  const selectedRoomType = useMemo(() => {
   329→    if (!roomsData || !selectedRoom) return null
   330→    return roomsData.roomTypes.find(rt => rt.id === selectedRoom.type.id) || null
   331→  }, [roomsData, selectedRoom])
   332→
   333→  const ratePlans = useMemo(() => {
   334→    if (!selectedRoomType) return []
   335→    return selectedRoomType.ratePlans || []
   336→  }, [selectedRoomType])
   337→
   338→  const selectedRatePlan = useMemo(() => {
   339→    return ratePlans.find(rp => rp.id === selectedRatePlanId) || null
   340→  }, [ratePlans, selectedRatePlanId])
   341→
   342→  // All available rooms of the selected room's type
   343→  const sameTypeRooms = useMemo(() => {
   344→    if (!roomsData || !selectedRoom) return []
   345→    return roomsData.rooms.filter(
   346→      r => r.type.id === selectedRoom.type.id && (r.status === 'vacant_clean' || r.status === 'inspected')
   347→    )
   348→  }, [roomsData, selectedRoom])
   349→
   350→  // Filtered rooms
   351→  const filteredRooms = useMemo(() => {
   352→    if (!roomsData) return []
   353→    let available = roomsData.rooms.filter(
   354→      r => r.status === 'vacant_clean' || r.status === 'inspected'
   355→    )
   356→    if (roomTypeFilter) {
   357→      available = available.filter(r => r.type.id === roomTypeFilter)
   358→    }
   359→    if (roomFloorFilter) {
   360→      available = available.filter(r => String(r.floor) === roomFloorFilter)
   361→    }
   362→    if (!roomSearchFilter) return available
   363→    const f = roomSearchFilter.toLowerCase()
   364→    return available.filter(r =>
   365→      r.number.toLowerCase().includes(f) ||
   366→      r.type.name.toLowerCase().includes(f) ||
   367→      r.type.code.toLowerCase().includes(f) ||
   368→      (r.wing && r.wing.toLowerCase().includes(f))
   369→    )
   370→  }, [roomsData, roomSearchFilter, roomTypeFilter, roomFloorFilter])
   371→
   372→  // Room counts by type
   373→  const roomCountsByType = useMemo(() => {
   374→    if (!roomsData) return {} as Record<string, number>
   375→    const counts: Record<string, number> = {}
   376→    for (const r of roomsData.rooms) {
   377→      if (r.status === 'vacant_clean' || r.status === 'inspected') {
   378→        counts[r.type.id] = (counts[r.type.id] || 0) + 1
   379→      }
   380→    }
   381→    return counts
   382→  }, [roomsData])
   383→
   384→  // Unique floors
   385→  const uniqueFloors = useMemo(() => {
   386→    if (!roomsData) return []
   387→    const floors = new Set(roomsData.rooms.map(r => r.floor))
   388→    return Array.from(floors).sort((a, b) => a - b)

   390→
   391→  // Occupancy check
   392→  const totalOccupancy = adults + children
   393→  const maxOccupancy = selectedRoom?.type.maxOccupancy || 0
   394→  const occupancyExceeded = maxOccupancy > 0 && totalOccupancy > maxOccupancy

   414→  }, [reservationData, isDirectWalkIn, guestFirstName, guestLastName])

   478→    }

   484→    const payload: Record<string, unknown> = {
   485→      source: confirmedReservationId ? 'reservation' : 'direct',
   486→      roomId: selectedRoomId,
   487→      roomTypeId: selectedRoom?.type.id,
   488→      ratePlanId: selectedRatePlanId || undefined,
   489→      adults,

   491→      roomRate,
   492→      advanceAmount: collectAdvance && advanceNum > 0 ? advanceNum : undefined,
   493→      advanceMethod: collectAdvance && advanceNum > 0 ? (advanceMethod || 'cash') : undefined,
   494→      advanceReference: collectAdvance && advanceNum > 0 ? advanceReference : undefined,
   495→      documents: collectDocuments && documents.length > 0
   496→        ? documents.map(d => ({
   497→            docType: d.docType,
   498→            docNumber: d.docNumber || undefined,
   499→            issueCountry: d.issueCountry || undefined,
   500→            issueDate: d.issueDate || undefined,
   501→            expiryDate: d.expiryDate || undefined,
   502→            placeOfIssue: d.placeOfIssue || undefined,
   503→          }))
   504→        : undefined,
   505→      checkedInBy: `${user?.firstName || ''} ${user?.lastName || ''}`.trim() || 'System',
   506→    }
   507→
   508→    if (confirmedReservationId) {
   509→      payload.reservationId = confirmedReservationId
   510→    } else {
   511→      payload.guest = {
   512→        firstName: guestFirstName,
   513→        lastName: guestLastName,
   514→        email: guestEmail || undefined,
   515→        phone: guestPhone || undefined,
   516→        nationality: guestNationality || undefined,
   517→        idType: guestIdType || undefined,
   518→        idNumber: guestIdNumber || undefined,
   519→      }
   520→      payload.checkIn = checkInDate
   521→      payload.checkOut = checkOutDate
   522→    }
   523→
   524→    checkInMutation.mutate(payload)
   525→  }, [
   526→    confirmedReservationId, selectedRoomId, selectedRoom, selectedRatePlanId,
   527→    adults, children, roomRate, collectAdvance, advanceNum, advanceMethod, advanceReference,
   528→    collectDocuments, documents, user, guestFirstName, guestLastName,
   529→    guestEmail, guestPhone, guestNationality, guestIdType, guestIdNumber,
   530→    checkInDate, checkOutDate, confirmChecked, checkInMutation,
   531→  ])
   532→
   533→  const handleStepClick = useCallback((step: number) => {
   534→    if (step < currentStep) setCurrentStep(step)
   535→  }, [currentStep])
   536→
   537→  const handleNext = useCallback(() => {
   538→    if (currentStep === 1) {
