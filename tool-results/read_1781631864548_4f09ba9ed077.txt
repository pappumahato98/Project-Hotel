   100→   100→  const { preferences } = usePreferencesStore()
   101→   101→  const { navigateTo } = useNavigationStore()

   104→   104→  // Dialog states

   111→   111→  // Selected departure

   114→   114→  // Late checkout

   117→   117→  // Payment

   120→   120→  const [paymentReference, setPaymentReference] = useState('')

   122→   122→  // Receipt data

   127→   127→    queryKey: ['departures', today],
   128→   128→    queryFn: async () => {
   129→   129→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })

   159→   159→      })
   160→   160→    },
   161→   161→    onSuccess: () => {
   162→   162→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   163→   163→      queryClient.invalidateQueries({ queryKey: ['reservations'] })

   165→   165→    },
   166→   166→  })

   194→   194→        method: 'PATCH',
   195→   195→        headers: { 'Content-Type': 'application/json' },
   196→   196→        body: JSON.stringify({ checkOut: newCheckOut }),
   197→   197→      })

   200→   200→        await apiFetch(`/api/folio`, {
   201→   201→          method: 'POST',
   202→   202→          headers: { 'Content-Type': 'application/json' },
   203→   203→          body: JSON.stringify({
   204→   204→            reservationId,
   205→   205→            type: 'charge',
   206→   206→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   207→   207→            amount: surcharge,
   208→   208→          }),
   209→   209→        })
   210→   210→      }
   211→   211→      return result
   212→   212→    },
   213→   213→    onSuccess: () => {
   214→   214→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   215→   215→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   216→   216→      queryClient.invalidateQueries({ queryKey: ['folios'] })

   219→   219→    },
   220→   220→    onError: () => {
   221→   221→      toast.error('Failed to process late checkout request')
   222→   222→    },
   223→   223→  })

   232→   232→          headers: { 'Content-Type': 'application/json' },

   239→   239→    },

   248→   248→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   249→   249→    },
   250→   250→  })
   251→   251→
   252→   252→  // ─── Handlers ─────────────────────────────────────────────────────
   253→   253→

   412→   412→
   413→   413→      {/* Departures List */}
   414→   414→      <Card className="py-0">

   416→   416→          <div className="max-h-[600px] overflow-y-auto">
   417→   417→            <Table>
   418→   418→              <TableHeader>
   419→   419→                <TableRow>

   421→   421→                  <TableHead>Guest</TableHead>
   422→   422→                  <TableHead className="w-[100px]">Check-out</TableHead>

   424→   424→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   425→   425→                  <TableHead className="w-[280px] text-right">Actions</TableHead>

   439→   439→                ) : departures.length === 0 ? (

   443→   443→                      No departures scheduled for today
   444→   444→                    </TableCell>
   445→   445→                  </TableRow>
   446→   446→                ) : (
   447→   447→                  departures.map((dep) => {
   448→   448→                    const balance = getBalance(dep)
   449→   449→                    const isPending = dep.status === 'checked_in'
   450→   450→                    const isZeroBalance = balance === 0
   451→   451→                    return (
   452→   452→                      <TableRow key={dep.id}>
   453→   453→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   454→   454→                        <TableCell>
