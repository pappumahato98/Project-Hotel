   157→// ─── Component ──────────────────────────────────────────────────────────
   158→
   159→export function ArrivalsView() {
   160→  const queryClient = useQueryClient()
   161→  const { settings } = useSettingsStore()
   162→  const { navigateTo } = useNavigationStore()
   163→  const today = getTodayString()
   164→
   165→  // Dialog states
   166→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   167→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   168→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   169→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   170→
   171→  // Selection states
   172→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   173→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   174→
   175→  // Check-in form states
   176→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   177→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   178→  const [roomPreference, setRoomPreference] = useState('')
   179→  const [pillowType, setPillowType] = useState('')
   180→  const [wakeupCall, setWakeupCall] = useState('')
   181→
   182→  // Key card states
   183→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   184→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   185→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   186→
   187→  // Mobile detail view state
   188→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   189→
   190→  // Walk-in form state
   191→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   192→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   193→
   194→  // ─── Queries ────────────────────────────────────────────────────────
   195→
   196→  const { data, isLoading } = useQuery({
   197→    queryKey: ['arrivals', today],
   198→    queryFn: async () => {
   199→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   200→      return apiFetch(`/api/reservations?${params.toString()}`)
   201→    },
   202→    refetchInterval: 30000,
   203→  })
   204→
   205→  const { data: roomsData } = useQuery({
   206→    queryKey: ['rooms', 'vacant'],
   207→    queryFn: () => apiFetch('/api/rooms'),
   208→  })
   209→
   210→  const arrivals: Arrival[] = data?.reservations || []
   211→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   212→  const availableRooms = allRooms.filter(
   213→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   214→  )
   215→
   216→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   217→
   218→  const sortedArrivals = useMemo(() => {
   219→    return [...arrivals].sort((a, b) => {
   220→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   221→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   222→      if (aVip && !bVip) return -1
   223→      if (!aVip && bVip) return 1
   224→      return 0
   225→    })
   226→  }, [arrivals])
   227→
   228→  // ─── Stats ──────────────────────────────────────────────────────────
   229→
   230→  const totalArrivals = arrivals.length
   231→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   232→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   233→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   234→
   235→  // ─── Mutations ──────────────────────────────────────────────────────
   236→
   237→  const assignRoomMutation = useMutation({
   238→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   239→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   240→      if (specialRequests !== undefined) {
   241→        payload.specialRequests = specialRequests
   242→      }
   243→      return apiFetch(`/api/reservations/${reservationId}`, {
   244→        method: 'PATCH',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify(payload),
   247→      })
   248→    },
   249→    onSuccess: (_data, variables) => {
   250→      invalidate.afterReservationChange(queryClient)
   251→      setRoomPickerOpen(false)
   252→      setCheckInDialogOpen(false)
   253→
   254→      // Find the room number for key card dialog
   255→      const room = availableRooms.find((r) => r.id === variables.roomId)
   256→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   257→      openKeyCardDialog(roomNumber)
   258→    },
   259→  })
   260→
   261→  const checkInMutation = useMutation({
   262→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   263→      const payload: Record<string, unknown> = { status: 'checked_in' }
   264→      if (specialRequests !== undefined) {
   265→        payload.specialRequests = specialRequests
   266→      }
   267→      return apiFetch(`/api/reservations/${reservationId}`, {
   268→        method: 'PATCH',
   269→        headers: { 'Content-Type': 'application/json' },
   270→        body: JSON.stringify(payload),
   271→      })
   272→    },
   273→    onSuccess: (_data, variables) => {
   274→      invalidate.afterCheckIn(queryClient)
   275→      setCheckInDialogOpen(false)
   276→
   277→      const roomNumber = selectedArrival?.room?.number || ''
   278→      openKeyCardDialog(roomNumber)
   279→    },
   280→  })
   281→
   282→  const walkInMutation = useMutation({
   283→    mutationFn: async (form: WalkInForm) => {
   284→      // 1. Create guest
   285→      const guestData = await apiFetch('/api/guests', {
   286→        method: 'POST',
   287→        headers: { 'Content-Type': 'application/json' },
   288→        body: JSON.stringify({
   289→          firstName: form.firstName,
   290→          lastName: form.lastName,
   291→          phone: form.phone || null,
   292→          email: form.email || null,
   293→          nationality: form.nationality || null,
   294→          idType: form.idType || null,
   295→          idNumber: form.idNumber || null,
   296→        }),
   297→      })
   298→      const guestId = guestData.guest.id
   299→
   300→      // 2. Find a vacant room
   301→      const vacantRoom = availableRooms[0]
   302→      if (!vacantRoom) throw new Error('No available rooms')
   303→
   304→      // 3. Create reservation as checked_in (walk-in)
   305→      const checkOutDate = form.checkOutDate || (() => {
   306→        const d = new Date()
   307→        d.setDate(d.getDate() + 1)
   308→        return d
   309→      })()
   310→
   311→      const resData = await apiFetch('/api/reservations', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({
   315→          guestId,
   316→          roomId: vacantRoom.id,
   317→          adults: form.adults,
   318→          children: form.children,
   319→          checkIn: new Date().toISOString(),
   320→          checkOut: checkOutDate.toISOString(),
   321→          roomRate: form.roomRate,
   322→          source: 'walk_in',
   323→          reservationType: 'walk_in',
   324→          guaranteed: false,
   325→        }),
   326→      })
   327→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   328→    },
   329→    onSuccess: (result) => {
   330→      invalidate.afterCheckIn(queryClient)
   331→      setWalkInDialogOpen(false)
   332→      setWalkInForm(initialWalkInForm)
   333→
   334→      // Open key card dialog
   335→      openKeyCardDialog(result.roomNumber)
   336→    },
   337→    onError: (error) => {
   338→      toast.error('Walk-in check-in failed', {
   339→        description: error instanceof Error ? error.message : 'Unknown error',
   340→      })
   341→    },
   342→  })
   343→
   344→  // ─── Handlers ──────────────────────────────────────────────────────
   345→
   346→  const handleAssignRoom = (arrival: Arrival) => {
   347→    setSelectedArrival(arrival)
   348→    setRoomPickerOpen(true)
   349→  }
   350→
   351→  const handleQuickCheckIn = (arrival: Arrival) => {
   352→    if (!arrival.room) {
   353→      setSelectedArrival(arrival)
   354→      setRoomPickerOpen(true)
   355→      return
   356→    }
   357→    setSelectedArrival(arrival)
   358→    resetCheckInForm()
   359→    setCheckInDialogOpen(true)
   360→  }
   361→
   362→  const handleOpenWalkIn = () => {
   363→    setWalkInForm(initialWalkInForm)
   364→    setWalkInDialogOpen(true)
   365→  }
   366→
   367→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   368→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   369→    navigateTo('front-desk', 'check-in')
   370→  }
   371→
   372→  const handleViewLedger = (arrival: Arrival) => {
   373→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   374→      guestId: arrival.guest?.id || '',
   375→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   376→    })
   377→    navigateTo('front-desk', 'guest-ledger')
   378→  }
   379→
   380→  const buildSpecialRequests = (arrival: Arrival): string => {
   381→    const parts: string[] = []
   382→
   383→    // Preserve existing requests
   384→    if (arrival.specialRequests) {
   385→      parts.push(arrival.specialRequests)
   386→    }
   387→
   388→    // Early check-in note
   389→    if (earlyCheckIn) {
   390→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   391→    }
   392→
   393→    // Check-out time override
   394→    if (checkOutTime !== settings.defaultCheckOut) {
   395→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   396→    }
   397→
   398→    // Room preference
   399→    if (roomPreference) {
   400→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   401→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   402→    }
   403→
   404→    // Pillow type
   405→    if (pillowType) {
   406→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   407→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   408→    }
   409→
   410→    // Wake-up call
   411→    if (wakeupCall) {
   412→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   413→    }
   414→
   415→    return parts.join(' | ')
   416→  }
   417→
   418→  const confirmCheckIn = () => {
   419→    if (!selectedArrival) return
   420→
   421→    const updatedRequests = buildSpecialRequests(selectedArrival)
   422→
   423→    if (selectedRoomId) {
   424→      assignRoomMutation.mutate({
   425→        reservationId: selectedArrival.id,
   426→        roomId: selectedRoomId,
   427→        specialRequests: updatedRequests,
   428→      })
   429→    } else if (selectedArrival.room) {
   430→      checkInMutation.mutate({
   431→        reservationId: selectedArrival.id,
   432→        specialRequests: updatedRequests,
   433→      })
   434→    }
   435→  }
   436→
   437→  const openKeyCardDialog = (roomNumber: string) => {
   438→    setKeyCardRoomNumber(roomNumber)
   439→    setKeyCardIssued(false)
   440→    setKeyCardProcessed(false)
   441→    setKeyCardDialogOpen(true)
   442→  }
   443→
   444→  const handleKeyCardConfirm = () => {
   445→    setKeyCardProcessed(true)
   446→    setKeyCardDialogOpen(false)
   447→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   448→      duration: 5000,
   449→      icon: <Mail className="size-4" />,
   450→    })
   451→    setSelectedArrival(null)
   452→    setSelectedRoomId('')
   453→  }
   454→
   455→  const resetCheckInForm = () => {
   456→    setEarlyCheckIn(false)
   457→    setCheckOutTime(settings.defaultCheckOut)
   458→    setRoomPreference('')
   459→    setPillowType('')
   460→    setWakeupCall('')
   461→  }
   462→
   463→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   464→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   465→    setWalkInForm((prev) => ({
   466→      ...prev,
   467→      roomTypeId,
   468→      roomRate: roomType?.baseRate || 0,
   469→    }))
   470→  }
   471→
   472→  const handleSelectArrival = (arrival: Arrival) => {
   473→    setSelectedArrival(arrival)
   474→    setSelectedRoomId('')
   475→    setMobileShowDetail(true)
   476→  }
   477→
   478→  const handleBackToList = () => {
   479→    setMobileShowDetail(false)
   480→  }
   481→
   482→  // ─── Render ─────────────────────────────────────────────────────────
   483→
   484→  return (
   485→    <div className="flex flex-col gap-3">
   486→      {/* Header */}
   487→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 shrink-0">
   488→        <div>
   489→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   490→          <p className="text-xs text-muted-foreground">
   491→            Guest check-ins scheduled for {formatDate(today)}
   492→          </p>
   493→        </div>
   494→        <div className="flex items-center gap-2 shrink-0">
   495→          <Badge variant="secondary" className="text-[10px]">
   496→            {totalArrivals} expected
   497→          </Badge>
   498→          {vipArrivals > 0 && (
   499→            <Badge variant="outline" className="text-[10px] border-amber-300 text-amber-700 dark:text-amber-400">
   500→              <Crown className="size-3 mr-0.5" />
   501→              {vipArrivals} VIP
   502→            </Badge>
   503→          )}
   504→          {unassignedArrivals > 0 && (
   505→            <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-600 dark:text-amber-400">
   506→              <AlertTriangle className="size-3 mr-0.5" />
   507→              {unassignedArrivals} unassigned
   508→            </Badge>
   509→          )}
   510→          <Button onClick={handleOpenWalkIn} size="sm" className="gap-1.5">
   511→            <Footprints className="size-3.5" />
   512→            <span className="hidden sm:inline">Walk-in</span>
   513→          </Button>
   514→        </div>
   515→      </div>
   516→
   517→      {/* ─── Split Screen Body ─── */}
   518→      <div className="h-[calc(100vh-14rem)] min-h-[400px] flex flex-col lg:flex-row gap-3 overflow-hidden">
   519→
   520→        {/* ─── Left Panel: Arrivals List ─── */}
   521→        <div className={cn(
   522→          'flex flex-col min-h-0 border rounded-lg bg-card overflow-hidden',
   523→          'w-full lg:w-[42%] xl:w-[38%]',
   524→          mobileShowDetail && selectedArrival && 'hidden lg:flex',
   525→        )}>
   526→          {/* List header */}
   527→          <div className="px-3 py-2 border-b shrink-0 flex items-center justify-between bg-muted/30">
   528→            <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   529→              Expected Arrivals
   530→            </span>
   531→            <span className="text-[10px] text-muted-foreground">{sortedArrivals.length} guests</span>
   532→          </div>
   533→          {/* List body */}
   534→          <div className="flex-1 overflow-y-auto">
   535→            {isLoading ? (
   536→              <div className="p-2 space-y-2">
   537→                {Array.from({ length: 4 }).map((_, i) => (
   538→                  <Skeleton key={i} className="h-16 w-full rounded-lg" />
   539→                ))}
   540→              </div>
   541→            ) : sortedArrivals.length === 0 ? (
   542→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   543→                <LogIn className="size-8 mb-2 opacity-50" />
   544→                <p className="text-sm">No arrivals scheduled for today</p>
   545→              </div>
   546→            ) : (
   547→              <div className="divide-y">
   548→                {sortedArrivals.map((arrival) => {
   549→                  const isUnassigned = !arrival.room
   550→                  const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   551→                  const isSelected = selectedArrival?.id === arrival.id
   552→                  return (
   553→                    <button
   554→                      key={arrival.id}
   555→                      type="button"
   556→                      onClick={() => handleSelectArrival(arrival)}
   557→                      className={cn(
