   159→// ─── Component ──────────────────────────────────────────────────────────
   160→
   161→export function ArrivalsView() {
   162→  const queryClient = useQueryClient()
   163→  const { settings } = useSettingsStore()
   164→  const { navigateTo } = useNavigationStore()
   165→  const today = getTodayString()
   166→
   167→  // Dialog states
   168→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   169→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   170→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   171→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   172→
   173→  // Selection states
   174→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   175→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   176→
   177→  // Check-in form states
   178→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   179→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   180→  const [roomPreference, setRoomPreference] = useState('')
   181→  const [pillowType, setPillowType] = useState('')
   182→  const [wakeupCall, setWakeupCall] = useState('')
   183→
   184→  // Key card states
   185→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   186→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   187→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   188→
   189→  // Mobile detail view state
   190→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   191→
   192→  // Cancel booking state
   193→  const [cancelDialogOpen, setCancelDialogOpen] = useState(false)
   194→  const [cancelReason, setCancelReason] = useState('')
   195→
   196→  // Walk-in form state
   197→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   198→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   199→
   200→  // ─── Queries ────────────────────────────────────────────────────────
   201→
   202→  const { data, isLoading } = useQuery({
   203→    queryKey: ['arrivals', today],
   204→    queryFn: async () => {
   205→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   206→      return apiFetch(`/api/reservations?${params.toString()}`)
   207→    },
   208→    refetchInterval: 30000,
   209→  })
   210→
   211→  const { data: roomsData } = useQuery({
   212→    queryKey: ['rooms', 'vacant'],
   213→    queryFn: () => apiFetch('/api/rooms'),
   214→  })
   215→
   216→  const arrivals: Arrival[] = data?.reservations || []
   217→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   218→  const availableRooms = allRooms.filter(
   219→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   220→  )
   221→
   222→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   223→
   224→  const sortedArrivals = useMemo(() => {
   225→    return [...arrivals].sort((a, b) => {
   226→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   227→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   228→      if (aVip && !bVip) return -1
   229→      if (!aVip && bVip) return 1
   230→      return 0
   231→    })
   232→  }, [arrivals])
   233→
   234→  // ─── Stats ──────────────────────────────────────────────────────────
   235→
   236→  const totalArrivals = arrivals.length
   237→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   238→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   239→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   240→
   241→  // ─── Mutations ──────────────────────────────────────────────────────
   242→
   243→  const assignRoomMutation = useMutation({
   244→    mutationFn: async ({ reservationId, roomId, roomTypeId, specialRequests }: { reservationId: string; roomId: string; roomTypeId: string; specialRequests?: string }) => {
   245→      return apiFetch('/api/check-in', {
   246→        method: 'POST',
   247→        headers: { 'Content-Type': 'application/json' },
   248→        body: JSON.stringify({
   249→          source: 'reservation',
   250→          reservationId,
   251→          roomId,
   252→          roomTypeId,
   253→          adults: selectedArrival?.adults || 1,
   254→          children: selectedArrival?.children || 0,
   255→          roomRate: selectedArrival?.roomRate || 0,
   256→          specialRequests,
   257→        }),
   258→      }) as Promise<{ reservation: { room?: { number: string } } }>
   259→    },
   260→    onSuccess: (_data, variables) => {
   261→      invalidate.afterCheckIn(queryClient)
   262→      setRoomPickerOpen(false)
   263→      setCheckInDialogOpen(false)
   264→
   265→      const room = availableRooms.find((r) => r.id === variables.roomId)
   266→      const roomNumber = room?.number || _data?.reservation?.room?.number || ''
   267→      openKeyCardDialog(roomNumber)
   268→    },
   269→    onError: (error) => {
   270→      toast.error('Check-in failed', {
   271→        description: error instanceof Error ? error.message : 'Unknown error',
   272→      })
   273→    },
   274→  })
   275→
   276→  const checkInMutation = useMutation({
   277→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   278→      const arrival = arrivals.find(a => a.id === reservationId)
   279→      if (!arrival?.room) throw new Error('No room assigned to this reservation')
   280→      return apiFetch('/api/check-in', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          source: 'reservation',
   285→          reservationId,
   286→          roomId: arrival.room.id,
   287→          roomTypeId: arrival.room.typeId,
   288→          adults: arrival.adults,
   289→          children: arrival.children,
   290→          roomRate: arrival.roomRate,
   291→          specialRequests,
   292→        }),
   293→      }) as Promise<{ reservation: { room?: { number: string } } }>
   294→    },
   295→    onSuccess: (_data) => {
   296→      invalidate.afterCheckIn(queryClient)
   297→      setCheckInDialogOpen(false)
   298→
   299→      const roomNumber = selectedArrival?.room?.number || _data?.reservation?.room?.number || ''
   300→      openKeyCardDialog(roomNumber)
   301→    },
   302→    onError: (error) => {
   303→      toast.error('Check-in failed', {
   304→        description: error instanceof Error ? error.message : 'Unknown error',
   305→      })
   306→    },
   307→  })
   308→
   309→  const walkInMutation = useMutation({
   310→    mutationFn: async (form: WalkInForm) => {
   311→      const vacantRoom = availableRooms[0]
   312→      if (!vacantRoom) throw new Error('No available rooms')
   313→
   314→      const checkOutDate = form.checkOutDate || (() => {
   315→        const d = new Date()
   316→        d.setDate(d.getDate() + 1)
   317→        return d
   318→      })()
   319→
   320→      return apiFetch('/api/check-in', {
   321→        method: 'POST',
   322→        headers: { 'Content-Type': 'application/json' },
   323→        body: JSON.stringify({
   324→          source: 'direct',
   325→          guest: {
   326→            firstName: form.firstName,
   327→            lastName: form.lastName,
   328→            phone: form.phone || undefined,
   329→            email: form.email || undefined,
   330→            nationality: form.nationality || undefined,
   331→            idType: form.idType || undefined,
   332→            idNumber: form.idNumber || undefined,
   333→          },
   334→          roomId: vacantRoom.id,
   335→          roomTypeId: vacantRoom.typeId,
   336→          adults: form.adults,
   337→          children: form.children,
   338→          roomRate: form.roomRate,
   339→          checkIn: new Date().toISOString(),
   340→          checkOut: checkOutDate.toISOString(),
   341→        }),
   342→      }) as Promise<{ reservation: { room?: { number: string } } }>
   343→    },
   344→    onSuccess: (result) => {
   345→      invalidate.afterCheckIn(queryClient)
   346→      setWalkInDialogOpen(false)
   347→      setWalkInForm(initialWalkInForm)
   348→
   349→      const roomNumber = result?.reservation?.room?.number || ''
   350→      openKeyCardDialog(roomNumber)
   351→    },
   352→    onError: (error) => {
   353→      toast.error('Walk-in check-in failed', {
   354→        description: error instanceof Error ? error.message : 'Unknown error',
   355→      })
   356→    },
   357→  })
   358→
   359→  const cancelBookingMutation = useMutation({
   360→    mutationFn: async ({ reservationId }: { reservationId: string }) => {
   361→      return apiFetch(`/api/reservations/${reservationId}`, {
   362→        method: 'PATCH',
   363→        headers: { 'Content-Type': 'application/json' },
   364→        body: JSON.stringify({ status: 'cancelled' }),
   365→      }) as Promise<{ reservation: Record<string, unknown> }>
   366→    },
   367→    onSuccess: () => {
   368→      invalidate.afterCheckIn(queryClient)
   369→      setCancelDialogOpen(false)
   370→      setSelectedArrival(null)
   371→      setCancelReason('')
   372→      toast.success('Booking cancelled successfully')
   373→    },
   374→    onError: (error) => {
   375→      toast.error('Failed to cancel booking', {
   376→        description: error instanceof Error ? error.message : 'Unknown error',
   377→      })
   378→    },
   379→  })
   380→
   381→  // ─── Handlers ──────────────────────────────────────────────────────
   382→
   383→  const handleAssignRoom = (arrival: Arrival) => {
   384→    setSelectedArrival(arrival)
   385→    setRoomPickerOpen(true)
   386→  }
   387→
   388→  const handleQuickCheckIn = (arrival: Arrival) => {
   389→    if (!arrival.room) {
   390→      setSelectedArrival(arrival)
   391→      setRoomPickerOpen(true)
   392→      return
   393→    }
   394→    setSelectedArrival(arrival)
   395→    resetCheckInForm()
   396→    setCheckInDialogOpen(true)
   397→  }
   398→
   399→  const handleOpenWalkIn = () => {
   400→    setWalkInForm(initialWalkInForm)
   401→    setWalkInDialogOpen(true)
   402→  }
   403→
   404→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   405→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   406→    navigateTo('front-desk', 'check-in')
   407→  }
   408→
   409→  const handleViewLedger = (arrival: Arrival) => {
   410→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   411→      guestId: arrival.guest?.id || '',
   412→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   413→    })
   414→    navigateTo('front-desk', 'guest-ledger')
   415→  }
   416→
   417→  const buildSpecialRequests = (arrival: Arrival): string => {
   418→    const parts: string[] = []
   419→
   420→    // Preserve existing requests
   421→    if (arrival.specialRequests) {
   422→      parts.push(arrival.specialRequests)
   423→    }
   424→
   425→    // Early check-in note
   426→    if (earlyCheckIn) {
   427→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   428→    }
   429→
   430→    // Check-out time override
   431→    if (checkOutTime !== settings.defaultCheckOut) {
   432→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   433→    }
   434→
   435→    // Room preference
   436→    if (roomPreference) {
   437→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   438→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   439→    }
   440→
   441→    // Pillow type
   442→    if (pillowType) {
   443→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   444→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   445→    }
   446→
   447→    // Wake-up call
   448→    if (wakeupCall) {
   449→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   450→    }
   451→
   452→    return parts.join(' | ')
   453→  }
   454→
   455→  const confirmCheckIn = () => {
   456→    if (!selectedArrival) return
   457→
   458→    const updatedRequests = buildSpecialRequests(selectedArrival)
   459→
   460→    if (selectedRoomId) {
   461→      const room = availableRooms.find((r) => r.id === selectedRoomId)
   462→      if (!room) {
   463→        toast.error('Please select a room')
   464→        return
   465→      }
   466→      assignRoomMutation.mutate({
   467→        reservationId: selectedArrival.id,
   468→        roomId: selectedRoomId,
   469→        roomTypeId: room.typeId,
   470→        specialRequests: updatedRequests,
   471→      })
   472→    } else if (selectedArrival.room) {
   473→      checkInMutation.mutate({
   474→        reservationId: selectedArrival.id,
   475→        specialRequests: updatedRequests,
   476→      })
   477→    }
   478→  }
   479→
   480→  const openKeyCardDialog = (roomNumber: string) => {
   481→    setKeyCardRoomNumber(roomNumber)
   482→    setKeyCardIssued(false)
   483→    setKeyCardProcessed(false)
   484→    setKeyCardDialogOpen(true)
   485→  }
   486→
   487→  const handleKeyCardConfirm = () => {
   488→    setKeyCardProcessed(true)
   489→    setKeyCardDialogOpen(false)
   490→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   491→      duration: 5000,
   492→      icon: <Mail className="size-4" />,
   493→    })
   494→    setSelectedArrival(null)
   495→    setSelectedRoomId('')
   496→  }
   497→
   498→  const resetCheckInForm = () => {
   499→    setEarlyCheckIn(false)
   500→    setCheckOutTime(settings.defaultCheckOut)
   501→    setRoomPreference('')
   502→    setPillowType('')
   503→    setWakeupCall('')
   504→  }
   505→
   506→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   507→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   508→    setWalkInForm((prev) => ({
   509→      ...prev,
   510→      roomTypeId,
   511→      roomRate: roomType?.baseRate || 0,
   512→    }))
   513→  }
   514→
   515→  const handleSelectArrival = (arrival: Arrival) => {
   516→    setSelectedArrival(arrival)
   517→    setSelectedRoomId('')
   518→    setMobileShowDetail(true)
   519→  }
   520→
   521→  const handleBackToList = () => {
   522→    setMobileShowDetail(false)
   523→  }
   524→
   525→  // ─── Render ─────────────────────────────────────────────────────────
   526→
   527→  return (
   528→    <div className="flex flex-col gap-3">
   529→      {/* Header */}
   530→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 shrink-0">
   531→        <div>
   532→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   533→          <p className="text-xs text-muted-foreground">
   534→            Guest check-ins scheduled for {formatDate(today)}
   535→          </p>
   536→        </div>
   537→        <div className="flex items-center gap-2 shrink-0">
   538→          <Badge variant="secondary" className="text-[10px]">
   539→            {totalArrivals} expected
   540→          </Badge>
   541→          {vipArrivals > 0 && (
   542→            <Badge variant="outline" className="text-[10px] border-amber-300 text-amber-700 dark:text-amber-400">
   543→              <Crown className="size-3 mr-0.5" />
   544→              {vipArrivals} VIP
   545→            </Badge>
   546→          )}
   547→          {unassignedArrivals > 0 && (
   548→            <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-600 dark:text-amber-400">
   549→              <AlertTriangle className="size-3 mr-0.5" />
   550→              {unassignedArrivals} unassigned
   551→            </Badge>
   552→          )}
   553→          <Button onClick={handleOpenWalkIn} size="sm" className="gap-1.5">
   554→            <Footprints className="size-3.5" />
   555→            <span className="hidden sm:inline">Walk-in</span>
   556→          </Button>
   557→        </div>
   558→      </div>
   559→
   560→      {/* ─── Split Screen Body ─── */}
   561→      <div className="h-[calc(100vh-14rem)] min-h-[400px] flex flex-col lg:flex-row gap-3 overflow-hidden">
   562→
   563→        {/* ─── Left Panel: Arrivals List ─── */}
   564→        <div className={cn(
   565→          'flex flex-col min-h-0 border rounded-lg bg-card overflow-hidden',
   566→          'w-full lg:w-[42%] xl:w-[38%]',
