   502→// ─── Discount Dialog ─────────────────────────────────────────────────
   503→function DiscountDialog({
   504→  open,
   505→  onClose,
   506→  orderId,
   507→  subtotal,
   508→  discountAmount: currentDiscount,
   509→}: {
   510→  open: boolean
   511→  onClose: () => void
   512→  orderId: string | null
   513→  subtotal: number
   514→  discountAmount: number
   515→}) {
   516→  const queryClient = useQueryClient()
   517→  const [discountType, setDiscountType] = useState<'percentage' | 'fixed'>('percentage')
   518→  const [discountValue, setDiscountValue] = useState('')
   519→  const [reason, setReason] = useState('')
   520→
   521→  const calculatedDiscount = useMemo(() => {
   522→    const val = parseFloat(discountValue) || 0
   523→    if (discountType === 'percentage') return Math.round(subtotal * (Math.min(val, 100) / 100))
   524→    return Math.min(val, subtotal)
   525→  }, [discountType, discountValue, subtotal])
   526→
   527→  const discountMutation = useMutation({
   528→    mutationFn: async (data: { orderId: string; discountType: string; discountValue: number; reason?: string }) => {
   529→      return apiFetch('/api/pos', {
   530→        method: 'POST',
   531→        headers: { 'Content-Type': 'application/json' },
   532→        body: JSON.stringify({ action: 'apply_discount', ...data }),

   636→// ─── Split Bill Dialog ────────────────────────────────────────────────
   637→function SplitBillDialog({
   638→  open,
   639→  onClose,
   640→  order,
   641→}: {
   642→  open: boolean
   643→  onClose: () => void
   644→  order: Order | null
   645→}) {
   646→  const queryClient = useQueryClient()
   647→  const [splitCount, setSplitCount] = useState(2)
   648→  const [assignments, setAssignments] = useState<Record<string, number>>({})
   649→
   650→  const splitSubtotals = useMemo(() => {
   651→    if (!order) return []
   652→    const subs = Array(splitCount).fill(0)
   653→    order.items.forEach((item) => {
   654→      const split = assignments[item.id] ?? 1
   655→      if (split >= 1 && split <= splitCount) {
   656→        subs[split - 1] += item.price * item.quantity
   657→      }
   658→    })
   659→    return subs
   660→  }, [order, splitCount, assignments])
   661→
   662→  const splitMutation = useMutation({
   663→    mutationFn: async (data: { orderId: string; assignments: Record<string, number>; splitSubtotals: number[] }) => {
   664→      return apiFetch('/api/pos', {
   665→        method: 'POST',
   666→        headers: { 'Content-Type': 'application/json' },
   667→        body: JSON.stringify({ action: 'split_bill', ...data }),
   668→      })
   669→    },
   670→    onSuccess: () => {
   671→      toast.success('Bill split successfully recorded')
   672→      queryClient.invalidateQueries({ queryKey: ['pos'] })
   673→      onClose()
   674→    },
   675→    onError: () => {
   676→      toast.error('Failed to split bill')
   677→    },
   678→  })
   679→
   680→  const handleApplySplit = () => {
   681→    if (!order) return
   682→    splitMutation.mutate({
   683→      orderId: order.id,
   684→      assignments,
   685→      splitSubtotals,
   686→    })
   687→  }
   688→
   689→  const handleReset = () => {
   690→    setAssignments({})
   691→  }
   692→
   693→  return (
   694→    <Dialog open={open} onOpenChange={onClose}>
   695→      <DialogContent className="sm:max-w-2xl max-h-[80vh] overflow-y-auto">
   696→        <DialogHeader>
   697→          <DialogTitle className="flex items-center gap-2">
   698→            <Scissors className="h-4 w-4" />
   699→            Split Bill — Order {order?.id}
   700→          </DialogTitle>
   701→        </DialogHeader>
   702→        <div className="space-y-4">
   703→          <div className="flex items-center gap-4">
   704→            <label className="text-sm font-medium">Number of Splits</label>
   705→            <div className="flex items-center gap-2">
   706→              <Button
   707→                variant="outline"
   708→                size="icon"
   709→                className="h-8 w-8"
   710→                onClick={() => { setSplitCount(Math.max(2, splitCount - 1)); handleReset() }}
   711→                disabled={splitCount <= 2}
   712→              >
   713→                <Minus className="h-3 w-3" />
   714→              </Button>
   715→              <span className="w-8 text-center font-bold">{splitCount}</span>
   716→              <Button
   717→                variant="outline"
   718→                size="icon"
   719→                className="h-8 w-8"
   720→                onClick={() => { setSplitCount(Math.min(5, splitCount + 1)); handleReset() }}
   721→                disabled={splitCount >= 5}
   722→              >
   723→                <Plus className="h-3 w-3" />
   724→              </Button>
   725→            </div>
   726→            <Button variant="ghost" size="sm" className="text-xs ml-auto" onClick={handleReset}>
   727→              Reset Assignments
   728→            </Button>
   729→          </div>
   730→
   731→          {/* Split subtotal cards */}
   732→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-2">
   733→            {splitSubtotals.map((sub, idx) => (
   734→              <div key={idx} className="rounded-lg border p-2 text-center">
   735→                <div className="flex items-center justify-center gap-1 text-[10px] text-muted-foreground">
   736→                  <Hash className="h-3 w-3" />
   737→                  Split {idx + 1}
   738→                </div>
   739→                <p className="text-sm font-bold mt-1">{formatNPR(sub)}</p>
   740→              </div>
   741→            ))}
   742→          </div>
   743→
   744→          {/* Item assignments */}
   745→          {order && (
   746→            <div className="space-y-2">
   747→              <p className="text-sm font-medium">Assign items to splits:</p>
   748→              {order.items.map((item) => {
   749→                const assigned = assignments[item.id] ?? 1
   750→                return (
   751→                  <div key={item.id} className="flex items-center gap-3 rounded-md border p-2.5">
   752→                    <div className="flex-1 min-w-0">
   753→                      <p className="text-sm font-medium truncate">
   754→                        {item.name} <span className="text-muted-foreground">×{item.quantity}</span>
   755→                      </p>
   756→                      <p className="text-xs text-muted-foreground">{formatNPR(item.price * item.quantity)}</p>
   757→                    </div>
   758→                    <div className="flex items-center gap-1">
   759→                      {Array.from({ length: splitCount }, (_, idx) => idx + 1).map((n) => (
   760→                        <Button
   761→                          key={n}
   762→                          variant={assigned === n ? 'default' : 'outline'}
   763→                          size="sm"
   764→                          className={`h-7 w-7 text-xs p-0 ${assigned === n ? '' : ''}`}
   765→                          onClick={() => setAssignments((prev) => ({ ...prev, [item.id]: n }))}
   766→                        >
   767→                          {n}
   768→                        </Button>
   769→                      ))}
   770→                    </div>
   771→                  </div>
   772→                )
   773→              })}
   774→            </div>
   775→          )}
   776→        </div>
   777→        <DialogFooter>
   778→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   779→          <Button onClick={handleApplySplit} disabled={splitMutation.isPending}>
   780→            {splitMutation.isPending ? (
   781→              <span className="h-3.5 w-3.5 animate-spin rounded-full border-2 border-current border-t-transparent mr-1.5" />
   782→            ) : (
   783→              <Receipt className="h-3.5 w-3.5 mr-1.5" />
   784→            )}
   785→            Apply Split
   786→          </Button>
   787→        </DialogFooter>
   788→      </DialogContent>
   789→    </Dialog>
   790→  )
   791→}
