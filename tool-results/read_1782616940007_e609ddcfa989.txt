    28→// ─── Types ──────────────────────────────────────────────────────────────
    29→
    30→interface RoomRatePostingDialogProps {
    31→  open: boolean
    32→  onOpenChange: (open: boolean) => void
    33→  reservationId: string
    34→  onPosted?: () => void
    35→  roomNumber?: string
    36→  roomTypeName?: string
    37→  roomRate?: number
    38→  checkIn?: string
    39→  checkOut?: string
    40→  reservationConfirmationNo?: string
    41→}
    42→
    43→interface ReservationDetail {
    44→  id: string
    45→  confirmationNo: string
    46→  status: string
    47→  checkIn: string
    48→  checkOut: string
    49→  roomRate: number
    50→  room: { number: string } | null
    51→  guest: { firstName: string; lastName: string } | null
    52→}
    53→
    54→interface RatePosting {
    55→  id: string
    56→  reservationId: string
    57→  postingDate: string
    58→  roomRate: number
    59→  tax: number
    60→  serviceCharge: number
    61→  total: number
    62→  status: 'posted' | 'pending'
    63→}
    64→
    65→interface NightRow {
    66→  date: Date
    67→  dateStr: string
    68→  dayName: string
    69→  formattedDate: string
    70→  roomRate: number
    71→  tax: number
    72→  serviceCharge: number
    73→  total: number
    74→  isFuture: boolean
    75→  isPosted: boolean
    76→  isPending: boolean
    77→}
    78→
    79→// ─── Constants ──────────────────────────────────────────────────────────
    80→
    81→const TAX_RATE = 0.13
    82→const SERVICE_CHARGE_RATE = 0.10
    83→
    84→const DAY_NAMES = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
    85→
    86→// ─── Helper: local date string (YYYY-MM-DD) ────────────────────────────
    87→
    88→function toLocalDateStr(d: Date): string {
    89→  const y = d.getFullYear()
    90→  const m = String(d.getMonth() + 1).padStart(2, '0')
    91→  const day = String(d.getDate()).padStart(2, '0')
    92→  return `${y}-${m}-${day}`
    93→}

    95→// ─── Component ──────────────────────────────────────────────────────────
    96→
    97→export function RoomRatePostingDialog({
    98→  open,
    99→  onOpenChange,
   100→  reservationId,
   101→  onPosted,
   102→}: RoomRatePostingDialogProps) {
   103→  const queryClient = useQueryClient()
   104→
   105→  // ── Fetch reservation details ────────────────────────────────────────
   106→  const { data: reservation, isLoading: isLoadingReservation } = useQuery({
   107→    queryKey: ['reservation', reservationId],
   108→    queryFn: async () => {
   109→      const raw = await apiFetch<{ reservation: ReservationDetail }>(`/api/reservations/${reservationId}`)
   110→      return raw?.reservation ?? null
   111→    },
   112→    enabled: open && !!reservationId,
   113→  })
   114→
   115→  // ── Fetch existing rate postings ──────────────────────────────────────
   116→  const { data: postingsRaw, isLoading: isLoadingPostings } = useQuery({
   117→    queryKey: ['room-rate-postings', reservationId],
   118→    queryFn: () => apiFetch<{ postings: RatePosting[]; count: number }>(`/api/room-rate-posting?reservationId=${reservationId}`),
   119→    enabled: open && !!reservationId,
   120→  })
   121→  const postings: RatePosting[] = Array.isArray(postingsRaw?.postings) ? postingsRaw.postings : []
   122→
   123→  // ── Build nights array ───────────────────────────────────────────────
   124→  const nights: NightRow[] = useMemo(() => {
   125→    if (!reservation) return []
   126→
   127→    const today = new Date()
   128→    const todayStr = toLocalDateStr(today)
   129→
   130→    const checkInDate = new Date(reservation.checkIn)
   131→    checkInDate.setHours(0, 0, 0, 0)
   132→    const checkOutDate = new Date(reservation.checkOut)
   133→    checkOutDate.setHours(0, 0, 0, 0)
   134→
   135→    const postedDateSet = new Set(
   136→      postings
   137→        .filter((p) => p.status === 'posted')
   138→        .map((p) => toLocalDateStr(new Date(p.postingDate))),
   139→    )
   140→
   141→    const roomRate = reservation.roomRate || 0
   142→    const rows: NightRow[] = []
   143→
   144→    let current = new Date(checkInDate)
   145→    while (current < checkOutDate) {
   146→      const currentStr = toLocalDateStr(current)
   147→      const isFuture = currentStr > todayStr
   148→      const isPosted = postedDateSet.has(currentStr)
   149→
   150→      const tax = Math.round(roomRate * TAX_RATE)
   151→      const serviceCharge = Math.round(roomRate * SERVICE_CHARGE_RATE)
   152→      const total = roomRate + tax + serviceCharge
   153→      const dayName = DAY_NAMES[current.getDay()]
   154→      const formattedDate = formatDate(current)
   155→
   156→      rows.push({
   157→        date: new Date(current),
   158→        dateStr: currentStr,
   159→        dayName,
   160→        formattedDate,
   161→        roomRate,
   162→        tax,
   163→        serviceCharge,
   164→        total,
   165→        isFuture,
   166→        isPosted,
   167→        isPending: !isPosted && !isFuture,
   168→      })
   169→
   170→      current.setDate(current.getDate() + 1)
   171→    }
   172→
   173→    return rows
   174→  }, [reservation, postings])
   175→
   176→  // ── Summary calculations ─────────────────────────────────────────────
   177→  const summary = useMemo(() => {
   178→    const totalNights = nights.length
   179→    const totalRoomCharges = nights.reduce((sum, n) => sum + n.roomRate, 0)
   180→    const totalTax = nights.reduce((sum, n) => sum + n.tax, 0)
   181→    const totalServiceCharge = nights.reduce((sum, n) => sum + n.serviceCharge, 0)
   182→    const grandTotal = totalRoomCharges + totalTax + totalServiceCharge
   183→    const pendingNights = nights.filter((n) => n.isPending)
   184→    const pendingCount = pendingNights.length
   185→    const pendingAmount = pendingNights.reduce((sum, n) => sum + n.total, 0)
   186→    const postedCount = nights.filter((n) => n.isPosted).length
   187→    const futureCount = nights.filter((n) => n.isFuture).length
   188→    const allPosted = pendingCount === 0
   189→
   190→    return { totalNights, totalRoomCharges, totalTax, totalServiceCharge, grandTotal, pendingCount, pendingAmount, postedCount, futureCount, allPosted }
   191→  }, [nights])
   192→
   193→  // ── Post all pending mutation ─────────────────────────────────────────
   194→  const postAllMutation = useMutation({
   195→    mutationFn: async () => {
   196→      // Send only pending dates to the backend
   197→      const pendingDates = nights
   198→        .filter((n) => n.isPending)
   199→        .map((n) => n.dateStr)
   200→
   201→      if (pendingDates.length === 0) return null
   202→
   203→      return apiFetch<{ message: string; newCount: number }>('/api/room-rate-posting', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({ reservationId, dates: pendingDates }),
   207→      })
   208→    },
   209→    onSuccess: (data) => {
   210→      if (data?.newCount === 0) {
   211→        toast.info('All nights already posted. No new charges created.')
   212→      } else {
   213→        toast.success(`${summary.pendingCount} room charge(s) posted to guest folio`)
   214→      }
   215→      invalidate.afterFolioChange(queryClient)
   216→      queryClient.invalidateQueries({ queryKey: ['room-rate-postings', reservationId] })
   217→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   218→      onPosted?.()
   219→    },
   220→    onError: (error) => {
   221→      toast.error(error.message || 'Failed to post room charges. Please try again.')
   222→    },
   223→  })
   224→
   225→  // ── Post single night mutation ────────────────────────────────────────
   226→  const postSingleMutation = useMutation({
   227→    mutationFn: async (dateStr: string) => {
   228→      return apiFetch<{ message: string }>('/api/room-rate-posting', {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({ reservationId, dates: [dateStr] }),
   232→      })
   233→    },
   234→    onSuccess: (_data, variables) => {
   235→      toast.success('Room charge posted for ' + formatDate(variables))
   236→      invalidate.afterFolioChange(queryClient)
   237→      queryClient.invalidateQueries({ queryKey: ['room-rate-postings', reservationId] })
   238→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   239→      onPosted?.()
   240→    },
   241→    onError: (error) => {
   242→      toast.error(error.message || 'Failed to post room charge. Please try again.')
   243→    },
   244→  })
   245→
   246→  // ── Handlers ──────────────────────────────────────────────────────────
   247→  const handlePostAll = () => {
   248→    postAllMutation.mutate()
   249→  }
   250→
   251→  const handlePostSingle = (night: NightRow) => {
   252→    postSingleMutation.mutate(night.dateStr)
   253→  }
   254→
   255→  const handleClose = () => {
   256→    onOpenChange(false)
   257→  }
   258→
   259→  // ── Computed ─────────────────────────────────────────────────────────
   260→  const isLoading = isLoadingReservation || isLoadingPostings
   261→  const isPosting = postAllMutation.isPending || postSingleMutation.isPending
   262→  const hasRoom = !!reservation?.room
   263→
   264→  // ── Render ───────────────────────────────────────────────────────────
   265→  return (
   266→    <Dialog open={open} onOpenChange={onOpenChange}>
   267→      <DialogContent className="max-w-4xl max-h-[90vh] flex flex-col p-0 gap-0 overflow-hidden">
   268→        {/* ── Header ─────────────────────────────────────────────────── */}
   269→        <div className="px-6 pt-6 pb-0">
   270→          <DialogHeader>
   271→            <DialogTitle className="flex items-center gap-2">
   272→              <DollarSign className="size-5 text-emerald-600" />
   273→              Room Rate Posting
   274→            </DialogTitle>
   275→            <DialogDescription>
   276→              Post daily room charges to guest folio
   277→            </DialogDescription>
   278→          </DialogHeader>
   279→        </div>
   280→
   281→        {/* ── Body (scrollable) ─────────────────────────────────────── */}
   282→        <div className="flex-1 overflow-y-auto px-6 py-4 space-y-4">
   283→          {isLoading ? (
   284→            <LoadingSkeleton />
   285→          ) : !reservation?.id ? (
   286→            <ErrorState />
   287→          ) : (
   288→            <>
   289→              {/* Reservation Summary Card — Enhanced */}
   290→              <ReservationSummaryCard
   291→                reservation={reservation}
   292→                totalNights={summary.totalNights}
   293→                pendingCount={summary.pendingCount}
   294→                pendingAmount={summary.pendingAmount}
   295→                postedCount={summary.postedCount}
   296→                futureCount={summary.futureCount}
   297→              />
   298→
   299→              {/* No room warning */}
   300→              {!hasRoom && (
   301→                <div className="flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 px-4 py-2.5 text-xs text-amber-800">
   302→                  <AlertCircle className="size-4 shrink-0" />
   303→                  <span>No room assigned to this reservation. Room charges cannot be posted without a room.</span>
   304→                </div>
   305→              )}
   306→
   307→              {/* Posting Schedule Table */}
   308→              <div className="rounded-lg border overflow-hidden">
   309→                <div className="max-h-[320px] overflow-y-auto">
   310→                  <Table>
   311→                    <TableHeader className="sticky top-0 z-10 bg-muted/95 backdrop-blur-sm">
   312→                      <TableRow>
   313→                        <TableHead className="text-xs font-semibold w-[150px]">Date</TableHead>
   314→                        <TableHead className="text-xs font-semibold text-right">Room Rate</TableHead>
   315→                        <TableHead className="text-xs font-semibold text-right">Tax (13%)</TableHead>
   316→                        <TableHead className="text-xs font-semibold text-right">Svc (10%)</TableHead>
   317→                        <TableHead className="text-xs font-semibold text-right">Total</TableHead>
   318→                        <TableHead className="text-xs font-semibold text-center w-[80px]">Status</TableHead>
   319→                        <TableHead className="text-xs font-semibold text-right w-[64px]">Action</TableHead>
   320→                      </TableRow>
   321→                    </TableHeader>
   322→                    <TableBody>
   323→                      {nights.length === 0 ? (
   324→                        <TableRow>
   325→                          <TableCell colSpan={7} className="text-center text-muted-foreground py-8">
   326→                            No nights found for this reservation.
   327→                          </TableCell>
   328→                        </TableRow>
   329→                      ) : (
   330→                        nights.map((night) => (
   331→                          <TableRow
   332→                            key={night.dateStr}
   333→                            className={cn(
   334→                              'text-xs',
   335→                              night.isPosted && 'bg-emerald-50/70 hover:bg-emerald-50',
   336→                              night.isPending && 'bg-amber-50/60 hover:bg-amber-50',
   337→                              night.isFuture && 'bg-muted/40 hover:bg-muted/40',
   338→                            )}
   339→                          >
   340→                            {/* Date */}
   341→                            <TableCell className="font-medium text-xs whitespace-nowrap">
   342→                              {night.formattedDate}
   343→                            </TableCell>
   344→
   345→                            {/* Room Rate */}
   346→                            <TableCell className="text-right font-mono tabular-nums">
   347→                              {formatCurrency(night.roomRate)}
   348→                            </TableCell>
   349→
   350→                            {/* Tax */}
   351→                            <TableCell className="text-right font-mono tabular-nums text-muted-foreground">
   352→                              {formatCurrency(night.tax)}
   353→                            </TableCell>
   354→
   355→                            {/* Service Charge */}
   356→                            <TableCell className="text-right font-mono tabular-nums text-muted-foreground">
   357→                              {formatCurrency(night.serviceCharge)}
   358→                            </TableCell>
   359→
   360→                            {/* Total */}
   361→                            <TableCell className="text-right font-mono tabular-nums font-semibold">
   362→                              {formatCurrency(night.total)}
   363→                            </TableCell>
   364→
   365→                            {/* Status */}
   366→                            <TableCell className="text-center">
   367→                              <StatusBadge
   368→                                status={night.isPosted ? 'posted' : night.isFuture ? 'future' : 'pending'}
   369→                              />
   370→                            </TableCell>
   371→
   372→                            {/* Action */}
   373→                            <TableCell className="text-right">
   374→                              {night.isPending ? (
   375→                                <Button
   376→                                  variant="outline"
   377→                                  size="sm"
   378→                                  className="h-6 text-[10px] px-2"
   379→                                  onClick={() => handlePostSingle(night)}
   380→                                  disabled={isPosting || !hasRoom}
   381→                                >
   382→                                  {postSingleMutation.isPending &&
   383→                                  postSingleMutation.variables === night.dateStr ? (
   384→                                    <Loader2 className="size-3 animate-spin" />
   385→                                  ) : (
   386→                                    <CheckCircle className="size-3" />
   387→                                  )}
   388→                                  Post
   389→                                </Button>
   390→                              ) : (
   391→                                <span className="sr-only">
   392→                                  {night.isPosted ? 'Already posted' : night.isFuture ? 'Future night' : '—'}
   393→                                </span>
   394→                              )}
   395→                            </TableCell>
   396→                          </TableRow>
   397→                        ))
   398→                      )}
   399→                    </TableBody>
   400→                  </Table>
   401→                </div>
   402→              </div>
   403→
   404→              {/* Summary Section */}
   405→              <Card className="py-3">
   406→                <CardContent className="px-4">
   407→                  <div className="grid grid-cols-2 sm:grid-cols-4 gap-x-4 gap-y-3 text-xs">
   408→                    <SummaryItem label="Total Nights" value={String(summary.totalNights)} />
   409→                    <SummaryItem
   410→                      label="Total Room Charges"
   411→                      value={formatCurrency(summary.totalRoomCharges)}
   412→                      align="right"
   413→                    />
   414→                    <SummaryItem
   415→                      label="Total Tax (13%)"
   416→                      value={formatCurrency(summary.totalTax)}
   417→                      align="right"
   418→                    />
   419→                    <SummaryItem
   420→                      label="Total Service (10%)"
   421→                      value={formatCurrency(summary.totalServiceCharge)}
   422→                      align="right"
   423→                    />
   424→                  </div>
   425→                  <Separator className="my-3" />
   426→                  <div className="flex items-center justify-between">
   427→                    <span className="text-xs font-semibold text-emerald-600 uppercase tracking-wider">
   428→                      Grand Total
   429→                    </span>
   430→                    <span className="text-lg font-bold text-emerald-700 tabular-nums">
   431→                      {formatCurrency(summary.grandTotal)}
   432→                    </span>
   433→                  </div>
   434→                </CardContent>
   435→              </Card>
   436→            </>
   437→          )}
   438→        </div>
   439→
   440→        {/* ── Footer ─────────────────────────────────────────────────── */}
   441→        <Separator />
   442→        <DialogFooter className="px-6 py-4 gap-2">
   443→          <Button variant="outline" onClick={handleClose} className="gap-1.5">
   444→            <X className="size-3.5" />
   445→            Skip for Now
   446→          </Button>
   447→          <Button
   448→            onClick={handlePostAll}
   449→            disabled={summary.allPosted || isPosting || nights.length === 0 || !hasRoom}
   450→            className={cn(
   451→              'gap-1.5 min-w-[180px]',
   452→              summary.allPosted && 'bg-emerald-600 hover:bg-emerald-600',
   453→            )}
   454→          >
   455→            {postAllMutation.isPending ? (
   456→              <>
   457→                <Loader2 className="size-4 animate-spin" />
   458→                Posting…
   459→              </>
   460→            ) : summary.allPosted ? (
   461→              <>
   462→                <CheckCircle className="size-4" />
   463→                All Posted ✓
   464→              </>
   465→            ) : (
   466→              <>
   467→                <DollarSign className="size-4" />
   468→                Post All Pending ({summary.pendingCount})
   469→              </>
   470→            )}
   471→          </Button>
   472→        </DialogFooter>
   473→      </DialogContent>
   474→    </Dialog>
   475→  )
   476→}

   480→function ReservationSummaryCard({
   481→  reservation,
   482→  totalNights,
   483→  pendingCount,
   484→  pendingAmount,
   485→  postedCount,
   486→  futureCount,
   487→}: {
   488→  reservation: ReservationDetail
   489→  totalNights: number
   490→  pendingCount: number
   491→  pendingAmount: number
   492→  postedCount: number
   493→  futureCount: number
   494→}) {
   495→  const guestName = reservation.guest
   496→    ? `${reservation.guest.firstName} ${reservation.guest.lastName}`
   497→    : '—'
   498→  const roomNumber = reservation.room?.number ?? 'Unassigned'
   499→
   500→  return (
   501→    <Card className="bg-muted/30 border-dashed overflow-hidden">
   502→      <CardContent className="p-0">
   503→        {/* Top row: Guest + Room + Confirmation — 3 equal columns */}
   504→        <div className="grid grid-cols-3 divide-x divide-border/60">
   505→          {/* Guest */}
   506→          <div className="px-4 py-3 flex flex-col justify-center">
   507→            <div className="flex items-center gap-1.5 mb-1">
   508→              <User className="size-3 text-muted-foreground shrink-0" />
   509→              <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">Guest</span>
   510→            </div>
   511→            <span className="text-sm font-semibold truncate" title={guestName}>{guestName}</span>
   512→          </div>
   513→
   514→          {/* Room */}
   515→          <div className="px-4 py-3 flex flex-col justify-center">
   516→            <div className="flex items-center gap-1.5 mb-1">
   517→              <BedDouble className="size-3 text-muted-foreground shrink-0" />
   518→              <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">Room</span>
   519→            </div>
   520→            <span className={cn(
   521→              'text-sm font-semibold',
   522→              !reservation.room && 'text-amber-600',
   523→            )}>
   524→              {roomNumber}
   525→            </span>
   526→          </div>
   527→
   528→          {/* Confirmation No */}
   529→          <div className="px-4 py-3 flex flex-col justify-center">
   530→            <div className="flex items-center gap-1.5 mb-1">
   531→              <CalendarDays className="size-3 text-muted-foreground shrink-0" />
   532→              <span className="text-[10px] text-muted-foreground font-medium uppercase tracking-wider">Confirmation</span>
   533→            </div>
   534→            <span className="text-sm font-semibold font-mono">{reservation.confirmationNo}</span>
   535→          </div>
   536→        </div>
   537→
   538→        {/* Bottom row: Stay dates + Rate/Night + Posting stats */}
   539→        <div className="border-t border-border/60 grid grid-cols-3 divide-x divide-border/60">
   540→          {/* Stay Period */}
   541→          <div className="px-4 py-2.5 flex items-center gap-2 text-xs">
   542→            <span className="text-muted-foreground">Stay:</span>
   543→            <span className="font-medium">{formatDate(reservation.checkIn)}</span>
   544→            <ArrowRight className="size-3 text-muted-foreground shrink-0" />
   545→            <span className="font-medium">{formatDate(reservation.checkOut)}</span>
   546→            <Badge variant="secondary" className="h-5 text-[10px] font-semibold px-1.5 ml-auto shrink-0">
   547→              <Moon className="size-2.5 mr-0.5" />
   548→              {totalNights}
   549→            </Badge>
   550→          </div>
   551→
   552→          {/* Rate per Night */}
   553→          <div className="px-4 py-2.5 flex items-center justify-between text-xs">
   554→            <span className="text-muted-foreground">Rate / Night</span>
   555→            <span className="font-semibold tabular-nums text-sm">{formatCurrency(reservation.roomRate)}</span>
   556→          </div>
