   111→   111→function isSameDay(dateStr: string, todayStr: string): boolean {
   112→   112→  const d = new Date(dateStr)
   113→   113→  const t = new Date(todayStr)
   114→   114→  return d.getFullYear() === t.getFullYear() && d.getMonth() === t.getMonth() && d.getDate() === t.getDate()
   115→   115→}
   116→   116→
   117→   117→function isBeforeDay(dateStr: string, todayStr: string): boolean {
   118→   118→  const d = new Date(dateStr)
   119→   119→  d.setHours(0, 0, 0, 0)
   120→   120→  const t = new Date(todayStr)
   121→   121→  t.setHours(0, 0, 0, 0)
   122→   122→  return d.getTime() < t.getTime()
   123→   123→}
