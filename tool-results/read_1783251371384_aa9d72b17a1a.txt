   777→            method: 'POST',

   790→        const occupiedIds = new Set(
   791→          roomIds
   792→            .map(id => tableRows.find(r => r.roomId === id))

   794→            .map(r => r.roomId)

   885→      rowStatusMutation.mutate({ taskId, status: newStatus })

   945→          assignedTo: row.assignedTo,

   949→          notes: row.taskNotes,

   951→        })
   952→      }
   953→    } else {
   954→      toast.info(`Room ${row?.roomNumber} — No active task`)
   955→    }
   956→  }
   957→
   958→  // Get unique floors for filter
   959→  const floors = useMemo(() => {
   960→    const f = new Set(tableRows.map((r) => r.floor))
   961→    return Array.from(f).sort((a, b) => a - b)
   962→  }, [tableRows])
   963→
   964→  // Get unique room types for filter

   972→
   973→  // Get unique reservation statuses
   974→  const uniqueResStatuses = useMemo(() => {
   975→    const s = new Set(tableRows.map((r) => r.reservationStatus))
   976→    return Array.from(s)
   977→  }, [tableRows])
   978→
   979→  // Toggle row selection
   980→  const toggleRow = (roomId: string) => {
   981→    setSelectedRows((prev) => {
   982→      const next = new Set(prev)
   983→      if (next.has(roomId)) next.delete(roomId)
   984→      else next.add(roomId)
   985→      return next
   986→    })
   987→  }
   988→
   989→  const toggleAllRows = () => {
   990→    if (selectedRows.size === filteredRows.length) {
   991→      setSelectedRows(new Set())
   992→    } else {
   993→      setSelectedRows(new Set(filteredRows.map((r) => r.roomId)))
   994→    }
   995→  }
   996→
   997→  const toggleColumn = (id: string) => {
   998→    setExpandedColumns((prev) => {
   999→      const next = new Set(prev)
  1000→      if (next.has(id)) next.delete(id)
  1001→      else next.add(id)
  1002→      return next
  1003→    })
  1004→  }
  1005→
  1006→  // Client-side filtered rows
  1007→  const filteredRows = (() => {
  1008→    let rows = tableRows
  1009→    if (filterRoomType) rows = rows.filter((r) => r.roomTypeCode === filterRoomType)
  1010→    if (filterRoomNumber) rows = rows.filter((r) => r.roomNumber.includes(filterRoomNumber))
  1011→    if (filterReservation) rows = rows.filter((r) => r.reservationStatus === filterReservation)
  1012→    return rows
  1013→  })()
  1014→
  1015→  // Unique values derived from tableRows (not filtered)
  1016→  const uniqueHkStatuses = useMemo(() => {
  1017→    const s = new Set(tableRows.map((r) => r.hkDisplayStatus))
  1018→    return Array.from(s)
  1019→  }, [tableRows])
  1020→
  1021→  const uniquePriorities = useMemo(() => {
  1022→    const s = new Set(tableRows.map((r) => r.priority))
  1023→    return Array.from(s)
  1024→  }, [tableRows])
  1025→
  1026→  const hasAnyFilter = filterHkStatus || filterPriority || filterFloor || filterRoomType || filterRoomNumber || filterReservation
  1027→
  1028→  // Loading state
  1029→  if ((viewMode === 'table' && tableLoading) || (viewMode !== 'table' && kanbanLoading)) {
  1030→    return (
  1031→      <div className="space-y-3">
  1032→        <div className="h-8 w-full rounded-md bg-muted animate-pulse" />
  1033→        <div className="h-96 rounded-lg bg-muted animate-pulse" />
  1034→      </div>
  1035→    )
  1036→  }
  1037→
  1038→  // ─── TABLE VIEW ──────────────────────────────────────────
  1039→  const renderTableView = () => (
  1040→    <div className="space-y-2">
  1041→      {/* Toolbar: Search */}
  1042→      <div className="flex items-center justify-between">
  1043→        <div className="relative flex-1 max-w-sm">
  1044→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
  1045→          <Input
  1046→            placeholder="Search by guest, reservation..."
  1047→            value={searchQuery}
  1048→            onChange={(e) => handleSearchChange(e.target.value)}
  1049→            className="h-8 pl-8 text-xs"
  1050→          />
  1051→          {searchQuery && (
  1052→            <button
  1053→              className="absolute right-2 top-1/2 -translate-y-1/2"
  1054→              onClick={() => { setSearchQuery(''); setDebouncedSearch('') }}
  1055→            >
  1056→              <X className="h-3.5 w-3.5 text-muted-foreground hover:text-foreground" />
  1057→            </button>
  1058→          )}
  1059→        </div>
  1060→        <div className="flex items-center gap-2 text-xs text-muted-foreground ml-3">
  1061→          <span>{filteredRows.length} room{filteredRows.length !== 1 ? 's' : ''}</span>
  1062→          {selectedRows.size > 0 && (
  1063→            <span className="font-medium text-foreground">· {selectedRows.size} selected</span>
  1064→          )}
  1065→          {forceMutatedRooms.size > 0 && (
  1066→            <Badge variant="outline" className="text-[10px] text-amber-600 border-amber-300 dark:border-amber-700 gap-0.5">
  1067→              <ShieldAlert className="h-2.5 w-2.5" /> {forceMutatedRooms.size} locked
  1068→            </Badge>
  1069→          )}
  1070→          {hasAnyFilter && (
  1071→            <Button
  1072→              variant="ghost"
  1073→              size="sm"
  1074→              className="h-6 px-2 text-[10px] text-muted-foreground"
  1075→              onClick={() => {
  1076→                setFilterHkStatus(''); setFilterPriority(''); setFilterFloor('')
  1077→                setFilterRoomType(''); setFilterRoomNumber(''); setFilterReservation('')
  1078→              }}
  1079→            >
  1080→              <X className="w-2.5 h-2.5 mr-0.5" /> Clear filters
  1081→            </Button>
  1082→          )}
  1083→        </div>
  1084→      </div>
  1085→
  1086→      {/* Bulk Action Bar (2+ selected) */}
  1087→      {selectedRows.size >= 2 && (
  1088→        <div className="flex items-center gap-2 rounded-lg border border-primary/30 bg-primary/5 px-3 py-2 flex-wrap">
  1089→          <span className="text-xs font-medium text-primary mr-1">{selectedRows.size} rooms selected</span>
  1090→          <div className="h-4 w-px bg-border" />
  1091→          <Button
  1092→            variant="outline" size="sm" className="h-7 text-[11px] gap-1"
  1093→            onClick={() => handleBulkAction('in_progress')}
  1094→            disabled={bulkStatusMutation.isPending}
  1095→          >
  1096→            <PlayCircle className="h-3 w-3" /> Start Cleaning
  1097→          </Button>
  1098→          <Button
  1099→            variant="outline" size="sm" className="h-7 text-[11px] gap-1"
  1100→            onClick={() => handleBulkAction('cleaned')}
  1101→            disabled={bulkStatusMutation.isPending}
  1102→          >
  1103→            <CheckCircle className="h-3 w-3" /> Mark Cleaned
  1104→          </Button>
  1105→          <Button
  1106→            variant="outline" size="sm" className="h-7 text-[11px] gap-1"
  1107→            onClick={() => handleBulkAction('inspected')}
  1108→            disabled={bulkStatusMutation.isPending}
  1109→          >
  1110→            <ShieldCheck className="h-3 w-3" /> Mark Inspected
  1111→          </Button>
  1112→          <Button
  1113→            variant="outline" size="sm" className="h-7 text-[11px] gap-1"
  1114→            onClick={() => handleBulkAction('rush')}
  1115→            disabled={bulkStatusMutation.isPending}
  1116→          >
  1117→            <AlertTriangle className="h-3 w-3" /> Set Rush
  1118→          </Button>
  1119→          <Button
  1120→            variant="outline" size="sm" className="h-7 text-[11px] gap-1"
  1121→            onClick={() => handleBulkAction('pending')}
  1122→            disabled={bulkStatusMutation.isPending}
  1123→          >
  1124→            <ArrowUpCircle className="h-3 w-3" /> Reset to Pending
  1125→          </Button>
  1126→          <Button
  1127→            variant="outline" size="sm" className="h-7 text-[11px] gap-1 text-red-600 hover:text-red-600"
  1128→            onClick={() => handleBulkAction('failed')}
  1129→            disabled={bulkStatusMutation.isPending}
  1130→          >
  1131→            <XCircle className="h-3 w-3" /> Mark Failed
  1132→          </Button>
  1133→          <div className="flex-1" />
  1134→          <Button
  1135→            variant="ghost" size="sm" className="h-7 text-[11px]"
  1136→            onClick={() => setSelectedRows(new Set())}
  1137→          >
  1138→            <X className="h-3 w-3 mr-1" /> Deselect All
  1139→          </Button>
  1140→        </div>
  1141→      )}
  1142→
  1143→      {/* Data Table with sticky header */}
  1144→      <div className="rounded-lg border bg-card overflow-hidden">
  1145→        <div className="max-h-[calc(100vh-280px)] overflow-auto">
  1146→          <Table>
  1147→            <TableHeader className="sticky top-0 z-10 bg-card">
  1148→              <TableRow className="hover:bg-muted/40">
  1149→                <TableHead className="w-10" />
  1150→                <TableHead className="w-10">
  1151→                  <Checkbox
  1152→                    checked={filteredRows.length > 0 && selectedRows.size === filteredRows.length}
  1153→                    onCheckedChange={toggleAllRows}
  1154→                  />
  1155→                </TableHead>
  1156→                <TableHead className="text-xs font-semibold">
  1157→                  <RoomFilterDropdown value={filterRoomNumber} onChange={setFilterRoomNumber} />
  1158→                </TableHead>
  1159→                <TableHead>
  1160→                  <FilterDropdown
  1161→                    label="Room Type"
  1162→                    value={filterRoomType}
  1163→                    onValueChange={setFilterRoomType}
  1164→                    options={roomTypes.map((rt) => ({ value: rt.value, label: rt.label }))}
  1165→                  />
  1166→                </TableHead>
  1167→                <TableHead>
  1168→                  <FilterDropdown
  1169→                    label="HK Status"
  1170→                    value={filterHkStatus}
  1171→                    onValueChange={setFilterHkStatus}
  1172→                    options={uniqueHkStatuses.map((s) => ({ value: s, label: s.split('_').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ') }))}
  1173→                  />
  1174→                </TableHead>
  1175→                <TableHead>
  1176→                  <FilterDropdown
  1177→                    label="Priority"
  1178→                    value={filterPriority}
  1179→                    onValueChange={setFilterPriority}
  1180→                    options={uniquePriorities.map((p) => ({ value: p, label: p.charAt(0).toUpperCase() + p.slice(1) }))}
  1181→                  />
  1182→                </TableHead>
  1183→                <TableHead>
  1184→                  <FilterDropdown
  1185→                    label="Floor"
  1186→                    value={filterFloor}
  1187→                    onValueChange={setFilterFloor}
  1188→                    options={floors.map((f) => ({ value: String(f), label: `Floor ${f}` }))}
  1189→                  />
  1190→                </TableHead>
  1191→                <TableHead>
  1192→                  <FilterDropdown
  1193→                    label="Reservation"
  1194→                    value={filterReservation}
  1195→                    onValueChange={setFilterReservation}
  1196→                    options={uniqueResStatuses.map((s) => ({ value: s, label: s.split('_').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ') }))}
  1197→                  />
  1198→                </TableHead>
  1199→                <TableHead className="text-xs font-semibold">Comments</TableHead>
  1200→              </TableRow>
  1201→            </TableHeader>
  1202→            <TableBody>
  1203→                {filteredRows.length === 0 ? (
  1204→                  <TableRow>
  1205→                    <TableCell colSpan={9} className="h-32 text-center text-muted-foreground text-sm">
  1206→                      No rooms found
  1207→                    </TableCell>
  1208→                  </TableRow>
  1209→                ) : (
  1210→                  filteredRows.map((row, idx) => (
  1211→                    <TableRow
  1212→                      key={row.roomId}
  1213→                      className={cn(
  1214→                        'cursor-pointer transition-colors',
  1215→                        selectedRows.has(row.roomId) && 'bg-primary/5',
  1216→                        idx % 2 === 1 && !selectedRows.has(row.roomId) && 'bg-muted/20',
  1217→                        forceMutatedRooms.has(row.roomId) && 'border-l-2 border-l-amber-400'
  1218→                      )}
  1219→                      onClick={() => toggleRow(row.roomId)}
  1220→                    >
  1221→                      <TableCell onClick={(e) => e.stopPropagation()}>
  1222→                        <InlineRowActions
  1223→                          row={row}
  1224→                          onStatusChange={handleRowStatusChange}
  1225→                          onViewDetail={handleViewDetail}
  1226→                          isForceMutated={forceMutatedRooms.has(row.roomId)}
  1227→                        />
  1228→                      </TableCell>
  1229→                      <TableCell onClick={(e) => e.stopPropagation()}>
  1230→                        <Checkbox
  1231→                          checked={selectedRows.has(row.roomId)}
  1232→                          onCheckedChange={() => toggleRow(row.roomId)}
  1233→                        />
  1234→                      </TableCell>
  1235→                      <TableCell>
  1236→                        <div className="flex items-center gap-2">
  1237→                          <BedDouble className="h-3.5 w-3.5 text-muted-foreground shrink-0" />
  1238→                          <span className="text-sm font-semibold">{row.roomNumber}</span>
  1239→                          {row.wing && (
  1240→                            <span className="text-[10px] text-muted-foreground">({row.wing})</span>
  1241→                          )}
  1242→                        </div>
  1243→                      </TableCell>
  1244→                      <TableCell>
  1245→                        <div className="text-xs">
  1246→                          <span className="font-medium">{row.roomTypeName}</span>
  1247→                          <span className="text-muted-foreground ml-1">({row.roomTypeCode})</span>
  1248→                        </div>
  1249→                      </TableCell>
  1250→                      <TableCell>
  1251→                        {hkStatusBadge(row.hkDisplayStatus)}
  1252→                      </TableCell>
  1253→                      <TableCell>
  1254→                        <PriorityDot priority={row.priority} />
  1255→                      </TableCell>
  1256→                      <TableCell>
  1257→                        <span className="text-xs text-muted-foreground">{row.floor}</span>
  1258→                      </TableCell>
  1259→                      <TableCell>
  1260→                        {resStatusBadge(row.reservationStatus)}
  1261→                      </TableCell>
  1262→                      <TableCell>
  1263→                        <div className="text-xs text-muted-foreground max-w-[180px] truncate">
  1264→                          {row.taskNotes || '—'}
  1265→                        </div>
  1266→                      </TableCell>
  1267→                    </TableRow>
  1268→                  ))
  1269→                )}
  1270→              </TableBody>
  1271→          </Table>
  1272→        </div>
  1273→      </div>
  1274→    </div>
  1275→  )
  1276→
  1277→  // ─── KANBAN VIEW ─────────────────────────────────────────
  1278→  const renderKanbanView = () => (
  1279→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-5">
  1280→      {KANBAN_COLUMNS.map((column) => {
  1281→        const columnTasks = kanbanTasks.filter((t) => t.status === column.id)
  1282→        const isExpanded = expandedColumns.has(column.id)
  1283→
  1284→        return (
  1285→          <div
  1286→            key={column.id}
  1287→            className={cn('rounded-lg border-2 border-t-4 bg-card', column.color)}
  1288→          >
  1289→            <button
  1290→              className="flex w-full items-center justify-between p-3 text-left"
  1291→              onClick={() => toggleColumn(column.id)}
  1292→            >
  1293→              <div className="flex items-center gap-2">
  1294→                <span className="font-semibold text-sm">{column.label}</span>
  1295→                <Badge variant="secondary" className="text-xs">{columnTasks.length}</Badge>
  1296→              </div>
  1297→            </button>
  1298→
  1299→            {isExpanded && (
  1300→              <ScrollArea className="h-[500px] px-3 pb-3">
  1301→                <div className="space-y-2">
  1302→                  {columnTasks.length === 0 ? (
  1303→                    <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
  1304→                      <ClipboardCheck className="h-8 w-8 opacity-20" />
  1305→                      <p className="mt-2 text-xs">No tasks</p>
  1306→                    </div>
  1307→                  ) : (
  1308→                    columnTasks.map((task) => (
  1309→                      <TaskCard key={task.id} task={task} onClick={() => setSelectedTask(task)} />
  1310→                    ))
  1311→                  )}
  1312→                </div>
  1313→              </ScrollArea>
  1314→            )}
  1315→          </div>
  1316→        )
  1317→      })}
  1318→    </div>
  1319→  )
  1320→
  1321→  return (
  1322→    <div className="space-y-3">
  1323→      {/* View Toggle */}
  1324→      <div className="flex items-center gap-2">
  1325→        <Button
  1326→          variant={viewMode === 'table' ? 'default' : 'outline'}
  1327→          size="sm"
  1328→          onClick={() => setViewMode('table')}
  1329→          className="gap-1.5"
  1330→        >
  1331→          <List className="h-4 w-4" />
  1332→          Board
  1333→        </Button>
  1334→        <Button
  1335→          variant={viewMode === 'kanban' ? 'default' : 'outline'}
  1336→          size="sm"
  1337→          onClick={() => setViewMode('kanban')}
  1338→          className="gap-1.5"
  1339→        >
  1340→          <LayoutGrid className="h-4 w-4" />
  1341→          Kanban
  1342→        </Button>
  1343→        <Button
  1344→          variant={viewMode === 'attendant' ? 'default' : 'outline'}
  1345→          size="sm"
  1346→          onClick={() => setViewMode('attendant')}
  1347→          className="gap-1.5"
  1348→        >
  1349→          <Users className="h-4 w-4" />
  1350→          Attendant
  1351→        </Button>
  1352→      </div>
  1353→
  1354→      {/* Views */}
  1355→      {viewMode === 'table' && renderTableView()}
  1356→      {viewMode === 'kanban' && renderKanbanView()}
  1357→      {viewMode === 'attendant' && <AttendantView tasks={kanbanTasks} />}
  1358→
  1359→      {/* Task Detail Dialog (for Kanban/Attendant/Table) */}
  1360→      <TaskDetailDialog
  1361→        task={selectedTask}
  1362→        open={!!selectedTask}
  1363→        onOpenChange={(open) => !open && setSelectedTask(null)}
  1364→      />
  1365→
  1366→      {/* Occupied Room Warning Dialog */}
  1367→      <OccupiedWarningDialog
  1368→        open={occupiedWarning.open}
  1369→        onOpenChange={(open) => !open && handleCancelWarning()}
  1370→        occupiedRooms={occupiedWarning.rooms}
  1371→        actionLabel={occupiedWarning.pendingAction.split('_').map(w => w.charAt(0).toUpperCase() + w.slice(1)).join(' ')}
  1372→        onForce={handleForceFromWarning}

  1374→      />
  1375→
  1376→      {/* Force-Mutated Room Blocked Dialog */}
  1377→      <ForceMutatedBlockDialog
  1378→        open={blockedDialog.open}

  1383→    </div>
