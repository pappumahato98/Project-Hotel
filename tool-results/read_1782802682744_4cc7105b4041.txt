   169→function SourceBadge({ source }: { source: string | null }) {
   170→  if (!source) return <span className="text-xs text-muted-foreground">—</span>
   171→
   172→  const colors: Record<string, string> = {
   173→    online: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   174→    walk_in: 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-300 border-violet-300 dark:border-violet-700',
   175→    phone: 'bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   176→    email: 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-300 border-cyan-300 dark:border-cyan-700',
   177→    agent: 'bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300 border-orange-300 dark:border-orange-700',
   178→    corporate: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300 border-slate-300 dark:border-slate-600',
   179→    group: 'bg-pink-100 text-pink-700 dark:bg-pink-950 dark:text-pink-300 border-pink-300 dark:border-pink-700',
   180→    booking_com: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   181→    direct: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-300 dark:border-emerald-700',
   182→  }
   183→
   184→  const colorClass = colors[source] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300 border-gray-300 dark:border-gray-600'
   185→
   186→  return (
   187→    <Badge variant="outline" className={cn('text-[10px] px-1.5 py-0', colorClass)}>
   188→      {source.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}
   189→    </Badge>
   190→  )
   191→}

   258→// ─── Main Component ───────────────────────────────────────────────
   259→
   260→export function FrontDeskDashboard() {
   261→  const { navigateTo, setActiveSubModule } = useNavigationStore()
   262→
   263→  const handleArrivalClick = (arrival: UpcomingArrival) => {
   264→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   265→    navigateTo('front-desk', 'check-in')
   266→  }
   267→
   268→  // ─── Dashboard data query ────────────────────────────────────
   269→  const {
   270→    data,
   271→    isLoading,
   272→    error,
   273→    refetch,
   274→    isFetching,
   275→  } = useQuery<DashboardData>({
   276→    queryKey: ['front-desk-dashboard'],
   277→    queryFn: () => apiFetch('/api/front-desk/dashboard'),
   278→    refetchInterval: 30000,
   279→  })
   280→
   281→  const snapshot = data?.snapshot
   282→  const timeline = data?.timeline || []
   283→  const upcomingArrivals = data?.upcomingArrivals || []
   284→
   285→  // ─── Quick action handler ───────────────────────────────────
   286→  const handleQuickAction = (actionId: string) => {
   287→    switch (actionId) {
   288→      case 'new-reservation':
   289→        setActiveSubModule('reservations')
   290→        toast.success('Navigated to Reservations', {
   291→          description: 'Create a new booking from the Reservations tab.',
   292→        })
   293→        break
   294→      case 'walk-in-checkin':
   295→        setActiveSubModule('arrivals')
   296→        toast.success('Navigated to Arrivals', {
   297→          description: 'Use the Walk-in Check-in button on the Arrivals page.',
   298→        })
   299→        break
   300→      case 'room-transfer':
   301→        toast.info('Room Transfer', {
   302→          description: 'Go to Calendar and drag a reservation to another room.',
   303→        })
   304→        break
   305→      case 'wake-up-call':
   306→        toast.info('Wake-Up Call', {
   307→          description: 'Schedule wake-up calls from the In-House guest details.',
   308→        })
   309→        break
   310→      case 'late-checkout':
   311→        toast.info('Late Checkout Request', {
   312→          description: 'Manage late checkout requests from the In-House view.',
   313→        })
   314→        break
   315→      case 'express-checkout':
   316→        setActiveSubModule('departures')
   317→        toast.success('Navigated to Departures', {
   318→          description: 'Process express checkout from the Departures tab.',
   319→        })
   320→        break
   321→      default:
   322→        break
   323→    }
   324→  }
   325→
   326→  // ─── Determine overbooking alert visibility ──────────────────
   327→  const showOverbookingAlert = useMemo(() => {
   328→    return snapshot ? snapshot.overbookingCount > 0 : false
   329→  }, [snapshot])
   330→
   331→  // ─── Snapshot cards data ────────────────────────────────────
   332→  const snapshotCards = useMemo(() => {
   333→    if (!snapshot) return []
   334→    const cards = [
   335→      {
   336→        label: 'Arrivals Today',
   337→        value: snapshot.arrivals,
   338→        icon: LogIn,
   339→        iconBg: 'bg-emerald-100 dark:bg-emerald-950',
   340→        iconColor: 'text-emerald-600 dark:text-emerald-400',
   341→      },
   342→      {
   343→        label: 'Departures Today',
   344→        value: snapshot.departures,
   345→        icon: LogOut,
   346→        iconBg: 'bg-rose-100 dark:bg-rose-950',
   347→        iconColor: 'text-rose-600 dark:text-rose-400',
   348→      },
   349→      {
   350→        label: 'In-House Guests',
   351→        value: snapshot.inHouse,
   352→        icon: BedDouble,
   353→        iconBg: 'bg-sky-100 dark:bg-sky-950',
   354→        iconColor: 'text-sky-600 dark:text-sky-400',
   355→      },
   356→      {
   357→        label: 'Available Rooms',
   358→        value: snapshot.available,
   359→        icon: Hotel,
   360→        iconBg: 'bg-teal-100 dark:bg-teal-950',
   361→        iconColor: 'text-teal-600 dark:text-teal-400',
   362→      },
   363→      {
   364→        label: 'Occupancy',
   365→        value: `${snapshot.occupancyPct}%`,
   366→        icon: Percent,
   367→        iconBg: 'bg-amber-100 dark:bg-amber-950',
   368→        iconColor: 'text-amber-600 dark:text-amber-400',
   369→        subtext: `${snapshot.inHouse} / ${snapshot.totalRooms} rooms`,
   370→      },
   371→    ]
   372→
   373→    // Only show overbooking alert if there are overbookings
   374→    if (snapshot.overbookingCount > 0) {
   375→      cards.push({
   376→        label: 'Overbooking Alert',
   377→        value: snapshot.overbookingCount,
   378→        icon: AlertTriangle,
   379→        iconBg: 'bg-red-100 dark:bg-red-950',
   380→        iconColor: 'text-red-600 dark:text-red-400',
   381→        subtext: `${snapshot.overbookingCount} room${snapshot.overbookingCount > 1 ? 's' : ''} double-booked`,
   382→      })
   383→    }
   384→
   385→    return cards
   386→  }, [snapshot])
   387→
   388→  // ─── Render ──────────────────────────────────────────────────
   389→  return (
   390→    <div className="flex flex-col gap-2">
   391→      {/* ─── Header ─────────────────────────────────────────── */}
   392→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   393→        <div>
   394→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Front Desk Dashboard</h2>
   395→          <p className="text-xs text-muted-foreground">
   396→            Real-time operational snapshot — {formatDate(new Date())}
   397→          </p>
   398→        </div>
   399→        <Button
   400→          variant="outline"
   401→          size="sm"
   402→          className="gap-1.5 text-xs text-muted-foreground shrink-0"
   403→          onClick={() => refetch()}
   404→          disabled={isFetching}
   405→        >
   406→          <RefreshCw className={cn('size-3.5', isFetching && 'animate-spin')} />
   407→          {isFetching ? 'Refreshing...' : 'Refresh'}
   408→        </Button>
   409→      </div>
   410→
   411→      {/* ─── Today's Snapshot Cards ──────────────────────────── */}
   412→      {isLoading ? (
   413→        <SnapshotSkeleton />
   414→      ) : error ? (
   415→        <Card>
   416→          <CardContent className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   417→            <AlertTriangle className="size-8 mb-2 opacity-40" />
   418→            <p className="text-sm">Failed to load dashboard data</p>
   419→            <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   420→              Try Again
   421→            </Button>
   422→          </CardContent>
   423→        </Card>
   424→      ) : (
   425→        <div className={cn(
   426→          'grid gap-2',
   427→          showOverbookingAlert
   428→            ? 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-6'
   429→            : 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-5',
   430→        )}>
   431→          {snapshotCards.map((card) => (
   432→            <Card key={card.label} className={cn(
   433→              card.label === 'Overbooking Alert' && 'border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20',
   434→            )}>
   435→              <CardContent className="p-2.5 flex items-center gap-3">
   436→                <div className={cn('flex size-10 items-center justify-center rounded-lg shrink-0', card.iconBg)}>
   437→                  <card.icon className={cn('size-5', card.iconColor)} />
   438→                </div>
   439→                <div className="min-w-0">
   440→                  <p className="text-lg font-bold leading-tight">{card.value}</p>
   441→                  <p className="text-xs text-muted-foreground truncate">{card.label}</p>
   442→                  {'subtext' in card && card.subtext && (
   443→                    <p className="text-[10px] text-muted-foreground mt-0.5 truncate">{card.subtext}</p>
   444→                  )}
   445→                </div>
   446→              </CardContent>
   447→            </Card>
   448→          ))}
   449→        </div>
   450→      )}
   451→
   452→      {/* ─── Quick Actions Panel ─────────────────────────────── */}
   453→      <div>
   454→        <h3 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   455→          <LayoutDashboard className="size-4 text-muted-foreground" />
   456→          Quick Actions
   457→        </h3>
   458→        <div className="grid gap-3 grid-cols-2 sm:grid-cols-3 lg:grid-cols-6">
   459→          {QUICK_ACTIONS.map((action) => (
   460→            <Card
   461→              key={action.id}
   462→              className="cursor-pointer transition-all hover:shadow-md hover:border-primary/30 active:scale-[0.98]"
   463→              onClick={() => handleQuickAction(action.id)}
   464→            >
   465→              <CardContent className="p-2.5 flex flex-col items-center text-center gap-2">
   466→                <div className={cn('flex size-10 items-center justify-center rounded-lg', action.bg)}>
   467→                  <action.icon className={cn('size-5', action.color)} />
   468→                </div>
   469→                <div>
   470→                  <p className="text-xs font-semibold leading-tight">{action.title}</p>
   471→                  <p className="text-[10px] text-muted-foreground mt-0.5 leading-tight">{action.description}</p>
   472→                </div>
   473→              </CardContent>
   474→            </Card>
   475→          ))}
   476→        </div>
   477→      </div>
   478→
   479→      <Separator />
   480→
   481→      {/* ─── Bottom Grid: Timeline + Upcoming Arrivals ──────── */}
   482→      <div className="grid gap-4 lg:grid-cols-2">
   483→        {/* ─── Today's Activity Timeline ──────────────────── */}
   484→        <div>
   485→          {isLoading ? (
   486→            <TimelineSkeleton />
   487→          ) : (
   488→            <Card className="h-full">
   489→              <CardHeader className="pb-3">
   490→                <CardTitle className="text-base flex items-center gap-2">
   491→                  <Clock className="size-4 text-muted-foreground" />
   492→                  Today&apos;s Activity
   493→                </CardTitle>
   494→                <CardDescription>Recent front desk operations for today</CardDescription>
   495→              </CardHeader>
   496→              <CardContent>
   497→                {timeline.length === 0 ? (
   498→                  <div className="flex flex-col items-center justify-center py-10 text-muted-foreground">
   499→                    <Clock className="size-8 mb-2 opacity-40" />
   500→                    <p className="text-sm">No activity recorded yet today</p>
   501→                  </div>
   502→                ) : (
   503→                  <div className="max-h-96 overflow-y-auto pr-1 space-y-0">
   504→                    {timeline.map((entry, idx) => (
   505→                      <div key={entry.id} className="flex gap-3 pb-4">
   506→                        {/* Timeline line + icon */}
   507→                        <div className="flex flex-col items-center shrink-0">
   508→                          <TimelineIcon type={entry.type} />
   509→                          {idx < timeline.length - 1 && (
   510→                            <div className="w-px flex-1 bg-border mt-1" />
   511→                          )}
   512→                        </div>
   513→
   514→                        {/* Content */}
   515→                        <div className="flex-1 min-w-0 pb-1">
   516→                          <div className="flex items-center gap-2 flex-wrap">
   517→                            <span className="text-sm font-medium truncate">
   518→                              {entry.guestName}
   519→                            </span>
   520→                            <Badge
   521→                              variant="outline"
   522→                              className={cn(
   523→                                'text-[10px] px-1.5 py-0 shrink-0',
   524→                                entry.type === 'check_in'
   525→                                  ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-300 dark:border-emerald-700'
   526→                                  : entry.type === 'check_out'
   527→                                    ? 'bg-rose-50 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-300 dark:border-rose-700'
   528→                                    : entry.type === 'room_move'
   529→                                      ? 'bg-violet-50 text-violet-700 dark:bg-violet-950 dark:text-violet-300 border-violet-300 dark:border-violet-700'
   530→                                      : 'bg-sky-50 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   531→                              )}
   532→                            >
   533→                              {entry.description}
