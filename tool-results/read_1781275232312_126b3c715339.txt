    33→// ─── Types ──────────────────────────────────────────────────────────────
    34→
    35→interface DepartureGuest {
    36→  id: string
    37→  firstName: string
    38→  lastName: string
    39→  vipLevel: string
    40→}
    41→
    42→interface DepartureRoom {
    43→  id: string
    44→  number: string
    45→  floor: number
    46→  type: { name: string; code: string }
    47→}
    48→
    49→interface FolioLineItem {
    50→  id: string
    51→  description: string
    52→  amount: number
    53→  type: 'charge' | 'payment'
    54→  date: string
    55→}
    56→
    57→interface DepartureFolio {
    58→  id: string
    59→  balance: number
    60→  status: string
    61→  items?: FolioLineItem[]
    62→}
    63→
    64→interface Departure {
    65→  id: string
    66→  confirmationNo: string
    67→  status: string
    68→  checkIn: string
    69→  checkOut: string
    70→  roomRate: number
    71→  totalAmount: number
    72→  paidAmount: number
    73→  creditLimit: number
    74→  guest: DepartureGuest
    75→  room: DepartureRoom
    76→  folios: DepartureFolio[]
    77→}
    78→
    79→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    80→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    81→
    82→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    83→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    84→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    85→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    86→}
    87→
    88→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    89→  cash: 'Cash',
    90→  card: 'Credit/Debit Card',
    91→  bank_transfer: 'Bank Transfer',
    92→}
    93→
    94→// ─── Component ──────────────────────────────────────────────────────────
    95→
    96→export function DeparturesView() {
    97→  const queryClient = useQueryClient()
    98→  const { settings } = useSettingsStore()
    99→  const { preferences } = usePreferencesStore()
   100→  const today = getTodayString()
   101→
   102→  // Dialog states
   103→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   104→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   105→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   106→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   107→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   108→
   109→  // Selected departure
   110→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   111→
   112→  // Late checkout
   113→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   114→
   115→  // Payment
   116→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   117→  const [paymentAmount, setPaymentAmount] = useState('')
   118→  const [paymentReference, setPaymentReference] = useState('')
   119→
   120→  // Receipt data
   121→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   122→
   123→  // Fetch today's departures (checked_in with checkOut = today)
   124→  const { data, isLoading } = useQuery({
   125→    queryKey: ['departures', today],
   126→    queryFn: async () => {
   127→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   128→      const res = await fetch(`/api/reservations?${params.toString()}`)
   129→      if (!res.ok) throw new Error('Failed to fetch departures')
   130→      return res.json()
   131→    },
   132→    refetchInterval: 30000,
   133→  })
   134→
   135→  const departures: Departure[] = data?.reservations || []
   136→
   137→  // Stats
   138→  const totalDepartures = departures.length
   139→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   140→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   141→  const outstandingBalance = departures.reduce((sum, d) => {
   142→    return sum + (d.folios[0]?.balance || 0)
   143→  }, 0)
   144→
   145→  // Zero-balance pending guests (for express & batch checkout)
   146→  const zeroBalancePending = departures.filter(
   147→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   148→  )
   149→
   150→  // ─── Mutations ─────────────────────────────────────────────────────
   151→
   152→  // Checkout mutation
   153→  const checkoutMutation = useMutation({
   154→    mutationFn: async (reservationId: string) => {
   155→      const res = await fetch(`/api/reservations/${reservationId}`, {
   156→        method: 'PATCH',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ status: 'checked_out' }),
   159→      })
   160→      if (!res.ok) throw new Error('Failed to checkout')
   161→      return res.json()
   162→    },
   163→    onSuccess: () => {
   164→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   165→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   166→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   167→    },
   168→  })
   169→
   170→  // Payment mutation
   171→  const paymentMutation = useMutation({
   172→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   173→      const res = await fetch(`/api/folio/${folioId}`, {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   177→      })
   178→      if (!res.ok) throw new Error('Failed to record payment')
   179→      return res.json()
   180→    },
   181→    onSuccess: () => {
   182→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   183→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   184→      setPaymentDialogOpen(false)
   185→      setPaymentAmount('')
   186→      setPaymentReference('')
   187→      toast.success('Payment recorded successfully')
   188→    },
   189→    onError: () => {
   190→      toast.error('Failed to record payment')
   191→    },
   192→  })
   193→
   194→  // Late checkout mutation
   195→  const lateCheckoutMutation = useMutation({
   196→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   197→      const res = await fetch(`/api/reservations/${reservationId}`, {
   198→        method: 'PATCH',
   199→        headers: { 'Content-Type': 'application/json' },
   200→        body: JSON.stringify({ checkOut: newCheckOut }),
   201→      })
   202→      if (!res.ok) throw new Error('Failed to update checkout time')
   203→      // Add surcharge to folio
   204→      if (surcharge > 0) {
   205→        const folioRes = await fetch(`/api/folio`, {
   206→          method: 'POST',
   207→          headers: { 'Content-Type': 'application/json' },
   208→          body: JSON.stringify({
   209→            reservationId,
   210→            type: 'charge',
   211→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   212→            amount: surcharge,
   213→          }),
   214→        })
   215→        if (!folioRes.ok) throw new Error('Failed to add surcharge')
   216→      }
   217→      return res.json()
   218→    },
   219→    onSuccess: () => {
   220→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   221→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   222→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   223→      setLateCheckoutDialogOpen(false)
   224→      toast.success('Late checkout request processed successfully')
   225→    },
   226→    onError: () => {
   227→      toast.error('Failed to process late checkout request')
   228→    },
   229→  })
   230→
   231→  // Batch checkout mutation
   232→  const batchCheckoutMutation = useMutation({
   233→    mutationFn: async (ids: string[]) => {
   234→      const results = []
   235→      for (let i = 0; i < ids.length; i++) {
   236→        const res = await fetch(`/api/reservations/${ids[i]}`, {
   237→          method: 'PATCH',
   238→          headers: { 'Content-Type': 'application/json' },
   239→          body: JSON.stringify({ status: 'checked_out' }),
   240→        })
   241→        if (!res.ok) throw new Error(`Failed to checkout room ${i + 1}`)
   242→        results.push(await res.json())
   243→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   244→      }
   245→      return results
   246→    },
   247→    onSuccess: (_data, ids) => {
   248→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   249→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   250→      queryClient.invalidateQueries({ queryKey: ['in-house'] })
   251→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   252→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   253→    },
   254→    onError: () => {
   255→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   256→    },
   257→  })
   258→
   259→  // ─── Handlers ─────────────────────────────────────────────────────
   260→
   261→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   262→
   263→  const handleExpressCheckout = (dep: Departure) => {
   264→    checkoutMutation.mutate(dep.id, {
   265→      onSuccess: () => {
   266→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   267→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   268→      },
   269→    })
   270→  }
   271→
   272→  const handleReviewFolio = (dep: Departure) => {
   273→    setSelectedDeparture(dep)
   274→    setFolioDialogOpen(true)
   275→  }
   276→
   277→  const handleFolioSettleCheckout = () => {
   278→    setFolioDialogOpen(false)
   279→    setCheckoutDialogOpen(true)
   280→  }
   281→
   282→  const handleQuickCheckout = (dep: Departure) => {
   283→    setSelectedDeparture(dep)
   284→    setCheckoutDialogOpen(true)
   285→  }
   286→
   287→  const handleRequestLateCheckout = (dep: Departure) => {
   288→    setSelectedDeparture(dep)
   289→    setLateCheckoutOption('14:00')
   290→    setLateCheckoutDialogOpen(true)
   291→  }
   292→
   293→  const handleConfirmLateCheckout = () => {
   294→    if (!selectedDeparture) return
   295→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   296→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   297→
   298→    // Build new checkout datetime with the selected time
   299→    const currentCheckout = new Date(selectedDeparture.checkOut)
   300→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   301→    currentCheckout.setHours(hours, minutes, 0, 0)
   302→
   303→    lateCheckoutMutation.mutate({
   304→      reservationId: selectedDeparture.id,
   305→      newCheckOut: currentCheckout.toISOString(),
   306→      surcharge,
   307→    })
   308→  }
   309→
   310→  const handleOpenPaymentDialog = () => {
   311→    if (!selectedDeparture) return
   312→    const balance = getBalance(selectedDeparture)
   313→    setPaymentAmount(String(balance))
   314→    setPaymentMethod('cash')
   315→    setPaymentReference('')
   316→    setPaymentDialogOpen(true)
   317→  }
   318→
   319→  const handleRecordPayment = () => {
   320→    if (!selectedDeparture) return
   321→    const folioId = selectedDeparture.folios[0]?.id
   322→    if (!folioId) {
   323→      toast.error('No folio found for this reservation')
   324→      return
   325→    }
   326→    const amount = Number(paymentAmount)
   327→    if (isNaN(amount) || amount <= 0) {
   328→      toast.error('Please enter a valid payment amount')
   329→      return
   330→    }
   331→    paymentMutation.mutate({
   332→      folioId,
   333→      amount,
   334→      method: paymentMethod,
   335→      reference: paymentReference,
   336→    })
   337→  }
   338→
   339→  const confirmCheckout = () => {
   340→    if (!selectedDeparture) return
   341→    const roomNumber = selectedDeparture.room.number
   342→    checkoutMutation.mutate(selectedDeparture.id, {
   343→      onSuccess: () => {
   344→        setCheckoutDialogOpen(false)
   345→        setReceiptData(selectedDeparture)
   346→        setReceiptDialogOpen(true)
   347→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   348→      },
   349→    })
   350→  }
   351→
   352→  const handleBatchCheckout = () => {
   353→    const ids = zeroBalancePending.map((d) => d.id)
   354→    if (ids.length === 0) {
   355→      toast.info('No guests with zero balance to batch checkout')
   356→      return
   357→    }
   358→    batchCheckoutMutation.mutate(ids)
   359→  }
   360→
   361→  const handlePrintReceipt = () => {
   362→    toast.success('Receipt sent to printer')
   363→  }
   364→
   365→  const handleEmailReceipt = () => {
   366→    toast.success('Receipt emailed to guest')
   367→  }
   368→
   369→  // ─── Computed values ──────────────────────────────────────────────
   370→
   371→  const lateCheckoutFee = selectedDeparture
   372→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   373→    : 0
   374→
   375→  // ─── Render ────────────────────────────────────────────────────────
   376→
   377→  return (
   378→    <div className="flex flex-col gap-2">
   379→      {/* Header */}
   380→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   381→        <div>
   382→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   383→          <p className="text-xs text-muted-foreground">
   384→            Guest check-outs scheduled for {formatDate(today)}
   385→          </p>
   386→        </div>
   387→        {zeroBalancePending.length > 0 && (
   388→          <Button
   389→            onClick={handleBatchCheckout}
   390→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   391→            disabled={batchCheckoutMutation.isPending}
   392→          >
   393→            <Zap className="size-4 mr-1.5" />
   394→            {batchCheckoutMutation.isPending
   395→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   396→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   397→          </Button>
   398→        )}
   399→      </div>
   400→
   401→      {/* Departures List */}
   402→      <Card className="py-0">
   403→        <CardContent className="p-0">
   404→          <div className="max-h-[600px] overflow-y-auto">
   405→            <Table>
   406→              <TableHeader>
   407→                <TableRow>
   408→                  <TableHead className="w-[70px]">Room</TableHead>
   409→                  <TableHead>Guest</TableHead>
   410→                  <TableHead className="w-[100px]">Check-out</TableHead>
   411→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   412→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   413→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   414→                </TableRow>
   415→              </TableHeader>
   416→              <TableBody>
   417→                {isLoading ? (
   418→                  Array.from({ length: 5 }).map((_, i) => (
   419→                    <TableRow key={i}>
   420→                      {Array.from({ length: 6 }).map((_, j) => (
   421→                        <TableCell key={j}>
   422→                          <Skeleton className="h-4 w-full" />
   423→                        </TableCell>
   424→                      ))}
   425→                    </TableRow>
   426→                  ))
   427→                ) : departures.length === 0 ? (
   428→                  <TableRow>
   429→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   430→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   431→                      No departures scheduled for today
   432→                    </TableCell>
   433→                  </TableRow>
   434→                ) : (
   435→                  departures.map((dep) => {
   436→                    const balance = getBalance(dep)
   437→                    const isPending = dep.status === 'checked_in'
   438→                    const isZeroBalance = balance === 0
   439→                    return (
   440→                      <TableRow key={dep.id}>
   441→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   442→                        <TableCell>
   443→                          <div className="flex items-center gap-2">
   444→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   445→                            {dep.guest.vipLevel !== 'none' && (
   446→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   447→                                VIP
   448→                              </Badge>
   449→                            )}
   450→                          </div>
   451→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   452→                        </TableCell>
   453→                        <TableCell className="text-xs">
   454→                          <div>{formatDate(dep.checkOut)}</div>
   455→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   456→                        </TableCell>
   457→                        <TableCell className="text-right">
   458→                          <div className="flex items-center justify-end gap-1.5">
   459→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   460→                              {formatCurrency(balance)}
   461→                            </span>
   462→                            {/* Outstanding Balance Alert Badges */}
   463→                            {isPending && balance > 5000 && (
   464→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   465→                                HIGH
   466→                              </Badge>
   467→                            )}
   468→                            {isPending && balance > 0 && balance <= 5000 && (
   469→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   470→                                DUE
   471→                              </Badge>
   472→                            )}
   473→                          </div>
   474→                        </TableCell>
   475→                        <TableCell className="text-right">
   476→                          {dep.status === 'checked_out' ? (
   477→                            <div className="flex items-center justify-end gap-1">
   478→                              <BedSingle className="size-3 text-yellow-500" />
   479→                              <StatusBadge status="vacant_dirty" />
   480→                            </div>
   481→                          ) : (
   482→                            <StatusBadge status="checked_in" />
   483→                          )}
   484→                        </TableCell>
   485→                        <TableCell className="text-right">
   486→                          <div className="flex justify-end gap-1 flex-wrap">
   487→                            {isPending && (
   488→                              <>
   489→                                {/* Express Checkout (zero balance) */}
   490→                                {isZeroBalance && (
   491→                                  <Button
   492→                                    size="sm"
   493→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   494→                                    onClick={() => handleExpressCheckout(dep)}
   495→                                    disabled={checkoutMutation.isPending}
   496→                                  >
   497→                                    <Zap className="size-3 mr-0.5" /> Express
   498→                                  </Button>
   499→                                )}
   500→                                {/* Review Folio */}
   501→                                <Button
   502→                                  size="sm"
   503→                                  variant="outline"
   504→                                  className="text-xs h-7"
   505→                                  onClick={() => handleReviewFolio(dep)}
   506→                                >
   507→                                  <Eye className="size-3 mr-0.5" /> Review Folio
   508→                                </Button>
   509→                                {/* Late Checkout */}
   510→                                <Button
   511→                                  size="sm"
   512→                                  variant="outline"
   513→                                  className="text-xs h-7"
   514→                                  onClick={() => handleRequestLateCheckout(dep)}
   515→                                >
   516→                                  <Clock className="size-3 mr-0.5" /> Late
   517→                                </Button>
   518→                                {/* Regular Checkout */}
   519→                                <Button
   520→                                  size="sm"
   521→                                  className="text-xs h-7"
   522→                                  onClick={() => handleQuickCheckout(dep)}
   523→                                >
   524→                                  <LogOut className="size-3 mr-0.5" /> Checkout
   525→                                </Button>
   526→                              </>
   527→                            )}
   528→                            {dep.status === 'checked_out' && (
   529→                              <Badge variant="outline" className="text-xs">
   530→                                <CheckCircle2 className="size-3 mr-1" />
   531→                                Checked Out
   532→                              </Badge>
   533→                            )}
   534→                          </div>
   535→                        </TableCell>
   536→                      </TableRow>
   537→                    )
   538→                  })
   539→                )}
   540→              </TableBody>
   541→            </Table>
   542→          </div>
   543→        </CardContent>
   544→      </Card>
   545→
   546→      {/* ─── Folio Review Dialog ─────────────────────────────────── */}
   547→      <Dialog open={folioDialogOpen} onOpenChange={setFolioDialogOpen}>
   548→        <DialogContent className="max-w-lg">
   549→          <DialogHeader>
   550→            <DialogTitle>Folio Review</DialogTitle>
   551→            <DialogDescription>
   552→              {selectedDeparture && (
   553→                <>Room {selectedDeparture.room.number} — {selectedDeparture.guest.firstName} {selectedDeparture.guest.lastName}</>
   554→              )}
   555→            </DialogDescription>
