   229→   229→
   230→   230→  // ─── Derived data ──────────────────────────────────────────────

   239→   239→

   244→   244→      id: rt.id,
   245→   245→      name: rt.name,
   246→   246→      code: rt.code,

   249→   249→  }, [roomsData])
   250→   250→
   251→   251→  // Available rooms (vacant_clean or inspected)
   252→   252→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   253→   253→  const availableRooms = allRooms.filter(
   254→   254→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'

   257→   257→  // Filter available rooms by search
   258→   258→  const filteredAvailableRooms = useMemo(() => {
   259→   259→    let filtered = availableRooms

   262→   262→      filtered = filtered.filter(
   263→   263→        (r: RoomInfo) =>
   264→   264→          r.number.toLowerCase().includes(q) ||
   265→   265→          r.type.name.toLowerCase().includes(q) ||
   266→   266→          String(r.floor).includes(q) ||
   267→   267→          (r.wing && r.wing.toLowerCase().includes(q))
   268→   268→      )
   269→   269→    }
   270→   270→    return filtered
   271→   271→  }, [availableRooms, roomSearchFilter])
   272→   272→
   273→   273→  // Group rooms by floor
   274→   274→  const roomsByFloor = useMemo(() => {
   275→   275→    const grouped = new Map<number, RoomInfo[]>()
   276→   276→    for (const room of filteredAvailableRooms) {
   277→   277→      const floor = room.floor
   278→   278→      if (!grouped.has(floor)) grouped.set(floor, [])
   279→   279→      grouped.get(floor)!.push(room)
   280→   280→    }
   281→   281→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   282→   282→  }, [filteredAvailableRooms])
   283→   283→
   284→   284→  // Get room type name by ID
   285→   285→  const getRoomTypeName = (typeId: string) => {

   287→   287→    return rt?.name || 'Unknown'
   288→   288→  }
   289→   289→
   290→   290→  // Get room rate by type ID
   291→   291→  const getRoomTypeRate = (typeId: string) => {
   292→   292→    const rt = roomTypes.find((t) => t.id === typeId)
   293→   293→    return rt?.baseRate || 0
   294→   294→  }
   295→   295→
   296→   296→  // ─── Initialize form when reservation loads ───────────────────
   297→   297→
   298→   298→  React.useEffect(() => {

   300→   300→      setSelectedReservationId(specificReservation.id)
   301→   301→      if (specificReservation.room?.id) {
   302→   302→        setSelectedRoomId(specificReservation.room.id)
   303→   303→      }
   304→   304→      setCurrentStep(specificReservation.guest ? 1 : 0)
   305→   305→    }
   306→   306→  }, [specificReservation, reservationId])
   307→   307→
   308→   308→  // Pre-fill registration fields when guest is selected
   309→   309→  React.useEffect(() => {
   310→   310→    if (activeReservation?.guest) {
   311→   311→      const g = activeReservation.guest
   312→   312→      setRegTitle('')
   313→   313→      setRegFirstName(g.firstName || '')
   314→   314→      setRegLastName(g.lastName || '')
   315→   315→      setRegContactNo(g.phone || '')
   316→   316→      setRegEmail(g.email || '')
   317→   317→      setRegNationality(g.nationality || '')
   318→   318→      setRegIdType(g.idType || 'passport')
   319→   319→      setRegIdNumber(g.idNumber || '')
   320→   320→      setRegAddress(g.address || '')
   321→   321→      setRegCity(g.city || '')
   322→   322→      setRegCountry(g.country || '')
   323→   323→    }
   324→   324→  }, [activeReservation?.guest])
   325→   325→
   326→   326→  // Auto-set deposit when room is selected
   327→   327→  React.useEffect(() => {
   328→   328→    if (activeReservation?.roomRate && !depositAmount) {
   329→   329→      setDepositAmount(String(activeReservation.roomRate))
   330→   330→    }
   331→   331→  }, [activeReservation?.roomRate, depositAmount])
   332→   332→
   333→   333→  // ─── Mutations ────────────────────────────────────────────────
   334→   334→
   335→   335→  const checkInMutation = useMutation({
   336→   336→    mutationFn: async ({
   337→   337→      reservationId: resId,
   338→   338→      roomId,
   339→   339→      specialRequests,
   340→   340→    }: {
   341→   341→      reservationId: string
   342→   342→      roomId: string
   343→   343→      specialRequests: string
   344→   344→    }) => {
   345→   345→      return apiFetch(`/api/reservations/${resId}`, {
   346→   346→        method: 'PATCH',
   347→   347→        headers: { 'Content-Type': 'application/json' },
   348→   348→        body: JSON.stringify({
   349→   349→          roomId,
   350→   350→          status: 'checked_in',
   351→   351→          specialRequests,
   352→   352→        }),
   353→   353→      })
   354→   354→    },
   355→   355→    onSuccess: () => {

   357→   357→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   358→   358→      toast.success('Guest checked in successfully!')
   359→   359→    },
   360→   360→    onError: (error) => {
   361→   361→      toast.error('Check-in failed', {
   362→   362→        description: error instanceof Error ? error.message : 'Unknown error',
   363→   363→      })
   364→   364→    },
   365→   365→  })
   366→   366→
   367→   367→  // ─── Special Requests Builder ─────────────────────────────────
   368→   368→

   371→   371→
   372→   372→    if (activeReservation?.specialRequests) {
   373→   373→      parts.push(activeReservation.specialRequests)

   379→   379→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)

   381→   381→    if (roomPreference) {
   382→   382→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   383→   383→      parts.push(`[PREFERENCE] Room: ${label}`)

   385→   385→    if (pillowType) {

   389→   389→    if (wakeupCall) {

   395→   395→    if (paymentMethod) {

   399→   399→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   400→   400→    }
   401→   401→
   402→   402→    return parts.join(' | ')
   403→   403→  }
   404→   404→
   405→   405→  // ─── Step Navigation ──────────────────────────────────────────
   406→   406→
   407→   407→  const canGoNext = (): boolean => {

   411→   411→      case 2: return !!selectedRoomId
   412→   412→      case 3: return true // Preferences are optional
   413→   413→      case 4: return true // Billing is optional
   414→   414→      case 5: return keyCardIssued
   415→   415→      default: return false
   416→   416→    }
   417→   417→  }
   418→   418→
   419→   419→  const handleNext = () => {
   420→   420→    if (currentStep === 5) {
   421→   421→      onComplete?.()
   422→   422→      return
   423→   423→    }
   424→   424→
   425→   425→    // On step 4 → 5, execute check-in
   426→   426→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   427→   427→      const requests = buildSpecialRequests()
   428→   428→      checkInMutation.mutate(
   429→   429→        {
   430→   430→          reservationId: activeReservation.id,
   431→   431→          roomId: selectedRoomId,
   432→   432→          specialRequests: requests,
   433→   433→        },
   434→   434→        {
   435→   435→          onSuccess: () => {
   436→   436→            // Find the selected room number for key card
   437→   437→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   438→   438→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   439→   439→            const guestName = activeReservation.guest
   440→   440→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   441→   441→              : 'Guest'
   442→   442→
   443→   443→            setKeyCardRoomNumber(roomNumber)
   444→   444→            setKeyCardGuestName(guestName)
   445→   445→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   446→   446→            setKeyCardIssued(false)
   447→   447→            setKeyCardDialogOpen(true)
   448→   448→            setCurrentStep(5)
   449→   449→          },
   450→   450→        }
   451→   451→      )
   452→   452→      return
   453→   453→    }
   454→   454→
   455→   455→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   456→   456→  }
   457→   457→
   458→   458→  const handleBack = () => {
   459→   459→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   460→   460→  }
   461→   461→
   462→   462→  // ─── Render: Step Indicator ────────────────────────────────────
   463→   463→
   464→   464→  const renderStepIndicator = () => (
   465→   465→    <div className="mb-4">
   466→   466→      <div className="flex items-center justify-between overflow-x-auto pb-2">
