    39→// ─── Types ──────────────────────────────────────────────────────────────

    41→interface SettlementGuest {
    42→  id: string
    43→  firstName: string
    44→  lastName: string
    45→  email: string
    46→  phone: string

    50→interface SettlementRoom {
    51→  id: string
    52→  number: string
    53→  floor: number
    54→  wing: string

    58→interface SettlementFolio {
    59→  id: string
    60→  folioType: string
    61→  balance: number
    62→  status: string
    63→  payments?: { id: string; amount: number; createdAt: string }[]
    64→}
    65→
    66→interface InHouseReservation {
    67→  id: string
    68→  confirmationNo: string
    69→  guest: SettlementGuest
    70→  room: SettlementRoom
    71→  adults: number
    72→  children: number
    73→  checkIn: string
    74→  checkOut: string
    75→  roomRate: number
    76→  totalAmount: number
    77→  paidAmount: number
    78→  folios: SettlementFolio[]
    79→}

    81→type PaymentMethod = 'cash' | 'card' | 'bank_transfer' | 'digital_wallet'
    82→
    83→type BalanceFilter = 'all' | 'over_10k' | 'over_50k' | 'over_100k'

    99→// ─── Helpers ────────────────────────────────────────────────────────────
   100→
   101→function getOutstandingBalance(res: InHouseReservation): number {
   102→  return res.folios.reduce((sum, f) => sum + (f.balance > 0 ? f.balance : 0), 0)
   103→    || (res.totalAmount - res.paidAmount)
   104→}
   105→
   106→function getLastPayment(res: InHouseReservation): string | null {
   107→  const allPayments = res.folios.flatMap((f) => f.payments || [])
   108→  if (allPayments.length === 0) return null
   109→  const latest = allPayments.reduce((a, b) =>
   110→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   111→  )
   112→  return new Date(latest.createdAt).toLocaleDateString()
   113→}
   114→
   115→function getVipBadge(vipLevel: string) {
   116→  if (vipLevel === 'none' || !vipLevel) return null
   117→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   118→  return (
   119→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   120→      {level}
   121→    </Badge>
   122→  )
   123→}

   157→      const next = new Set(prev)

   168→  const handlePrint = () => {

   171→

   174→  const { data, isLoading } = useQuery({

   185→

   191→    // Search filter

   193→      const q = searchQuery.toLowerCase().trim()

   195→        const guestName = r.guest ? `${r.guest.firstName} ${r.guest.lastName}`.toLowerCase() : ''
   196→        const roomNumber = r.room ? r.room.number.toLowerCase() : ''

   198→        return guestName.includes(q) || roomNumber.includes(q) || confirmationNo.includes(q)

   202→    // Balance filter

   211→        result = result.filter((r) => getOutstandingBalance(r) > 100000)
   212→        break

   219→    (sum, r) => sum + getOutstandingBalance(r),
   220→    0
   221→  )
   222→  const guestCount = filteredReservations.length
   223→

   225→  const allOutstanding = reservations.filter((r) => getOutstandingBalance(r) > 0)

   228→    0
   229→  )
   230→

   232→
   233→  // Single payment mutation — uses existing API pattern: POST /api/folio/{folioId}

   235→    mutationFn: async ({

   238→      method,
   239→      reference,

   242→      amount: number

   244→      reference: string

   248→        headers: { 'Content-Type': 'application/json' },

   252→    onSuccess: () => {
   253→      invalidate.afterFolioChange(queryClient)
   254→      setSettleDialogOpen(false)
   255→      setPaymentAmount('')

   258→    },

   260→      toast.error('Failed to process payment. Please try again.')
   261→    },

   269→        const res = allOutstanding[i]

   271→        const foliosToSettle = res.folios.filter((f) => f.balance > 0)
   272→        if (foliosToSettle.length === 0) {
   273→          results.push({ reservation: res, success: false, error: 'No folio with balance found' })
   274→          continue
   275→        }
   276→        let allSuccess = true
   277→        for (const folio of foliosToSettle) {
   278→          try {
   279→            await apiFetch(`/api/folio/${folio.id}`, {
   280→              method: 'POST',
   281→              headers: { 'Content-Type': 'application/json' },
   282→              body: JSON.stringify({ type: 'payment', paymentMethod: method, amount: folio.balance, reference: '' }),
   283→            })
   284→          } catch {
   285→            allSuccess = false
   286→          }
   287→        }
   288→        if (allSuccess) {
   289→          results.push({ reservation: res, success: true })
   290→          toast.success(`Settled ${i + 1} of ${allOutstanding.length}: Room ${res.room.number}`)
   291→        } else {
   292→          results.push({ reservation: res, success: false, error: 'Payment failed on one or more folios' })
   293→        }
   294→      }
   295→      return results
   296→    },

   298→      invalidate.afterFolioChange(queryClient)
   299→      setBatchDialogOpen(false)
   300→      const successCount = results.filter((r) => r.success).length

   302→      if (failCount === 0) {
   303→        toast.success(`All ${successCount} accounts settled successfully!`)
   304→      } else {
   305→        toast.warning(`Settled ${successCount} accounts. ${failCount} failed.`)
   306→      }
   307→    },
   308→    onError: () => {
   309→      toast.error('Batch settlement failed. Please try again.')
   310→    },
   311→  })

   335→    if (isNaN(amount) || amount <= 0) {
   336→      toast.error('Please enter a valid payment amount')
   337→      return
   338→    }
   339→    paymentMutation.mutate({
   340→      folioId,
   341→      amount,
   342→      method: paymentMethod,
   343→      reference: paymentReference,
   344→    })

   348→    if (allOutstanding.length === 0) {
   349→      toast.info('No outstanding balances to settle')

   353→    setBatchDialogOpen(true)

   363→    } else {
   364→      setSelectedIds(new Set(filteredReservations.map((r) => r.id)))

   379→    const blob = new Blob([csv], { type: 'text/csv' })
   380→    const url = URL.createObjectURL(blob)
   381→    const a = document.createElement('a')
   382→    a.href = url
   383→    a.download = 'settlement.csv'
   384→    a.click()
   385→    URL.revokeObjectURL(url)
   386→  }
   387→

   392→    const first = filteredReservations.find((r) => r.id === ids[0])

   395→      const balance = getOutstandingBalance(first)
   396→      setPaymentAmount(String(balance))
   397→      setPaymentMethod('cash')
   398→      setPaymentReference('')
   399→      setSettleDialogOpen(true)

   409→      confirmationNo: res.confirmationNo,

   412→  }
   413→
   414→  // ─── Render ─────────────────────────────────────────────────────────
   415→
   416→  const selectedBalance = selectedReservation
   417→    ? (selectedFolioId
   418→        ? selectedReservation.folios.find((f) => f.id === selectedFolioId)?.balance ?? 0
   419→        : getOutstandingBalance(selectedReservation))
   420→    : 0

   423→    <div className="flex flex-col gap-2">
   424→      {/* Header */}
   425→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   426→        <div>
   427→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">
   428→            Settlement
   429→          </h2>
   430→          <p className="text-xs text-muted-foreground">

   432→          </p>
   433→        </div>

   469→            </Button>
   470→          )}
   471→        </div>
   472→      </div>
   473→

   479→              <Wallet className="size-5 text-red-600 dark:text-red-400" />

   501→          <CardContent className="p-4 flex items-center gap-3">
   502→            <div className="flex items-center justify-center size-10 rounded-lg bg-green-100 dark:bg-green-950">
   503→              <Banknote className="size-5 text-green-600 dark:text-green-400" />

   543→            </Select>

   545→        </CardContent>
   546→      </Card>

   548→      {/* Outstanding Accounts Table */}
   549→      <div className="relative">

   551→        <CardContent className="p-0">
