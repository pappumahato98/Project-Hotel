   180→// ─── Debounce hook ────────────────────────────────────────────────────
   181→function useDebounce<T>(value: T, delay: number): T {
   182→  const [debounced, setDebounced] = useState(value)
   183→  useEffect(() => {
   184→    const timer = setTimeout(() => setDebounced(value), delay)
   185→    return () => clearTimeout(timer)
   186→  }, [value, delay])
   187→  return debounced
   188→}

   190→// ─── Date range presets ──────────────────────────────────────────────
   191→function getDatePreset(preset: string): { from: Date; to: Date } {
   192→  const today = new Date()
   193→  today.setHours(0, 0, 0, 0)
   194→  switch (preset) {
   195→    case 'today':
   196→      return { from: today, to: today }
   197→    case 'this-week': {
   198→      const from = new Date(today)
   199→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   200→      const to = new Date(from)
   201→      to.setDate(from.getDate() + 6) // Sunday
   202→      return { from, to }
   203→    }
   204→    case 'this-month': {
   205→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   206→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   207→      return { from, to }
   208→    }
   209→    case 'next-7': {
   210→      const to = new Date(today)
   211→      to.setDate(today.getDate() + 6)
   212→      return { from: today, to }
   213→    }
   214→    case 'next-14': {
   215→      const to = new Date(today)
   216→      to.setDate(today.getDate() + 13)
   217→      return { from: today, to }
   218→    }
   219→    case 'next-30': {
   220→      const to = new Date(today)
   221→      to.setDate(today.getDate() + 29)
   222→      return { from: today, to }
   223→    }
   224→    default:
   225→      return { from: today, to: today }
   226→  }
   227→}

   229→function toDateString(d: Date): string {
   230→  const y = d.getFullYear()
   231→  const m = String(d.getMonth() + 1).padStart(2, '0')
   232→  const day = String(d.getDate()).padStart(2, '0')
   233→  return `${y}-${m}-${day}`
   234→}
   235→
   236→function formatShortDate(dateStr: string): string {
   237→  if (!dateStr) return ''
   238→  const d = new Date(dateStr + 'T00:00:00')
   239→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   240→}
