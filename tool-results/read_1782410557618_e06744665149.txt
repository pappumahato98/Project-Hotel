    63→    63→// ─── Quick Search Dialog ────────────────────────────────────────────
    64→    64→function QuickSearchDialog() {
    65→    65→  const { searchOpen, setSearchOpen, navigateTo } = useNavigationStore()
    66→    66→  const [query, setQuery] = React.useState('')
    67→    67→
    68→    68→  const filteredItems = React.useMemo(() => {
    69→    69→    if (!query.trim()) return NAV_ITEMS
    70→    70→    const lower = query.toLowerCase()
    71→    71→    return NAV_ITEMS.filter(
    72→    72→      (item) =>
    73→    73→        item.label.toLowerCase().includes(lower) ||
    74→    74→        item.children?.some((c) => c.label.toLowerCase().includes(lower))
    75→    75→    )
    76→    76→  }, [query])
    77→    77→
    78→    78→  React.useEffect(() => {
    79→    79→    const down = (e: KeyboardEvent) => {
    80→    80→      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
    81→    81→        e.preventDefault()
    82→    82→        setSearchOpen(!searchOpen)
    83→    83→      }
    84→    84→    }
    85→    85→    document.addEventListener('keydown', down)
    86→    86→    return () => document.removeEventListener('keydown', down)
    87→    87→  }, [searchOpen, setSearchOpen])
    88→    88→
    89→    89→  return (
    90→    90→    <CommandDialog open={searchOpen} onOpenChange={setSearchOpen}>
    91→    91→      <CommandInput
    92→    92→        placeholder="Search modules, pages..."
    93→    93→        value={query}
    94→    94→        onValueChange={setQuery}
    95→    95→      />
    96→    96→      <CommandList>
    97→    97→        <CommandEmpty>No results found.</CommandEmpty>
    98→    98→        {filteredItems.map((item) => (
    99→    99→          <React.Fragment key={item.id}>
   100→   100→            <CommandGroup heading={item.label}>
   101→   101→              <CommandItem
   102→   102→                onSelect={() => {
   103→   103→                  navigateTo(item.id)
   104→   104→                  setSearchOpen(false)
   105→   105→                  setQuery('')
   106→   106→                }}
   107→   107→              >
   108→   108→                <item.icon className="mr-2 size-4 text-muted-foreground" />
   109→   109→                <span>{item.label}</span>
   110→   110→              </CommandItem>
   111→   111→              {item.children?.map((child) => (
   112→   112→                <CommandItem
   113→   113→                  key={child.id}
   114→   114→                  onSelect={() => {
   115→   115→                    navigateTo(item.id, child.id)
   116→   116→                    setSearchOpen(false)
   117→   117→                    setQuery('')
   118→   118→                  }}
   119→   119→                  className="pl-8"
   120→   120→                >
   121→   121→                  <span>{child.label}</span>
   122→   122→                </CommandItem>
   123→   123→              ))}
   124→   124→            </CommandGroup>
   125→   125→            <CommandSeparator />
   126→   126→          </React.Fragment>
   127→   127→        ))}
   128→   128→      </CommandList>
   129→   129→    </CommandDialog>
   130→   130→  )
   131→   131→}
