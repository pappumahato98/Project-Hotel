   157→
   158→export function CheckInView({ reservationId, onComplete }: CheckInViewProps) {
   159→  const queryClient = useQueryClient()
   160→  const { settings } = useSettingsStore()
   161→  const today = getTodayString()
   162→
   163→  // ─── Step state ────────────────────────────────────────────────
   164→  const [currentStep, setCurrentStep] = useState(0)
   165→
   166→  // ─── Selection states ────────────────────────────────────────
   167→  const [selectedReservationId, setSelectedReservationId] = useState<string>('')
   168→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   169→  const [roomSearchFilter, setRoomSearchFilter] = useState('')
   170→
   171→  // ─── Registration card form ────────────────────────────────────
   172→  const [regTitle, setRegTitle] = useState('')
   173→  const [regFirstName, setRegFirstName] = useState('')
   174→  const [regLastName, setRegLastName] = useState('')
   175→  const [regContactNo, setRegContactNo] = useState('')
   176→  const [regEmail, setRegEmail] = useState('')
   177→  const [regAddress, setRegAddress] = useState('')
   178→  const [regNationality, setRegNationality] = useState('')
   179→  const [regIdType, setRegIdType] = useState('passport')
   180→  const [regIdNumber, setRegIdNumber] = useState('')
   181→  const [regCity, setRegCity] = useState('')
   182→  const [regCountry, setRegCountry] = useState('')
   183→
   184→  // ─── Stay preferences ─────────────────────────────────────────
   185→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   186→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   187→  const [pillowType, setPillowType] = useState('')
   188→  const [roomPreference, setRoomPreference] = useState('')
   189→  const [wakeupCall, setWakeupCall] = useState('')
   190→
   191→  // ─── Billing ──────────────────────────────────────────────────
   192→  const [depositAmount, setDepositAmount] = useState('')
   193→  const [paymentMethod, setPaymentMethod] = useState('')
   194→  const [billingNotes, setBillingNotes] = useState('')
   195→
   196→  // ─── Key card dialog ───────────────────────────────────────────
   197→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   198→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   199→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   200→  const [keyCardGuestName, setKeyCardGuestName] = useState('')
   201→  const [keyCardCheckOut, setKeyCardCheckOut] = useState('')
   202→
   203→  // ─── Queries ─────────────────────────────────────────────────
   204→
   205→  // Fetch specific reservation by ID (if prop provided)
   206→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   207→    queryKey: ['reservation', reservationId],
   208→    queryFn: () => apiFetch(`/api/reservations/${reservationId}`),
   209→    enabled: !!reservationId,
   210→  })
   211→
   212→  // Fetch today's confirmed arrivals
   213→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   214→    queryKey: ['arrivals', today],
   215→    queryFn: async () => {
   216→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   217→      return apiFetch(`/api/reservations?${params.toString()}`)
   218→    },
   219→    enabled: !reservationId,
   220→  })
   221→
   222→  // Fetch all rooms with types
   223→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   224→    queryKey: ['rooms', 'all'],
   225→    queryFn: () => apiFetch('/api/rooms'),
   226→  })
   227→
   228→  // ─── Derived data ──────────────────────────────────────────────
   229→
   230→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   231→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   232→
   233→  // The reservation we're working with
   234→  const activeReservation: ReservationInfo | null = reservationId
   235→    ? specificReservation
   236→    : arrivals.find((a) => a.id === selectedReservationId) || null
   237→
   238→  // Room type data from API (not hardcoded)
   239→  const roomTypes: RoomTypeData[] = useMemo(() => {
   240→    if (!roomsData?.roomTypes) return []
   241→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   242→      id: rt.id,
   243→      name: rt.name,
   244→      code: rt.code,
   245→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   246→    }))
   247→  }, [roomsData])
   248→
   249→  // Available rooms (vacant_clean or inspected)
   250→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   251→  const availableRooms = allRooms.filter(
   252→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   253→  )
   254→
   255→  // Filter available rooms by search
   256→  const filteredAvailableRooms = useMemo(() => {
   257→    let filtered = availableRooms
   258→    if (roomSearchFilter.trim()) {
   259→      const q = roomSearchFilter.toLowerCase()
   260→      filtered = filtered.filter(
   261→        (r: RoomInfo) =>
   262→          r.number.toLowerCase().includes(q) ||
   263→          r.type.name.toLowerCase().includes(q) ||
   264→          String(r.floor).includes(q) ||
   265→          (r.wing && r.wing.toLowerCase().includes(q))
   266→      )
   267→    }
   268→    return filtered
   269→  }, [availableRooms, roomSearchFilter])
   270→
   271→  // Group rooms by floor
   272→  const roomsByFloor = useMemo(() => {
   273→    const grouped = new Map<number, RoomInfo[]>()
   274→    for (const room of filteredAvailableRooms) {
   275→      const floor = room.floor
   276→      if (!grouped.has(floor)) grouped.set(floor, [])
   277→      grouped.get(floor)!.push(room)
   278→    }
   279→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   280→  }, [filteredAvailableRooms])
   281→
   282→  // Get room type name by ID
   283→  const getRoomTypeName = (typeId: string) => {
   284→    const rt = roomTypes.find((t) => t.id === typeId)
   285→    return rt?.name || 'Unknown'
   286→  }
   287→
   288→  // Get room rate by type ID
   289→  const getRoomTypeRate = (typeId: string) => {
   290→    const rt = roomTypes.find((t) => t.id === typeId)
   291→    return rt?.baseRate || 0
   292→  }
   293→
   294→  // ─── Initialize form when reservation loads ───────────────────
   295→
   296→  React.useEffect(() => {
   297→    if (specificReservation && reservationId) {
   298→      setSelectedReservationId(specificReservation.id)
   299→      if (specificReservation.room?.id) {
   300→        setSelectedRoomId(specificReservation.room.id)
   301→      }
   302→      setCurrentStep(specificReservation.guest ? 1 : 0)
   303→    }
   304→  }, [specificReservation, reservationId])
   305→
   306→  // Pre-fill registration fields when guest is selected
   307→  React.useEffect(() => {
   308→    if (activeReservation?.guest) {
   309→      const g = activeReservation.guest
   310→      setRegTitle('')
   311→      setRegFirstName(g.firstName || '')
   312→      setRegLastName(g.lastName || '')
   313→      setRegContactNo(g.phone || '')
   314→      setRegEmail(g.email || '')
   315→      setRegNationality(g.nationality || '')
   316→      setRegIdType(g.idType || 'passport')
   317→      setRegIdNumber(g.idNumber || '')
   318→      setRegAddress(g.address || '')
   319→      setRegCity(g.city || '')
   320→      setRegCountry(g.country || '')
   321→    }
   322→  }, [activeReservation?.guest])
   323→
   324→  // Auto-set deposit when room is selected
   325→  React.useEffect(() => {
   326→    if (activeReservation?.roomRate && !depositAmount) {
   327→      setDepositAmount(String(activeReservation.roomRate))
   328→    }
   329→  }, [activeReservation?.roomRate, depositAmount])
   330→
   331→  // ─── Mutations ────────────────────────────────────────────────
   332→
   333→  const checkInMutation = useMutation({
   334→    mutationFn: async ({
   335→      reservationId: resId,
   336→      roomId,
   337→      specialRequests,
   338→    }: {
   339→      reservationId: string
   340→      roomId: string
   341→      specialRequests: string
   342→    }) => {
   343→      return apiFetch(`/api/reservations/${resId}`, {
   344→        method: 'PATCH',
   345→        headers: { 'Content-Type': 'application/json' },
   346→        body: JSON.stringify({
   347→          roomId,
   348→          status: 'checked_in',
   349→          specialRequests,
   350→        }),
   351→      })
   352→    },
   353→    onSuccess: () => {
   354→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   355→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   356→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   357→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   358→      toast.success('Guest checked in successfully!')
   359→    },
   360→    onError: (error) => {
   361→      toast.error('Check-in failed', {
   362→        description: error instanceof Error ? error.message : 'Unknown error',
   363→      })
   364→    },
   365→  })
   366→
   367→  // ─── Special Requests Builder ─────────────────────────────────
   368→
   369→  const buildSpecialRequests = (): string => {
   370→    const parts: string[] = []
   371→
   372→    if (activeReservation?.specialRequests) {
   373→      parts.push(activeReservation.specialRequests)
   374→    }
   375→    if (earlyCheckIn) {
   376→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   377→    }
   378→    if (checkOutTime !== settings.defaultCheckOut) {
   379→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   380→    }
   381→    if (roomPreference) {
   382→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   383→      parts.push(`[PREFERENCE] Room: ${label}`)
   384→    }
   385→    if (pillowType) {
   386→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   387→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   388→    }
   389→    if (wakeupCall) {
   390→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   391→    }
   392→    if (billingNotes) {
   393→      parts.push(`[BILLING] ${billingNotes}`)
   394→    }
   395→    if (paymentMethod) {
   396→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   397→    }
   398→    if (depositAmount && parseFloat(depositAmount) > 0) {
   399→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   400→    }
   401→
   402→    return parts.join(' | ')
   403→  }
   404→
   405→  // ─── Step Navigation ──────────────────────────────────────────
   406→
   407→  const canGoNext = (): boolean => {
   408→    switch (currentStep) {
   409→      case 0: return !!activeReservation
   410→      case 1: return true // Guest verification is info-only
   411→      case 2: return !!selectedRoomId
   412→      case 3: return true // Preferences are optional
   413→      case 4: return true // Billing is optional
   414→      case 5: return keyCardIssued
   415→      default: return false
   416→    }
   417→  }
   418→
   419→  const handleNext = () => {
   420→    if (currentStep === 5) {
   421→      onComplete?.()
   422→      return
   423→    }
   424→
   425→    // On step 4 → 5, execute check-in
   426→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   427→      const requests = buildSpecialRequests()
   428→      checkInMutation.mutate(
   429→        {
   430→          reservationId: activeReservation.id,
   431→          roomId: selectedRoomId,
   432→          specialRequests: requests,
   433→        },
   434→        {
   435→          onSuccess: () => {
   436→            // Find the selected room number for key card
   437→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   438→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   439→            const guestName = activeReservation.guest
   440→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   441→              : 'Guest'
   442→
   443→            setKeyCardRoomNumber(roomNumber)
   444→            setKeyCardGuestName(guestName)
   445→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   446→            setKeyCardIssued(false)
   447→            setKeyCardDialogOpen(true)
   448→            setCurrentStep(5)
   449→          },
   450→        }
   451→      )
   452→      return
   453→    }
   454→
   455→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   456→  }
   457→
   458→  const handleBack = () => {
   459→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   460→  }
   461→
   462→  // ─── Render: Step Indicator ────────────────────────────────────
   463→
   464→  const renderStepIndicator = () => (
   465→    <div className="mb-4">
   466→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   467→        {STEP_LABELS.map((label, idx) => (
   468→          <React.Fragment key={idx}>
   469→            <div className="flex flex-col items-center gap-1 min-w-0">
   470→              <div
   471→                className={cn(
   472→                  'flex size-7 shrink-0 items-center justify-center rounded-full border-2 text-[10px] font-bold transition-colors',
   473→                  idx < currentStep && 'border-green-500 bg-green-500 text-white',
   474→                  idx === currentStep && 'border-primary bg-primary text-primary-foreground',
   475→                  idx > currentStep && 'border-muted-foreground/30 bg-background text-muted-foreground'
   476→                )}
   477→              >
   478→                {idx < currentStep ? <Check className="size-4" /> : idx + 1}
   479→              </div>
   480→              <span
   481→                className={cn(
   482→                  'text-[10px] sm:text-xs font-medium whitespace-nowrap',
   483→                  idx === currentStep ? 'text-primary' : 'text-muted-foreground'
   484→                )}
   485→              >
   486→                {label}
   487→              </span>
   488→            </div>
   489→            {idx < STEP_LABELS.length - 1 && (
   490→              <div
   491→                className={cn(
   492→                  'flex-1 h-0.5 mx-1 min-w-[12px] max-w-[40px] transition-colors',
   493→                  idx < currentStep ? 'bg-green-500' : 'bg-muted-foreground/20'
   494→                )}
   495→              />
   496→            )}
   497→          </React.Fragment>
   498→        ))}
   499→      </div>
   500→    </div>
   501→  )
   502→
   503→  // ─── Render: Step 0 - Select Arrival ─────────────────────────
   504→
   505→  const renderSelectArrival = () => {
   506→    if (loadingArrivals) {
   507→      return (
   508→        <div className="space-y-3 p-4">
   509→          {Array.from({ length: 4 }).map((_, i) => (
   510→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   511→          ))}
   512→        </div>
   513→      )
   514→    }
   515→
   516→    if (arrivals.length === 0) {
   517→      return (
   518→        <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   519→          <UserCheck className="size-12 mb-3 opacity-30" />
   520→          <p className="text-sm font-medium">No arrivals scheduled for today</p>
   521→          <p className="text-xs mt-1">Check-in date: {formatDate(today)}</p>
   522→        </div>
   523→      )
   524→    }
   525→
   526→    return (
   527→      <div className="space-y-2">
   528→        <div className="flex items-center gap-2 mb-3">
   529→          <Search className="size-4 text-muted-foreground" />
   530→          <p className="text-sm text-muted-foreground">
   531→            {arrivals.length} confirmed arrival{arrivals.length !== 1 ? 's' : ''} for today
   532→          </p>
   533→        </div>
   534→        <div className="max-h-96 overflow-y-auto space-y-2">
   535→          {arrivals.map((arrival) => {
   536→            const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   537→            const isUnassigned = !arrival.room
   538→            const isSelected = arrival.id === selectedReservationId
   539→
   540→            return (
   541→              <button
   542→                key={arrival.id}
   543→                onClick={() => {
   544→                  setSelectedReservationId(arrival.id)
   545→                  // Pre-fill room if already assigned
   546→                  if (arrival.room?.id) {
   547→                    setSelectedRoomId(arrival.room.id)
   548→                  }
   549→                  setCurrentStep(1)
   550→                }}
   551→                className={cn(
   552→                  'w-full text-left rounded-lg border p-3 sm:p-4 transition-all',
   553→                  isSelected && 'ring-2 ring-primary',
   554→                  isSelected
   555→                    ? 'border-primary bg-primary/5'
   556→                    : 'border-border hover:border-primary/40 hover:bg-muted/50',
   557→                  isUnassigned && 'border-l-4 border-l-amber-400'
   558→                )}
   559→              >
   560→                <div className="flex items-start gap-3">
