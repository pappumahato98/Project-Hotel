   106→function NumberStepper({ label, value, min, max, onChange }: {
   107→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   108→}) {
   109→  return (
   110→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   111→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   112→      <div className="flex items-center gap-0 ml-auto">
   113→        <button
   114→          type="button"
   115→          onClick={() => onChange(Math.max(min, value - 1))}
   116→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   117→        >
   118→          <Minus className="w-3 h-3" />
   119→        </button>
   120→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   121→        <button
   122→          type="button"
   123→          onClick={() => onChange(Math.min(max, value + 1))}
   124→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   125→        >
   126→          <Plus className="w-3 h-3" />
   127→        </button>
   128→      </div>
   129→    </div>
   130→  )
   131→}

   133→function sourceLabel(source: string | null) {
   134→  switch (source) {
   135→    case 'walk_in': return 'Walk-in'
   136→    case 'online': return 'Online'
   137→    case 'phone': return 'Phone'
   138→    case 'travel_agent': return 'Travel Agent'
   139→    case 'corporate': return 'Corporate'
   140→    default: return source || 'Direct'
   141→  }
   142→}

   171→  const [guestPhone, setGuestPhone] = useState('')
   172→  const [guestNationality, setGuestNationality] = useState('')
   173→

   198→  })
   199→
   200→  // Guest stay history

   202→    queryKey: ['guest-stays', reservationData?.guest?.id],
   203→    queryFn: async () => {
   204→      if (!reservationData?.guest?.id) return null

   206→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>

   215→  const {
   216→    data: arrivalsData,
   217→    isLoading: arrivalsLoading,
   218→  } = useQuery({
   219→    queryKey: ['arrivals', today],
   220→    queryFn: async () => {
   221→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   222→      return data.reservations || []
   223→    },
   224→    enabled: !isDirectWalkIn && !prefillReservationId,
   225→  })
   226→
   227→  // Reservation search (debounced)
   228→  const {
   229→    data: searchResults,

   232→    queryKey: ['reservation-search', debouncedQuery],

   238→    enabled: debouncedQuery.length >= 2,
   239→  })
   240→
   241→  // Handlers ────────────────────────────────────────────────────────-----------------------------------------------
   242→

   252→      reservationData: reservationData as unknown as Record<string, unknown>,

   254→      walkInGuest: null,
   255→      walkInDates: null,
   256→      startAtStep: 1,

   264→      reservationData: res as unknown as Record<string, unknown>,
   265→      isDirectWalkIn: false,
   266→      walkInGuest: null,
   267→      walkInDates: null,
   268→      startAtStep: 2,

   272→
   273→  const handleWalkInContinue = useCallback(() => {

   276→      return
   277→    }

   279→      reservationId: null,
   280→      reservationData: null,
   281→      isDirectWalkIn: true,

   283→        firstName: guestFirstName,
   284→        lastName: guestLastName,
   285→        email: guestEmail,
   286→        phone: guestPhone,
   287→        nationality: guestNationality,
   288→      },
   289→      walkInDates: {
   290→        checkIn: walkInDate,
   291→        checkOut: walkInCheckOut,
   292→        adults: walkInAdults,
   293→        children: walkInChildren,
   294→      },
   295→      startAtStep: 1,
   296→    })

   298→

   326→              {res.source && (
   327→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   328→                  {sourceLabel(res.source)}
   329→                </Badge>

   332→            <p className="text-sm font-medium truncate">
   333→              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   334→            </p>

   336→              {res.room && <span>Room {res.room.number}</span>}
   337→              <span>·</span>
   338→              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   339→              <span>·</span>

   357→                <Zap className="w-3 h-3 mr-1" />
   358→                Express
   359→              </Button>

   361→          </div>
   362→        </div>
   363→      </Card>

   372→
   373→          {/* Prefilled reservation mode */}
   374→          {prefillReservationId ? (

   376→              {/* Header for prefill mode */}

   685→                              <CalendarDays className="mr-2 h-3.5 w-3.5" />
