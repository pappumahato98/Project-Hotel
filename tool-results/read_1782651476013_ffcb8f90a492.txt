    41→// Use shared formatDateLong and formatDateTime from @/lib/format (en-GB/DD-MM-YYYY)
    42→// Re-export for convenience since component references them as formatDate/formatDateTime
    43→const formatDate = (dateStr: string | null | undefined): string => {
    44→  if (!dateStr) return '—'
    45→  try {
    46→    const d = new Date(dateStr)
    47→    if (isNaN(d.getTime())) return dateStr ?? ''
    48→    return d.toLocaleDateString('en-GB', { year: 'numeric', month: 'long', day: 'numeric' })
    49→  } catch {
    50→    return dateStr ?? ''
    51→  }
    52→}
    53→
    54→const formatDateTime = (dateStr: string | null | undefined): string => {
    55→  if (!dateStr) return '—'
    56→  try {
    57→    const d = new Date(dateStr)
    58→    if (isNaN(d.getTime())) return dateStr ?? ''
    59→    return d.toLocaleDateString('en-GB', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
    60→  } catch {
    61→    return dateStr ?? ''
    62→  }
    63→}
    64→
    65→function toDateInputValue(dateStr: string | null | undefined): string {
    66→  if (!dateStr) return ''
    67→  try {
    68→    const d = new Date(dateStr)
    69→    if (isNaN(d.getTime())) return ''
    70→    return d.toISOString().split('T')[0]
    71→  } catch {
    72→    return ''
    73→  }
    74→}

   673→    setPasswords((prev) => ({ ...prev, [field]: value }))

   676→  const toggleShowPassword = (field: 'current' | 'new' | 'confirm') => {

   682→      toast.error('Please fill in all password fields')

   684→    }
   685→    if (passwords.newPassword.length < 8) {
   686→      toast.error('New password must be at least 8 characters')

   688→    }
   689→    if (passwords.newPassword !== passwords.confirmPassword) {
   690→      toast.error('New passwords do not match')

   692→    }

   704→          email: user?.email,
   705→          currentPassword: passwords.currentPassword,
   706→          newPassword: passwords.newPassword,

   726→    },

   746→          action: 'Clear Sessions',
   747→          module: 'Security',
   748→          details: 'User cleared all sessions and logged out',
   749→        }),
   750→      })

   765→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-rose-100 text-rose-600">

   767→            </div>

   770→              <CardDescription>Update your account password regularly for security</CardDescription>
   771→            </div>
   772→          </div>
   773→        </CardHeader>
   774→        <CardContent className="space-y-4">
