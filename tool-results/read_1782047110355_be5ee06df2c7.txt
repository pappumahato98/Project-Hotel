    97→    97→// ─── Component ──────────────────────────────────────────────────────────
    98→    98→
    99→    99→export function DeparturesView() {
   100→   100→  const queryClient = useQueryClient()
   101→   101→  const { settings } = useSettingsStore()
   102→   102→  const { preferences } = usePreferencesStore()
   103→   103→  const { navigateTo } = useNavigationStore()
   104→   104→  const today = getTodayString()
   105→   105→
   106→   106→  // Dialog states
   107→   107→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   108→   108→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   109→   109→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   110→   110→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   111→   111→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   112→   112→
   113→   113→  // Selected departure
   114→   114→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   115→   115→
   116→   116→  // Late checkout
   117→   117→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   118→   118→
   119→   119→  // Payment
   120→   120→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   121→   121→  const [paymentAmount, setPaymentAmount] = useState('')
   122→   122→  const [paymentReference, setPaymentReference] = useState('')
   123→   123→
   124→   124→  // Receipt data
   125→   125→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   126→   126→
   127→   127→  // Solo Settlement dialog state
   128→   128→  const [soloDialogOpen, setSoloDialogOpen] = useState(false)
   129→   129→  const [soloDeparture, setSoloDeparture] = useState<Departure | null>(null)
   130→   130→
   131→   131→  // Fetch today's departures: checked_in with checkOut=today, AND checked_out today
   132→   132→  // Uses status=checked_in,checked_out with checkOutDate filter
   133→   133→  const { data, isLoading } = useQuery({
   134→   134→    queryKey: ['departures', today],
   135→   135→    queryFn: async () => {
   136→   136→      const params = new URLSearchParams({
   137→   137→        status: 'checked_in,checked_out',
   138→   138→        checkOutDate: today,
   139→   139→      })
   140→   140→      return apiFetch(`/api/reservations?${params.toString()}`)
   141→   141→    },
   142→   142→    refetchInterval: 30000,
   143→   143→  })
   144→   144→
   145→   145→  const departures: Departure[] = data?.reservations || []
   146→   146→
   147→   147→  // Stats
   148→   148→  const totalDepartures = departures.length
   149→   149→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   150→   150→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   151→   151→  const outstandingBalance = departures.reduce((sum, d) => {
   152→   152→    return sum + (d.folios[0]?.balance || 0)
   153→   153→  }, 0)
   154→   154→
   155→   155→  // Zero-balance pending guests (for express & batch checkout)
   156→   156→  const zeroBalancePending = departures.filter(
   157→   157→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   158→   158→  )
   159→   159→
   160→   160→  // ─── Mutations ─────────────────────────────────────────────────────
   161→   161→
   162→   162→  // Checkout mutation
   163→   163→  const checkoutMutation = useMutation({
   164→   164→    mutationFn: async (reservationId: string) => {
   165→   165→      return apiFetch(`/api/reservations/${reservationId}`, {
   166→   166→        method: 'PATCH',
   167→   167→        headers: { 'Content-Type': 'application/json' },
   168→   168→        body: JSON.stringify({ status: 'checked_out' }),
   169→   169→      })
   170→   170→    },
   171→   171→    onSuccess: () => {
   172→   172→      invalidate.afterCheckout(queryClient)
   173→   173→    },
   174→   174→  })
   175→   175→
   176→   176→  // Payment mutation
   177→   177→  const paymentMutation = useMutation({
   178→   178→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   179→   179→      return apiFetch(`/api/folio/${folioId}`, {
   180→   180→        method: 'POST',
   181→   181→        headers: { 'Content-Type': 'application/json' },
   182→   182→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   183→   183→      })
   184→   184→    },
   185→   185→    onSuccess: () => {
   186→   186→      invalidate.afterFolioChange(queryClient)
   187→   187→      setPaymentDialogOpen(false)
   188→   188→      setPaymentAmount('')
   189→   189→      setPaymentReference('')
   190→   190→      toast.success('Payment recorded successfully')
   191→   191→    },
   192→   192→    onError: () => {
   193→   193→      toast.error('Failed to record payment')
   194→   194→    },
   195→   195→  })
   196→   196→
   197→   197→  // Late checkout mutation
   198→   198→  const lateCheckoutMutation = useMutation({
   199→   199→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   200→   200→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   201→   201→        method: 'PATCH',
   202→   202→        headers: { 'Content-Type': 'application/json' },
   203→   203→        body: JSON.stringify({ checkOut: newCheckOut }),
   204→   204→      })
   205→   205→      // Add surcharge to folio
   206→   206→      if (surcharge > 0) {
   207→   207→        await apiFetch(`/api/folio`, {
   208→   208→          method: 'POST',
   209→   209→          headers: { 'Content-Type': 'application/json' },
   210→   210→          body: JSON.stringify({
   211→   211→            reservationId,
   212→   212→            type: 'charge',
   213→   213→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   214→   214→            amount: surcharge,
   215→   215→          }),
   216→   216→        })
   217→   217→      }
   218→   218→      return result
   219→   219→    },
   220→   220→    onSuccess: () => {
   221→   221→      invalidate.afterReservationChange(queryClient)
   222→   222→      invalidate.afterFolioChange(queryClient)
   223→   223→      setLateCheckoutDialogOpen(false)
   224→   224→      toast.success('Late checkout request processed successfully')
   225→   225→    },
   226→   226→    onError: () => {
   227→   227→      toast.error('Failed to process late checkout request')
   228→   228→    },
   229→   229→  })
   230→   230→
   231→   231→  // Batch checkout mutation
   232→   232→  const batchCheckoutMutation = useMutation({
   233→   233→    mutationFn: async (ids: string[]) => {
   234→   234→      const results = []
   235→   235→      for (let i = 0; i < ids.length; i++) {
   236→   236→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   237→   237→          method: 'PATCH',
   238→   238→          headers: { 'Content-Type': 'application/json' },
   239→   239→          body: JSON.stringify({ status: 'checked_out' }),
   240→   240→        })
   241→   241→        results.push(result)
   242→   242→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   243→   243→      }
   244→   244→      return results
   245→   245→    },
   246→   246→    onSuccess: (_data, ids) => {
   247→   247→      invalidate.afterCheckout(queryClient)
   248→   248→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   249→   249→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   250→   250→    },
   251→   251→    onError: () => {
   252→   252→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   253→   253→    },
   254→   254→  })
   255→   255→
   256→   256→  // ─── Handlers ─────────────────────────────────────────────────────
   257→   257→
   258→   258→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   259→   259→
   260→   260→  const handleExpressCheckout = (dep: Departure) => {
   261→   261→    checkoutMutation.mutate(dep.id, {
   262→   262→      onSuccess: () => {
   263→   263→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   264→   264→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   265→   265→      },
   266→   266→    })
   267→   267→  }
   268→   268→
   269→   269→  const handleReviewFolio = (dep: Departure) => {
   270→   270→    setSelectedDeparture(dep)
   271→   271→    setFolioDialogOpen(true)
   272→   272→  }
   273→   273→
   274→   274→  const handleNavigateToFolio = (dep: Departure) => {
   275→   275→    useFolioContextStore.getState().setFolioContext({
   276→   276→      reservationId: dep.id,
   277→   277→      guestId: dep.guest?.id || '',
   278→   278→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   279→   279→      roomNumber: dep.room?.number || '',
   280→   280→      confirmationNo: dep.confirmationNo,
   281→   281→    })
   282→   282→    navigateTo('front-desk', 'folio')
   283→   283→  }
   284→   284→
   285→   285→  const handleViewLedger = (dep: Departure) => {
   286→   286→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   287→   287→      guestId: dep.guest?.id || '',
   288→   288→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   289→   289→    })
   290→   290→    navigateTo('front-desk', 'guest-ledger')
   291→   291→  }
   292→   292→
   293→   293→  const handleFolioSettleCheckout = () => {
   294→   294→    setFolioDialogOpen(false)
   295→   295→    setCheckoutDialogOpen(true)
   296→   296→  }
   297→   297→
   298→   298→  const handleQuickCheckout = (dep: Departure) => {
   299→   299→    setSelectedDeparture(dep)
   300→   300→    setCheckoutDialogOpen(true)
   301→   301→  }
   302→   302→
   303→   303→  const handleRequestLateCheckout = (dep: Departure) => {
   304→   304→    setSelectedDeparture(dep)
   305→   305→    setLateCheckoutOption('14:00')
   306→   306→    setLateCheckoutDialogOpen(true)
   307→   307→  }
   308→   308→
   309→   309→  const handleConfirmLateCheckout = () => {
   310→   310→    if (!selectedDeparture) return
   311→   311→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   312→   312→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   313→   313→
   314→   314→    // Build new checkout datetime with the selected time
   315→   315→    const currentCheckout = new Date(selectedDeparture.checkOut)
   316→   316→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   317→   317→    currentCheckout.setHours(hours, minutes, 0, 0)
   318→   318→
   319→   319→    lateCheckoutMutation.mutate({
   320→   320→      reservationId: selectedDeparture.id,
   321→   321→      newCheckOut: currentCheckout.toISOString(),
   322→   322→      surcharge,
   323→   323→    })
   324→   324→  }
   325→   325→
   326→   326→  const handleOpenPaymentDialog = () => {
   327→   327→    if (!selectedDeparture) return
   328→   328→    const balance = getBalance(selectedDeparture)
   329→   329→    setPaymentAmount(String(balance))
   330→   330→    setPaymentMethod('cash')
   331→   331→    setPaymentReference('')
   332→   332→    setPaymentDialogOpen(true)
   333→   333→  }
   334→   334→
   335→   335→  const handleRecordPayment = () => {
   336→   336→    if (!selectedDeparture) return
   337→   337→    const folioId = selectedDeparture.folios[0]?.id
   338→   338→    if (!folioId) {
   339→   339→      toast.error('No folio found for this reservation')
   340→   340→      return
   341→   341→    }
   342→   342→    const amount = Number(paymentAmount)
   343→   343→    if (isNaN(amount) || amount <= 0) {
   344→   344→      toast.error('Please enter a valid payment amount')
   345→   345→      return
   346→   346→    }
   347→   347→    paymentMutation.mutate({
   348→   348→      folioId,
   349→   349→      amount,
   350→   350→      method: paymentMethod,
   351→   351→      reference: paymentReference,
   352→   352→    })
   353→   353→  }
   354→   354→
   355→   355→  const confirmCheckout = () => {
   356→   356→    if (!selectedDeparture) return
   357→   357→    const roomNumber = selectedDeparture.room.number
   358→   358→    checkoutMutation.mutate(selectedDeparture.id, {
   359→   359→      onSuccess: () => {
   360→   360→        setCheckoutDialogOpen(false)
   361→   361→        setReceiptData(selectedDeparture)
   362→   362→        setReceiptDialogOpen(true)
   363→   363→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   364→   364→      },
   365→   365→    })
   366→   366→  }
   367→   367→
   368→   368→  const handleBatchCheckout = () => {
   369→   369→    const ids = zeroBalancePending.map((d) => d.id)
   370→   370→    if (ids.length === 0) {
   371→   371→      toast.info('No guests with zero balance to batch checkout')
   372→   372→      return
   373→   373→    }
   374→   374→    batchCheckoutMutation.mutate(ids)
   375→   375→  }
   376→   376→
   377→   377→  const handlePrintReceipt = () => {
   378→   378→    toast.success('Receipt sent to printer')
   379→   379→  }
   380→   380→
   381→   381→  const handleEmailReceipt = () => {
   382→   382→    toast.success('Receipt emailed to guest')
   383→   383→  }
   384→   384→
   385→   385→  // Solo Departure — opens settlement page for a single guest
   386→   386→  const handleSoloDeparture = (dep: Departure) => {
   387→   387→    setSoloDeparture(dep)
   388→   388→    setSoloDialogOpen(true)
   389→   389→  }
   390→   390→
   391→   391→  // Force Checkout — marks a checked_in guest as checked_out regardless of balance
   392→   392→  const handleForceCheckout = (dep: Departure) => {
   393→   393→    setSelectedDeparture(dep)
   394→   394→    setCheckoutDialogOpen(true)
   395→   395→  }
   396→   396→
   397→   397→  // Solo settlement complete (settle + checkout)
   398→   398→  const handleSoloSettleCheckout = () => {
   399→   399→    if (!soloDeparture) return
   400→   400→    const roomNumber = soloDeparture.room.number
   401→   401→    checkoutMutation.mutate(soloDeparture.id, {
   402→   402→      onSuccess: () => {
   403→   403→        setSoloDialogOpen(false)
   404→   404→        setReceiptData(soloDeparture)
   405→   405→        setReceiptDialogOpen(true)
   406→   406→        toast.success(`Solo checkout complete for Room ${roomNumber}`)
   407→   407→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   408→   408→      },
   409→   409→    })
   410→   410→  }
   411→   411→
   412→   412→  // Solo record payment during settlement
   413→   413→  const handleSoloRecordPayment = () => {
   414→   414→    if (!soloDeparture) return
   415→   415→    const folioId = soloDeparture.folios[0]?.id
   416→   416→    if (!folioId) {
   417→   417→      toast.error('No folio found for this reservation')
   418→   418→      return
   419→   419→    }
   420→   420→    const amount = Number(paymentAmount)
   421→   421→    if (isNaN(amount) || amount <= 0) {
   422→   422→      toast.error('Please enter a valid payment amount')
   423→   423→      return
   424→   424→    }
   425→   425→    paymentMutation.mutate(
   426→   426→      { folioId, amount, method: paymentMethod, reference: paymentReference },
   427→   427→      {
   428→   428→        onSuccess: () => {
   429→   429→          // Refresh solo departure data after payment
   430→   430→          queryClient.invalidateQueries({ queryKey: ['departures', today] })
   431→   431→        },
   432→   432→      },
   433→   433→    )
   434→   434→  }
   435→   435→
   436→   436→  // ─── Computed values ──────────────────────────────────────────────
   437→   437→
   438→   438→  const lateCheckoutFee = selectedDeparture
   439→   439→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   440→   440→    : 0
   441→   441→
   442→   442→  // ─── Render ────────────────────────────────────────────────────────
   443→   443→
   444→   444→  return (
   445→   445→    <div className="flex flex-col gap-2">
   446→   446→      {/* Header */}
   447→   447→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   448→   448→        <div>
   449→   449→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   450→   450→          <p className="text-xs text-muted-foreground">
   451→   451→            Guest check-outs scheduled for {formatDate(today)}
   452→   452→          </p>
   453→   453→        </div>
   454→   454→        {zeroBalancePending.length > 0 && (
   455→   455→          <Button
   456→   456→            onClick={handleBatchCheckout}
   457→   457→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   458→   458→            disabled={batchCheckoutMutation.isPending}
   459→   459→          >
   460→   460→            <Zap className="size-4 mr-1.5" />
   461→   461→            {batchCheckoutMutation.isPending
   462→   462→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   463→   463→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   464→   464→          </Button>
   465→   465→        )}
   466→   466→      </div>
   467→   467→
   468→   468→      {/* Departures List */}
   469→   469→      <Card className="py-0">
   470→   470→        <CardContent className="p-0">
