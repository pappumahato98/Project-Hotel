    95→    95→
    96→    96→function getOutstandingBalance(res: InHouseReservation): number {
    97→    97→  return res.folios[0]?.balance ?? (res.totalAmount - res.paidAmount)
    98→    98→}
    99→    99→
   100→   100→function getLastPayment(res: InHouseReservation): string | null {
   101→   101→  const payments = res.folios[0]?.payments
   102→   102→  if (!payments || payments.length === 0) return null
   103→   103→  const latest = payments.reduce((a, b) =>
   104→   104→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   105→   105→  )
   106→   106→  return new Date(latest.createdAt).toLocaleDateString()
   107→   107→}
   108→   108→
   109→   109→function getVipBadge(vipLevel: string) {
   110→   110→  if (vipLevel === 'none' || !vipLevel) return null
   111→   111→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   112→   112→  return (
   113→   113→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   114→   114→      {level}
   115→   115→    </Badge>
   116→   116→  )
   117→   117→}
