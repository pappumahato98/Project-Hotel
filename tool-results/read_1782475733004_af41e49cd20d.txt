    62→// ─── Quick Search Dialog ────────────────────────────────────────────
    63→function QuickSearchDialog() {
    64→  const { searchOpen, setSearchOpen, navigateTo } = useNavigationStore()
    65→  const [query, setQuery] = React.useState('')
    66→
    67→  const filteredItems = React.useMemo(() => {
    68→    if (!query.trim()) return NAV_ITEMS
    69→    const lower = query.toLowerCase()
    70→    return NAV_ITEMS.filter(
    71→      (item) =>
    72→        item.label.toLowerCase().includes(lower) ||
    73→        item.children?.some((c) => c.label.toLowerCase().includes(lower))
    74→    )
    75→  }, [query])
    76→
    77→  React.useEffect(() => {
    78→    const down = (e: KeyboardEvent) => {
    79→      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
    80→        e.preventDefault()
    81→        setSearchOpen(!searchOpen)
    82→      }
    83→    }
    84→    document.addEventListener('keydown', down)
    85→    return () => document.removeEventListener('keydown', down)
    86→  }, [searchOpen, setSearchOpen])
    87→
    88→  return (
    89→    <CommandDialog open={searchOpen} onOpenChange={setSearchOpen}>
    90→      <CommandInput
    91→        placeholder="Search modules, pages..."
    92→        value={query}
    93→        onValueChange={setQuery}
    94→      />
    95→      <CommandList>
    96→        <CommandEmpty>No results found.</CommandEmpty>
    97→        {filteredItems.map((item) => (
    98→          <React.Fragment key={item.id}>
    99→            <CommandGroup heading={item.label}>
   100→              <CommandItem
   101→                onSelect={() => {
   102→                  navigateTo(item.id)
   103→                  setSearchOpen(false)
   104→                  setQuery('')
   105→                }}
   106→              >
   107→                <item.icon className="mr-2 size-4 text-muted-foreground" />
   108→                <span>{item.label}</span>
   109→              </CommandItem>
   110→              {item.children?.map((child) => (
   111→                <CommandItem
   112→                  key={child.id}
   113→                  onSelect={() => {
   114→                    navigateTo(item.id, child.id)
   115→                    setSearchOpen(false)
   116→                    setQuery('')
   117→                  }}
   118→                  className="pl-8"
   119→                >
   120→                  <span>{child.label}</span>
   121→                </CommandItem>
   122→              ))}
   123→            </CommandGroup>
   124→            <CommandSeparator />
   125→          </React.Fragment>
   126→        ))}
   127→      </CommandList>
   128→    </CommandDialog>
   129→  )
   130→}

   565→// ─── Page Breadcrumb ──────────────────────────────────────────────
   566→function PageBreadcrumb({ activeModule, activeSubModule }: { activeModule: string; activeSubModule: string | null }) {
   567→  const navigateTo = useNavigationStore((s) => s.navigateTo)
   568→  const navItem = NAV_ITEMS.find(n => n.id === activeModule)
   569→
   570→  if (!navItem) return null
   571→
   572→  const hasChildren = navItem.children && navItem.children.length > 0
   573→  const subLabel = activeSubModule ? getSubModuleLabel(activeModule, activeSubModule) : null
   574→
   575→  // Clicking the module name navigates to its default sub-module (or just the module for no-children items)
   576→  const handleModuleClick = () => {
   577→    if (hasChildren) {
   578→      const defaultSub = getDefaultSubModule(activeModule)
   579→      if (defaultSub && activeSubModule !== defaultSub) {
   580→        navigateTo(activeModule, defaultSub)
   581→      }
   582→    }
   583→  }
   584→
   585→  // For items with children, clicking the sub-module name could navigate to the parent
   586→  // But since the sub-module IS the current page, we leave it as display-only text
   587→
   588→  return (
   589→    <nav aria-label="Page breadcrumb" className="flex items-center gap-1.5 text-sm min-w-0">
   590→      <navItem.icon className={cn('size-4 shrink-0', navItem.color)} />
   591→      {hasChildren && subLabel ? (
   592→        <>
   593→          <button
   594→            type="button"
   595→            onClick={handleModuleClick}
   596→            className={cn(
   597→              'font-semibold truncate hover:underline underline-offset-2 transition-colors',
   598→              navItem.color,
   599→              activeSubModule !== getDefaultSubModule(activeModule) ? 'cursor-pointer' : 'cursor-default'
   600→            )}
   601→            tabIndex={activeSubModule !== getDefaultSubModule(activeModule) ? 0 : -1}
   602→            disabled={activeSubModule === getDefaultSubModule(activeModule)}
   603→          >
   604→            {navItem.label}
   605→          </button>
   606→          <ChevronRight className="size-3.5 text-muted-foreground shrink-0" />
   607→          <span className="text-foreground font-medium truncate">
   608→            {subLabel}
   609→          </span>
   610→        </>
   611→      ) : (
   612→        <span className={cn('font-semibold truncate', navItem.color)}>
   613→          {navItem.label}
   614→        </span>
   615→      )}
   616→    </nav>
   617→  )
   618→}
