   163→   163→function guestFullName(g: FolioGuest | null | undefined): string {
   164→   164→  if (!g) return '—'
   165→   165→  return `${g.firstName} ${g.lastName}`
   166→   166→}
   167→   167→
   168→   168→function folioCharges(f: Folio): number {
   169→   169→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   170→   170→}
   171→   171→
   172→   172→function folioPayments(f: Folio): number {
   173→   173→  return f.payments.reduce((s, p) => s + p.amount, 0)
   174→   174→}
   175→   175→
   176→   176→function folioOutstanding(f: Folio): number {
   177→   177→  return folioCharges(f) - folioPayments(f)
   178→   178→}
   179→   179→
   180→   180→function isVoidedTransaction(t: FolioTransaction): boolean {
   181→   181→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   182→   182→}
   183→   183→
   184→   184→function isVoidedPayment(p: FolioPayment): boolean {
   185→   185→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   186→   186→}
   187→   187→
   188→   188→function useDebounce<T>(value: T, delay: number): T {
   189→   189→  const [debounced, setDebounced] = useState(value)
   190→   190→  useEffect(() => {
   191→   191→    const timer = setTimeout(() => setDebounced(value), delay)
   192→   192→    return () => clearTimeout(timer)
   193→   193→  }, [value, delay])
   194→   194→  return debounced
   195→   195→}
