   163→function generatePreviewConfirmation(): string {
   164→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
   165→  let result = 'MRD-'
   166→  for (let i = 0; i < 6; i++) {
   167→    result += chars.charAt(Math.floor(Math.random() * chars.length))
   168→  }
   169→  return result
   170→}
