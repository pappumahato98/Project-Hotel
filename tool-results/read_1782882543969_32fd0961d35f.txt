    41→interface SettlementGuest {

    43→  firstName: string
    44→  lastName: string

    58→interface SettlementFolio {

    65→interface InHouseReservation {
    66→  id: string

    68→  guest: SettlementGuest

    82→type BalanceFilter = 'all' | 'over_10k' | 'over_50k' | 'over_100k'

    98→// ─── Helpers ────────────────────────────────────────────────────────────
    99→
   100→function getOutstandingBalance(res: InHouseReservation): number {
   101→  return res.folios[0]?.balance ?? (res.totalAmount - res.paidAmount)
   102→}
   103→
   104→function getLastPayment(res: InHouseReservation): string | null {
   105→  const payments = res.folios[0]?.payments
   106→  if (!payments || payments.length === 0) return null
   107→  const latest = payments.reduce((a, b) =>
   108→    new Date(a.createdAt) > new Date(b.createdAt) ? a : b
   109→  )
   110→  return new Date(latest.createdAt).toLocaleDateString()
   111→}
   112→
   113→function getVipBadge(vipLevel: string) {
   114→  if (vipLevel === 'none' || !vipLevel) return null
   115→  const level = vipLevel.charAt(0).toUpperCase() + vipLevel.slice(1)
   116→  return (
   117→    <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-0">
   118→      {level}
   119→    </Badge>
   120→  )
   121→}

   413→            Settlement

   416→            View and settle outstanding guest folio balances

   668→                        </TableCell>
   669→                      </TableRow>

   671→                  })
   672→                )}
   673→              </TableBody>
   674→            </Table>
   675→          </div>
   676→        </CardContent>
   677→      </Card>
   678→
