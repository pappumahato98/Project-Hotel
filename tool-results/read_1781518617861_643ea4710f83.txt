   525→
   526→    if (confirmedReservationId) {
   527→      payload.reservationId = confirmedReservationId

   530→        firstName: guestFirstName,

   545→    adults, children, roomRate, collectAdvance, advanceNum, advanceMethod, advanceReference,
   546→    collectDocuments, documents, user, guestFirstName, guestLastName,
