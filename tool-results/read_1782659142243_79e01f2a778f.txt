   103→function NumberStepper({ label, value, min, max, onChange }: {
   104→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   105→}) {
   106→  return (
   107→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   108→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   109→      <div className="flex items-center gap-0 ml-auto">
   110→        <button
   111→          type="button"
   112→          onClick={() => onChange(Math.max(min, value - 1))}
   113→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   114→        >
   115→          <Minus className="w-3 h-3" />
   116→        </button>
   117→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   118→        <button
   119→          type="button"
   120→          onClick={() => onChange(Math.min(max, value + 1))}
   121→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   122→        >
   123→          <Plus className="w-3 h-3" />
   124→        </button>
   125→      </div>
   126→    </div>
   127→  )
   128→}

   130→function sourceLabel(source: string | null) {
   131→  switch (source) {
   132→    case 'walk_in': return 'Walk-in'
   133→    case 'online': return 'Online'
   134→    case 'phone': return 'Phone'
   135→    case 'travel_agent': return 'Travel Agent'
   136→    case 'corporate': return 'Corporate'
   137→    default: return source || 'Direct'
   138→  }
   139→}

   141→// Main Component ──────────────────────────────────────────────────--------------------------------------------------
   142→
   143→export function CheckInLookup({ onBack, prefillReservationId }: CheckInLookupProps) {
   144→  const { setActiveSubModule } = useNavigationStore()
   145→  const { setCheckInSession, setPrefillReservationId } = useFrontDeskContextStore()
   146→
   147→  const today = getTodayString()
   148→
   149→  // State ───────────────────────────────────────────────────────────-------------------------------------------------
   150→  const [expressMode, setExpressMode] = useState(false)
   151→  const [isDirectWalkIn, setIsDirectWalkIn] = useState(false)
   152→  const [searchQuery, setSearchQuery] = useState('')
   153→  const [debouncedQuery, setDebouncedQuery] = useState('')
   154→  const [searchedReservationId, setSearchedReservationId] = useState<string | null>(null)
   155→
   156→  // Walk-in fields
   157→  const [walkInDate, setWalkInDate] = useState(today)
   158→  const [walkInCheckOut, setWalkInCheckOut] = useState(() => {
   159→    const d = new Date()
   160→    d.setDate(d.getDate() + 1)
   161→    return toDateOnly(d)
   162→  })
   163→  const [walkInAdults, setWalkInAdults] = useState(2)
   164→  const [walkInChildren, setWalkInChildren] = useState(0)
   165→  const [guestFirstName, setGuestFirstName] = useState('')
   166→  const [guestLastName, setGuestLastName] = useState('')
   167→  const [guestEmail, setGuestEmail] = useState('')
   168→  const [guestPhone, setGuestPhone] = useState('')
   169→  const [guestNationality, setGuestNationality] = useState('')
   170→
   171→  // Mobile detail view state
   172→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   173→
   174→  // Clear prefill on unmount to avoid stale prefills
   175→  useEffect(() => {
   176→    return () => { setPrefillReservationId(null) }
   177→  }, [setPrefillReservationId])
   178→
   179→  // Debounced search ────────────────────────────────────────────────---------------------------------------
   180→  useEffect(() => {
   181→    const timer = setTimeout(() => {
   182→      setDebouncedQuery(searchQuery)
   183→    }, 300)
   184→    return () => clearTimeout(timer)
   185→  }, [searchQuery])
   186→
   187→  // Data Queries ────────────────────────────────────────────────────------------------------------------------
   188→
   189→  const activeReservationId = prefillReservationId || searchedReservationId
   190→
   191→  const {
   192→    data: reservationData,
   193→    isLoading: reservationLoading,
   194→    error: reservationError,
   195→  } = useQuery({
   196→    queryKey: ['reservation', activeReservationId],
   197→    queryFn: async () => {
   198→      if (!activeReservationId) return null
   199→      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   200→      return data.reservation
   201→    },
   202→    enabled: !!activeReservationId,
   203→  })
   204→
   205→  // Guest stay history
   206→  const { data: guestStayHistory } = useQuery({
   207→    queryKey: ['guest-stays', reservationData?.guest?.id],
   208→    queryFn: async () => {
   209→      if (!reservationData?.guest?.id) return null
   210→      try {
   211→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   212→      } catch {
   213→        return null
   214→      }
   215→    },
   216→    enabled: !!reservationData?.guest?.id && !!searchedReservationId,
   217→  })
   218→
   219→  // Today's expected arrivals
   220→  const {
   221→    data: arrivalsData,
   222→    isLoading: arrivalsLoading,
   223→  } = useQuery({
   224→    queryKey: ['arrivals', today],
   225→    queryFn: async () => {
   226→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   227→      return data.reservations || []
   228→    },
   229→    enabled: !isDirectWalkIn && !prefillReservationId,
   230→  })
   231→
   232→  // Reservation search (debounced)
   233→  const {
   234→    data: searchResults,
   235→    isLoading: searchLoading,
   236→  } = useQuery({
   237→    queryKey: ['reservation-search', debouncedQuery],
   238→    queryFn: async () => {
   239→      if (!debouncedQuery || debouncedQuery.length < 2) return null
   240→      const data = await apiFetch(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   241→      return (data.reservations || []) as ReservationData[]
   242→    },
   243→    enabled: debouncedQuery.length >= 2,
   244→  })
   245→
   246→  // Handlers ────────────────────────────────────────────────────────-----------------------------------------------
   247→
   248→  const navigateToProcess = useCallback((session: CheckInSession) => {
   249→    setCheckInSession(session)
   250→    setActiveSubModule('check-in-process')
   251→  }, [setCheckInSession, setActiveSubModule])
   252→
   253→  const handleUseReservation = useCallback(() => {
   254→    if (!reservationData) return
   255→    navigateToProcess({
   256→      reservationId: reservationData.id,
   257→      reservationData: reservationData as unknown as Record<string, unknown>,
   258→      isDirectWalkIn: false,
   259→      walkInGuest: null,
   260→      walkInDates: null,
   261→      startAtStep: 1,
   262→    })
   263→    toast.success('Reservation confirmed! Proceeding to check-in.')
   264→  }, [reservationData, navigateToProcess])
   265→
   266→  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   267→    navigateToProcess({
   268→      reservationId: res.id,
   269→      reservationData: res as unknown as Record<string, unknown>,
   270→      isDirectWalkIn: false,
   271→      walkInGuest: null,
   272→      walkInDates: null,
   273→      startAtStep: 2,
   274→    })
   275→    toast.success('Express check-in - skipping to Room & Rate')
   276→  }, [navigateToProcess])
   277→
   278→  const handleWalkInContinue = useCallback(() => {
   279→    if (!guestFirstName || !guestLastName) {
   280→      toast.error('Please enter guest first and last name')
   281→      return
   282→    }
   283→    navigateToProcess({
   284→      reservationId: null,
   285→      reservationData: null,
   286→      isDirectWalkIn: true,
   287→      walkInGuest: {
   288→        firstName: guestFirstName,
   289→        lastName: guestLastName,
   290→        email: guestEmail,
   291→        phone: guestPhone,
   292→        nationality: guestNationality,
   293→      },
   294→      walkInDates: {
   295→        checkIn: walkInDate,
   296→        checkOut: walkInCheckOut,
   297→        adults: walkInAdults,
   298→        children: walkInChildren,
   299→      },
   300→      startAtStep: 1,
   301→    })
   302→  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   303→
   304→  // Reservation Card Renderer ───────────────────────────────────────-----------------------------
   305→
   306→  const renderReservationCard = useCallback((res: ReservationData) => {
   307→    const isSelected = searchedReservationId === res.id
   308→    return (
   309→      <Card
   310→        key={res.id}
   311→        className={cn(
   312→          'p-2.5 cursor-pointer transition-all border',
   313→          isSelected
   314→            ? 'border-teal-400 ring-2 ring-teal-400/20 bg-teal-50/50 dark:bg-teal-950/20'
   315→            : 'hover:bg-muted/50'
   316→        )}
   317→        onClick={() => { setSearchedReservationId(res.id); setMobileShowDetail(true) }}
   318→      >
   319→        <div className="flex items-start justify-between gap-2">
   320→          <div className="space-y-1 min-w-0 flex-1">
   321→            <div className="flex items-center gap-2 flex-wrap">
   322→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   323→                {res.confirmationNo}
   324→              </Badge>
   325→              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   326→                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   327→                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   328→                  VIP
   329→                </Badge>
   330→              )}
   331→              {res.source && (
   332→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   333→                  {sourceLabel(res.source)}
   334→                </Badge>
   335→              )}
   336→            </div>
   337→            <p className="text-sm font-medium truncate">
   338→              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   339→            </p>
   340→            <div className="flex items-center gap-2 text-xs text-muted-foreground flex-wrap">
   341→              {res.room && <span>Room {res.room.number}</span>}
   342→              <span>·</span>
   343→              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   344→              <span>·</span>
   345→              <span className="font-medium">{formatCurrency(res.roomRate)}/night</span>
   346→            </div>
   347→          </div>
   348→          <div className="flex flex-col items-end gap-1.5 shrink-0">
   349→            {isSelected && (
   350→              <Check className="w-4 h-4 text-teal-600" />
   351→            )}
   352→            {expressMode && (
   353→              <Button
   354→                size="sm"
   355→                className="h-7 text-[11px] px-2 bg-emerald-500 hover:bg-emerald-600 text-white"
   356→                onClick={(e) => {
   357→                  e.stopPropagation()
   358→                  setSearchedReservationId(res.id)
   359→                  setMobileShowDetail(true)
   360→                  handleExpressCheckIn(res)
   361→                }}
   362→              >
   363→                <Zap className="w-3 h-3 mr-1" />
   364→                Express
   365→              </Button>
   366→            )}
   367→          </div>
   368→        </div>
   369→      </Card>
   370→    )
   371→  }, [searchedReservationId, expressMode, handleExpressCheckIn])
   372→
   373→  // Determine list items for left panel (Find mode)
   374→  const leftPanelItems = useMemo(() => {
   375→    if (isDirectWalkIn || prefillReservationId) return []
   376→    if (searchResults && debouncedQuery.length >= 2) return searchResults
   377→    return arrivalsData || []
   378→  }, [isDirectWalkIn, prefillReservationId, searchResults, debouncedQuery, arrivalsData])
   379→
   380→  const leftPanelLabel = useMemo(() => {
   381→    if (searchResults && debouncedQuery.length >= 2) return 'Search Results'
   382→    return 'Expected Arrivals'
   383→  }, [searchResults, debouncedQuery])
   384→
   385→  // Render ──────────────────────────────────────────────────────────
   386→  return (
   387→    <div className="flex flex-col h-full bg-background">
   388→      {/* ─── Header Bar ─── */}
   389→      <div className="shrink-0 px-3 sm:px-4 py-2.5 border-b bg-card">
   390→        <div className="flex items-center gap-2">
   391→          <button
   392→            type="button"
   393→            onClick={onBack}
   394→            className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors shrink-0"
   395→          >
   396→            <ArrowLeft className="w-4 h-4" />
   397→            <span className="hidden sm:inline">Back</span>
   398→          </button>
   399→          <Separator orientation="vertical" className="h-5 hidden sm:block" />
   400→          {!prefillReservationId && (
   401→            <>
   402→              <div className="flex-1 flex items-center bg-muted/60 rounded-lg p-0.5 gap-0.5">
   403→                <button
   404→                  type="button"
   405→                  onClick={() => { setIsDirectWalkIn(false); setMobileShowDetail(false) }}
   406→                  className={cn(
   407→                    'flex-1 py-1.5 px-3 rounded-md text-sm font-medium transition-all text-center',
   408→                    !isDirectWalkIn
   409→                      ? 'bg-white dark:bg-card shadow-sm text-foreground'
   410→                      : 'text-muted-foreground hover:text-foreground'
   411→                  )}
   412→                >
   413→                  <Search className="w-3.5 h-3.5 inline mr-1.5" />
   414→                  Find Reservation
   415→                </button>
   416→                <button
   417→                  type="button"
   418→                  onClick={() => { setIsDirectWalkIn(true); setMobileShowDetail(false) }}
   419→                  className={cn(
   420→                    'flex-1 py-1.5 px-3 rounded-md text-sm font-medium transition-all text-center',
   421→                    isDirectWalkIn
   422→                      ? 'bg-white dark:bg-card shadow-sm text-foreground'
   423→                      : 'text-muted-foreground hover:text-foreground'
   424→                  )}
   425→                >
   426→                  <User className="w-3.5 h-3.5 inline mr-1.5" />
   427→                  Direct Walk-in
   428→                </button>
   429→              </div>
   430→              <div className="flex items-center gap-2 shrink-0 ml-auto">
   431→                <Zap className={cn('w-3.5 h-3.5', expressMode ? 'text-amber-500' : 'text-muted-foreground')} />
   432→                <span className="text-xs text-muted-foreground hidden sm:inline">Express</span>
   433→                <Switch
   434→                  id="express-mode"
   435→                  checked={expressMode}
   436→                  onCheckedChange={setExpressMode}
   437→                  className="scale-90"
   438→                />
   439→              </div>
   440→            </>
   441→          )}
   442→          {prefillReservationId && (
   443→            <h1 className="text-base sm:text-lg font-semibold text-foreground">Guest Check-In</h1>
   444→          )}
   445→        </div>
   446→      </div>
   447→
   448→      {/* ─── Split Screen Body ─── */}
   449→      <div className="flex flex-col lg:flex-row h-[calc(100vh-8rem)] min-h-[400px] gap-4 overflow-hidden p-3 sm:p-4">
   450→
   451→        {/* ─── Left Panel ─── */}
   452→        <div className={cn(
   453→          'flex flex-col min-h-0 border rounded-lg bg-card overflow-hidden',
   454→          // Direct Walk-in: wider left (form-heavy), Find: narrower left (card list)
   455→          isDirectWalkIn
   456→            ? 'w-full lg:w-[55%] xl:w-[52%] 2xl:w-[50%]'
   457→            : 'w-full lg:w-[42%] xl:w-[38%]',
   458→          prefillReservationId && 'hidden lg:hidden',
   459→          // Mobile: hide when showing detail in Find mode
   460→          !prefillReservationId && !isDirectWalkIn && mobileShowDetail && searchedReservationId && 'hidden lg:flex',
   461→        )}>
   462→          {/* Find Reservation Mode - Left Panel */}
   463→          {!isDirectWalkIn ? (
   464→            <>
   465→              {/* Search input */}
   466→              <div className="shrink-0 p-2 border-b">
   467→                <div className="relative">
   468→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   469→                  <Input
   470→                    placeholder="Search by confirmation #, guest name, or room..."
   471→                    value={searchQuery}
   472→                    onChange={e => setSearchQuery(e.target.value)}
   473→                    className="pl-9 pr-9 h-10 text-sm"
   474→                  />
   475→                  {searchQuery && (
   476→                    <button
   477→                      type="button"
   478→                      onClick={() => { setSearchQuery(''); setDebouncedQuery(''); setSearchedReservationId(null); setMobileShowDetail(false) }}
   479→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   480→                    >
   481→                      <X className="w-3.5 h-3.5" />
   482→                    </button>
   483→                  )}
   484→                  {searchLoading && (
   485→                    <Loader2 className="absolute right-9 top-1/2 -translate-y-1/2 h-3.5 w-3.5 animate-spin text-muted-foreground" />
   486→                  )}
   487→                </div>
   488→              </div>
   489→
   490→              {/* List header */}
   491→              <div className="px-3 py-2 border-b shrink-0 flex items-center justify-between bg-muted/30">
   492→                <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   493→                  {leftPanelLabel}
   494→                </span>
   495→                <span className="text-[10px] text-muted-foreground">{leftPanelItems.length} guests</span>
   496→              </div>
   497→
   498→              {/* List body */}
   499→              <div className="flex-1 overflow-y-auto p-2 space-y-2">
   500→                {arrivalsLoading && !searchResults && (
   501→                  <div className="space-y-2">
   502→                    {Array.from({ length: 3 }).map((_, i) => (
   503→                      <Skeleton key={i} className="h-16 w-full rounded-lg" />
   504→                    ))}
   505→                  </div>
   506→                )}
   507→                {searchResults && debouncedQuery.length >= 2 && searchResults.length === 0 && (
   508→                  <p className="text-sm text-muted-foreground text-center py-2 bg-muted/30 rounded-lg">
   509→                    No reservations found matching &quot;{debouncedQuery}&quot;
   510→                  </p>
   511→                )}
   512→                {arrivalsData && !debouncedQuery && arrivalsData.length === 0 && (
   513→                  <p className="text-sm text-muted-foreground text-center py-2 bg-muted/30 rounded-lg">
   514→                    No expected arrivals for today
   515→                  </p>
   516→                )}
   517→                {leftPanelItems.map(res => renderReservationCard(res))}
   518→              </div>
   519→            </>
   520→          ) : (
   521→            /* Direct Walk-in Mode - Left Panel */
   522→            <div className="flex-1 overflow-y-auto p-3 space-y-3">
   523→              <Badge variant="outline" className="border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-900/40 dark:text-amber-300">
   524→                <User className="w-3 h-3 mr-1" />
