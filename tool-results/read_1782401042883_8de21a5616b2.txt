   171→   171→

   179→   179→  const [regNationality, setRegNationality] = useState('')

   189→   189→  const [roomPreference, setRoomPreference] = useState('')
   190→   190→  const [wakeupCall, setWakeupCall] = useState('')
   191→   191→
   192→   192→  // ─── Billing ──────────────────────────────────────────────────
   193→   193→  const [depositAmount, setDepositAmount] = useState('')
   194→   194→  const [paymentMethod, setPaymentMethod] = useState('')
   195→   195→  const [billingNotes, setBillingNotes] = useState('')
   196→   196→
   197→   197→  // ─── Key card dialog ───────────────────────────────────────────
   198→   198→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   199→   199→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   200→   200→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')

   203→   203→
   204→   204→  // ─── Queries ─────────────────────────────────────────────────
   205→   205→
   206→   206→  // Fetch specific reservation by ID (if prop provided)
   207→   207→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   208→   208→    queryKey: ['reservation', reservationId],

   211→   211→  })
   212→   212→
   213→   213→  // Fetch today's confirmed arrivals
   214→   214→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   215→   215→    queryKey: ['arrivals', today],
   216→   216→    queryFn: async () => {
   217→   217→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   218→   218→      return apiFetch(`/api/reservations?${params.toString()}`)
   219→   219→    },
   220→   220→    enabled: !reservationId,
   221→   221→  })
   222→   222→
   223→   223→  // Fetch all rooms with types
   224→   224→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   225→   225→    queryKey: ['rooms', 'all'],
   226→   226→    queryFn: () => apiFetch('/api/rooms'),
   227→   227→  })
   228→   228→
   229→   229→  // ─── Derived data ──────────────────────────────────────────────
   230→   230→
   231→   231→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   232→   232→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   233→   233→
   234→   234→  // The reservation we're working with
   235→   235→  const activeReservation: ReservationInfo | null = reservationId
   236→   236→    ? specificReservation
   237→   237→    : arrivals.find((a) => a.id === selectedReservationId) || null
   238→   238→
   239→   239→  // Room type data from API (not hardcoded)
   240→   240→  const roomTypes: RoomTypeData[] = useMemo(() => {
   241→   241→    if (!roomsData?.roomTypes) return []
   242→   242→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   243→   243→      id: rt.id,
   244→   244→      name: rt.name,
   245→   245→      code: rt.code,
   246→   246→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   247→   247→    }))
   248→   248→  }, [roomsData])
   249→   249→
   250→   250→  // Available rooms (vacant_clean or inspected)
   251→   251→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   252→   252→  const availableRooms = allRooms.filter(
   253→   253→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   254→   254→  )
   255→   255→
   256→   256→  // Filter available rooms by search
   257→   257→  const filteredAvailableRooms = useMemo(() => {
   258→   258→    let filtered = availableRooms
   259→   259→    if (roomSearchFilter.trim()) {
   260→   260→      const q = roomSearchFilter.toLowerCase()
   261→   261→      filtered = filtered.filter(
   262→   262→        (r: RoomInfo) =>
   263→   263→          r.number.toLowerCase().includes(q) ||
   264→   264→          r.type.name.toLowerCase().includes(q) ||
   265→   265→          String(r.floor).includes(q) ||
   266→   266→          (r.wing && r.wing.toLowerCase().includes(q))
   267→   267→      )
   268→   268→    }
   269→   269→    return filtered
   270→   270→  }, [availableRooms, roomSearchFilter])
   271→   271→
   272→   272→  // Group rooms by floor
   273→   273→  const roomsByFloor = useMemo(() => {
   274→   274→    const grouped = new Map<number, RoomInfo[]>()
   275→   275→    for (const room of filteredAvailableRooms) {
   276→   276→      const floor = room.floor
   277→   277→      if (!grouped.has(floor)) grouped.set(floor, [])
   278→   278→      grouped.get(floor)!.push(room)
   279→   279→    }
   280→   280→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   281→   281→  }, [filteredAvailableRooms])
   282→   282→
   283→   283→  // Get room type name by ID
   284→   284→  const getRoomTypeName = (typeId: string) => {
   285→   285→    const rt = roomTypes.find((t) => t.id === typeId)
   286→   286→    return rt?.name || 'Unknown'
   287→   287→  }
   288→   288→
   289→   289→  // Get room rate by type ID
   290→   290→  const getRoomTypeRate = (typeId: string) => {
   291→   291→    const rt = roomTypes.find((t) => t.id === typeId)
   292→   292→    return rt?.baseRate || 0
   293→   293→  }
   294→   294→
   295→   295→  // ─── Initialize form when reservation loads ───────────────────
   296→   296→
   297→   297→  React.useEffect(() => {
   298→   298→    if (specificReservation && reservationId) {
   299→   299→      setSelectedReservationId(specificReservation.id)
   300→   300→      if (specificReservation.room?.id) {
   301→   301→        setSelectedRoomId(specificReservation.room.id)
   302→   302→      }
   303→   303→      setCurrentStep(specificReservation.guest ? 1 : 0)
   304→   304→    }
   305→   305→  }, [specificReservation, reservationId])
   306→   306→
   307→   307→  // Pre-fill registration fields when guest is selected
   308→   308→  React.useEffect(() => {
   309→   309→    if (activeReservation?.guest) {
   310→   310→      const g = activeReservation.guest
   311→   311→      setRegTitle('')
   312→   312→      setRegFirstName(g.firstName || '')
   313→   313→      setRegLastName(g.lastName || '')
   314→   314→      setRegContactNo(g.phone || '')
   315→   315→      setRegEmail(g.email || '')
   316→   316→      setRegNationality(g.nationality || '')
   317→   317→      setRegIdType(g.idType || 'passport')
   318→   318→      setRegIdNumber(g.idNumber || '')
   319→   319→      setRegAddress(g.address || '')
   320→   320→      setRegCity(g.city || '')
   321→   321→      setRegCountry(g.country || '')
   322→   322→    }
   323→   323→  }, [activeReservation?.guest])
   324→   324→
   325→   325→  // Auto-set deposit when room is selected
   326→   326→  React.useEffect(() => {
   327→   327→    if (activeReservation?.roomRate && !depositAmount) {
   328→   328→      setDepositAmount(String(activeReservation.roomRate))
   329→   329→    }
   330→   330→  }, [activeReservation?.roomRate, depositAmount])
   331→   331→
   332→   332→  // ─── Mutations ────────────────────────────────────────────────
   333→   333→
   334→   334→  const checkInMutation = useMutation({
   335→   335→    mutationFn: async ({
   336→   336→      reservationId: resId,
   337→   337→      roomId,
   338→   338→      specialRequests,
   339→   339→    }: {
   340→   340→      reservationId: string
   341→   341→      roomId: string
   342→   342→      specialRequests: string
   343→   343→    }) => {
   344→   344→      return apiFetch(`/api/reservations/${resId}`, {
   345→   345→        method: 'PATCH',
   346→   346→        headers: { 'Content-Type': 'application/json' },
   347→   347→        body: JSON.stringify({
   348→   348→          roomId,
   349→   349→          status: 'checked_in',
   350→   350→          specialRequests,
   351→   351→        }),
   352→   352→      })
   353→   353→    },
   354→   354→    onSuccess: () => {
   355→   355→      invalidate.afterCheckIn(queryClient)
   356→   356→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   357→   357→      toast.success('Guest checked in successfully!')
   358→   358→    },
   359→   359→    onError: (error) => {
   360→   360→      toast.error('Check-in failed', {
   361→   361→        description: error instanceof Error ? error.message : 'Unknown error',
   362→   362→      })
   363→   363→    },
   364→   364→  })
   365→   365→
   366→   366→  // ─── Special Requests Builder ─────────────────────────────────
   367→   367→
   368→   368→  const buildSpecialRequests = (): string => {
   369→   369→    const parts: string[] = []
   370→   370→
   371→   371→    if (activeReservation?.specialRequests) {
   372→   372→      parts.push(activeReservation.specialRequests)
   373→   373→    }
   374→   374→    if (earlyCheckIn) {
   375→   375→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   376→   376→    }
   377→   377→    if (checkOutTime !== settings.defaultCheckOut) {
   378→   378→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   379→   379→    }
   380→   380→    if (roomPreference) {
   381→   381→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   382→   382→      parts.push(`[PREFERENCE] Room: ${label}`)
   383→   383→    }
   384→   384→    if (pillowType) {
   385→   385→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   386→   386→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   387→   387→    }
   388→   388→    if (wakeupCall) {
   389→   389→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   390→   390→    }
   391→   391→    if (billingNotes) {
   392→   392→      parts.push(`[BILLING] ${billingNotes}`)
   393→   393→    }
   394→   394→    if (paymentMethod) {
   395→   395→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   396→   396→    }
   397→   397→    if (depositAmount && parseFloat(depositAmount) > 0) {
   398→   398→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   399→   399→    }
   400→   400→
   401→   401→    return parts.join(' | ')
   402→   402→  }
   403→   403→
   404→   404→  // ─── Step Navigation ──────────────────────────────────────────
   405→   405→
   406→   406→  const canGoNext = (): boolean => {
   407→   407→    switch (currentStep) {
   408→   408→      case 0: return !!activeReservation
   409→   409→      case 1: return true // Guest verification is info-only
   410→   410→      case 2: return !!selectedRoomId
   411→   411→      case 3: return true // Preferences are optional
   412→   412→      case 4: return true // Billing is optional
   413→   413→      case 5: return keyCardIssued
   414→   414→      default: return false
   415→   415→    }
   416→   416→  }
   417→   417→
   418→   418→  const handleNext = () => {
   419→   419→    if (currentStep === 5) {
   420→   420→      onComplete?.()
   421→   421→      return
   422→   422→    }
   423→   423→
   424→   424→    // On step 4 → 5, execute check-in
   425→   425→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   426→   426→      const requests = buildSpecialRequests()
   427→   427→      checkInMutation.mutate(
   428→   428→        {
   429→   429→          reservationId: activeReservation.id,
   430→   430→          roomId: selectedRoomId,
   431→   431→          specialRequests: requests,
   432→   432→        },
   433→   433→        {
   434→   434→          onSuccess: () => {
   435→   435→            // Find the selected room number for key card
   436→   436→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   437→   437→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   438→   438→            const guestName = activeReservation.guest
   439→   439→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   440→   440→              : 'Guest'
   441→   441→
   442→   442→            setKeyCardRoomNumber(roomNumber)
   443→   443→            setKeyCardGuestName(guestName)
   444→   444→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   445→   445→            setKeyCardIssued(false)
   446→   446→            setKeyCardDialogOpen(true)
   447→   447→            setCurrentStep(5)
   448→   448→          },
   449→   449→        }
   450→   450→      )
   451→   451→      return
   452→   452→    }
   453→   453→
   454→   454→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   455→   455→  }
   456→   456→
   457→   457→  const handleBack = () => {
   458→   458→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   459→   459→  }
   460→   460→
   461→   461→  // ─── Render: Step Indicator ────────────────────────────────────
   462→   462→
   463→   463→  const renderStepIndicator = () => (
   464→   464→    <div className="mb-4">
   465→   465→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   466→   466→        {STEP_LABELS.map((label, idx) => (
   467→   467→          <React.Fragment key={idx}>
   468→   468→            <div className="flex flex-col items-center gap-1 min-w-0">
