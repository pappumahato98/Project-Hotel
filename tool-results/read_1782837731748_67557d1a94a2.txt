   178→function SourceBadge({ source }: { source: string | null }) {
   179→  if (!source) return <span className="text-xs text-muted-foreground">—</span>
   180→
   181→  const colors: Record<string, string> = {
   182→    online: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   183→    walk_in: 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-300 border-violet-300 dark:border-violet-700',
   184→    phone: 'bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-300 border-sky-300 dark:border-sky-700',
   185→    email: 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-300 border-cyan-300 dark:border-cyan-700',
   186→    agent: 'bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300 border-orange-300 dark:border-orange-700',
   187→    corporate: 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300 border-slate-300 dark:border-slate-600',
   188→    group: 'bg-pink-100 text-pink-700 dark:bg-pink-950 dark:text-pink-300 border-pink-300 dark:border-pink-700',
   189→    booking_com: 'bg-teal-100 text-teal-700 dark:bg-teal-950 dark:text-teal-300 border-teal-300 dark:border-teal-700',
   190→    direct: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-300 dark:border-emerald-700',
   191→  }
   192→
   193→  const colorClass = colors[source] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300 border-gray-300 dark:border-gray-600'
   194→
   195→  return (
   196→    <Badge variant="outline" className={cn('text-[10px] px-1.5 py-0', colorClass)}>
   197→      {source.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())}
   198→    </Badge>
   199→  )
   200→}

   267→// ─── Main Component ───────────────────────────────────────────────

   269→export function FrontDeskDashboard() {
   270→  const { navigateTo, setActiveSubModule } = useNavigationStore()
   271→  const { setTriggerWalkInDialog } = useFrontDeskContextStore()
   272→
   273→  const handleArrivalClick = (arrival: UpcomingArrival) => {
   274→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   275→    navigateTo('front-desk', 'check-in')
   276→  }
   277→
   278→  // ─── Dashboard data query ────────────────────────────────────
   279→  const {
   280→    data,
   281→    isLoading,
   282→    error,
   283→    refetch,
   284→    isFetching,
   285→  } = useQuery<DashboardData>({
   286→    queryKey: ['front-desk-dashboard'],
   287→    queryFn: () => apiFetch('/api/front-desk/dashboard'),
   288→    refetchInterval: 30000,
   289→  })
   290→
   291→  const snapshot = data?.snapshot
   292→  const timeline = data?.timeline || []
   293→  const upcomingArrivals = data?.upcomingArrivals || []
   294→
   295→  // ─── Quick action handler ───────────────────────────────────
   296→  const handleQuickAction = (actionId: string) => {
   297→    switch (actionId) {
   298→      case 'new-reservation':
   299→        setActiveSubModule('reservations')
   300→        toast.success('Navigated to Reservations', {
   301→          description: 'Create a new booking from the Reservations tab.',
   302→        })
   303→        break
   304→      case 'walk-in-checkin':
   305→        setActiveSubModule('arrivals')
   306→        setTriggerWalkInDialog(true)
   307→        toast.success('Walk-In Check-In', {
   308→          description: 'Register a walk-in guest directly.',
   309→        })
   310→        break
   311→      case 'room-transfer':
   312→        setActiveSubModule('in-house')
   313→        toast.success('Navigated to In-House Guests', {
   314→          description: 'Select a guest and use Room Transfer from their details.',
   315→        })
   316→        break
   317→      case 'wake-up-call':
   318→        setActiveSubModule('wake-up-calls')
   319→        toast.success('Navigated to Wake-up Calls', {
   320→          description: 'Schedule and manage guest wake-up calls.',
   321→        })
   322→        break
   323→      case 'late-checkout':
   324→        setActiveSubModule('departures')
   325→        toast.success('Navigated to Departures', {
   326→          description: 'Manage late checkout requests from the Departures tab.',
   327→        })
   328→        break
   329→      case 'express-checkout':
   330→        setActiveSubModule('departures')
   331→        toast.success('Navigated to Departures', {
   332→          description: 'Process express checkout from the Departures tab.',
   333→        })
   334→        break
   335→      case 'settlement':
   336→        setActiveSubModule('settlement')
   337→        toast.success('Navigated to Settlement', {
   338→          description: 'Settle outstanding guest folio balances.',
   339→        })
   340→        break
   341→      default:
   342→        break
   343→    }
   344→  }
   345→
   346→  // ─── Determine overbooking alert visibility ──────────────────
   347→  const showOverbookingAlert = useMemo(() => {
   348→    return snapshot ? snapshot.overbookingCount > 0 : false
   349→  }, [snapshot])
   350→
   351→  // ─── Snapshot cards data ────────────────────────────────────
   352→  const snapshotCards = useMemo(() => {
   353→    if (!snapshot) return []
   354→    const cards = [
   355→      {
   356→        label: 'Arrivals Today',
   357→        value: snapshot.arrivals,
   358→        icon: LogIn,
   359→        iconBg: 'bg-emerald-100 dark:bg-emerald-950',
   360→        iconColor: 'text-emerald-600 dark:text-emerald-400',
   361→      },
   362→      {
   363→        label: 'Departures Today',
   364→        value: snapshot.departures,
   365→        icon: LogOut,
   366→        iconBg: 'bg-rose-100 dark:bg-rose-950',
   367→        iconColor: 'text-rose-600 dark:text-rose-400',
   368→      },
   369→      {
   370→        label: 'In-House Guests',
   371→        value: snapshot.inHouse,
   372→        icon: BedDouble,
   373→        iconBg: 'bg-sky-100 dark:bg-sky-950',
   374→        iconColor: 'text-sky-600 dark:text-sky-400',
   375→      },
   376→      {
   377→        label: 'Available Rooms',
   378→        value: snapshot.available,
   379→        icon: Hotel,
   380→        iconBg: 'bg-teal-100 dark:bg-teal-950',
   381→        iconColor: 'text-teal-600 dark:text-teal-400',
   382→      },
   383→      {
   384→        label: 'Occupancy',
   385→        value: `${snapshot.occupancyPct}%`,
   386→        icon: Percent,
   387→        iconBg: 'bg-amber-100 dark:bg-amber-950',
   388→        iconColor: 'text-amber-600 dark:text-amber-400',
   389→        subtext: `${snapshot.inHouse} / ${snapshot.totalRooms} rooms`,
   390→      },
   391→    ]
   392→
   393→    // Only show overbooking alert if there are overbookings
   394→    if (snapshot.overbookingCount > 0) {
   395→      cards.push({
   396→        label: 'Overbooking Alert',
   397→        value: snapshot.overbookingCount,
   398→        icon: AlertTriangle,
   399→        iconBg: 'bg-red-100 dark:bg-red-950',
   400→        iconColor: 'text-red-600 dark:text-red-400',
   401→        subtext: `${snapshot.overbookingCount} room${snapshot.overbookingCount > 1 ? 's' : ''} double-booked`,
   402→      })
   403→    }
   404→
   405→    return cards
   406→  }, [snapshot])
   407→
   408→  // ─── Render ──────────────────────────────────────────────────
   409→  return (
   410→    <div className="flex flex-col gap-2">
   411→      {/* ─── Header ─────────────────────────────────────────── */}
   412→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   413→        <div>
   414→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Front Desk Dashboard</h2>
   415→          <p className="text-xs text-muted-foreground">
   416→            Real-time operational snapshot — {formatDate(new Date())}
   417→          </p>
   418→        </div>
   419→        <Button
   420→          variant="outline"
   421→          size="sm"
   422→          className="gap-1.5 text-xs text-muted-foreground shrink-0"
   423→          onClick={() => refetch()}
   424→          disabled={isFetching}
   425→        >
   426→          <RefreshCw className={cn('size-3.5', isFetching && 'animate-spin')} />
   427→          {isFetching ? 'Refreshing...' : 'Refresh'}
   428→        </Button>
   429→      </div>
   430→
   431→      {/* ─── Today's Snapshot Cards ──────────────────────────── */}
   432→      {isLoading ? (
   433→        <SnapshotSkeleton />
   434→      ) : error ? (
   435→        <Card>
   436→          <CardContent className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   437→            <AlertTriangle className="size-8 mb-2 opacity-40" />
   438→            <p className="text-sm">Failed to load dashboard data</p>
   439→            <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   440→              Try Again
   441→            </Button>
   442→          </CardContent>
   443→        </Card>
   444→      ) : (
   445→        <div className={cn(
   446→          'grid gap-2',
   447→          showOverbookingAlert
   448→            ? 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-6'
   449→            : 'grid-cols-2 sm:grid-cols-3 lg:grid-cols-5',
   450→        )}>
   451→          {snapshotCards.map((card) => (
   452→            <Card key={card.label} className={cn(
   453→              card.label === 'Overbooking Alert' && 'border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20',
   454→            )}>
   455→              <CardContent className="p-2.5 flex items-center gap-3">
   456→                <div className={cn('flex size-10 items-center justify-center rounded-lg shrink-0', card.iconBg)}>
   457→                  <card.icon className={cn('size-5', card.iconColor)} />
   458→                </div>
   459→                <div className="min-w-0">
   460→                  <p className="text-lg font-bold leading-tight">{card.value}</p>
   461→                  <p className="text-xs text-muted-foreground truncate">{card.label}</p>
   462→                  {'subtext' in card && card.subtext && (
   463→                    <p className="text-[10px] text-muted-foreground mt-0.5 truncate">{card.subtext}</p>
   464→                  )}
   465→                </div>
   466→              </CardContent>
   467→            </Card>
   468→          ))}
   469→        </div>
   470→      )}
   471→
   472→      {/* ─── Quick Actions Panel ─────────────────────────────── */}
   473→      <div>
   474→        <h3 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   475→          <LayoutDashboard className="size-4 text-muted-foreground" />
   476→          Quick Actions
   477→        </h3>
   478→        <div className="grid gap-3 grid-cols-2 sm:grid-cols-4 lg:grid-cols-7">
   479→          {QUICK_ACTIONS.map((action) => (
   480→            <Card
   481→              key={action.id}
   482→              className="cursor-pointer transition-all hover:shadow-md hover:border-primary/30 active:scale-[0.98]"
   483→              onClick={() => handleQuickAction(action.id)}
   484→            >
   485→              <CardContent className="p-2.5 flex flex-col items-center text-center gap-2">
   486→                <div className={cn('flex size-10 items-center justify-center rounded-lg', action.bg)}>
   487→                  <action.icon className={cn('size-5', action.color)} />
   488→                </div>
   489→                <div>
   490→                  <p className="text-xs font-semibold leading-tight">{action.title}</p>
   491→                  <p className="text-[10px] text-muted-foreground mt-0.5 leading-tight">{action.description}</p>
   492→                </div>
   493→              </CardContent>
   494→            </Card>
   495→          ))}
   496→        </div>
   497→      </div>
   498→
   499→      <Separator />
   500→
   501→      {/* ─── Bottom Grid: Timeline + Upcoming Arrivals ──────── */}
   502→      <div className="grid gap-4 lg:grid-cols-2">
   503→        {/* ─── Today's Activity Timeline ──────────────────── */}
   504→        <div>
   505→          {isLoading ? (
   506→            <TimelineSkeleton />
   507→          ) : (
   508→            <Card className="h-full">
   509→              <CardHeader className="pb-3">
   510→                <CardTitle className="text-base flex items-center gap-2">
   511→                  <Clock className="size-4 text-muted-foreground" />
   512→                  Today&apos;s Activity
   513→                </CardTitle>
   514→                <CardDescription>Recent front desk operations for today</CardDescription>
   515→              </CardHeader>
   516→              <CardContent>
   517→                {timeline.length === 0 ? (
   518→                  <div className="flex flex-col items-center justify-center py-10 text-muted-foreground">
   519→                    <Clock className="size-8 mb-2 opacity-40" />
   520→                    <p className="text-sm">No activity recorded yet today</p>
   521→                  </div>
   522→                ) : (
   523→                  <div className="max-h-96 overflow-y-auto pr-1 space-y-0">
   524→                    {timeline.map((entry, idx) => (
   525→                      <div key={entry.id} className="flex gap-3 pb-4">
   526→                        {/* Timeline line + icon */}
   527→                        <div className="flex flex-col items-center shrink-0">
   528→                          <TimelineIcon type={entry.type} />
   529→                          {idx < timeline.length - 1 && (
   530→                            <div className="w-px flex-1 bg-border mt-1" />
   531→                          )}
   532→                        </div>
   533→
   534→                        {/* Content */}
   535→                        <div className="flex-1 min-w-0 pb-1">
   536→                          <div className="flex items-center gap-2 flex-wrap">
   537→                            <span className="text-sm font-medium truncate">
   538→                              {entry.guestName}
   539→                            </span>
   540→                            <Badge
   541→                              variant="outline"
   542→                              className={cn(
