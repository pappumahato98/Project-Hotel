   178→// ─── Debounce hook ────────────────────────────────────────────────────
   179→function useDebounce<T>(value: T, delay: number): T {
   180→  const [debounced, setDebounced] = useState(value)
   181→  useEffect(() => {
   182→    const timer = setTimeout(() => setDebounced(value), delay)
   183→    return () => clearTimeout(timer)
   184→  }, [value, delay])
   185→  return debounced
   186→}

   188→// ─── Date range presets ──────────────────────────────────────────────
   189→function getDatePreset(preset: string): { from: Date; to: Date } {
   190→  const today = new Date()
   191→  today.setHours(0, 0, 0, 0)
   192→  switch (preset) {
   193→    case 'today':
   194→      return { from: today, to: today }
   195→    case 'this-week': {
   196→      const from = new Date(today)
   197→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   198→      const to = new Date(from)
   199→      to.setDate(from.getDate() + 6) // Sunday
   200→      return { from, to }
   201→    }
   202→    case 'this-month': {
   203→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   204→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   205→      return { from, to }
   206→    }
   207→    case 'next-7': {
   208→      const to = new Date(today)
   209→      to.setDate(today.getDate() + 6)
   210→      return { from: today, to }
   211→    }
   212→    case 'next-14': {
   213→      const to = new Date(today)
   214→      to.setDate(today.getDate() + 13)
   215→      return { from: today, to }
   216→    }
   217→    case 'next-30': {
   218→      const to = new Date(today)
   219→      to.setDate(today.getDate() + 29)
   220→      return { from: today, to }
   221→    }
   222→    default:
   223→      return { from: today, to: today }
   224→  }
   225→}

   227→function toDateString(d: Date): string {
   228→  const y = d.getFullYear()
   229→  const m = String(d.getMonth() + 1).padStart(2, '0')
   230→  const day = String(d.getDate()).padStart(2, '0')
   231→  return `${y}-${m}-${day}`
   232→}
   233→
   234→function formatShortDate(dateStr: string): string {
   235→  if (!dateStr) return ''
   236→  const d = new Date(dateStr + 'T00:00:00')
   237→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   238→}
