   176→// ─── Debounce hook ────────────────────────────────────────────────────
   177→function useDebounce<T>(value: T, delay: number): T {
   178→  const [debounced, setDebounced] = useState(value)
   179→  useEffect(() => {
   180→    const timer = setTimeout(() => setDebounced(value), delay)
   181→    return () => clearTimeout(timer)
   182→  }, [value, delay])
   183→  return debounced
   184→}

   186→// ─── Date range presets ──────────────────────────────────────────────
   187→function getDatePreset(preset: string): { from: Date; to: Date } {
   188→  const today = new Date()
   189→  today.setHours(0, 0, 0, 0)
   190→  switch (preset) {
   191→    case 'today':
   192→      return { from: today, to: today }
   193→    case 'this-week': {
   194→      const from = new Date(today)
   195→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   196→      const to = new Date(from)
   197→      to.setDate(from.getDate() + 6) // Sunday
   198→      return { from, to }
   199→    }
   200→    case 'this-month': {
   201→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   202→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   203→      return { from, to }
   204→    }
   205→    case 'next-7': {
   206→      const to = new Date(today)
   207→      to.setDate(today.getDate() + 6)
   208→      return { from: today, to }
   209→    }
   210→    case 'next-14': {
   211→      const to = new Date(today)
   212→      to.setDate(today.getDate() + 13)
   213→      return { from: today, to }
   214→    }
   215→    case 'next-30': {
   216→      const to = new Date(today)
   217→      to.setDate(today.getDate() + 29)
   218→      return { from: today, to }
   219→    }
   220→    default:
   221→      return { from: today, to: today }
   222→  }
   223→}

   225→function toDateString(d: Date): string {
   226→  const y = d.getFullYear()
   227→  const m = String(d.getMonth() + 1).padStart(2, '0')
   228→  const day = String(d.getDate()).padStart(2, '0')
   229→  return `${y}-${m}-${day}`
   230→}
   231→
   232→function formatShortDate(dateStr: string): string {
   233→  if (!dateStr) return ''
   234→  const d = new Date(dateStr + 'T00:00:00')
   235→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   236→}
