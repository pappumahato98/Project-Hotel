    44→// ─── Types ──────────────────────────────────────────────────────────────
    45→
    46→interface PendingReservation {
    47→  reservationId: string
    48→  confirmationNo: string
    49→  guest: { id: string; firstName: string; lastName: string } | null
    50→  room: { id: string; number: string; type: { name: string } | null } | null
    51→  checkIn: string
    52→  checkOut: string
    53→  roomRate: number
    54→  folio: { id: string; balance: number; status: string } | null
    55→  pendingNights: string[]
    56→  postedNights: string[]
    57→  futureNights: string[]
    58→  totalNights: number
    59→  pendingAmount: number
    60→}
    61→
    62→interface PendingResponse {
    63→  pendingReservations: PendingReservation[]
    64→  summary: { totalReservations: number; totalPendingNights: number; totalPendingAmount: number }
    65→}

    85→// ─── Component ──────────────────────────────────────────────────────────
    86→
    87→export function RoomRatePostingPage() {
    88→  const queryClient = useQueryClient()
    89→  const navigateTo = useNavigationStore((s) => s.navigateTo)
    90→  const setFolioContext = useFolioContextStore((s) => s.setFolioContext)
    91→
    92→  // State
    93→  const [search, setSearch] = useState('')
    94→  const [searchInput, setSearchInput] = useState('')
    95→  const [postingType, setPostingType] = useState<PostingType>('all')
    96→  const [sortBy, setSortBy] = useState('room-asc')
    97→  const [dateFilter, setDateFilter] = useState('')
    98→  const [expandedRow, setExpandedRow] = useState<string | null>(null)
    99→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   100→
   101→  // Dialogs
   102→  const [postConfirmOpen, setPostConfirmOpen] = useState(false)
   103→  const [postTarget, setPostTarget] = useState<PendingReservation | null>(null)
   104→  const [bulkPostOpen, setBulkPostOpen] = useState(false)
   105→  const [rateOverride, setRateOverride] = useState('')
   106→  const [showQuickActions, setShowQuickActions] = useState(false)
   107→
   108→  // ── Fetch pending reservations ───────────────────────────────────
   109→  const { data: pendingData, isLoading, refetch } = useQuery({
   110→    queryKey: ['rate-posting-pending'],
   111→    queryFn: () => apiFetch<PendingResponse>('/api/room-rate-posting/pending'),
   112→  })
   113→
   114→  // ── Post single reservation's pending nights ────────────────────
   115→  const postMutation = useMutation({
   116→    mutationFn: async ({ reservationId, dates, customRate }: { reservationId: string; dates: string[]; customRate?: number }) => {
   117→      const body: Record<string, unknown> = { reservationId, dates }
   118→      if (customRate && customRate > 0) body.customRate = customRate
   119→      return apiFetch('/api/room-rate-posting', {
   120→        method: 'POST',
   121→        headers: { 'Content-Type': 'application/json' },
   122→        body: JSON.stringify(body),
   123→      })
   124→    },
   125→    onSuccess: () => {
   126→      toast.success('Pending charges posted successfully')
   127→      invalidate.afterFolioChange(queryClient)
   128→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   129→      setPostConfirmOpen(false)
   130→      setPostTarget(null)
   131→      setRateOverride('')
   132→      setSelectedIds(new Set())
   133→    },
   134→    onError: (error) => {
   135→      toast.error(error.message || 'Failed to post charges')
   136→      setPostConfirmOpen(false)
   137→      setPostTarget(null)
   138→      setRateOverride('')
   139→    },
   140→  })
   141→
   142→  // ── Post selected (batch) ───────────────────────────────────────
   143→  const postSelectedMutation = useMutation({
   144→    mutationFn: async (items: { reservationId: string; dates: string[] }[]) => {
   145→      const results: { reservationId: string; success: boolean; error?: string }[] = []
   146→      for (const item of items) {
   147→        try {
   148→          await apiFetch('/api/room-rate-posting', {
   149→            method: 'POST',
   150→            headers: { 'Content-Type': 'application/json' },
   151→            body: JSON.stringify({ reservationId: item.reservationId, dates: item.dates }),
   152→          })
   153→          results.push({ reservationId: item.reservationId, success: true })
   154→        } catch (e) {
   155→          results.push({ reservationId: item.reservationId, success: false, error: (e as Error).message })
   156→        }
   157→      }
   158→      return results
   159→    },
   160→    onSuccess: (results) => {
   161→      const ok = results.filter((r) => r.success).length
   162→      const fail = results.filter((r) => !r.success).length
   163→      if (fail > 0) toast.warning(`${ok} posted, ${fail} failed`)
   164→      else toast.success(`${ok} reservation(s) posted successfully`)
   165→      invalidate.afterFolioChange(queryClient)
   166→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   167→      setSelectedIds(new Set())
   168→    },
   169→    onError: (error) => toast.error(error.message || 'Batch post failed'),
   170→  })
   171→
   172→  // ── Bulk post all pending ───────────────────────────────────────
   173→  const bulkPostMutation = useMutation({
   174→    mutationFn: async () => {
   175→      return apiFetch('/api/room-rate-posting/bulk-post', {
   176→        method: 'POST',
   177→        headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify({ postAll: true }),
   179→      })
   180→    },
   181→    onSuccess: (data) => {
   182→      const result = data as { message?: string }
   183→      toast.success(result?.message || 'Bulk post completed')
   184→      invalidate.afterFolioChange(queryClient)
   185→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   186→      setBulkPostOpen(false)
   187→    },
   188→    onError: (error) => {
   189→      toast.error(error.message || 'Bulk post failed')
   190→      setBulkPostOpen(false)
   191→    },
   192→  })
   193→
   194→  // ── Derived data ────────────────────────────────────────────────
   195→  const pendingSummary = pendingData?.summary
   196→  const pendingReservations = pendingData?.pendingReservations || []
   197→
   198→  const filteredPendingReservations = useMemo(() => {
   199→    let list = [...pendingReservations]
   200→
   201→    // Posting type filter
   202→    if (postingType === 'no-room') list = list.filter((r) => !r.room)
   203→    else if (postingType === 'urgent') list = list.filter((r) => r.pendingNights.length >= 3)
   204→
   205→    // Search (name, room, conf#)
   206→    if (search) {
   207→      const s = search.toLowerCase()
   208→      list = list.filter((r) =>
   209→        r.guest?.firstName?.toLowerCase().includes(s) ||
   210→        r.guest?.lastName?.toLowerCase().includes(s) ||
   211→        r.confirmationNo?.toLowerCase().includes(s) ||
   212→        r.room?.number?.toLowerCase().includes(s)
   213→      )
   214→    }
   215→
   216→    // Date filter
   217→    if (dateFilter) {
   218→      list = list.filter((r) => r.pendingNights.includes(dateFilter))
   219→    }
   220→
   221→    // Sort
   222→    list.sort((a, b) => {
   223→      switch (sortBy) {
   224→        case 'room-desc': return (b.room?.number || '').localeCompare(a.room?.number || '', undefined, { numeric: true })
   225→        case 'guest-asc': {
   226→          const ga = a.guest ? `${a.guest.firstName} ${a.guest.lastName}` : ''
   227→          const gb = b.guest ? `${b.guest.firstName} ${b.guest.lastName}` : ''
   228→          return ga.localeCompare(gb)
   229→        }
   230→        case 'amount-desc': return b.pendingAmount - a.pendingAmount
   231→        case 'pending-desc': return b.pendingNights.length - a.pendingNights.length
   232→        default: return (a.room?.number || '').localeCompare(b.room?.number || '', undefined, { numeric: true })
   233→      }
   234→    })
   235→    return list
   236→  }, [pendingReservations, search, sortBy, postingType, dateFilter])
   237→
   238→  const pendingStats = useMemo(() => {
   239→    const noRoom = filteredPendingReservations.filter((r) => !r.room).length
   240→    const avgRate = filteredPendingReservations.length > 0
   241→      ? Math.round(filteredPendingReservations.reduce((s, r) => s + r.roomRate, 0) / filteredPendingReservations.length)
   242→      : 0
   243→    const totalFolioBalance = filteredPendingReservations.reduce((s, r) => s + (r.folio?.balance || 0), 0)
   244→    const totalPendingNights = filteredPendingReservations.reduce((s, r) => s + r.pendingNights.length, 0)
   245→    const totalPendingAmount = filteredPendingReservations.reduce((s, r) => s + r.pendingAmount, 0)
   246→    return { noRoom, avgRate, totalFolioBalance, totalPendingNights, totalPendingAmount }
   247→  }, [filteredPendingReservations])
   248→
   249→  const isWorking = postMutation.isPending || bulkPostMutation.isPending || postSelectedMutation.isPending
   250→
   251→  // ── Handlers ─────────────────────────────────────────────────────
   252→  const handleSearch = useCallback(() => setSearch(searchInput), [searchInput])
   253→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => { if (e.key === 'Enter') handleSearch() }, [handleSearch])
   254→
   255→  const clearFilters = useCallback(() => {
   256→    setSearchInput(''); setSearch(''); setPostingType('all'); setDateFilter('')
   257→  }, [])
   258→
   259→  const handlePostReservation = useCallback(() => {
   260→    if (!postTarget) return
   261→    const customRate = rateOverride ? parseFloat(rateOverride) : undefined
   262→    postMutation.mutate({ reservationId: postTarget.reservationId, dates: postTarget.pendingNights, customRate })
   263→  }, [postTarget, rateOverride, postMutation])
   264→
   265→  const handleBulkPost = useCallback(() => bulkPostMutation.mutate(), [bulkPostMutation])
   266→
   267→  const handlePostSelected = useCallback(() => {
   268→    const items = filteredPendingReservations
   269→      .filter((r) => selectedIds.has(r.reservationId))
   270→      .map((r) => ({ reservationId: r.reservationId, dates: r.pendingNights }))
   271→    if (items.length === 0) return
   272→    postSelectedMutation.mutate(items)
   273→  }, [filteredPendingReservations, selectedIds, postSelectedMutation])
   274→
   275→  const toggleExpand = useCallback((id: string) => {
   276→    setExpandedRow((prev) => (prev === id ? null : id))
   277→  }, [])
   278→
   279→  const toggleSelectAll = useCallback(() => {
   280→    if (selectedIds.size === filteredPendingReservations.length) setSelectedIds(new Set())
   281→    else setSelectedIds(new Set(filteredPendingReservations.map((r) => r.reservationId)))
   282→  }, [selectedIds.size, filteredPendingReservations])
   283→
   284→  const toggleSelect = useCallback((id: string) => {
   285→    setSelectedIds((prev) => { const n = new Set(prev); if (n.has(id)) n.delete(id); else n.add(id); return n })
   286→  }, [])
   287→
   288→  // ── Navigation actions ───────────────────────────────────────────
   289→  const handleViewFolio = useCallback((res: PendingReservation) => {
   290→    if (res.guest && res.room && res.folio) {
   291→      setFolioContext({
   292→        reservationId: res.reservationId, guestId: res.guest.id,
   293→        guestName: `${res.guest.firstName} ${res.guest.lastName}`,
   294→        roomNumber: res.room.number, confirmationNo: res.confirmationNo, folioId: res.folio.id,
   295→      })
   296→      navigateTo('front-desk', 'folio')
   297→    } else { toast.error('Cannot open folio — missing guest, room, or folio data') }
   298→  }, [navigateTo, setFolioContext])
   299→
   300→  const handleViewReservation = useCallback((res: PendingReservation) => {
   301→    navigateTo('front-desk', 'reservations')
   302→    toast.info(`Viewing reservation ${res.confirmationNo}`)
   303→  }, [navigateTo])
   304→
   305→  const handleViewGuest = useCallback((res: PendingReservation) => {
   306→    navigateTo('crm', 'profiles')
   307→    toast.info('Viewing guest profile')
   308→  }, [navigateTo])
   309→
   310→  // ── Export CSV ──────────────────────────────────────────────────
   311→  const handleExport = useCallback(() => {
   312→    const rows = filteredPendingReservations.map((r) => ({
   313→      Room: r.room?.number || '', Guest: r.guest ? `${r.guest.firstName} ${r.guest.lastName}` : '',
   314→      'Check-In': formatDate(r.checkIn), 'Check-Out': formatDate(r.checkOut),
   315→      'Rate/Night': r.roomRate, 'Pending Nights': r.pendingNights.length,
   316→      'Posted Nights': r.postedNights.length, 'Future Nights': r.futureNights.length,
   317→      'Pending Amount': r.pendingAmount, 'Folio Balance': r.folio?.balance || 0,
   318→      'Confirmation #': r.confirmationNo,
   319→    }))
   320→    if (rows.length === 0) { toast.info('No data to export'); return }
   321→    const headers = Object.keys(rows[0])
   322→    const csv = [headers.join(','), ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h]}"`).join(','))].join('\n')
   323→    const blob = new Blob([csv], { type: 'text/csv' })
   324→    const url = URL.createObjectURL(blob)
   325→    const a = document.createElement('a'); a.href = url; a.download = `rate-posting-${new Date().toISOString().split('T')[0]}.csv`
   326→    a.click(); URL.revokeObjectURL(url); toast.success(`Exported ${rows.length} records`)
   327→  }, [filteredPendingReservations])
   328→
   329→  const hasActiveFilters = search || postingType !== 'all' || dateFilter
   330→
   331→  // ── Render ───────────────────────────────────────────────────────
   332→  return (
   333→    <TooltipProvider delayDuration={300}>
   334→    <div className="space-y-4">
   335→      {/* ═══════════════ HEADER ═══════════════ */}
   336→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   337→        <div>
   338→          <h2 className="text-lg font-semibold flex items-center gap-2">
   339→            <DollarSign className="size-5 text-emerald-600" />
   340→            Room Rate Posting
   341→          </h2>
   342→          <p className="text-sm text-muted-foreground mt-0.5">
   343→            Manage daily room charge postings for in-house guests
   344→          </p>
   345→        </div>
   346→        <div className="flex items-center gap-2">
   347→          <Button variant="outline" size="sm" onClick={() => refetch()} className="gap-1.5">
   348→            <RefreshCw className="size-3.5" /> Refresh
   349→          </Button>
   350→          <Button variant="outline" size="sm" onClick={handleExport} className="gap-1.5">
   351→            <Download className="size-3.5" /> Export
   352→          </Button>
   353→          <DropdownMenu open={showQuickActions} onOpenChange={setShowQuickActions}>
   354→            <DropdownMenuTrigger asChild>
   355→              <Button size="sm" className="gap-1.5 bg-emerald-600 hover:bg-emerald-700">
   356→                <Zap className="size-3.5" /> Quick Actions <ChevronDown className="size-3" />
   357→              </Button>
   358→            </DropdownMenuTrigger>
   359→            <DropdownMenuContent align="end" className="w-64">
   360→              <DropdownMenuLabel className="text-xs">Posting Actions</DropdownMenuLabel>
   361→              <DropdownMenuSeparator />
   362→              <DropdownMenuItem
   363→                onClick={() => { setShowQuickActions(false); setBulkPostOpen(true) }}
   364→                disabled={isWorking || pendingReservations.length === 0} className="gap-2.5 py-2"
   365→              >
   366→                <Zap className="size-4 text-emerald-600" />
   367→                <div className="flex-1">
   368→                  <p className="font-medium text-sm">Post All Pending</p>
   369→                  <p className="text-[10px] text-muted-foreground">Post charges for all {pendingSummary?.totalReservations || 0} reservations</p>
   370→                </div>
   371→              </DropdownMenuItem>
   372→              {selectedIds.size > 0 && (
   373→                <DropdownMenuItem
   374→                  onClick={() => { setShowQuickActions(false); handlePostSelected() }}
   375→                  disabled={isWorking} className="gap-2.5 py-2"
   376→                >
   377→                  <ListChecks className="size-4 text-blue-600" />
   378→                  <div className="flex-1">
   379→                    <p className="font-medium text-sm">Post Selected ({selectedIds.size})</p>
   380→                    <p className="text-[10px] text-muted-foreground">Post charges for selected reservations</p>
   381→                  </div>
   382→                </DropdownMenuItem>
   383→              )}
   384→              <DropdownMenuSeparator />
   385→              <DropdownMenuLabel className="text-xs">Navigate To</DropdownMenuLabel>
   386→              <DropdownMenuSeparator />
   387→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'in-house') }} className="gap-2.5 py-2">
   388→                <BedDouble className="size-4 text-teal-600" />
   389→                <div className="flex-1"><p className="font-medium text-sm">In-House Guests</p></div>
   390→              </DropdownMenuItem>
   391→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'folio') }} className="gap-2.5 py-2">
   392→                <Receipt className="size-4 text-violet-600" />
   393→                <div className="flex-1"><p className="font-medium text-sm">Guest Folio</p></div>
   394→              </DropdownMenuItem>
   395→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'calendar') }} className="gap-2.5 py-2">
   396→                <CalendarDays className="size-4 text-sky-600" />
   397→                <div className="flex-1"><p className="font-medium text-sm">Reservation Calendar</p></div>
   398→              </DropdownMenuItem>
   399→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('front-desk', 'reports') }} className="gap-2.5 py-2">
   400→                <FileSpreadsheet className="size-4 text-amber-600" />
   401→                <div className="flex-1"><p className="font-medium text-sm">Reports</p></div>
   402→              </DropdownMenuItem>
   403→              <DropdownMenuItem onClick={() => { setShowQuickActions(false); navigateTo('accounting', 'ledger') }} className="gap-2.5 py-2">
   404→                <CreditCard className="size-4 text-purple-600" />
   405→                <div className="flex-1"><p className="font-medium text-sm">Accounting Ledger</p></div>
   406→              </DropdownMenuItem>
   407→            </DropdownMenuContent>
   408→          </DropdownMenu>
   409→        </div>
   410→      </div>
   411→
   412→      {/* ═══════════════ STAT CARDS ═══════════════ */}
   413→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   414→        <StatCard icon={<Clock className="size-4 text-amber-600" />} label="Pending" value={String(pendingSummary?.totalReservations ?? 0)} sublabel={`${pendingSummary?.totalPendingNights ?? 0} nights`} bgClass="bg-amber-50 border-amber-200" />
   415→        <StatCard icon={<Wallet className="size-4 text-emerald-600" />} label="Pending Amount" value={pendingSummary ? formatCurrency(pendingSummary.totalPendingAmount) : 'NPR 0'} sublabel="Total charges" bgClass="bg-emerald-50 border-emerald-200" />
   416→        <StatCard icon={<Users className="size-4 text-blue-600" />} label="In-House" value={String(filteredPendingReservations.length)} sublabel="With pending" bgClass="bg-blue-50 border-blue-200" />
   417→        <StatCard icon={<BarChart3 className="size-4 text-violet-600" />} label="Avg Rate/Night" value={pendingStats.avgRate > 0 ? formatCurrency(pendingStats.avgRate) : 'NPR 0'} sublabel="Of pending" bgClass="bg-violet-50 border-violet-200" />
   418→        <StatCard icon={<Building2 className="size-4 text-sky-600" />} label="Folio Balance" value={pendingStats.totalFolioBalance > 0 ? formatCurrency(pendingStats.totalFolioBalance) : 'NPR 0'} sublabel="Combined balance" bgClass="bg-sky-50 border-sky-200" />
   419→        <StatCard icon={<AlertCircle className="size-4 text-red-600" />} label="Issues" value={String(pendingStats.noRoom)} sublabel={pendingStats.noRoom > 0 ? 'No room assigned' : 'All assigned'} bgClass="bg-red-50 border-red-200" />
   420→      </div>
   421→
   422→      {/* ═══════════════ FILTER BAR ═══════════════ */}
   423→      <div className="flex items-center gap-2 flex-wrap">
   424→        {/* Search box — searches by name, room, conf# */}
   425→        <div className="relative">
   426→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground pointer-events-none" />
   427→          <Input
   428→            placeholder="Search guest, room, conf#..."
   429→            value={searchInput}
   430→            onChange={(e) => setSearchInput(e.target.value)}
   431→            onKeyDown={handleKeyDown}
   432→            className="h-8 w-[200px] pl-8 text-xs"
   433→          />
