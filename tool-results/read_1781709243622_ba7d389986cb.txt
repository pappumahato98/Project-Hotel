    38→// ─── Types ──────────────────────────────────────────────────────────────
    39→

    41→  id: string
    42→  reservationId: string
    43→  folioId: string
    44→  roomId: string
    45→  postingDate: string
    46→  roomRate: number
    47→  taxAmount: number
    48→  serviceCharge: number
    49→  totalAmount: number
    50→  status: string
    51→  postedBy: string | null
    52→  voidedBy: string | null
    53→  voidReason: string | null
    54→  createdAt: string
    55→  reservation: {
    56→    id: string
    57→    confirmationNo: string
    58→    status: string
    59→    checkIn: string
    60→    checkOut: string
    61→    roomRate: number
    62→    guest: { id: string; firstName: string; lastName: string } | null
    63→    room: { id: string; number: string; type: { name: string } | null } | null
    64→  } | null
    65→  folio: { id: string; balance: number; status: string } | null
    66→}
    67→
    68→interface PendingReservation {
    69→  reservationId: string
    70→  confirmationNo: string
    71→  guest: { id: string; firstName: string; lastName: string } | null
    72→  room: { id: string; number: string; type: { name: string } | null } | null
    73→  checkIn: string
    74→  checkOut: string
    75→  roomRate: number
    76→  folio: { id: string; balance: number; status: string } | null
    77→  pendingNights: string[]
    78→  postedNights: string[]
    79→  futureNights: string[]
    80→  totalNights: number
    81→  pendingAmount: number
    82→}
    83→
    84→interface PostingListResponse {
    85→  postings: PostingRow[]
    86→  pagination: { page: number; limit: number; total: number; totalPages: number }
    87→  stats: {
    88→    totalPosted: number
    89→    postedToday: number
    90→    totalRevenue: number
    91→    todayRevenue: number
    92→    inHouseReservations: number
    93→    voidedCount: number
    94→  }
    95→}
    96→
    97→interface PendingResponse {
    98→  pendingReservations: PendingReservation[]
    99→  summary: { totalReservations: number; totalPendingNights: number; totalPendingAmount: number }
   100→}

   103→
   104→type TabValue = 'all' | 'pending' | 'posted' | 'voided'
   105→
   106→// ─── Component ──────────────────────────────────────────────────────────
   107→
   108→export function RoomRatePostingPage() {
   109→  const queryClient = useQueryClient()
   110→
   111→  // State
   112→  const [activeTab, setActiveTab] = useState<TabValue>('pending')
   113→  const [search, setSearch] = useState('')
   114→  const [searchInput, setSearchInput] = useState('')
   115→  const [statusFilter, setStatusFilter] = useState('all')
   116→  const [dateFrom, setDateFrom] = useState('')
   117→  const [dateTo, setDateTo] = useState('')
   118→  const [page, setPage] = useState(1)
   119→  const [expandedRow, setExpandedRow] = useState<string | null>(null)
   120→  const [voidDialogOpen, setVoidDialogOpen] = useState(false)
   121→  const [voidTarget, setVoidTarget] = useState<PostingRow | null>(null)
   122→  const [voidReason, setVoidReason] = useState('')
   123→  const [postConfirmOpen, setPostConfirmOpen] = useState(false)
   124→  const [postTarget, setPostTarget] = useState<PendingReservation | null>(null)
   125→
   126→  // ── Fetch all postings list ──────────────────────────────────────────
   127→  const { data: listData, isLoading: isLoadingList, refetch: refetchList } = useQuery({
   128→    queryKey: ['rate-posting-list', { activeTab, search, statusFilter, dateFrom, dateTo, page }],
   129→    queryFn: async () => {
   130→      const params = new URLSearchParams()
   131→      if (activeTab !== 'pending') {
   132→        params.set('status', activeTab === 'all' ? 'all' : activeTab)
   133→      }
   134→      if (search) params.set('search', search)
   135→      if (dateFrom) params.set('dateFrom', dateFrom)
   136→      if (dateTo) params.set('dateTo', dateTo)
   137→      params.set('page', String(page))
   138→      params.set('limit', '50')
   139→
   140→      return apiFetch<PostingListResponse>(`/api/room-rate-posting/list?${params}`)
   141→    },
   142→    enabled: activeTab !== 'pending',
   143→  })
   144→
   145→  // ── Fetch pending reservations ───────────────────────────────────────
   146→  const { data: pendingData, isLoading: isLoadingPending, refetch: refetchPending } = useQuery({
   147→    queryKey: ['rate-posting-pending'],
   148→    queryFn: () => apiFetch<PendingResponse>('/api/room-rate-posting/pending'),
   149→    enabled: activeTab === 'pending',
   150→  })
   151→
   152→  // ── Void mutation ───────────────────────────────────────────────────
   153→  const voidMutation = useMutation({
   154→    mutationFn: async ({ id, reason }: { id: string; reason: string }) => {
   155→      return apiFetch(`/api/room-rate-posting/${id}`, {
   156→        method: 'PATCH',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ voidReason: reason }),
   159→      })
   160→    },
   161→    onSuccess: () => {
   162→      toast.success('Rate posting voided successfully')
   163→      invalidate.afterFolioChange(queryClient)
   164→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   165→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   166→      setVoidDialogOpen(false)
   167→      setVoidTarget(null)
   168→      setVoidReason('')
   169→    },
   170→    onError: () => toast.error('Failed to void posting'),
   171→  })
   172→
   173→  // ── Post single reservation's pending nights ────────────────────────
   174→  const postMutation = useMutation({
   175→    mutationFn: async (reservationId: string) => {
   176→      return apiFetch('/api/room-rate-posting', {
   177→        method: 'POST',
   178→        headers: { 'Content-Type': 'application/json' },
   179→        body: JSON.stringify({ reservationId }),
   180→      })
   181→    },
   182→    onSuccess: () => {
   183→      toast.success('Pending charges posted successfully')
   184→      invalidate.afterFolioChange(queryClient)
   185→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   186→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   187→      setPostConfirmOpen(false)
   188→      setPostTarget(null)
   189→    },
   190→    onError: () => toast.error('Failed to post charges'),
   191→  })
   192→
   193→  // ── Bulk post all pending ───────────────────────────────────────────
   194→  const bulkPostMutation = useMutation({
   195→    mutationFn: async () => {
   196→      return apiFetch('/api/room-rate-posting/bulk-post', {
   197→        method: 'POST',
   198→        headers: { 'Content-Type': 'application/json' },
   199→        body: JSON.stringify({ postAll: true }),
   200→      })
   201→    },
   202→    onSuccess: (data) => {
   203→      const result = data as { message?: string; summary?: { totalPostedNights: number; successCount: number } }
   204→      toast.success(result?.message || 'Bulk post completed')
   205→      invalidate.afterFolioChange(queryClient)
   206→      queryClient.invalidateQueries({ queryKey: ['rate-posting-pending'] })
   207→      queryClient.invalidateQueries({ queryKey: ['rate-posting-list'] })
   208→    },
   209→    onError: () => toast.error('Bulk post failed'),
   210→  })
   211→
   212→  // ── Handlers ─────────────────────────────────────────────────────────
   213→  const handleSearch = useCallback(() => {
   214→    setSearch(searchInput)
   215→    setPage(1)
   216→  }, [searchInput])
   217→
   218→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
   219→    if (e.key === 'Enter') handleSearch()
   220→  }, [handleSearch])
   221→
   222→  const clearFilters = useCallback(() => {
   223→    setSearchInput('')
   224→    setSearch('')
   225→    setStatusFilter('all')
   226→    setDateFrom('')
   227→    setDateTo('')
   228→    setPage(1)
   229→  }, [])
   230→
   231→  const handleVoid = useCallback(() => {
   232→    if (!voidTarget || !voidReason.trim()) return
   233→    voidMutation.mutate({ id: voidTarget.id, reason: voidReason.trim() })
   234→  }, [voidTarget, voidReason, voidMutation])
   235→
   236→  const handlePostReservation = useCallback(() => {
   237→    if (!postTarget) return
   238→    postMutation.mutate(postTarget.reservationId)
   239→  }, [postTarget, postMutation])
   240→
   241→  const handleBulkPost = useCallback(() => {
   242→    bulkPostMutation.mutate()
   243→  }, [bulkPostMutation])
   244→
   245→  const toggleExpand = useCallback((id: string) => {
   246→    setExpandedRow((prev) => (prev === id ? null : id))
   247→  }, [])
   248→
   249→  // ── Derived data ────────────────────────────────────────────────────
   250→  const stats = listData?.stats
   251→  const pendingSummary = pendingData?.summary
   252→  const postings = listData?.postings || []
   253→  const pendingReservations = pendingData?.pendingReservations || []
   254→  const pagination = listData?.pagination
   255→
   256→  const isWorking = voidMutation.isPending || postMutation.isPending || bulkPostMutation.isPending
   257→
   258→  // ── Render ───────────────────────────────────────────────────────────
   259→  return (
   260→    <div className="space-y-4">
   261→      {/* Header */}
   262→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   263→        <div>
   264→          <h2 className="text-lg font-semibold flex items-center gap-2">
   265→            <DollarSign className="size-5 text-emerald-600" />
   266→            Room Rate Posting
   267→          </h2>
   268→          <p className="text-sm text-muted-foreground mt-0.5">
   269→            Manage daily room charge postings across all reservations
   270→          </p>
   271→        </div>
   272→        <div className="flex items-center gap-2">
   273→          <Button
   274→            variant="outline"
   275→            size="sm"
   276→            onClick={() => {
   277→              refetchList()
   278→              refetchPending()
   279→            }}
   280→            className="gap-1.5"
   281→          >
   282→            <RefreshCw className="size-3.5" />
   283→            Refresh
   284→          </Button>
   285→          {activeTab === 'pending' && pendingReservations.length > 0 && (
   286→            <Button
   287→              size="sm"
   288→              onClick={handleBulkPost}
   289→              disabled={isWorking}
   290→              className="gap-1.5 bg-emerald-600 hover:bg-emerald-700"
   291→            >
   292→              {bulkPostMutation.isPending ? (
   293→                <Loader2 className="size-3.5 animate-spin" />
   294→              ) : (
   295→                <Zap className="size-3.5" />
   296→              )}
   297→              Post All Pending
   298→            </Button>
   299→          )}
   300→        </div>
   301→      </div>
   302→
   303→      {/* Stat Cards */}
   304→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   305→        <StatCard
   306→          icon={<DollarSign className="size-4 text-emerald-600" />}
   307→          label="Today's Posting"
   308→          value={String(stats?.postedToday ?? pendingSummary?.totalReservations ?? 0)}
   309→          sublabel={stats ? `${stats.todayRevenue > 0 ? formatCurrency(stats.todayRevenue) : 'No revenue today'}` : pendingSummary ? `${pendingSummary.totalPendingNights} nights pending` : '—'}
   310→          bgClass="bg-emerald-50 border-emerald-200"
   311→        />
   312→        <StatCard
   313→          icon={<TrendingUp className="size-4 text-blue-600" />}
   314→          label="Total Posted"
   315→          value={String(stats?.totalPosted ?? 0)}
   316→          sublabel={stats ? formatCurrency(stats.totalRevenue) : '—'}
   317→          bgClass="bg-blue-50 border-blue-200"
   318→        />
   319→        <StatCard
   320→          icon={<Users className="size-4 text-orange-600" />}
   321→          label="In-House Guests"
   322→          value={String(stats?.inHouseReservations ?? 0)}
   323→          sublabel="Active reservations"
   324→          bgClass="bg-orange-50 border-orange-200"
   325→        />
   326→        <StatCard
   327→          icon={<AlertCircle className="size-4 text-red-600" />}
   328→          label="Voided"
   329→          value={String(stats?.voidedCount ?? 0)}
   330→          sublabel="Cancelled postings"
   331→          bgClass="bg-red-50 border-red-200"
   332→        />
   333→      </div>
   334→
   335→      {/* Tabs + Filters */}
   336→      <Tabs value={activeTab} onValueChange={(v) => { setActiveTab(v as TabValue); setPage(1) }}>
   337→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   338→          <TabsList>
   339→            <TabsTrigger value="pending" className="gap-1.5">
   340→              <Clock className="size-3.5" />
   341→              Pending
   342→              {pendingSummary && pendingSummary.totalReservations > 0 && (
   343→                <Badge variant="destructive" className="h-4 min-w-4 text-[10px] px-1 ml-1">
   344→                  {pendingSummary.totalReservations}
   345→                </Badge>
   346→              )}
   347→            </TabsTrigger>
   348→            <TabsTrigger value="all" className="gap-1.5">
   349→              <DollarSign className="size-3.5" />
   350→              All
   351→            </TabsTrigger>
   352→            <TabsTrigger value="posted" className="gap-1.5">
   353→              <CheckCircle className="size-3.5" />
   354→              Posted
   355→            </TabsTrigger>
   356→            <TabsTrigger value="voided" className="gap-1.5">
   357→              <Ban className="size-3.5" />
   358→              Voided
   359→            </TabsTrigger>
   360→          </TabsList>
   361→
   362→          {/* Filters (for All/Posted/Voided tabs) */}
   363→          {activeTab !== 'pending' && (
   364→            <div className="flex items-center gap-2 flex-wrap">
   365→              <div className="relative">
   366→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   367→                <Input
   368→                  placeholder="Guest, Room, Conf#..."
   369→                  value={searchInput}
   370→                  onChange={(e) => setSearchInput(e.target.value)}
   371→                  onKeyDown={handleKeyDown}
   372→                  className="h-8 w-[180px] pl-8 text-xs"
   373→                />
   374→              </div>
   375→              <Input
   376→                type="date"
   377→                value={dateFrom}
   378→                onChange={(e) => { setDateFrom(e.target.value); setPage(1) }}
   379→                className="h-8 w-[130px] text-xs"
   380→              />
   381→              <span className="text-xs text-muted-foreground">to</span>
   382→              <Input
   383→                type="date"
   384→                value={dateTo}
   385→                onChange={(e) => { setDateTo(e.target.value); setPage(1) }}
   386→                className="h-8 w-[130px] text-xs"
   387→              />
   388→              <Button variant="ghost" size="sm" onClick={handleSearch} className="h-8 text-xs">
   389→                Search
   390→              </Button>
   391→              {(search || dateFrom || dateTo) && (
   392→                <Button variant="ghost" size="sm" onClick={clearFilters} className="h-8 text-xs text-muted-foreground">
   393→                  Clear
   394→                </Button>
   395→              )}
   396→            </div>
   397→          )}
   398→        </div>
   399→
   400→        {/* ── PENDING TAB ──────────────────────────────────────────── */}
   401→        <TabsContent value="pending" className="mt-4">
   402→          {isLoadingPending ? (
   403→            <Card>
   404→              <CardContent className="p-4 space-y-3">
   405→                {Array.from({ length: 4 }).map((_, i) => (
   406→                  <Skeleton key={i} className="h-14 w-full" />
   407→                ))}
   408→              </CardContent>
   409→            </Card>
   410→          ) : pendingReservations.length === 0 ? (
   411→            <EmptyState
   412→              icon={<CheckCircle className="size-12 text-emerald-400" />}
   413→              title="All Caught Up!"
   414→              description="No pending room charges to post. All in-house guests are up to date."
   415→            />
   416→          ) : (
   417→            <>
   418→              {/* Pending Summary Bar */}
   419→              <Card className="py-2.5 px-4">
   420→                <div className="flex items-center justify-between text-xs">
   421→                  <div className="flex items-center gap-4">
   422→                    <span className="text-muted-foreground">
   423→                      <span className="font-semibold text-foreground">{pendingSummary?.totalReservations}</span> reservations
   424→                    </span>
   425→                    <span className="text-muted-foreground">
   426→                      <span className="font-semibold text-amber-600">{pendingSummary?.totalPendingNights}</span> pending nights
   427→                    </span>
   428→                    <span className="font-semibold tabular-nums">
   429→                      {pendingSummary && formatCurrency(pendingSummary.totalPendingAmount)}
   430→                    </span>
   431→                  </div>
   432→                  <Button
   433→                    size="sm"
   434→                    variant="outline"
   435→                    onClick={handleBulkPost}
   436→                    disabled={isWorking}
   437→                    className="h-7 text-xs gap-1.5"
   438→                  >
   439→                    {bulkPostMutation.isPending ? <Loader2 className="size-3 animate-spin" /> : <Zap className="size-3" />}
   440→                    Post All
   441→                  </Button>
   442→                </div>
   443→              </Card>
   444→
   445→              {/* Pending Table */}
   446→              <Card className="overflow-hidden">
   447→                <div className="max-h-[calc(100vh-380px)] overflow-y-auto">
   448→                  <Table>
   449→                    <TableHeader className="sticky top-0 z-10 bg-muted/95 backdrop-blur-sm">
   450→                      <TableRow>
   451→                        <TableHead className="text-xs w-[40px]"></TableHead>
   452→                        <TableHead className="text-xs">Room</TableHead>
   453→                        <TableHead className="text-xs">Guest</TableHead>
   454→                        <TableHead className="text-xs">Stay</TableHead>
   455→                        <TableHead className="text-xs text-right">Rate/Night</TableHead>
   456→                        <TableHead className="text-xs text-center">Pending</TableHead>
   457→                        <TableHead className="text-xs text-center">Posted</TableHead>
   458→                        <TableHead className="text-xs text-center">Future</TableHead>
   459→                        <TableHead className="text-xs text-right">Pending Amt</TableHead>
   460→                        <TableHead className="text-xs text-right w-[80px]">Action</TableHead>
   461→                      </TableRow>
   462→                    </TableHeader>
   463→                    <TableBody>
   464→                      {pendingReservations.map((res) => (
   465→                        <PendingRow
   466→                          key={res.reservationId}
   467→                          reservation={res}
   468→                          isExpanded={expandedRow === res.reservationId}
   469→                          onToggle={() => toggleExpand(res.reservationId)}
   470→                          onPost={() => { setPostTarget(res); setPostConfirmOpen(true) }}
   471→                          isPosting={postMutation.isPending && postMutation.variables === res.reservationId}
   472→                        />
   473→                      ))}
   474→                    </TableBody>
   475→                  </Table>
   476→                </div>
   477→              </Card>
   478→            </>
   479→          )}
   480→        </TabsContent>
   481→
   482→        {/* ── ALL / POSTED / VOIDED TABS ───────────────────────────── */}
   483→        <TabsContent value={activeTab} className="mt-4">
   484→          {isLoadingList ? (
   485→            <Card>
   486→              <CardContent className="p-4 space-y-3">
   487→                {Array.from({ length: 6 }).map((_, i) => (
   488→                  <Skeleton key={i} className="h-10 w-full" />
   489→                ))}
   490→              </CardContent>
   491→            </Card>
   492→          ) : postings.length === 0 ? (
   493→            <EmptyState
   494→              icon={<DollarSign className="size-12 text-slate-300" />}
   495→              title="No Postings Found"
   496→              description={
   497→                activeTab === 'voided'
   498→                  ? 'No voided postings in the system.'
   499→                  : 'No room rate postings match your filters.'
   500→              }
   501→            />
   502→          ) : (
   503→            <>
   504→              <Card className="overflow-hidden">
   505→                <div className="max-h-[calc(100vh-380px)] overflow-y-auto">
   506→                  <Table>
   507→                    <TableHeader className="sticky top-0 z-10 bg-muted/95 backdrop-blur-sm">
   508→                      <TableRow>
   509→                        <TableHead className="text-xs w-[40px]"></TableHead>
   510→                        <TableHead className="text-xs">Date</TableHead>
   511→                        <TableHead className="text-xs">Room</TableHead>
   512→                        <TableHead className="text-xs">Guest</TableHead>
   513→                        <TableHead className="text-xs">Reservation</TableHead>
   514→                        <TableHead className="text-xs text-right">Room Rate</TableHead>
   515→                        <TableHead className="text-xs text-right">Tax</TableHead>
   516→                        <TableHead className="text-xs text-right">Svc</TableHead>
   517→                        <TableHead className="text-xs text-right">Total</TableHead>
   518→                        <TableHead className="text-xs text-center">Status</TableHead>
   519→                        <TableHead className="text-xs">Posted By</TableHead>
   520→                        <TableHead className="text-xs w-[50px]"></TableHead>
   521→                      </TableRow>
   522→                    </TableHeader>
   523→                    <TableBody>
   524→                      {postings.map((posting) => (
   525→                        <PostingListRow
   526→                          key={posting.id}
   527→                          posting={posting}
   528→                          isExpanded={expandedRow === posting.id}
   529→                          onToggle={() => toggleExpand(posting.id)}
   530→                          onVoid={() => { setVoidTarget(posting); setVoidDialogOpen(true) }}
   531→                        />
   532→                      ))}
   533→                    </TableBody>
   534→                  </Table>
   535→                </div>
   536→              </Card>
   537→
   538→              {/* Pagination */}
   539→              {pagination && pagination.totalPages > 1 && (
   540→                <div className="flex items-center justify-between text-xs">
   541→                  <span className="text-muted-foreground">
   542→                    Showing {((pagination.page - 1) * pagination.limit) + 1}–{Math.min(pagination.page * pagination.limit, pagination.total)} of {pagination.total}
   543→                  </span>
   544→                  <div className="flex items-center gap-1">
   545→                    <Button
   546→                      variant="outline"
   547→                      size="sm"
   548→                      className="h-7 text-xs"
   549→                      disabled={pagination.page <= 1}
   550→                      onClick={() => setPage((p) => p - 1)}
   551→                    >
   552→                      Previous
   553→                    </Button>
   554→                    {Array.from({ length: Math.min(5, pagination.totalPages) }, (_, i) => {
   555→                      let pageNum: number
   556→                      if (pagination.totalPages <= 5) {
   557→                        pageNum = i + 1
   558→                      } else if (pagination.page <= 3) {
   559→                        pageNum = i + 1
   560→                      } else if (pagination.page >= pagination.totalPages - 2) {
   561→                        pageNum = pagination.totalPages - 4 + i
   562→                      } else {
   563→                        pageNum = pagination.page - 2 + i
   564→                      }

   566→                        <Button
   567→                          key={pageNum}
   568→                          variant={pageNum === pagination.page ? 'default' : 'outline'}
   569→                          size="sm"
   570→                          className="h-7 w-7 text-xs p-0"
   571→                          onClick={() => setPage(pageNum)}
   572→                        >
   573→                          {pageNum}
   574→                        </Button>
   575→                      )
   576→                    })}
   577→                    <Button
   578→                      variant="outline"
   579→                      size="sm"
   580→                      className="h-7 text-xs"
   581→                      disabled={pagination.page >= pagination.totalPages}
   582→                      onClick={() => setPage((p) => p + 1)}
   583→                    >
   584→                      Next
   585→                    </Button>
   586→                  </div>
   587→                </div>
   588→              )}
   589→            </>
   590→          )}
   591→        </TabsContent>
   592→      </Tabs>
   593→
   594→      {/* ── Void Confirmation Dialog ────────────────────────────────── */}
   595→      <Dialog open={voidDialogOpen} onOpenChange={setVoidDialogOpen}>
   596→        <DialogContent className="max-w-md">
   597→          <DialogHeader>
   598→            <DialogTitle className="flex items-center gap-2 text-red-600">
   599→              <Ban className="size-5" />
   600→              Void Rate Posting
   601→            </DialogTitle>
   602→            <DialogDescription>
   603→              This will void the room charge and reverse the folio transaction. This action cannot be undone.
   604→            </DialogDescription>
   605→          </DialogHeader>
   606→          {voidTarget && (
   607→            <div className="space-y-3 py-2">
   608→              <div className="grid grid-cols-2 gap-2 text-xs">
   609→                <div><span className="text-muted-foreground">Date:</span> <span className="font-medium">{formatDate(voidTarget.postingDate)}</span></div>
   610→                <div><span className="text-muted-foreground">Room:</span> <span className="font-medium">{voidTarget.reservation?.room?.number || '—'}</span></div>
   611→                <div><span className="text-muted-foreground">Guest:</span> <span className="font-medium">{voidTarget.reservation?.guest ? `${voidTarget.reservation.guest.firstName} ${voidTarget.reservation.guest.lastName}` : '—'}</span></div>
   612→                <div><span className="text-muted-foreground">Amount:</span> <span className="font-semibold text-red-600">{formatCurrency(voidTarget.totalAmount)}</span></div>
   613→              </div>
   614→              <div>
   615→                <label className="text-xs font-medium text-muted-foreground">Reason for voiding (required)</label>
   616→                <Input
   617→                  value={voidReason}
   618→                  onChange={(e) => setVoidReason(e.target.value)}
   619→                  placeholder="e.g. Incorrect rate, duplicate posting..."
   620→                  className="mt-1 h-8 text-sm"
   621→                />
   622→              </div>
   623→            </div>
   624→          )}
   625→          <DialogFooter className="gap-2">
   626→            <Button variant="outline" onClick={() => setVoidDialogOpen(false)} className="gap-1.5">
   627→              <X className="size-3.5" />
   628→              Cancel
   629→            </Button>
   630→            <Button
   631→              variant="destructive"
   632→              onClick={handleVoid}
   633→              disabled={!voidReason.trim() || voidMutation.isPending}
   634→              className="gap-1.5"
   635→            >
   636→              {voidMutation.isPending ? <Loader2 className="size-3.5 animate-spin" /> : <Ban className="size-3.5" />}
   637→              Void Posting
   638→            </Button>
   639→          </DialogFooter>
   640→        </DialogContent>
   641→      </Dialog>
   642→
   643→      {/* ── Post Confirmation Dialog ────────────────────────────────── */}
   644→      <Dialog open={postConfirmOpen} onOpenChange={setPostConfirmOpen}>
   645→        <DialogContent className="max-w-md">
   646→          <DialogHeader>
   647→            <DialogTitle className="flex items-center gap-2">
   648→              <Zap className="size-5 text-emerald-600" />
   649→              Post Pending Charges
   650→            </DialogTitle>
   651→            <DialogDescription>
   652→              Post all pending room charges for this reservation to the guest folio.
   653→            </DialogDescription>
   654→          </DialogHeader>
   655→          {postTarget && (
   656→            <div className="space-y-2 py-2 text-xs">
   657→              <div className="grid grid-cols-2 gap-2">
   658→                <div><span className="text-muted-foreground">Room:</span> <span className="font-medium">{postTarget.room?.number || '—'}</span></div>
   659→                <div><span className="text-muted-foreground">Guest:</span> <span className="font-medium">{postTarget.guest ? `${postTarget.guest.firstName} ${postTarget.guest.lastName}` : '—'}</span></div>
   660→                <div><span className="text-muted-foreground">Pending Nights:</span> <span className="font-semibold text-amber-600">{postTarget.pendingNights.length}</span></div>
   661→                <div><span className="text-muted-foreground">Amount:</span> <span className="font-semibold">{formatCurrency(postTarget.pendingAmount)}</span></div>
   662→              </div>
   663→              <div className="text-muted-foreground mt-2">
   664→                Nights to post: {postTarget.pendingNights.join(', ')}
   665→              </div>
   666→            </div>
   667→          )}
   668→          <DialogFooter className="gap-2">
   669→            <Button variant="outline" onClick={() => setPostConfirmOpen(false)} className="gap-1.5">
   670→              <X className="size-3.5" />
   671→              Cancel
   672→            </Button>
   673→            <Button
   674→              onClick={handlePostReservation}
   675→              disabled={postMutation.isPending}
   676→              className="gap-1.5 bg-emerald-600 hover:bg-emerald-700"
   677→            >
   678→              {postMutation.isPending ? <Loader2 className="size-3.5 animate-spin" /> : <CheckCircle className="size-3.5" />}
   679→              Post Charges
   680→            </Button>
   681→          </DialogFooter>
   682→        </DialogContent>
   683→      </Dialog>
   684→    </div>
   685→  )
   686→}
