    82→// ── Format helpers ────────────────────────────────────────────
    83→function formatAuditAction(action: string): { label: string; color: string } {
    84→  switch (action) {
    85→    case 'reject_reassign': return { label: 'Rejected & Reassigned', color: 'text-red-600 bg-red-50 border-red-200 dark:text-red-400 dark:bg-red-950/30 dark:border-red-800' }
    86→    case 'approve': return { label: 'Approved', color: 'text-green-600 bg-green-50 border-green-200 dark:text-green-400 dark:bg-green-950/30 dark:border-green-800' }
    87→    case 'force_mutation': return { label: 'Force Mutated', color: 'text-amber-600 bg-amber-50 border-amber-200 dark:text-amber-400 dark:bg-amber-950/30 dark:border-amber-800' }
    88→    default: return { label: action, color: 'text-muted-foreground bg-muted border-border' }
    89→  }
    90→}

    92→function formatDateTime(iso: string): string {
    93→  const d = new Date(iso)
    94→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
    95→    + ' ' + d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
    96→}

   141→// ── Photo Capture Component ──────────────────────────────────
   142→function PhotoCaptureSection({
   143→  photos,
   144→  onCapture,
   145→  onRemove,
   146→}: {
   147→  photos: string[]
   148→  onCapture: (dataUrl: string) => void
   149→  onRemove: (index: number) => void
   150→}) {
   151→  const videoRef = useRef<HTMLVideoElement>(null)
   152→  const streamRef = useRef<MediaStream | null>(null)
   153→  const [cameraOpen, setCameraOpen] = useState(false)
   154→  const [cameraError, setCameraError] = useState<string | null>(null)
   155→
   156→  const startCamera = useCallback(async () => {
   157→    try {
   158→      setCameraError(null)
   159→      const stream = await navigator.mediaDevices.getUserMedia({
   160→        video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } },
   161→      })
   162→      streamRef.current = stream
   163→      if (videoRef.current) {
   164→        videoRef.current.srcObject = stream
   165→      }
   166→      setCameraOpen(true)
   167→    } catch (_err) {
   168→      setCameraError('Camera access denied or unavailable')
   169→    }
   170→  }, [])
   171→
   172→  const stopCamera = useCallback(() => {
   173→    if (streamRef.current) {
   174→      streamRef.current.getTracks().forEach((t) => t.stop())
   175→      streamRef.current = null
   176→    }
   177→    setCameraOpen(false)
   178→  }, [])
   179→
   180→  const capturePhoto = useCallback(() => {
   181→    const video = videoRef.current
   182→    if (!video) return
   183→    const canvas = document.createElement('canvas')
   184→    canvas.width = video.videoWidth
   185→    canvas.height = video.videoHeight
   186→    const ctx = canvas.getContext('2d')
   187→    if (!ctx) return
   188→    ctx.drawImage(video, 0, 0)
   189→    const dataUrl = canvas.toDataURL('image/jpeg', 0.85)
   190→    onCapture(dataUrl)
   191→    stopCamera()
   192→  }, [onCapture, stopCamera])
   193→
   194→  return (
   195→    <div className="space-y-2">
   196→      <p className="text-sm font-semibold">Photo Evidence</p>
   197→
   198→      {/* Camera Preview */}
   199→      {cameraOpen && (
   200→        <div className="relative rounded-lg border overflow-hidden bg-black max-h-36 sm:max-h-48">
   201→          <video
   202→            ref={videoRef}
   203→            autoPlay
   204→            playsInline
   205→            muted
   206→            className="w-full h-36 sm:h-48 object-cover"
   207→          />
   208→          <div className="absolute bottom-2 left-1/2 -translate-x-1/2 flex gap-2">
   209→            <Button
   210→              size="sm"
   211→              className="gap-1.5 bg-white text-black hover:bg-gray-200"
   212→              onClick={capturePhoto}
   213→            >
   214→              <Camera className="h-4 w-4" /> Capture
   215→            </Button>
   216→            <Button
   217→              size="sm"
   218→              variant="outline"
   219→              className="gap-1.5 bg-white/90 text-black hover:bg-gray-200 border-gray-300"
   220→              onClick={stopCamera}
   221→            >
   222→              <X className="h-4 w-4" /> Cancel
   223→            </Button>
   224→          </div>
   225→        </div>
   226→      )}
   227→
   228→      {cameraError && (
   229→        <p className="text-xs text-red-500 flex items-center gap-1">
   230→          <AlertCircle className="h-3 w-3" /> {cameraError}
   231→        </p>
   232→      )}
   233→
   234→      {/* Photo Grid */}
   235→      {photos.length > 0 && (
   236→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   237→          {photos.map((photo, idx) => (
   238→            <div key={idx} className="relative group rounded-md border overflow-hidden">
   239→              <img src={photo} alt={`Evidence ${idx + 1}`} className="w-full h-auto max-h-24 sm:max-h-28 object-cover" />
   240→              <button
   241→                onClick={() => onRemove(idx)}
   242→                className="absolute top-1 right-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-70 sm:opacity-0 sm:group-hover:opacity-100 transition-opacity"
   243→              >
   244→                <Trash2 className="h-3 w-3" />
   245→              </button>
   246→            </div>
   247→          ))}
   248→        </div>
   249→      )}
   250→
   251→      {/* Open Camera Button */}
   252→      {!cameraOpen && (
   253→        <Button variant="outline" className="w-full gap-2" onClick={startCamera}>
   254→          <Camera className="h-4 w-4" />
   255→          {photos.length > 0 ? `Add More Photo (${photos.length} captured)` : 'Capture Photo Evidence'}
   256→        </Button>
   257→      )}
   258→    </div>
   259→  )
   260→}

   262→// ── Audit Trail Section ──────────────────────────────────────
   263→function AuditTrailSection({ roomId }: { roomId: string }) {
   264→  const { data: audits, isLoading } = useQuery<AuditRecord[]>({
   265→    queryKey: ['inspection-audit', roomId],
   266→    queryFn: () => apiFetch(`/api/housekeeping?section=inspection-audit&roomId=${roomId}`) as Promise<AuditRecord[]>,
   267→  })
   268→
   269→  if (isLoading) {
   270→    return (
   271→      <div className="flex items-center gap-2 py-3 text-xs text-muted-foreground">
   272→        <Loader2 className="h-3 w-3 animate-spin" /> Loading audit history...
   273→      </div>
   274→    )
   275→  }
   276→
   277→  if (!audits || audits.length === 0) {
   278→    return (
   279→      <div className="py-3 text-center">
   280→        <History className="h-6 w-6 mx-auto text-muted-foreground/30" />
   281→        <p className="mt-1 text-xs text-muted-foreground">No inspection history for this room</p>
   282→      </div>
   283→    )
   284→  }
   285→
   286→  return (
   287→    <div className="space-y-2 max-h-48 overflow-y-auto">
   288→      {audits.map((audit) => {
   289→        const { label, color } = formatAuditAction(audit.action)
   290→        return (
   291→          <div key={audit.id} className="rounded-md border p-2.5 text-xs space-y-1">
   292→            <div className="flex items-center justify-between flex-wrap gap-1">
   293→              <Badge variant="outline" className={cn('text-[10px] font-semibold', color)}>
   294→                {label}
   295→              </Badge>
   296→              <span className="text-muted-foreground">{formatDateTime(audit.createdAt)}</span>
   297→            </div>
   298→            <div className="flex items-center gap-2 flex-wrap text-muted-foreground">
   299→              <span className="flex items-center gap-1"><User className="h-3 w-3" /> {audit.performedBy}</span>
   300→            </div>
   301→            {audit.reason && (
   302→              <div className="flex items-start gap-1 text-muted-foreground">
   303→                <FileText className="h-3 w-3 mt-0.5 shrink-0" />
   304→                <span className="italic">&quot;{audit.reason}&quot;</span>
   305→              </div>
   306→            )}
   307→          </div>
   308→        )
   309→      })}
   310→    </div>
   311→  )
   312→}

   314→// ── Inspection Dialog ────────────────────────────────────────
   315→function InspectionDialog({
   316→  task,
   317→  open,
   318→  onOpenChange,
   319→}: {
   320→  task: HkTask | null
   321→  open: boolean
   322→  onOpenChange: (open: boolean) => void
   323→}) {
   324→  const queryClient = useQueryClient()
   325→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   326→  const [photos, setPhotos] = useState<string[]>([])
   327→  const [rejectReason, setRejectReason] = useState('')
   328→  const [showRejectForm, setShowRejectForm] = useState(false)
   329→  const [showAuditTrail, setShowAuditTrail] = useState(false)
   330→
   331→  const approveMutation = useMutation({
   332→    mutationFn: async () => {
   333→      if (!task) return
   334→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   335→        if (v) acc[k] = true
   336→        return acc
   337→      }, {} as Record<string, boolean>)
   338→      return apiFetch('/api/housekeeping', {
   339→        method: 'POST',
   340→        headers: { 'Content-Type': 'application/json' },
   341→        body: JSON.stringify({
   342→          action: 'approve-inspection',
   343→          id: task.id,
   344→          performedBy: 'Inspector',
   345→          checklist,
   346→          photos: photos.length > 0 ? photos : undefined,
   347→        }),
   348→      })
   349→    },
   350→    onSuccess: () => {
   351→      invalidate.afterRoomStatusChange(queryClient)
   352→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   353→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   354→      toast.success(`Room ${task?.room.number} inspection passed`)
   355→      resetAndClose()
   356→    },
   357→    onError: (err: Error) => {
   358→      toast.error(err.message || 'Failed to approve inspection')
   359→    },
   360→  })
   361→
   362→  const rejectMutation = useMutation({
   363→    mutationFn: async () => {
   364→      if (!task) return
   365→      if (!rejectReason.trim()) {
   366→        toast.error('Please provide a reason for rejection')
   367→        throw new Error('Reason required')
   368→      }
   369→      const checklist = Object.entries(checkedItems).reduce((acc, [k, v]) => {
   370→        if (v) acc[k] = true
   371→        return acc
   372→      }, {} as Record<string, boolean>)
   373→      return apiFetch('/api/housekeeping', {
   374→        method: 'POST',
   375→        headers: { 'Content-Type': 'application/json' },
   376→        body: JSON.stringify({
   377→          action: 'reject-inspection',
   378→          id: task.id,
   379→          performedBy: 'Inspector',
   380→          reason: rejectReason.trim(),
   381→          checklist,
   382→          photos: photos.length > 0 ? photos : undefined,
   383→        }),
   384→      })
   385→    },
   386→    onSuccess: () => {
   387→      invalidate.afterRoomStatusChange(queryClient)
   388→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   389→      queryClient.invalidateQueries({ queryKey: ['inspection-audit'] })
   390→      toast.info(`Room ${task?.room.number} rejected & reassigned for cleaning`)
   391→      resetAndClose()
   392→    },
   393→    onError: (err: Error) => {
   394→      if (err.message !== 'Reason required') {
   395→        toast.error(err.message || 'Failed to reject inspection')
   396→      }
   397→    },
   398→  })
   399→
   400→  const resetAndClose = () => {
   401→    setCheckedItems({})
   402→    setPhotos([])
   403→    setRejectReason('')
   404→    setShowRejectForm(false)
   405→    setShowAuditTrail(false)
   406→    onOpenChange(false)
   407→  }
   408→
   409→  if (!task) return null
   410→
   411→  const toggleCheck = (id: string) => {
   412→    setCheckedItems((prev) => ({ ...prev, [id]: !prev[id] }))
   413→  }
   414→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   415→  const allChecked = checkedCount === INSPECTION_ITEMS.length
   416→  const passCount = INSPECTION_ITEMS.length
   417→  const passPercent = Math.round((checkedCount / passCount) * 100)
   418→
   419→  const handleCapturePhoto = (dataUrl: string) => {
   420→    setPhotos((prev) => [...prev, dataUrl])
   421→    toast.success('Photo captured')
   422→  }
   423→
   424→  const handleRemovePhoto = (index: number) => {
   425→    setPhotos((prev) => prev.filter((_, i) => i !== index))
   426→  }
   427→
   428→  const isSubmitting = approveMutation.isPending || rejectMutation.isPending
   429→
   430→  return (
   431→    <Dialog open={open} onOpenChange={(v) => { if (!v) resetAndClose(); else onOpenChange(true) }}>
   432→      <DialogContent className="w-[calc(100vw-0.5rem)] sm:w-auto sm:max-w-lg max-h-[calc(100dvh-1rem)] sm:max-h-[calc(100vh-2rem)] flex flex-col p-0 gap-0 rounded-2xl sm:rounded-lg">
   433→        {/* Fixed Header */}
   434→        <div className="shrink-0 px-4 sm:px-6 pt-4 sm:pt-5 pb-2">
   435→          <DialogHeader>
   436→            <DialogTitle className="flex items-center gap-1.5 sm:gap-2 text-sm sm:text-lg">
   437→              <ShieldCheck className="h-5 w-5 text-purple-600 shrink-0" />
   438→              <span>Inspect Room {task.room.number}</span>
   439→            </DialogTitle>
   440→            <DialogDescription className="mt-1.5 text-xs sm:text-sm leading-relaxed">
   441→              <span className="inline-flex flex-wrap gap-x-1.5 gap-y-0.5">
   442→                <span>Floor {task.room.floor}</span>
   443→                {task.room.wing && <span>· {task.room.wing} Wing</span>}
   444→                <span>· {task.room.type.name}</span>
   445→                {task.assignedTo && <span>· by {task.assignedTo}</span>}
   446→              </span>
   447→            </DialogDescription>
   448→          </DialogHeader>
   449→        </div>
   450→
   451→        {/* Scrollable Body */}
   452→        <ScrollArea className="flex-1 min-h-0 px-3 sm:px-6">
   453→          <div className="space-y-3 sm:space-y-4 pb-4 pr-1">
   454→            {/* Progress */}
   455→            <div className="rounded-lg border p-2.5 sm:p-3">
   456→              <div className="flex items-center justify-between mb-2">
   457→                <p className="text-sm font-medium">Inspection Progress</p>
   458→                <span className="text-sm text-muted-foreground">{checkedCount}/{passCount} items</span>
   459→              </div>
   460→              <div className="h-2 rounded-full bg-muted overflow-hidden">
   461→                <div
   462→                  className={cn(
   463→                    'h-full rounded-full transition-all',
   464→                    allChecked ? 'bg-green-500' : passPercent >= 50 ? 'bg-amber-500' : 'bg-red-500'
   465→                  )}
   466→                  style={{ width: `${passPercent}%` }}
   467→                />
   468→              </div>
   469→            </div>
   470→
   471→            {/* Checklist */}
   472→            <div className="space-y-1.5">
   473→              <p className="text-sm font-semibold">Inspection Checklist</p>
   474→              <div className="rounded-lg border divide-y">
   475→                {INSPECTION_ITEMS.map((item) => (
   476→                  <label
   477→                    key={item.id}
   478→                    className="flex items-center gap-2 sm:gap-3 px-3 sm:px-4 py-2 sm:py-2.5 text-sm hover:bg-muted/50 cursor-pointer transition-colors"
   479→                  >
   480→                    <Checkbox
   481→                      checked={checkedItems[item.id] || false}
   482→                      onCheckedChange={() => toggleCheck(item.id)}
   483→                    />
   484→                    <span className={cn(
   485→                      'flex-1',
   486→                      checkedItems[item.id] && 'text-green-700 dark:text-green-400'
   487→                    )}>
   488→                      {item.label}
   489→                    </span>
   490→                    {checkedItems[item.id] && (
   491→                      <Check className="h-4 w-4 text-green-600" />
   492→                    )}
   493→                  </label>
   494→                ))}
   495→              </div>
   496→            </div>
   497→
   498→            {/* Photo Capture */}
   499→            <PhotoCaptureSection
   500→              photos={photos}
   501→              onCapture={handleCapturePhoto}
   502→              onRemove={handleRemovePhoto}
   503→            />
   504→
   505→            {/* Reject Reason (shown when Reject is clicked) */}
   506→            {showRejectForm && (
   507→              <div className="rounded-lg border border-red-200 bg-red-50 dark:border-red-800 dark:bg-red-950/20 p-3 space-y-2">
   508→                <p className="text-sm font-semibold text-red-700 dark:text-red-400">
   509→                  <AlertCircle className="h-4 w-4 inline mr-1.5" />
   510→                  Reason for Rejection (required)
   511→                </p>
   512→                <Textarea
   513→                  placeholder="Describe why this room failed inspection and needs to be recleaned..."
   514→                  value={rejectReason}
   515→                  onChange={(e) => setRejectReason(e.target.value)}
   516→                  className="min-h-[80px] sm:min-h-[60px] text-sm"
   517→                  autoFocus
   518→                />
   519→              </div>
   520→            )}
   521→
   522→            {/* Audit Trail Toggle */}
   523→            <div>
   524→              <button
   525→                className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground transition-colors w-full"
   526→                onClick={() => setShowAuditTrail(!showAuditTrail)}
   527→              >
   528→                <History className="h-4 w-4" />
   529→                <span className="font-medium">Inspection History</span>
   530→                <ChevronRight className={cn('h-4 w-4 transition-transform', showAuditTrail && 'rotate-90')} />
   531→              </button>
   532→              {showAuditTrail && (
   533→                <div className="mt-2 rounded-lg border p-3">
   534→                  <AuditTrailSection roomId={task.roomId} />
   535→                </div>
   536→              )}
   537→            </div>
   538→          </div>
   539→        </ScrollArea>
   540→
   541→        {/* Fixed Footer Actions */}
   542→        <div className="shrink-0 px-3 sm:px-6 pb-3 sm:pb-4 pt-2">
   543→          <Separator className="mb-3" />
   544→          <div className="flex flex-col sm:flex-row gap-2 sm:gap-3">
   545→            {!showRejectForm ? (
   546→              <>
   547→                <Button
   548→                  variant="outline"
   549→                  className="flex-1 gap-2 border-red-200 text-red-600 hover:bg-red-50 dark:border-red-800 dark:text-red-400 dark:hover:bg-red-950/40"
   550→                  onClick={() => setShowRejectForm(true)}
   551→                  disabled={isSubmitting}
   552→                >
   553→                  <X className="h-4 w-4" />
   554→                  Reject &amp; Reassign
   555→                </Button>
   556→                <Button
   557→                  className="flex-1 gap-2"
   558→                  onClick={() => approveMutation.mutate()}
   559→                  disabled={isSubmitting}
   560→                >
   561→                  {approveMutation.isPending && <Loader2 className="h-4 w-4 animate-spin" />}
   562→                  <Check className="h-4 w-4" />
   563→                  Approve
   564→                </Button>
   565→              </>
   566→            ) : (
   567→              <>
   568→                <Button
   569→                  variant="outline"
   570→                  className="flex-1 gap-2"
   571→                  onClick={() => { setShowRejectForm(false); setRejectReason('') }}
   572→                  disabled={isSubmitting}
   573→                >
   574→                  <X className="h-4 w-4" />
   575→                  Cancel
   576→                </Button>
   577→                <Button
   578→                  variant="destructive"
   579→                  className="flex-1 gap-2"
   580→                  onClick={() => rejectMutation.mutate()}
   581→                  disabled={isSubmitting || !rejectReason.trim()}
   582→                >
   583→                  {rejectMutation.isPending && <Loader2 className="h-4 w-4 animate-spin" />}
   584→                  <AlertCircle className="h-4 w-4" />
   585→                  Confirm Reject
   586→                </Button>
   587→              </>
   588→            )}
   589→          </div>
   590→        </div>
   591→      </DialogContent>
   592→    </Dialog>
   593→  )
   594→}

   596→// ── Main Component ───────────────────────────────────────────
   597→export function InspectionView() {
   598→  const [selectedTask, setSelectedTask] = useState<HkTask | null>(null)
   599→
   600→  const { data, isLoading } = useQuery<{
   601→    tasks: HkTask[]
   602→    summary: HkSummary
   603→  }>({
   604→    queryKey: ['housekeeping-tasks'],
   605→    queryFn: () => apiFetch('/api/housekeeping'),
   606→  })
   607→
   608→  const tasks = data?.tasks || []
   609→  const summary = data?.summary
   610→  const cleanedTasks = tasks.filter((t) => t.status === 'cleaned')
   611→  const inspectedTasks = tasks.filter((t) => t.status === 'inspected')
   612→  const totalInspectable = cleanedTasks.length + inspectedTasks.length
   613→  const inspectionRate = totalInspectable > 0
   614→    ? Math.round((inspectedTasks.length / totalInspectable) * 100)
   615→    : 0
   616→
   617→  if (isLoading) {
   618→    return (
   619→      <div className="space-y-4">
   620→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   621→          {Array.from({ length: 4 }).map((_, i) => (
   622→            <div key={i} className="h-24 rounded-lg bg-muted animate-pulse" />
   623→          ))}
   624→        </div>
   625→        <div className="h-64 rounded-lg bg-muted animate-pulse" />
   626→      </div>
   627→    )
   628→  }
   629→
   630→  return (
   631→    <div className="space-y-4">
   632→      {/* Stats Bar */}
   633→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-4">
   634→        {[
   635→          { label: 'Awaiting Inspection', value: cleanedTasks.length, icon: ShieldCheck, color: 'text-green-600' },
   636→          { label: 'Inspected Today', value: inspectedTasks.length, icon: Check, color: 'text-purple-600' },
   637→          { label: 'Inspection Rate', value: `${inspectionRate}%`, icon: BadgePercent, color: 'text-amber-600' },
   638→          { label: 'Failed Today', value: summary?.failed ?? 0, icon: X, color: 'text-red-600' },
   639→        ].map((stat) => (
   640→          <Card key={stat.label} className="p-2.5">
   641→            <div className="flex items-center justify-between">
   642→              <div>
   643→                <p className="text-xs font-medium text-muted-foreground">{stat.label}</p>
   644→                <p className="text-lg font-bold tracking-tight">{stat.value}</p>
   645→              </div>
   646→              <stat.icon className={cn('h-7 w-7 sm:h-8 sm:w-8 opacity-20', stat.color)} />
   647→            </div>
   648→          </Card>
   649→        ))}
   650→      </div>
   651→
   652→      {/* Awaiting Inspection */}
   653→      <div className="space-y-3">
   654→        <h3 className="text-sm font-semibold flex items-center gap-2">
   655→          <ShieldCheck className="h-4 w-4 text-purple-600" />
   656→          Rooms Awaiting Inspection
   657→          <Badge variant="secondary" className="text-xs">{cleanedTasks.length}</Badge>
   658→        </h3>
   659→        {cleanedTasks.length === 0 ? (
   660→          <Card className="p-8 text-center text-muted-foreground">
   661→            <ShieldCheck className="h-10 w-10 mx-auto opacity-20" />
   662→            <p className="mt-2 text-sm">No rooms awaiting inspection</p>
   663→          </Card>
   664→        ) : (
   665→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   666→            {cleanedTasks.map((task) => (
   667→              <RoomInspectionCard
   668→                key={task.id}
   669→                task={task}
   670→                onInspect={setSelectedTask}
   671→              />
   672→            ))}
   673→          </div>
   674→        )}
   675→      </div>
   676→
   677→      {/* Recently Inspected */}
   678→      {inspectedTasks.length > 0 && (
   679→        <div className="space-y-3">
   680→          <h3 className="text-sm font-semibold flex items-center gap-2">
   681→            <ClipboardCheck className="h-4 w-4 text-green-600" />
   682→            Recently Inspected
   683→            <Badge variant="secondary" className="text-xs">{inspectedTasks.length}</Badge>
   684→          </h3>
   685→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   686→            {inspectedTasks.slice(0, 6).map((task) => (
   687→              <RoomInspectionCard
   688→                key={task.id}
   689→                task={task}
   690→                onInspect={setSelectedTask}
   691→              />
   692→            ))}
   693→          </div>
   694→        </div>
   695→      )}
   696→
   697→      <InspectionDialog
   698→        task={selectedTask}
   699→        open={!!selectedTask}
   700→        onOpenChange={(open) => !open && setSelectedTask(null)}
   701→      />
   702→    </div>
   703→  )
