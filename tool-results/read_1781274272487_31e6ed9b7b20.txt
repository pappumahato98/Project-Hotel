   176→   176→// ─── Debounce hook ────────────────────────────────────────────────────
   177→   177→function useDebounce<T>(value: T, delay: number): T {
   178→   178→  const [debounced, setDebounced] = useState(value)
   179→   179→  useEffect(() => {
   180→   180→    const timer = setTimeout(() => setDebounced(value), delay)
   181→   181→    return () => clearTimeout(timer)
   182→   182→  }, [value, delay])
   183→   183→  return debounced
   184→   184→}

   186→   186→// ─── Date range presets ──────────────────────────────────────────────
   187→   187→function getDatePreset(preset: string): { from: Date; to: Date } {
   188→   188→  const today = new Date()
   189→   189→  today.setHours(0, 0, 0, 0)
   190→   190→  switch (preset) {
   191→   191→    case 'today':
   192→   192→      return { from: today, to: today }
   193→   193→    case 'this-week': {
   194→   194→      const from = new Date(today)
   195→   195→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   196→   196→      const to = new Date(from)
   197→   197→      to.setDate(from.getDate() + 6) // Sunday
   198→   198→      return { from, to }
   199→   199→    }
   200→   200→    case 'this-month': {
   201→   201→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   202→   202→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   203→   203→      return { from, to }
   204→   204→    }
   205→   205→    case 'next-7': {
   206→   206→      const to = new Date(today)
   207→   207→      to.setDate(today.getDate() + 6)
   208→   208→      return { from: today, to }
   209→   209→    }
   210→   210→    case 'next-14': {
   211→   211→      const to = new Date(today)
   212→   212→      to.setDate(today.getDate() + 13)
   213→   213→      return { from: today, to }
   214→   214→    }
   215→   215→    case 'next-30': {
   216→   216→      const to = new Date(today)
   217→   217→      to.setDate(today.getDate() + 29)
   218→   218→      return { from: today, to }
   219→   219→    }
   220→   220→    default:
   221→   221→      return { from: today, to: today }
   222→   222→  }
   223→   223→}
   224→   224→
   225→   225→function toDateString(d: Date): string {
   226→   226→  const y = d.getFullYear()
   227→   227→  const m = String(d.getMonth() + 1).padStart(2, '0')
   228→   228→  const day = String(d.getDate()).padStart(2, '0')
   229→   229→  return `${y}-${m}-${day}`
   230→   230→}
   231→   231→
   232→   232→function formatShortDate(dateStr: string): string {
   233→   233→  if (!dateStr) return ''
   234→   234→  const d = new Date(dateStr + 'T00:00:00')
   235→   235→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   236→   236→}
