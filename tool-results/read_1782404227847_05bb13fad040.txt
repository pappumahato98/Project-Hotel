   106→function NumberStepper({ label, value, min, max, onChange }: {
   107→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   108→}) {
   109→  return (
   110→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   111→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   112→      <div className="flex items-center gap-0 ml-auto">
   113→        <button
   114→          type="button"
   115→          onClick={() => onChange(Math.max(min, value - 1))}
   116→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   117→        >
   118→          <Minus className="w-3 h-3" />
   119→        </button>
   120→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   121→        <button
   122→          type="button"
   123→          onClick={() => onChange(Math.min(max, value + 1))}
   124→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   125→        >
   126→          <Plus className="w-3 h-3" />
   127→        </button>
   128→      </div>
   129→    </div>
   130→  )
   131→}

   133→function sourceLabel(source: string | null) {
   134→  switch (source) {
   135→    case 'walk_in': return 'Walk-in'
   136→    case 'online': return 'Online'
   137→    case 'phone': return 'Phone'
   138→    case 'travel_agent': return 'Travel Agent'
   139→    case 'corporate': return 'Corporate'
   140→    default: return source || 'Direct'
   141→  }
   142→}

   146→export function CheckInLookup({ onBack, prefillReservationId }: CheckInLookupProps) {
   147→  const { setActiveSubModule } = useNavigationStore()

   149→
   150→  const today = getTodayString()
   151→
   152→  // State ───────────────────────────────────────────────────────────-------------------------------------------------

   154→  const [isDirectWalkIn, setIsDirectWalkIn] = useState(false)
   155→  const [searchQuery, setSearchQuery] = useState('')

   157→  const [searchedReservationId, setSearchedReservationId] = useState<string | null>(null)
   158→

   160→  const [walkInDate, setWalkInDate] = useState(today)
   161→  const [walkInCheckOut, setWalkInCheckOut] = useState(() => {
   162→    const d = new Date()
   163→    d.setDate(d.getDate() + 1)
   164→    return toDateOnly(d)
   165→  })
   166→  const [walkInAdults, setWalkInAdults] = useState(2)
   167→  const [walkInChildren, setWalkInChildren] = useState(0)
   168→  const [guestFirstName, setGuestFirstName] = useState('')
   169→  const [guestLastName, setGuestLastName] = useState('')
   170→  const [guestEmail, setGuestEmail] = useState('')
   171→  const [guestPhone, setGuestPhone] = useState('')
   172→  const [guestNationality, setGuestNationality] = useState('')
   173→
   174→  // Debounced search ────────────────────────────────────────────────---------------------------------------
   175→  useEffect(() => {
   176→    const timer = setTimeout(() => {
   177→      setDebouncedQuery(searchQuery)
   178→    }, 300)
   179→    return () => clearTimeout(timer)
   180→  }, [searchQuery])
   181→
   182→  // Data Queries ────────────────────────────────────────────────────------------------------------------------
   183→
   184→  const activeReservationId = prefillReservationId || searchedReservationId
   185→
   186→  const {
   187→    data: reservationData,
   188→    isLoading: reservationLoading,
   189→    error: reservationError,
   190→  } = useQuery({
   191→    queryKey: ['reservation', activeReservationId],
   192→    queryFn: async () => {
   193→      if (!activeReservationId) return null
   194→      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   195→      return data.reservation
   196→    },
   197→    enabled: !!activeReservationId,
   198→  })
   199→
   200→  // Guest stay history
   201→  const { data: guestStayHistory } = useQuery({
   202→    queryKey: ['guest-stays', reservationData?.guest?.id],
   203→    queryFn: async () => {
   204→      if (!reservationData?.guest?.id) return null
   205→      try {
   206→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   207→      } catch {
   208→        return null
   209→      }
   210→    },
   211→    enabled: !!reservationData?.guest?.id && !!searchedReservationId,
   212→  })
   213→
   214→  // Today's expected arrivals
   215→  const {
   216→    data: arrivalsData,
   217→    isLoading: arrivalsLoading,
   218→  } = useQuery({
   219→    queryKey: ['arrivals', today],
   220→    queryFn: async () => {
   221→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   222→      return data.reservations || []
   223→    },
   224→    enabled: !isDirectWalkIn && !prefillReservationId,
   225→  })
   226→
   227→  // Reservation search (debounced)
   228→  const {
   229→    data: searchResults,
   230→    isLoading: searchLoading,
   231→  } = useQuery({
   232→    queryKey: ['reservation-search', debouncedQuery],
   233→    queryFn: async () => {
   234→      if (!debouncedQuery || debouncedQuery.length < 2) return null
   235→      const data = await apiFetch(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   236→      return (data.reservations || []) as ReservationData[]
   237→    },
   238→    enabled: debouncedQuery.length >= 2,
   239→  })
   240→
   241→  // Handlers ────────────────────────────────────────────────────────-----------------------------------------------
   242→
   243→  const navigateToProcess = useCallback((session: CheckInSession) => {
   244→    setCheckInSession(session)
   245→    setActiveSubModule('check-in-process')
   246→  }, [setCheckInSession, setActiveSubModule])
   247→
   248→  const handleUseReservation = useCallback(() => {
   249→    if (!reservationData) return
   250→    navigateToProcess({
   251→      reservationId: reservationData.id,
   252→      reservationData: reservationData as unknown as Record<string, unknown>,
   253→      isDirectWalkIn: false,
   254→      walkInGuest: null,
   255→      walkInDates: null,
   256→      startAtStep: 1,
   257→    })
   258→    toast.success('Reservation confirmed! Proceeding to check-in.')
   259→  }, [reservationData, navigateToProcess])
   260→
   261→  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   262→    navigateToProcess({
   263→      reservationId: res.id,
   264→      reservationData: res as unknown as Record<string, unknown>,
   265→      isDirectWalkIn: false,
   266→      walkInGuest: null,
   267→      walkInDates: null,
   268→      startAtStep: 2,
   269→    })
   270→    toast.success('Express check-in - skipping to Room & Rate')
   271→  }, [navigateToProcess])
   272→
   273→  const handleWalkInContinue = useCallback(() => {
   274→    if (!guestFirstName || !guestLastName) {
   275→      toast.error('Please enter guest first and last name')
   276→      return
   277→    }
   278→    navigateToProcess({
   279→      reservationId: null,
   280→      reservationData: null,
   281→      isDirectWalkIn: true,
   282→      walkInGuest: {
   283→        firstName: guestFirstName,
   284→        lastName: guestLastName,
   285→        email: guestEmail,
   286→        phone: guestPhone,
   287→        nationality: guestNationality,
   288→      },
   289→      walkInDates: {
   290→        checkIn: walkInDate,
   291→        checkOut: walkInCheckOut,
   292→        adults: walkInAdults,
   293→        children: walkInChildren,
   294→      },
   295→      startAtStep: 1,
   296→    })
   297→  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   298→
   299→  // Reservation Card Renderer ───────────────────────────────────────-----------------------------
   300→
   301→  const renderReservationCard = useCallback((res: ReservationData) => {
   302→    const isSelected = searchedReservationId === res.id
   303→    return (
   304→      <Card
   305→        key={res.id}
   306→        className={cn(
   307→          'p-2.5 cursor-pointer transition-all border',
   308→          isSelected
   309→            ? 'border-teal-400 ring-2 ring-teal-400/20 bg-teal-50/50 dark:bg-teal-950/20'
   310→            : 'hover:bg-muted/50'
   311→        )}
   312→        onClick={() => setSearchedReservationId(res.id)}
   313→      >
   314→        <div className="flex items-start justify-between gap-2">
   315→          <div className="space-y-1 min-w-0 flex-1">
   316→            <div className="flex items-center gap-2 flex-wrap">
   317→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   318→                {res.confirmationNo}
   319→              </Badge>
   320→              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   321→                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   322→                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   323→                  VIP
   324→                </Badge>
   325→              )}
   326→              {res.source && (
   327→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">
   328→                  {sourceLabel(res.source)}
   329→                </Badge>
   330→              )}
   331→            </div>
   332→            <p className="text-sm font-medium truncate">
   333→              {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
   334→            </p>
   335→            <div className="flex items-center gap-2 text-xs text-muted-foreground flex-wrap">
   336→              {res.room && <span>Room {res.room.number}</span>}
   337→              <span>·</span>
   338→              <span>{formatDate(res.checkIn)} → {formatDate(res.checkOut)}</span>
   339→              <span>·</span>
   340→              <span className="font-medium">{formatCurrency(res.roomRate)}/night</span>
   341→            </div>
   342→          </div>
   343→          <div className="flex flex-col items-end gap-1.5 shrink-0">
   344→            {isSelected && (
   345→              <Check className="w-4 h-4 text-teal-600" />
   346→            )}
   347→            {expressMode && (
   348→              <Button
   349→                size="sm"
   350→                className="h-7 text-[11px] px-2 bg-emerald-500 hover:bg-emerald-600 text-white"
   351→                onClick={(e) => {
   352→                  e.stopPropagation()
   353→                  setSearchedReservationId(res.id)
   354→                  handleExpressCheckIn(res)
   355→                }}
   356→              >
   357→                <Zap className="w-3 h-3 mr-1" />
   358→                Express
   359→              </Button>
   360→            )}
   361→          </div>
   362→        </div>
   363→      </Card>
   364→    )
   365→  }, [searchedReservationId, expressMode, handleExpressCheckIn])
   366→
   367→  // Render ──────────────────────────────────────────────────────────-----------------------------------------------
   368→  return (
   369→    <div className="flex flex-col h-full bg-background">
   370→      {/* Header */}
   371→      <div className="shrink-0 border-b bg-card">
   372→        <div className="flex items-center justify-between px-3 sm:px-6 py-2.5">
   373→          <button
   374→            type="button"
   375→            onClick={onBack}
   376→            className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors"
   377→          >
   378→            <ArrowLeft className="w-4 h-4" />
   379→            <span className="hidden sm:inline">Back</span>
   380→          </button>
   381→          <h1 className="text-base sm:text-lg font-semibold text-foreground">Guest Check-In</h1>
   382→          <div className="flex items-center gap-2">
   383→            <Zap className={cn('w-3.5 h-3.5', expressMode ? 'text-amber-500' : 'text-muted-foreground')} />
   384→            <span className="text-xs text-muted-foreground hidden sm:inline">Express</span>

   386→              id="express-mode"
   387→              checked={expressMode}

   389→              className="scale-90"
   390→            />
   391→          </div>
   392→        </div>
   393→      </div>
   394→
   395→      {/* Content */}
   396→      <div className="flex-1 overflow-y-auto p-3 sm:p-4">
   397→        <div className="max-w-2xl mx-auto space-y-2">
   398→
   399→          {/* Prefilled reservation mode */}
   400→          {prefillReservationId ? (
   401→            <>
   402→              {reservationLoading && (
   403→                <div className="space-y-2">
   404→                  <Skeleton className="h-4 w-3/4" />
   405→                  <Skeleton className="h-40 w-full" />
   406→                </div>
   407→              )}
   408→              {reservationError && (
   409→                <Card className="border-red-200 dark:border-red-800 p-2.5">
   410→                  <div className="flex items-start gap-2 text-red-600 dark:text-red-400">
   411→                    <svg className="h-5 w-5 mt-0.5 shrink-0" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>
   412→                    <div>
   413→                      <p className="font-medium text-sm">Reservation Not Found</p>
   414→                      <p className="text-sm text-muted-foreground mt-1">
   415→                        The reservation could not be found. Please search or use direct walk-in.
   416→                      </p>
   417→                    </div>
   418→                  </div>
   419→                  <Button variant="outline" size="sm" className="mt-2" onClick={() => { setPrefillReservationId(null) }}>
   420→                    Use Direct Walk-in Instead
   421→                  </Button>
   422→                </Card>
   423→              )}
   424→              {reservationData && (
   425→                <>
   426→                  <div className="flex items-center gap-2">
   427→                    <Badge variant="outline" className="border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">
   428→                      <Check className="w-3 h-3 mr-1" />
   429→                      Reservation Found
   430→                    </Badge>
   431→                  </div>
   432→                  <Card className="p-2.5">
   433→                    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   434→                      <InfoItem label="Confirmation #" value={reservationData.confirmationNo} />
   435→                      <div className="space-y-0.5">
   436→                        <span className="text-xs text-muted-foreground">Status</span>
   437→                        <p className="text-sm font-medium flex items-center gap-1"><StatusBadge status={reservationData.status} /></p>
   438→                      </div>

   440→                      <InfoItem label="Room" value={reservationData.room ? `${reservationData.room.number} - ${reservationData.room.type.name}` : '-'} />
   441→                      <InfoItem label="Stay" value={`${formatDate(reservationData.checkIn)} → ${formatDate(reservationData.checkOut)}`} />

   444→                    {/* Guest stay history */}
   445→                    {guestStayHistory && guestStayHistory.length > 0 && (

   447→                        <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground mb-1">

   449→                        </p>
   450→                        <div className="flex flex-wrap gap-1">

   453→                              {stay.roomNumber} · {formatDateShort(stay.checkIn)}
   454→                            </Badge>
   455→                          ))}
   456→                        </div>
   457→                      </div>
   458→                    )}
   459→                  </Card>
   460→                  <div className="flex gap-2">
   461→                    <Button onClick={handleUseReservation} className="flex-1">
   462→                      <Check className="w-4 h-4 mr-2" />
   463→                      Use Reservation
   464→                    </Button>
   465→                    {expressMode && (
   466→                      <Button variant="outline" className="flex-1" onClick={() => handleExpressCheckIn(reservationData)}>
   467→                        <Zap className="w-4 h-4 mr-2 text-amber-500" />
   468→                        Express Check-In
   469→                      </Button>
   470→                    )}
   471→                  </div>
   472→                </>
   473→              )}
   474→            </>
   475→          ) : (
   476→            <>
   477→              {/* Mode Toggle */}

   479→                <button

   481→                  onClick={() => setIsDirectWalkIn(false)}
   482→                  className={cn(

   489→                  <Search className="w-3.5 h-3.5 inline mr-1.5" />

   493→                  type="button"

   496→                    'flex-1 py-2 px-3 rounded-md text-sm font-medium transition-all text-center',

   506→
   507→              {!isDirectWalkIn ? (

   509→                <div className="space-y-2">

   511→                  <div className="relative">
   512→                    <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   513→                    <Input
   514→                      placeholder="Search by confirmation #, guest name, or room..."
   515→                      value={searchQuery}
   516→                      onChange={e => setSearchQuery(e.target.value)}
   517→                      className="pl-9 pr-9 h-10 text-sm"
   518→                    />
   519→                    {searchQuery && (
   520→                      <button
   521→                        type="button"
   522→                        onClick={() => { setSearchQuery(''); setDebouncedQuery(''); setSearchedReservationId(null) }}
