     6→import { apiFetch } from '@/lib/api'

     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,

    11→} from 'lucide-react'

    17→import { Input } from '@/components/ui/input'
    18→import { Label } from '@/components/ui/label'

    36→// ─── Types ──────────────────────────────────────────────────────────────
    37→
    38→interface DepartureGuest {
    39→  id: string
    40→  firstName: string
    41→  lastName: string
    42→  vipLevel: string
    43→}
    44→
    45→interface DepartureRoom {
    46→  id: string
    47→  number: string
    48→  floor: number
    49→  type: { name: string; code: string }
    50→}
    51→
    52→interface FolioLineItem {
    53→  id: string
    54→  description: string
    55→  amount: number
    56→  type: 'charge' | 'payment'
    57→  date: string
    58→}
    59→
    60→interface DepartureFolio {
    61→  id: string
    62→  balance: number
    63→  status: string
    64→  items?: FolioLineItem[]
    65→}
    66→
    67→interface Departure {
    68→  id: string
    69→  confirmationNo: string
    70→  status: string
    71→  checkIn: string
    72→  checkOut: string
    73→  roomRate: number
    74→  totalAmount: number
    75→  paidAmount: number
    76→  creditLimit: number
    77→  guest: DepartureGuest
    78→  room: DepartureRoom
    79→  folios: DepartureFolio[]
    80→}
    81→
    82→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    83→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    84→
    85→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    86→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    87→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    88→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    89→}
    90→
    91→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    92→  cash: 'Cash',
    93→  card: 'Credit/Debit Card',
    94→  bank_transfer: 'Bank Transfer',
    95→}
    96→
    97→// ─── Component ──────────────────────────────────────────────────────────
    98→
    99→export function DeparturesView() {
   100→  const queryClient = useQueryClient()
   101→  const { settings } = useSettingsStore()
   102→  const { preferences } = usePreferencesStore()
   103→  const { navigateTo } = useNavigationStore()
   104→  const today = getTodayString()
   105→
   106→  // Dialog states
   107→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   108→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   109→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   110→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   111→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   112→
   113→  // Selected departure
   114→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   115→
   116→  // Late checkout
   117→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   118→
   119→  // Payment
   120→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   121→  const [paymentAmount, setPaymentAmount] = useState('')
   122→  const [paymentReference, setPaymentReference] = useState('')
   123→
   124→  // Receipt data
   125→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   126→
   127→  // Solo Settlement dialog state
   128→  const [soloDialogOpen, setSoloDialogOpen] = useState(false)
   129→  const [soloDeparture, setSoloDeparture] = useState<Departure | null>(null)
   130→
   131→  // Fetch today's departures: checked_in with checkOut=today, AND checked_out today
   132→  // Uses status=checked_in,checked_out with checkOutDate filter
   133→  const { data, isLoading } = useQuery({
   134→    queryKey: ['departures', today],
   135→    queryFn: async () => {
   136→      const params = new URLSearchParams({
   137→        status: 'checked_in,checked_out',
   138→        checkOutDate: today,
   139→      })
   140→      return apiFetch(`/api/reservations?${params.toString()}`)
   141→    },
   142→    refetchInterval: 30000,
   143→  })
   144→
   145→  const departures: Departure[] = data?.reservations || []
   146→
   147→  // Stats
   148→  const totalDepartures = departures.length
   149→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   150→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   151→  const outstandingBalance = departures.reduce((sum, d) => {
   152→    return sum + (d.folios[0]?.balance || 0)
   153→  }, 0)
   154→
   155→  // Zero-balance pending guests (for express & batch checkout)
   156→  const zeroBalancePending = departures.filter(
   157→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   158→  )
   159→
   160→  // ─── Mutations ─────────────────────────────────────────────────────
   161→
   162→  // Checkout mutation
   163→  const checkoutMutation = useMutation({
   164→    mutationFn: async (reservationId: string) => {
   165→      return apiFetch(`/api/reservations/${reservationId}`, {
   166→        method: 'PATCH',
   167→        headers: { 'Content-Type': 'application/json' },
   168→        body: JSON.stringify({ status: 'checked_out' }),
   169→      })
   170→    },
   171→    onSuccess: () => {
   172→      invalidate.afterCheckout(queryClient)
   173→    },
   174→  })
   175→
   176→  // Payment mutation
   177→  const paymentMutation = useMutation({
   178→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   179→      return apiFetch(`/api/folio/${folioId}`, {
   180→        method: 'POST',
   181→        headers: { 'Content-Type': 'application/json' },
   182→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   183→      })
   184→    },
   185→    onSuccess: () => {
   186→      invalidate.afterFolioChange(queryClient)
   187→      setPaymentDialogOpen(false)
   188→      setPaymentAmount('')
   189→      setPaymentReference('')
   190→      toast.success('Payment recorded successfully')
   191→    },
   192→    onError: () => {
   193→      toast.error('Failed to record payment')
   194→    },
   195→  })
   196→
   197→  // Late checkout mutation
   198→  const lateCheckoutMutation = useMutation({
   199→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   200→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   201→        method: 'PATCH',
   202→        headers: { 'Content-Type': 'application/json' },
   203→        body: JSON.stringify({ checkOut: newCheckOut }),
   204→      })
   205→      // Add surcharge to folio
   206→      if (surcharge > 0) {
   207→        await apiFetch(`/api/folio`, {
   208→          method: 'POST',
   209→          headers: { 'Content-Type': 'application/json' },
   210→          body: JSON.stringify({
   211→            reservationId,
   212→            type: 'charge',
   213→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   214→            amount: surcharge,
   215→          }),
   216→        })
   217→      }
   218→      return result
   219→    },
   220→    onSuccess: () => {
   221→      invalidate.afterReservationChange(queryClient)
   222→      invalidate.afterFolioChange(queryClient)
   223→      setLateCheckoutDialogOpen(false)
   224→      toast.success('Late checkout request processed successfully')
   225→    },
   226→    onError: () => {
   227→      toast.error('Failed to process late checkout request')
   228→    },
   229→  })
   230→
   231→  // Batch checkout mutation
   232→  const batchCheckoutMutation = useMutation({
   233→    mutationFn: async (ids: string[]) => {
   234→      const results = []
   235→      for (let i = 0; i < ids.length; i++) {
   236→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   237→          method: 'PATCH',
   238→          headers: { 'Content-Type': 'application/json' },
   239→          body: JSON.stringify({ status: 'checked_out' }),
   240→        })
   241→        results.push(result)
   242→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   243→      }
   244→      return results
   245→    },
   246→    onSuccess: (_data, ids) => {
   247→      invalidate.afterCheckout(queryClient)
   248→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   249→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   250→    },
   251→    onError: () => {
   252→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   253→    },
   254→  })
   255→
   256→  // ─── Handlers ─────────────────────────────────────────────────────
   257→
   258→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   259→
   260→  const handleExpressCheckout = (dep: Departure) => {
   261→    checkoutMutation.mutate(dep.id, {
   262→      onSuccess: () => {
   263→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   264→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   265→      },
   266→    })
   267→  }
   268→
   269→  const handleReviewFolio = (dep: Departure) => {
   270→    setSelectedDeparture(dep)
   271→    setFolioDialogOpen(true)
   272→  }
   273→
   274→  const handleNavigateToFolio = (dep: Departure) => {
   275→    useFolioContextStore.getState().setFolioContext({
   276→      reservationId: dep.id,
   277→      guestId: dep.guest?.id || '',
   278→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   279→      roomNumber: dep.room?.number || '',
   280→      confirmationNo: dep.confirmationNo,
   281→    })
   282→    navigateTo('front-desk', 'folio')
   283→  }
   284→
   285→  const handleViewLedger = (dep: Departure) => {
   286→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   287→      guestId: dep.guest?.id || '',
   288→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   289→    })
   290→    navigateTo('front-desk', 'guest-ledger')
   291→  }
   292→
   293→  const handleFolioSettleCheckout = () => {
   294→    setFolioDialogOpen(false)
   295→    setCheckoutDialogOpen(true)
   296→  }
   297→
   298→  const handleQuickCheckout = (dep: Departure) => {
   299→    setSelectedDeparture(dep)
   300→    setCheckoutDialogOpen(true)
   301→  }
   302→
   303→  const handleRequestLateCheckout = (dep: Departure) => {
   304→    setSelectedDeparture(dep)
   305→    setLateCheckoutOption('14:00')
   306→    setLateCheckoutDialogOpen(true)
   307→  }
   308→
   309→  const handleConfirmLateCheckout = () => {
   310→    if (!selectedDeparture) return
   311→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   312→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   313→
   314→    // Build new checkout datetime with the selected time
   315→    const currentCheckout = new Date(selectedDeparture.checkOut)
   316→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   317→    currentCheckout.setHours(hours, minutes, 0, 0)
   318→
   319→    lateCheckoutMutation.mutate({
   320→      reservationId: selectedDeparture.id,
   321→      newCheckOut: currentCheckout.toISOString(),
   322→      surcharge,
   323→    })
   324→  }
   325→
   326→  const handleOpenPaymentDialog = () => {
   327→    if (!selectedDeparture) return
   328→    const balance = getBalance(selectedDeparture)
   329→    setPaymentAmount(String(balance))
   330→    setPaymentMethod('cash')
   331→    setPaymentReference('')
   332→    setPaymentDialogOpen(true)
   333→  }
   334→
   335→  const handleRecordPayment = () => {
   336→    if (!selectedDeparture) return
   337→    const folioId = selectedDeparture.folios[0]?.id
   338→    if (!folioId) {
   339→      toast.error('No folio found for this reservation')
   340→      return
   341→    }
   342→    const amount = Number(paymentAmount)
   343→    if (isNaN(amount) || amount <= 0) {
   344→      toast.error('Please enter a valid payment amount')
   345→      return
   346→    }
   347→    paymentMutation.mutate({
   348→      folioId,
   349→      amount,
   350→      method: paymentMethod,
   351→      reference: paymentReference,
   352→    })
   353→  }
   354→
   355→  const confirmCheckout = () => {
   356→    if (!selectedDeparture) return
   357→    const roomNumber = selectedDeparture.room.number
   358→    checkoutMutation.mutate(selectedDeparture.id, {
   359→      onSuccess: () => {
   360→        setCheckoutDialogOpen(false)
   361→        setReceiptData(selectedDeparture)
   362→        setReceiptDialogOpen(true)
   363→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   364→      },
   365→    })
   366→  }
   367→
   368→  const handleBatchCheckout = () => {
   369→    const ids = zeroBalancePending.map((d) => d.id)
   370→    if (ids.length === 0) {
   371→      toast.info('No guests with zero balance to batch checkout')
   372→      return
   373→    }
   374→    batchCheckoutMutation.mutate(ids)
   375→  }
   376→
   377→  const handlePrintReceipt = () => {
   378→    toast.success('Receipt sent to printer')
   379→  }
   380→
   381→  const handleEmailReceipt = () => {
   382→    toast.success('Receipt emailed to guest')
   383→  }
   384→
   385→  // Solo Departure — opens settlement page for a single guest
   386→  const handleSoloDeparture = (dep: Departure) => {
   387→    setSoloDeparture(dep)
   388→    setSoloDialogOpen(true)
   389→  }
   390→
   391→  // Force Checkout — marks a checked_in guest as checked_out regardless of balance
   392→  const handleForceCheckout = (dep: Departure) => {
   393→    setSelectedDeparture(dep)
   394→    setCheckoutDialogOpen(true)
   395→  }
   396→
   397→  // Solo settlement complete (settle + checkout)
   398→  const handleSoloSettleCheckout = () => {
   399→    if (!soloDeparture) return
   400→    const roomNumber = soloDeparture.room.number
   401→    checkoutMutation.mutate(soloDeparture.id, {
   402→      onSuccess: () => {
   403→        setSoloDialogOpen(false)
   404→        setReceiptData(soloDeparture)
   405→        setReceiptDialogOpen(true)
   406→        toast.success(`Solo checkout complete for Room ${roomNumber}`)
   407→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   408→      },
   409→    })
   410→  }
   411→
   412→  // Solo record payment during settlement
   413→  const handleSoloRecordPayment = () => {
   414→    if (!soloDeparture) return
   415→    const folioId = soloDeparture.folios[0]?.id
   416→    if (!folioId) {
   417→      toast.error('No folio found for this reservation')
   418→      return
   419→    }
   420→    const amount = Number(paymentAmount)
   421→    if (isNaN(amount) || amount <= 0) {
   422→      toast.error('Please enter a valid payment amount')
   423→      return
   424→    }
   425→    paymentMutation.mutate(
   426→      { folioId, amount, method: paymentMethod, reference: paymentReference },
   427→      {
   428→        onSuccess: () => {
   429→          // Refresh solo departure data after payment
   430→          queryClient.invalidateQueries({ queryKey: ['departures', today] })
   431→        },
   432→      },
   433→    )
   434→  }
   435→
   436→  // ─── Computed values ──────────────────────────────────────────────
   437→
   438→  const lateCheckoutFee = selectedDeparture
   439→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   440→    : 0
   441→
   442→  // ─── Render ────────────────────────────────────────────────────────
   443→
   444→  return (
   445→    <div className="flex flex-col gap-2">
   446→      {/* Header */}
   447→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   448→        <div>
   449→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   450→          <p className="text-xs text-muted-foreground">
   451→            Guest check-outs scheduled for {formatDate(today)}
   452→          </p>
   453→        </div>
   454→        {zeroBalancePending.length > 0 && (
   455→          <Button
   456→            onClick={handleBatchCheckout}
   457→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   458→            disabled={batchCheckoutMutation.isPending}
   459→          >
   460→            <Zap className="size-4 mr-1.5" />
   461→            {batchCheckoutMutation.isPending
   462→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   463→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   464→          </Button>
   465→        )}
   466→      </div>
   467→
   468→      {/* Departures List */}
   469→      <Card className="py-0">
   470→        <CardContent className="p-0">
   471→          <div className="max-h-[600px] overflow-y-auto">
   472→            <Table>
   473→              <TableHeader>
   474→                <TableRow>
   475→                  <TableHead className="w-[70px]">Room</TableHead>
   476→                  <TableHead>Guest</TableHead>
   477→                  <TableHead className="w-[100px]">Check-out</TableHead>
   478→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   479→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   480→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   481→                </TableRow>
   482→              </TableHeader>
   483→              <TableBody>
   484→                {isLoading ? (
   485→                  Array.from({ length: 5 }).map((_, i) => (
   486→                    <TableRow key={i}>
   487→                      {Array.from({ length: 6 }).map((_, j) => (
   488→                        <TableCell key={j}>
   489→                          <Skeleton className="h-4 w-full" />
   490→                        </TableCell>
   491→                      ))}
   492→                    </TableRow>
   493→                  ))
   494→                ) : departures.length === 0 ? (
   495→                  <TableRow>
   496→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   497→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   498→                      No departures scheduled for today
   499→                    </TableCell>
   500→                  </TableRow>
   501→                ) : (
   502→                  departures.map((dep) => {
   503→                    const balance = getBalance(dep)
   504→                    const isPending = dep.status === 'checked_in'
   505→                    const isZeroBalance = balance === 0
   506→                    return (
   507→                      <TableRow key={dep.id}>
   508→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   509→                        <TableCell>
   510→                          <div className="flex items-center gap-2">
   511→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   512→                            {dep.guest.vipLevel !== 'none' && (
   513→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   514→                                VIP
   515→                              </Badge>
   516→                            )}
   517→                          </div>
   518→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   519→                        </TableCell>
   520→                        <TableCell className="text-xs">
   521→                          <div>{formatDate(dep.checkOut)}</div>
   522→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   523→                        </TableCell>
   524→                        <TableCell className="text-right">
   525→                          <div className="flex items-center justify-end gap-1.5">
   526→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   527→                              {formatCurrency(balance)}
   528→                            </span>
   529→                            {/* Outstanding Balance Alert Badges */}
   530→                            {isPending && balance > 5000 && (
   531→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   532→                                HIGH
   533→                              </Badge>
   534→                            )}
   535→                            {isPending && balance > 0 && balance <= 5000 && (
   536→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   537→                                DUE
   538→                              </Badge>
   539→                            )}
   540→                          </div>
   541→                        </TableCell>
   542→                        <TableCell className="text-right">
   543→                          {dep.status === 'checked_out' ? (
   544→                            <div className="flex items-center justify-end gap-1">
   545→                              <BedSingle className="size-3 text-yellow-500" />
   546→                              <StatusBadge status="vacant_dirty" />
   547→                            </div>
   548→                          ) : (
   549→                            <StatusBadge status="checked_in" />
   550→                          )}
   551→                        </TableCell>
   552→                        <TableCell className="text-right">
   553→                          <div className="flex justify-end gap-1 flex-wrap">
   554→                            {isPending && (
   555→                              <>
