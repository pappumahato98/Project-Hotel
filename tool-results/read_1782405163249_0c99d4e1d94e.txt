   146→function NumberStepper({ label, value, min, max, onChange }: {
   147→  label: string; value: number; min: number; max: number; onChange: (v: number) => void
   148→}) {
   149→  return (
   150→    <div className="flex items-center h-10 rounded-lg border border-border bg-background px-3 gap-2 flex-1">
   151→      <span className="text-xs text-muted-foreground font-medium whitespace-nowrap">{label}</span>
   152→      <div className="flex items-center gap-0 ml-auto">
   153→        <button
   154→          type="button"
   155→          onClick={() => onChange(Math.max(min, value - 1))}
   156→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   157→        >
   158→          <Minus className="w-3 h-3" />
   159→        </button>
   160→        <span className="w-8 text-center text-sm font-semibold tabular-nums">{value}</span>
   161→        <button
   162→          type="button"
   163→          onClick={() => onChange(Math.min(max, value + 1))}
   164→          className="w-7 h-7 rounded-md flex items-center justify-center hover:bg-muted transition-colors"
   165→        >
   166→          <Plus className="w-3 h-3" />
   167→        </button>
   168→      </div>
   169→    </div>
   170→  )
   171→}

   173→function sourceLabel(source: string | null) {
   174→  switch (source) {
   175→    case 'walk_in': return 'Walk-in'
   176→    case 'online': return 'Online'
   177→    case 'phone': return 'Phone'
   178→    case 'travel_agent': return 'Travel Agent'
   179→    case 'corporate': return 'Corporate'
   180→    default: return source || 'Direct'
   181→  }
   182→}

   184→function getRoomStatusColor(status: string) {
   185→  switch (status) {
   186→    case 'vacant_clean':
   187→    case 'inspected':
   188→      return 'bg-emerald-500'
   189→    case 'occupied':
   190→      return 'bg-red-500'
   191→    case 'vacant_dirty':
   192→    case 'cleaning':
   193→    case 'on_change':
   194→      return 'bg-yellow-400'
   195→    case 'out_of_order':
   196→      return 'bg-gray-400'
   197→    default:
   198→      return 'bg-gray-300'
   199→  }

   244→  const [guestPhone, setGuestPhone] = useState('')
   245→  const [guestNationality, setGuestNationality] = useState('')
   246→
   247→  // ─── Debounced search ──────────────────────────────────────────────────────
   248→  useEffect(() => {
   249→    const timer = setTimeout(() => {
   250→      setDebouncedQuery(searchQuery)
   251→    }, 300)
   252→    return () => clearTimeout(timer)
   253→  }, [searchQuery])
   254→
   255→  // ─── Data Queries ──────────────────────────────────────────────────────────
   256→
   257→  const activeReservationId = prefillReservationId || selectedReservationId
   258→

   262→    error: reservationError,
   263→  } = useQuery({
   264→    queryKey: ['reservation', activeReservationId],
   265→    queryFn: async () => {
   266→      if (!activeReservationId) return null
   267→      const data = await apiFetch(`/api/reservations/${activeReservationId}`) as { reservation: ReservationData }
   268→      return data.reservation
   269→    },
   270→    enabled: !!activeReservationId,
   271→  })
   272→
   273→  // Guest stay history
   274→  const { data: guestStayHistory } = useQuery({
   275→    queryKey: ['guest-stays', reservationData?.guest?.id],
   276→    queryFn: async () => {
   277→      if (!reservationData?.guest?.id) return null
   278→      try {
   279→        return await apiFetch(`/api/guests/${reservationData.guest.id}/stays`) as Array<{ id: string; checkIn: string; checkOut: string; roomNumber: string }>
   280→      } catch {
   281→        return null
   282→      }
   283→    },
   284→    enabled: !!reservationData?.guest?.id && !!selectedReservationId,
   285→  })
   286→
   287→  // Today's expected arrivals
   288→  const {
   289→    data: arrivalsData,
   290→    isLoading: arrivalsLoading,
   291→  } = useQuery({
   292→    queryKey: ['arrivals', today],
   293→    queryFn: async () => {
   294→      const data = await apiFetch(`/api/reservations?date=${today}&status=confirmed`) as { reservations: ReservationData[]; total: number }
   295→      return data.reservations || []
   296→    },
   297→    enabled: activeTab === 'find-reservation' && !prefillReservationId,
   298→  })
   299→
   300→  // Reservation search (debounced)
   301→  const {
   302→    data: searchResults,
   303→    isLoading: searchLoading,
   304→  } = useQuery({
   305→    queryKey: ['reservation-search', debouncedQuery],
   306→    queryFn: async () => {
   307→      if (!debouncedQuery || debouncedQuery.length < 2) return null
   308→      const data = await apiFetch(`/api/reservations?search=${encodeURIComponent(debouncedQuery)}&status=confirmed,tentative`)
   309→      return (data.reservations || []) as ReservationData[]
   310→    },
   311→    enabled: debouncedQuery.length >= 2,
   312→  })
   313→
   314→  // Room data for Direct Check-in right panel
   315→  const {
   316→    data: roomsData,
   317→    isLoading: roomsLoading,
   318→  } = useQuery({
   319→    queryKey: ['rooms-summary'],
   320→    queryFn: async () => {
   321→      const data = await apiFetch('/api/rooms') as RoomsApiResponse
   322→      return data
   323→    },
   324→    enabled: activeTab === 'direct-checkin' && !prefillReservationId,
   325→  })
   326→
   327→  // ─── Handlers ──────────────────────────────────────────────────────────────
   328→
   329→  const navigateToProcess = useCallback((session: CheckInSession) => {
   330→    setCheckInSession(session)
   331→    setActiveSubModule('check-in-process')

   333→

   336→    navigateToProcess({
   337→      reservationId: reservationData.id,
   338→      reservationData: reservationData as unknown as Record<string, unknown>,
   339→      isDirectWalkIn: false,
   340→      walkInGuest: null,
   341→      walkInDates: null,
   342→      startAtStep: 1,
   343→    })

   345→  }, [reservationData, navigateToProcess])
   346→
   347→  const handleExpressCheckIn = useCallback((res: ReservationData) => {
   348→    navigateToProcess({
   349→      reservationId: res.id,
   350→      reservationData: res as unknown as Record<string, unknown>,
   351→      isDirectWalkIn: false,
   352→      walkInGuest: null,
   353→      walkInDates: null,
   354→      startAtStep: 2,
   355→    })
   356→    toast.success('Express check-in - skipping to Room & Rate')
   357→  }, [navigateToProcess])
   358→
   359→  const handleWalkInContinue = useCallback(() => {
   360→    if (!guestFirstName || !guestLastName) {
   361→      toast.error('Please enter guest first and last name')
   362→      return
   363→    }

   365→      reservationId: null,
   366→      reservationData: null,
   367→      isDirectWalkIn: true,
   368→      walkInGuest: {
   369→        firstName: guestFirstName,
   370→        lastName: guestLastName,
   371→        email: guestEmail,
   372→        phone: guestPhone,
   373→        nationality: guestNationality,
   374→      },
   375→      walkInDates: {
   376→        checkIn: walkInDate,
   377→        checkOut: walkInCheckOut,
   378→        adults: walkInAdults,
   379→        children: walkInChildren,
   380→      },
   381→      startAtStep: 1,
   382→    })
   383→  }, [guestFirstName, guestLastName, guestEmail, guestPhone, guestNationality, walkInDate, walkInCheckOut, walkInAdults, walkInChildren, navigateToProcess])
   384→

   386→

   398→        onClick={() => setSelectedReservationId(res.id)}

   400→        <div className="flex items-start justify-between gap-2">
   401→          <div className="space-y-1.5 min-w-0 flex-1">
   402→            <div className="flex items-center gap-2 flex-wrap">
   403→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">
   404→                {res.confirmationNo}
   405→              </Badge>
   406→              {res.guest?.vipLevel && res.guest.vipLevel !== 'none' && (
   407→                <Badge className="text-[9px] h-4 px-1.5 bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800">
   408→                  <Crown className="w-2.5 h-2.5 mr-0.5" />
   409→                  VIP

   413→                <Badge variant="secondary" className="text-[9px] h-4 px-1.5">

   459→          <div className="w-16 h-16 rounded-full bg-muted flex items-center justify-center mb-4">
   460→            <Eye className="w-7 h-7 text-muted-foreground" />
   461→          </div>
   462→          <p className="text-sm font-medium text-muted-foreground">
   463→            Select a reservation to view details

   465→          <p className="text-xs text-muted-foreground/70 mt-1">
   466→            Click on any reservation card from the list

   494→
   495→    const res = reservationData
   496→    const nights = nightsBetween(res.checkIn, res.checkOut)
   497→
   498→    return (
   499→      <div className="p-4 space-y-4 overflow-y-auto h-full">
   500→        {/* Guest Info Section */}
   501→        <div>
   502→          <h4 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">Guest Information</h4>

   504→            <div className="space-y-2.5">
   505→              <div className="flex items-center gap-2">
   506→                <div className="w-9 h-9 rounded-full bg-teal-100 dark:bg-teal-900/40 flex items-center justify-center shrink-0">
   507→                  <User className="w-4 h-4 text-teal-700 dark:text-teal-300" />
   508→                </div>
   509→                <div className="min-w-0">
   510→                  <p className="text-sm font-semibold truncate">
   511→                    {res.guest ? `${res.guest.firstName} ${res.guest.lastName}` : 'Guest unknown'}
