   179→// ─── Debounce hook ────────────────────────────────────────────────────
   180→function useDebounce<T>(value: T, delay: number): T {
   181→  const [debounced, setDebounced] = useState(value)
   182→  useEffect(() => {
   183→    const timer = setTimeout(() => setDebounced(value), delay)
   184→    return () => clearTimeout(timer)
   185→  }, [value, delay])
   186→  return debounced
   187→}

   189→// ─── Date range presets ──────────────────────────────────────────────
   190→function getDatePreset(preset: string): { from: Date; to: Date } {
   191→  const today = new Date()
   192→  today.setHours(0, 0, 0, 0)
   193→  switch (preset) {
   194→    case 'today':
   195→      return { from: today, to: today }
   196→    case 'this-week': {
   197→      const from = new Date(today)
   198→      from.setDate(today.getDate() - today.getDay() + 1) // Monday
   199→      const to = new Date(from)
   200→      to.setDate(from.getDate() + 6) // Sunday
   201→      return { from, to }
   202→    }
   203→    case 'this-month': {
   204→      const from = new Date(today.getFullYear(), today.getMonth(), 1)
   205→      const to = new Date(today.getFullYear(), today.getMonth() + 1, 0)
   206→      return { from, to }
   207→    }
   208→    case 'next-7': {
   209→      const to = new Date(today)
   210→      to.setDate(today.getDate() + 6)
   211→      return { from: today, to }
   212→    }
   213→    case 'next-14': {
   214→      const to = new Date(today)
   215→      to.setDate(today.getDate() + 13)
   216→      return { from: today, to }
   217→    }
   218→    case 'next-30': {
   219→      const to = new Date(today)
   220→      to.setDate(today.getDate() + 29)
   221→      return { from: today, to }
   222→    }
   223→    default:
   224→      return { from: today, to: today }
   225→  }
   226→}

   228→function toDateString(d: Date): string {
   229→  const y = d.getFullYear()
   230→  const m = String(d.getMonth() + 1).padStart(2, '0')
   231→  const day = String(d.getDate()).padStart(2, '0')
   232→  return `${y}-${m}-${day}`
   233→}
   234→
   235→function formatShortDate(dateStr: string | Date): string {
   236→  const d = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
   237→  if (isNaN(d.getTime())) return '—'
   238→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: '2-digit' })
   239→}
