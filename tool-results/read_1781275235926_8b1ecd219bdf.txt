   102→   102→  // Dialog states

   108→   108→
   109→   109→  // Selected departure
   110→   110→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   111→   111→
   112→   112→  // Late checkout

   120→   120→  // Receipt data

   125→   125→    queryKey: ['departures', today],
   126→   126→    queryFn: async () => {

   147→   147→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,

   155→   155→      const res = await fetch(`/api/reservations/${reservationId}`, {
   156→   156→        method: 'PATCH',
   157→   157→        headers: { 'Content-Type': 'application/json' },

   164→   164→      queryClient.invalidateQueries({ queryKey: ['departures'] })

   175→   175→        headers: { 'Content-Type': 'application/json' },

   179→   179→      return res.json()

   182→   182→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   183→   183→      queryClient.invalidateQueries({ queryKey: ['folios'] })

   185→   185→      setPaymentAmount('')

   198→   198→        method: 'PATCH',
   199→   199→        headers: { 'Content-Type': 'application/json' },

   205→   205→        const folioRes = await fetch(`/api/folio`, {
   206→   206→          method: 'POST',
   207→   207→          headers: { 'Content-Type': 'application/json' },
   208→   208→          body: JSON.stringify({
   209→   209→            reservationId,
   210→   210→            type: 'charge',
   211→   211→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   212→   212→            amount: surcharge,
   213→   213→          }),
   214→   214→        })
   215→   215→        if (!folioRes.ok) throw new Error('Failed to add surcharge')
   216→   216→      }
   217→   217→      return res.json()
   218→   218→    },
   219→   219→    onSuccess: () => {
   220→   220→      queryClient.invalidateQueries({ queryKey: ['departures'] })

   222→   222→      queryClient.invalidateQueries({ queryKey: ['folios'] })
   223→   223→      setLateCheckoutDialogOpen(false)

   225→   225→    },
   226→   226→    onError: () => {
   227→   227→      toast.error('Failed to process late checkout request')
   228→   228→    },
   229→   229→  })

   233→   233→    mutationFn: async (ids: string[]) => {

   236→   236→        const res = await fetch(`/api/reservations/${ids[i]}`, {
   237→   237→          method: 'PATCH',
   238→   238→          headers: { 'Content-Type': 'application/json' },
   239→   239→          body: JSON.stringify({ status: 'checked_out' }),
   240→   240→        })
   241→   241→        if (!res.ok) throw new Error(`Failed to checkout room ${i + 1}`)
   242→   242→        results.push(await res.json())
   243→   243→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   244→   244→      }
   245→   245→      return results
   246→   246→    },
   247→   247→    onSuccess: (_data, ids) => {
   248→   248→      queryClient.invalidateQueries({ queryKey: ['departures'] })
   249→   249→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   250→   250→      queryClient.invalidateQueries({ queryKey: ['in-house'] })

   252→   252→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   253→   253→    },
   254→   254→    onError: () => {
   255→   255→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   256→   256→    },
   257→   257→  })
   258→   258→
   259→   259→  // ─── Handlers ─────────────────────────────────────────────────────
   260→   260→

   262→   262→

   264→   264→    checkoutMutation.mutate(dep.id, {
   265→   265→      onSuccess: () => {
   266→   266→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   267→   267→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   268→   268→      },
   269→   269→    })
   270→   270→  }
   271→   271→
   272→   272→  const handleReviewFolio = (dep: Departure) => {

   289→   289→    setLateCheckoutOption('14:00')

   296→   296→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   297→   297→

   299→   299→    const currentCheckout = new Date(selectedDeparture.checkOut)
   300→   300→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   301→   301→    currentCheckout.setHours(hours, minutes, 0, 0)
   302→   302→
   303→   303→    lateCheckoutMutation.mutate({
   304→   304→      reservationId: selectedDeparture.id,
   305→   305→      newCheckOut: currentCheckout.toISOString(),

   307→   307→    })
   308→   308→  }

   330→   330→    }
   331→   331→    paymentMutation.mutate({
   332→   332→      folioId,
   333→   333→      amount,
   334→   334→      method: paymentMethod,
   335→   335→      reference: paymentReference,
   336→   336→    })

   367→   367→  }
   368→   368→
   369→   369→  // ─── Computed values ──────────────────────────────────────────────
   370→   370→
   371→   371→  const lateCheckoutFee = selectedDeparture
   372→   372→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   373→   373→    : 0
   374→   374→

   395→   395→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   396→   396→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}

   398→   398→        )}
   399→   399→      </div>
   400→   400→
   401→   401→      {/* Departures List */}
   402→   402→      <Card className="py-0">

   404→   404→          <div className="max-h-[600px] overflow-y-auto">
   405→   405→            <Table>
   406→   406→              <TableHeader>
   407→   407→                <TableRow>
   408→   408→                  <TableHead className="w-[70px]">Room</TableHead>
   409→   409→                  <TableHead>Guest</TableHead>
   410→   410→                  <TableHead className="w-[100px]">Check-out</TableHead>
   411→   411→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   412→   412→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   413→   413→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   414→   414→                </TableRow>
   415→   415→              </TableHeader>
   416→   416→              <TableBody>

   418→   418→                  Array.from({ length: 5 }).map((_, i) => (
   419→   419→                    <TableRow key={i}>
   420→   420→                      {Array.from({ length: 6 }).map((_, j) => (
   421→   421→                        <TableCell key={j}>
   422→   422→                          <Skeleton className="h-4 w-full" />
   423→   423→                        </TableCell>
   424→   424→                      ))}
   425→   425→                    </TableRow>
   426→   426→                  ))
   427→   427→                ) : departures.length === 0 ? (
   428→   428→                  <TableRow>
   429→   429→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   430→   430→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   431→   431→                      No departures scheduled for today
   432→   432→                    </TableCell>
   433→   433→                  </TableRow>
   434→   434→                ) : (
   435→   435→                  departures.map((dep) => {
   436→   436→                    const balance = getBalance(dep)
   437→   437→                    const isPending = dep.status === 'checked_in'
   438→   438→                    const isZeroBalance = balance === 0
   439→   439→                    return (

   441→   441→                        <TableCell className="font-bold font-mono">{dep.room.number}</TableCell>
   442→   442→                        <TableCell>

   444→   444→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>

   447→   447→                                VIP
   448→   448→                              </Badge>
   449→   449→                            )}

   452→   452→                        </TableCell>
