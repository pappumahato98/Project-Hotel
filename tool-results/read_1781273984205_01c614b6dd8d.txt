   253→// ─── Date Helpers ────────────────────────────────────────────────────────
   254→
   255→function addDays(date: Date, days: number): Date {
   256→  const d = new Date(date)
   257→  d.setDate(d.getDate() + days)
   258→  return d
   259→}
   260→
   261→function dateToKey(date: Date): string {
   262→  const y = date.getFullYear()
   263→  const m = String(date.getMonth() + 1).padStart(2, '0')
   264→  const d = String(date.getDate()).padStart(2, '0')
   265→  return `${y}-${m}-${d}`
   266→}
   267→
   268→function toDateOnly(dateStr: string): Date {
   269→  const d = new Date(dateStr)
   270→  d.setHours(0, 0, 0, 0)
   271→  return d
   272→}
   273→
   274→function isSameDay(a: Date, b: Date): boolean {
   275→  return (
   276→    a.getFullYear() === b.getFullYear() &&
   277→    a.getMonth() === b.getMonth() &&
   278→    a.getDate() === b.getDate()
   279→  )
   280→}
   281→
   282→function isWeekend(date: Date): boolean {
   283→  const day = date.getDay()
   284→  return day === 0 || day === 6
   285→}
   286→
   287→function formatDateShort(dateStr: string): string {
   288→  const d = new Date(dateStr)
   289→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   290→}
   291→
   292→// Determine the display category for a reservation block
   293→function getReservationCategory(res: CalendarReservation, todayStr: string): string {
   294→  const checkInStr = dateToKey(toDateOnly(res.checkIn))
   295→  const checkOutStr = dateToKey(toDateOnly(res.checkOut))
   296→
   297→  if (res.status === 'tentative') return 'tentative'
   298→  if (checkInStr === todayStr) return 'arrival_today'
   299→  if (checkOutStr === todayStr && res.status === 'checked_in') return 'departure_today'
   300→  if (res.status === 'checked_in') return 'checked_in'
   301→  return 'confirmed'
   302→}
