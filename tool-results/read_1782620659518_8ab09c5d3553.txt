   159→// ─── Component ──────────────────────────────────────────────────────────
   160→
   161→export function ArrivalsView() {
   162→  const queryClient = useQueryClient()
   163→  const { settings } = useSettingsStore()
   164→  const { navigateTo } = useNavigationStore()
   165→  const today = getTodayString()
   166→
   167→  // Dialog states
   168→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   169→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   170→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   171→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   172→
   173→  // Selection states
   174→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   175→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   176→
   177→  // Check-in form states
   178→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   179→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   180→  const [roomPreference, setRoomPreference] = useState('')
   181→  const [pillowType, setPillowType] = useState('')
   182→  const [wakeupCall, setWakeupCall] = useState('')
   183→
   184→  // Key card states
   185→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   186→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   187→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   188→
   189→  // Mobile detail view state
   190→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   191→
   192→  // Walk-in form state
   193→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   194→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   195→
   196→  // ─── Queries ────────────────────────────────────────────────────────
   197→
   198→  const { data, isLoading } = useQuery({
   199→    queryKey: ['arrivals', today],
   200→    queryFn: async () => {
   201→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   202→      return apiFetch(`/api/reservations?${params.toString()}`)
   203→    },
   204→    refetchInterval: 30000,
   205→  })
   206→
   207→  const { data: roomsData } = useQuery({
   208→    queryKey: ['rooms', 'vacant'],
   209→    queryFn: () => apiFetch('/api/rooms'),
   210→  })
   211→
   212→  const arrivals: Arrival[] = data?.reservations || []
   213→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   214→  const availableRooms = allRooms.filter(
   215→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   216→  )
   217→
   218→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   219→
   220→  const sortedArrivals = useMemo(() => {
   221→    return [...arrivals].sort((a, b) => {
   222→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   223→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   224→      if (aVip && !bVip) return -1
   225→      if (!aVip && bVip) return 1
   226→      return 0
   227→    })
   228→  }, [arrivals])
   229→
   230→  // ─── Stats ──────────────────────────────────────────────────────────
   231→
   232→  const totalArrivals = arrivals.length
   233→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   234→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   235→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   236→
   237→  // ─── Mutations ──────────────────────────────────────────────────────
   238→
   239→  const assignRoomMutation = useMutation({
   240→    mutationFn: async ({ reservationId, roomId, roomTypeId, specialRequests }: { reservationId: string; roomId: string; roomTypeId: string; specialRequests?: string }) => {
   241→      return apiFetch('/api/check-in', {
   242→        method: 'POST',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify({
   245→          source: 'reservation',
   246→          reservationId,
   247→          roomId,
   248→          roomTypeId,
   249→          adults: selectedArrival?.adults || 1,
   250→          children: selectedArrival?.children || 0,
   251→          roomRate: selectedArrival?.roomRate || 0,
   252→          specialRequests,
   253→        }),
   254→      }) as Promise<{ reservation: { room?: { number: string } } }>
   255→    },
   256→    onSuccess: (_data, variables) => {
   257→      invalidate.afterCheckIn(queryClient)
   258→      setRoomPickerOpen(false)
   259→      setCheckInDialogOpen(false)
   260→
   261→      const room = availableRooms.find((r) => r.id === variables.roomId)
   262→      const roomNumber = room?.number || _data?.reservation?.room?.number || ''
   263→      openKeyCardDialog(roomNumber)
   264→    },
   265→    onError: (error) => {
   266→      toast.error('Check-in failed', {
   267→        description: error instanceof Error ? error.message : 'Unknown error',
   268→      })
   269→    },
   270→  })
   271→
   272→  const checkInMutation = useMutation({
   273→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   274→      const arrival = arrivals.find(a => a.id === reservationId)
   275→      if (!arrival?.room) throw new Error('No room assigned to this reservation')
   276→      return apiFetch('/api/check-in', {
   277→        method: 'POST',
   278→        headers: { 'Content-Type': 'application/json' },
   279→        body: JSON.stringify({
   280→          source: 'reservation',
   281→          reservationId,
   282→          roomId: arrival.room.id,
   283→          roomTypeId: arrival.room.typeId,
   284→          adults: arrival.adults,
   285→          children: arrival.children,
   286→          roomRate: arrival.roomRate,
   287→          specialRequests,
   288→        }),
   289→      }) as Promise<{ reservation: { room?: { number: string } } }>
   290→    },
   291→    onSuccess: (_data) => {
   292→      invalidate.afterCheckIn(queryClient)
   293→      setCheckInDialogOpen(false)
   294→
   295→      const roomNumber = selectedArrival?.room?.number || _data?.reservation?.room?.number || ''
   296→      openKeyCardDialog(roomNumber)
   297→    },
   298→    onError: (error) => {
   299→      toast.error('Check-in failed', {
   300→        description: error instanceof Error ? error.message : 'Unknown error',
   301→      })
   302→    },
   303→  })
   304→
   305→  const walkInMutation = useMutation({
   306→    mutationFn: async (form: WalkInForm) => {
   307→      const vacantRoom = availableRooms[0]
   308→      if (!vacantRoom) throw new Error('No available rooms')
   309→
   310→      const checkOutDate = form.checkOutDate || (() => {
   311→        const d = new Date()
   312→        d.setDate(d.getDate() + 1)
   313→        return d
   314→      })()
   315→
   316→      return apiFetch('/api/check-in', {
   317→        method: 'POST',
   318→        headers: { 'Content-Type': 'application/json' },
   319→        body: JSON.stringify({
   320→          source: 'direct',
   321→          guest: {
   322→            firstName: form.firstName,
   323→            lastName: form.lastName,
   324→            phone: form.phone || undefined,
   325→            email: form.email || undefined,
   326→            nationality: form.nationality || undefined,
   327→            idType: form.idType || undefined,
   328→            idNumber: form.idNumber || undefined,
   329→          },
   330→          roomId: vacantRoom.id,
   331→          roomTypeId: vacantRoom.typeId,
   332→          adults: form.adults,
   333→          children: form.children,
   334→          roomRate: form.roomRate,
   335→          checkIn: new Date().toISOString(),
   336→          checkOut: checkOutDate.toISOString(),
   337→        }),
   338→      }) as Promise<{ reservation: { room?: { number: string } } }>
   339→    },
   340→    onSuccess: (result) => {
   341→      invalidate.afterCheckIn(queryClient)
   342→      setWalkInDialogOpen(false)
   343→      setWalkInForm(initialWalkInForm)
   344→
   345→      const roomNumber = result?.reservation?.room?.number || ''
   346→      openKeyCardDialog(roomNumber)
   347→    },
   348→    onError: (error) => {
   349→      toast.error('Walk-in check-in failed', {
   350→        description: error instanceof Error ? error.message : 'Unknown error',
   351→      })
   352→    },
   353→  })
   354→
   355→  // ─── Handlers ──────────────────────────────────────────────────────
   356→
   357→  const handleAssignRoom = (arrival: Arrival) => {
   358→    setSelectedArrival(arrival)
   359→    setRoomPickerOpen(true)
   360→  }
   361→
   362→  const handleQuickCheckIn = (arrival: Arrival) => {
   363→    if (!arrival.room) {
   364→      setSelectedArrival(arrival)
   365→      setRoomPickerOpen(true)
   366→      return
   367→    }
   368→    setSelectedArrival(arrival)
   369→    resetCheckInForm()
   370→    setCheckInDialogOpen(true)
   371→  }
   372→
   373→  const handleOpenWalkIn = () => {
   374→    setWalkInForm(initialWalkInForm)
   375→    setWalkInDialogOpen(true)
   376→  }
   377→
   378→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   379→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   380→    navigateTo('front-desk', 'check-in')
   381→  }
   382→
   383→  const handleViewLedger = (arrival: Arrival) => {
   384→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   385→      guestId: arrival.guest?.id || '',
   386→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   387→    })
   388→    navigateTo('front-desk', 'guest-ledger')
   389→  }
   390→
   391→  const buildSpecialRequests = (arrival: Arrival): string => {
   392→    const parts: string[] = []
   393→
   394→    // Preserve existing requests
   395→    if (arrival.specialRequests) {
   396→      parts.push(arrival.specialRequests)
   397→    }
   398→
   399→    // Early check-in note
   400→    if (earlyCheckIn) {
   401→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   402→    }
   403→
   404→    // Check-out time override
   405→    if (checkOutTime !== settings.defaultCheckOut) {
   406→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   407→    }
   408→
   409→    // Room preference
   410→    if (roomPreference) {
   411→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   412→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   413→    }
   414→
   415→    // Pillow type
   416→    if (pillowType) {
   417→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   418→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   419→    }
   420→
   421→    // Wake-up call
   422→    if (wakeupCall) {
   423→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   424→    }
   425→
   426→    return parts.join(' | ')
   427→  }
   428→
   429→  const confirmCheckIn = () => {
   430→    if (!selectedArrival) return
   431→
   432→    const updatedRequests = buildSpecialRequests(selectedArrival)
   433→
   434→    if (selectedRoomId) {
   435→      const room = availableRooms.find((r) => r.id === selectedRoomId)
   436→      if (!room) {
   437→        toast.error('Please select a room')
   438→        return
   439→      }
   440→      assignRoomMutation.mutate({
   441→        reservationId: selectedArrival.id,
   442→        roomId: selectedRoomId,
   443→        roomTypeId: room.typeId,
   444→        specialRequests: updatedRequests,
   445→      })
   446→    } else if (selectedArrival.room) {
   447→      checkInMutation.mutate({
   448→        reservationId: selectedArrival.id,
   449→        specialRequests: updatedRequests,
   450→      })
   451→    }
   452→  }
   453→
   454→  const openKeyCardDialog = (roomNumber: string) => {
   455→    setKeyCardRoomNumber(roomNumber)
   456→    setKeyCardIssued(false)
   457→    setKeyCardProcessed(false)
   458→    setKeyCardDialogOpen(true)
   459→  }
   460→
   461→  const handleKeyCardConfirm = () => {
   462→    setKeyCardProcessed(true)
   463→    setKeyCardDialogOpen(false)
   464→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   465→      duration: 5000,
   466→      icon: <Mail className="size-4" />,
   467→    })
   468→    setSelectedArrival(null)
   469→    setSelectedRoomId('')
   470→  }
   471→
   472→  const resetCheckInForm = () => {
   473→    setEarlyCheckIn(false)
   474→    setCheckOutTime(settings.defaultCheckOut)
   475→    setRoomPreference('')
   476→    setPillowType('')
   477→    setWakeupCall('')
   478→  }
   479→
   480→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   481→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   482→    setWalkInForm((prev) => ({
   483→      ...prev,
   484→      roomTypeId,
   485→      roomRate: roomType?.baseRate || 0,
   486→    }))
   487→  }
   488→
   489→  const handleSelectArrival = (arrival: Arrival) => {
   490→    setSelectedArrival(arrival)
   491→    setSelectedRoomId('')
   492→    setMobileShowDetail(true)
   493→  }
   494→
   495→  const handleBackToList = () => {
   496→    setMobileShowDetail(false)
   497→  }
   498→
   499→  // ─── Render ─────────────────────────────────────────────────────────
   500→
   501→  return (
   502→    <div className="flex flex-col gap-3">
   503→      {/* Header */}
   504→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 shrink-0">
   505→        <div>
   506→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   507→          <p className="text-xs text-muted-foreground">
   508→            Guest check-ins scheduled for {formatDate(today)}
   509→          </p>
   510→        </div>
   511→        <div className="flex items-center gap-2 shrink-0">
   512→          <Badge variant="secondary" className="text-[10px]">
   513→            {totalArrivals} expected
   514→          </Badge>
   515→          {vipArrivals > 0 && (
   516→            <Badge variant="outline" className="text-[10px] border-amber-300 text-amber-700 dark:text-amber-400">
   517→              <Crown className="size-3 mr-0.5" />
   518→              {vipArrivals} VIP
   519→            </Badge>
   520→          )}
   521→          {unassignedArrivals > 0 && (
   522→            <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-600 dark:text-amber-400">
   523→              <AlertTriangle className="size-3 mr-0.5" />
   524→              {unassignedArrivals} unassigned
   525→            </Badge>
   526→          )}
   527→          <Button onClick={handleOpenWalkIn} size="sm" className="gap-1.5">
   528→            <Footprints className="size-3.5" />
   529→            <span className="hidden sm:inline">Walk-in</span>
   530→          </Button>
   531→        </div>
   532→      </div>
   533→
   534→      {/* ─── Split Screen Body ─── */}
   535→      <div className="h-[calc(100vh-14rem)] min-h-[400px] flex flex-col lg:flex-row gap-3 overflow-hidden">
   536→
   537→        {/* ─── Left Panel: Arrivals List ─── */}
   538→        <div className={cn(
   539→          'flex flex-col min-h-0 border rounded-lg bg-card overflow-hidden',
   540→          'w-full lg:w-[42%] xl:w-[38%]',
   541→          mobileShowDetail && selectedArrival && 'hidden lg:flex',
   542→        )}>
   543→          {/* List header */}
   544→          <div className="px-3 py-2 border-b shrink-0 flex items-center justify-between bg-muted/30">
   545→            <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   546→              Expected Arrivals
   547→            </span>
   548→            <span className="text-[10px] text-muted-foreground">{sortedArrivals.length} guests</span>
   549→          </div>
   550→          {/* List body */}
   551→          <div className="flex-1 overflow-y-auto">
   552→            {isLoading ? (
   553→              <div className="p-2 space-y-2">
   554→                {Array.from({ length: 4 }).map((_, i) => (
   555→                  <Skeleton key={i} className="h-16 w-full rounded-lg" />
   556→                ))}
   557→              </div>
   558→            ) : sortedArrivals.length === 0 ? (
   559→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   560→                <LogIn className="size-8 mb-2 opacity-50" />
   561→                <p className="text-sm">No arrivals scheduled for today</p>
