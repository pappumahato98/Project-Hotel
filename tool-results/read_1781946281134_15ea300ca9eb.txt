   108→function formatDate(date: Date): string {
   109→  return date.toLocaleDateString('en-US', {
   110→    weekday: 'long',
   111→    year: 'numeric',
   112→    month: 'long',
   113→    day: 'numeric',
   114→  })
   115→}
   116→
   117→function getGreeting(): string {
   118→  const hour = new Date().getHours()
   119→  if (hour < 12) return 'Good Morning'
   120→  if (hour < 17) return 'Good Afternoon'
   121→  return 'Good Evening'
   122→}
   123→
   124→function getTimeAgo(timestamp: string): string {
   125→  const now = new Date()
   126→  const then = new Date(timestamp)
   127→  const diffMs = now.getTime() - then.getTime()
   128→  const diffMins = Math.floor(diffMs / 60000)
   129→  const diffHours = Math.floor(diffMins / 60)
   130→  const diffDays = Math.floor(diffHours / 24)
   131→
   132→  if (diffMins < 1) return 'Just now'
   133→  if (diffMins < 60) return `${diffMins}m ago`
   134→  if (diffHours < 24) return `${diffHours}h ago`
   135→  return `${diffDays}d ago`
   136→}
   137→
   138→function formatChartDate(dateStr: string): string {
   139→  const d = new Date(dateStr + 'T00:00:00')
   140→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   141→}

   829→// ─── Error State ────────────────────────────────────────────────────────
   830→function DashboardError({ error, refetch }: { error: Error; refetch: () => void }) {
   831→  const isServerDown = error.message?.includes('Server unavailable') || error.message?.includes('Failed to fetch')
   832→  return (
   833→    <div className="flex flex-1 flex-col items-center justify-center gap-4 p-6 overflow-y-auto">
   834→      <div className="flex flex-col items-center gap-2 text-center">
   835→        <div className="flex size-12 items-center justify-center rounded-full bg-muted">
   836→          {isServerDown ? <WifiOff className="size-6 text-muted-foreground" /> : <AlertCircle className="size-6 text-destructive" />}
   837→        </div>
   838→        <AlertTitle className="text-base">
   839→          {isServerDown ? 'Connecting to server...' : 'Failed to load dashboard'}
   840→        </AlertTitle>
   841→        <AlertDescription className="text-sm text-muted-foreground max-w-md">
   842→          {isServerDown
   843→            ? 'The server is starting up. This may take a moment.'
   844→            : error.message || 'An unexpected error occurred.'}
   845→        </AlertDescription>
   846→      </div>
   847→      <Button variant="outline" size="sm" onClick={() => refetch()}>
   848→        <RefreshCw className="size-4 mr-2" />
   849→        Retry
   850→      </Button>
   851→    </div>
   852→  )
   853→}

   855→// ─── Main Dashboard Module ──────────────────────────────────────────────
   856→export function DashboardModule() {
   857→  const { data, isLoading, isError, error, refetch } = useQuery<DashboardData>({
   858→    queryKey: ['dashboard'],
   859→    queryFn: () => apiFetch('/api/dashboard'),
   860→    refetchInterval: 30000, // refresh every 30s
   861→    retry: 3,
   862→    retryDelay: (attempt) => Math.min(1000 * 2 ** attempt, 10000),
   863→  })
   864→
   865→  // Defensive defaults — guard against partial API responses
   866→  const safeData: DashboardData = {
   867→    kpis: data?.kpis ?? {
   868→      totalRooms: 0, occupiedRooms: 0, occupancy: 0, occupancyTrend: 0,
   869→      arrivals: 0, departures: 0, vacantClean: 0,
   870→      totalRevenue: 0, roomRevenue: 0, fAndBRevenue: 0, otherRevenue: 0,

   886→
   887→  return (
