     5→import { toast } from 'sonner'
     6→import { apiFetch } from '@/lib/api'
     7→import {
     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,
     9→  Zap, Eye, Clock, Banknote, Mail, X, ArrowRight, BedSingle, FileText, BookOpen,
    10→} from 'lucide-react'
    11→
    12→import { Button } from '@/components/ui/button'
    13→import { Badge } from '@/components/ui/badge'
    14→import { Card, CardContent } from '@/components/ui/card'
    15→import { Skeleton } from '@/components/ui/skeleton'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import { Separator } from '@/components/ui/separator'
    19→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group'
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select'

    29→import { StatusBadge } from '@/components/shared/status-badge'
    30→import { RoomTypeBedBadge } from '@/components/shared/room-type-bed-badge'
    31→import { formatDate, formatTime, formatCurrency, getTodayString } from '@/lib/format'
    32→import { cn } from '@/lib/utils'

    36→// ─── Types ──────────────────────────────────────────────────────────────
    37→
    38→interface DepartureGuest {
    39→  id: string
    40→  firstName: string
    41→  lastName: string
    42→  vipLevel: string
    43→}
    44→
    45→interface DepartureRoom {
    46→  id: string
    47→  number: string
    48→  floor: number
    49→  type: { name: string; code: string; bedConfig?: string }
    50→}
    51→
    52→interface FolioLineItem {
    53→  id: string
    54→  description: string
    55→  amount: number
    56→  type: 'charge' | 'payment'
    57→  date: string
    58→}
    59→
    60→interface DepartureFolio {
    61→  id: string
    62→  balance: number
    63→  status: string
    64→  items?: FolioLineItem[]
    65→}
    66→
    67→interface Departure {
    68→  id: string
    69→  confirmationNo: string
    70→  status: string
    71→  checkIn: string
    72→  checkOut: string
    73→  roomRate: number
    74→  totalAmount: number
    75→  paidAmount: number
    76→  creditLimit: number
    77→  guest: DepartureGuest
    78→  room: DepartureRoom
    79→  adults?: number
    80→  children?: number
    81→  folios: DepartureFolio[]
    82→}
    83→
    84→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    85→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    86→
    87→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    88→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    89→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    90→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    91→}
    92→
    93→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    94→  cash: 'Cash',
    95→  card: 'Credit/Debit Card',
    96→  bank_transfer: 'Bank Transfer',
    97→}
    98→
    99→// ─── Component ──────────────────────────────────────────────────────────
   100→
   101→export function DeparturesView() {
   102→  const queryClient = useQueryClient()
   103→  const { settings } = useSettingsStore()
   104→  const { preferences } = usePreferencesStore()
   105→  const { navigateTo } = useNavigationStore()
   106→  const today = getTodayString()
   107→
   108→  // Dialog states
   109→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   110→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   111→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   112→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   113→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   114→
   115→  // Selected departure
   116→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   117→
   118→  // Late checkout
   119→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   120→
   121→  // Payment
   122→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   123→  const [paymentAmount, setPaymentAmount] = useState('')
   124→  const [paymentReference, setPaymentReference] = useState('')
   125→
   126→  // Receipt data
   127→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   128→
   129→  // Fetch today's departures (checked_in with checkOut = today)
   130→  const { data, isLoading } = useQuery({
   131→    queryKey: ['departures', today],
   132→    queryFn: async () => {
   133→      const params = new URLSearchParams({ status: 'checked_in', checkOutDate: today })
   134→      return apiFetch(`/api/reservations?${params.toString()}`)
   135→    },
   136→    refetchInterval: 30000,
   137→  })
   138→
   139→  const departures: Departure[] = data?.reservations || []
   140→
   141→  // Stats
   142→  const totalDepartures = departures.length
   143→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   144→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   145→  const outstandingBalance = departures.reduce((sum, d) => {
   146→    return sum + (d.folios[0]?.balance || 0)
   147→  }, 0)
   148→
   149→  // Zero-balance pending guests (for express & batch checkout)
   150→  const zeroBalancePending = departures.filter(
   151→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   152→  )
   153→
   154→  // ─── Mutations ─────────────────────────────────────────────────────
   155→
   156→  // Checkout mutation
   157→  const checkoutMutation = useMutation({
   158→    mutationFn: async (reservationId: string) => {
   159→      return apiFetch(`/api/reservations/${reservationId}`, {
   160→        method: 'PATCH',
   161→        headers: { 'Content-Type': 'application/json' },
   162→        body: JSON.stringify({ status: 'checked_out' }),
   163→      })
   164→    },
   165→    onSuccess: () => {
   166→      invalidate.afterCheckout(queryClient)
   167→    },
   168→  })
   169→
   170→  // Payment mutation
   171→  const paymentMutation = useMutation({
   172→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   173→      return apiFetch(`/api/folio/${folioId}`, {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ type: 'payment', amount, method, reference }),
   177→      })
   178→    },
   179→    onSuccess: () => {
   180→      invalidate.afterFolioChange(queryClient)
   181→      setPaymentDialogOpen(false)
   182→      setPaymentAmount('')
   183→      setPaymentReference('')
   184→      toast.success('Payment recorded successfully')
   185→    },
   186→    onError: () => {
   187→      toast.error('Failed to record payment')
   188→    },
   189→  })
   190→
   191→  // Late checkout mutation
   192→  const lateCheckoutMutation = useMutation({
   193→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   194→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   195→        method: 'PATCH',
   196→        headers: { 'Content-Type': 'application/json' },
   197→        body: JSON.stringify({ checkOut: newCheckOut }),
   198→      })
   199→      // Add surcharge to folio
   200→      if (surcharge > 0) {
   201→        await apiFetch(`/api/folio`, {
   202→          method: 'POST',
   203→          headers: { 'Content-Type': 'application/json' },
   204→          body: JSON.stringify({
   205→            reservationId,
   206→            type: 'charge',
   207→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   208→            amount: surcharge,
   209→          }),
   210→        })
   211→      }
   212→      return result
   213→    },
   214→    onSuccess: () => {
   215→      invalidate.afterReservationChange(queryClient)
   216→      invalidate.afterFolioChange(queryClient)
   217→      setLateCheckoutDialogOpen(false)
   218→      toast.success('Late checkout request processed successfully')
   219→    },
   220→    onError: () => {
   221→      toast.error('Failed to process late checkout request')
   222→    },
   223→  })
   224→
   225→  // Batch checkout mutation
   226→  const batchCheckoutMutation = useMutation({
   227→    mutationFn: async (ids: string[]) => {
   228→      const results = []
   229→      for (let i = 0; i < ids.length; i++) {
   230→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   231→          method: 'PATCH',
   232→          headers: { 'Content-Type': 'application/json' },
   233→          body: JSON.stringify({ status: 'checked_out' }),
   234→        })
   235→        results.push(result)
   236→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   237→      }
   238→      return results
   239→    },
   240→    onSuccess: (_data, ids) => {
   241→      invalidate.afterCheckout(queryClient)
   242→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   243→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   244→    },
   245→    onError: () => {
   246→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   247→    },
   248→  })
   249→
   250→  // ─── Handlers ─────────────────────────────────────────────────────
   251→
   252→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   253→
   254→  const handleExpressCheckout = (dep: Departure) => {
   255→    checkoutMutation.mutate(dep.id, {
   256→      onSuccess: () => {
   257→        toast.success(`Express checkout complete for Room ${dep.room.number}`)
   258→        toast.info(`Room ${dep.room.number} marked for housekeeping - Vacant Dirty`)
   259→      },
   260→    })
   261→  }
   262→
   263→  const handleReviewFolio = (dep: Departure) => {
   264→    setSelectedDeparture(dep)
   265→    setFolioDialogOpen(true)
   266→  }
   267→
   268→  const handleNavigateToFolio = (dep: Departure) => {
   269→    useFolioContextStore.getState().setFolioContext({
   270→      reservationId: dep.id,
   271→      guestId: dep.guest?.id || '',
   272→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   273→      roomNumber: dep.room?.number || '',
   274→      confirmationNo: dep.confirmationNo,
   275→    })
   276→    navigateTo('front-desk', 'folio')
   277→  }
   278→
   279→  const handleViewLedger = (dep: Departure) => {
   280→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   281→      guestId: dep.guest?.id || '',
   282→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   283→    })
   284→    navigateTo('front-desk', 'guest-ledger')
   285→  }
   286→
   287→  const handleFolioSettleCheckout = () => {
   288→    setFolioDialogOpen(false)
   289→    setCheckoutDialogOpen(true)
   290→  }
   291→
   292→  const handleQuickCheckout = (dep: Departure) => {
   293→    setSelectedDeparture(dep)
   294→    setCheckoutDialogOpen(true)
   295→  }
   296→
   297→  const handleRequestLateCheckout = (dep: Departure) => {
   298→    setSelectedDeparture(dep)
   299→    setLateCheckoutOption('14:00')
   300→    setLateCheckoutDialogOpen(true)
   301→  }
   302→
   303→  const handleConfirmLateCheckout = () => {
   304→    if (!selectedDeparture) return
   305→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   306→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   307→
   308→    // Build new checkout datetime with the selected time
   309→    const currentCheckout = new Date(selectedDeparture.checkOut)
   310→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   311→    currentCheckout.setHours(hours, minutes, 0, 0)
   312→
   313→    lateCheckoutMutation.mutate({
   314→      reservationId: selectedDeparture.id,
   315→      newCheckOut: currentCheckout.toISOString(),
   316→      surcharge,
   317→    })
   318→  }
   319→
   320→  const handleOpenPaymentDialog = () => {
   321→    if (!selectedDeparture) return
   322→    const balance = getBalance(selectedDeparture)
   323→    setPaymentAmount(String(balance))
   324→    setPaymentMethod('cash')
   325→    setPaymentReference('')
   326→    setPaymentDialogOpen(true)
   327→  }
   328→
   329→  const handleRecordPayment = () => {
   330→    if (!selectedDeparture) return
   331→    const folioId = selectedDeparture.folios[0]?.id
   332→    if (!folioId) {
   333→      toast.error('No folio found for this reservation')
   334→      return
   335→    }
   336→    const amount = Number(paymentAmount)
   337→    if (isNaN(amount) || amount <= 0) {
   338→      toast.error('Please enter a valid payment amount')
   339→      return
   340→    }
   341→    paymentMutation.mutate({
   342→      folioId,
   343→      amount,
   344→      method: paymentMethod,
   345→      reference: paymentReference,
   346→    })
   347→  }
   348→
   349→  const confirmCheckout = () => {
   350→    if (!selectedDeparture) return
   351→    const roomNumber = selectedDeparture.room.number
   352→    checkoutMutation.mutate(selectedDeparture.id, {
   353→      onSuccess: () => {
   354→        setCheckoutDialogOpen(false)
   355→        setReceiptData(selectedDeparture)
   356→        setReceiptDialogOpen(true)
   357→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   358→      },
   359→    })
   360→  }
   361→
   362→  const handleBatchCheckout = () => {
   363→    const ids = zeroBalancePending.map((d) => d.id)
   364→    if (ids.length === 0) {
   365→      toast.info('No guests with zero balance to batch checkout')
   366→      return
   367→    }
   368→    batchCheckoutMutation.mutate(ids)
   369→  }
   370→
   371→  const handlePrintReceipt = () => {
   372→    toast.success('Receipt sent to printer')
   373→  }
   374→
   375→  const handleEmailReceipt = () => {
   376→    toast.success('Receipt emailed to guest')
   377→  }
   378→
   379→  // ─── Computed values ──────────────────────────────────────────────
   380→
   381→  const lateCheckoutFee = selectedDeparture
   382→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   383→    : 0
   384→
   385→  // ─── Render ────────────────────────────────────────────────────────
   386→
   387→  return (
   388→    <div className="flex flex-col gap-2">
   389→      {/* Header */}
   390→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   391→        <div>
   392→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   393→          <p className="text-xs text-muted-foreground">
   394→            Guest check-outs scheduled for {formatDate(today)}
   395→          </p>
   396→        </div>
   397→        {zeroBalancePending.length > 0 && (
   398→          <Button
   399→            onClick={handleBatchCheckout}
   400→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   401→            disabled={batchCheckoutMutation.isPending}
   402→          >
   403→            <Zap className="size-4 mr-1.5" />
   404→            {batchCheckoutMutation.isPending
   405→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   406→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   407→          </Button>
   408→        )}
   409→      </div>
   410→
   411→      {/* Departures List */}
   412→      <Card className="py-0">
   413→        <CardContent className="p-0">
   414→          <div className="max-h-[600px] overflow-y-auto">
   415→            <Table>
   416→              <TableHeader>
   417→                <TableRow>
   418→                  <TableHead className="w-[70px]">Room</TableHead>
   419→                  <TableHead>Guest</TableHead>
   420→                  <TableHead className="w-[100px]">Check-out</TableHead>
   421→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   422→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   423→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   424→                </TableRow>
   425→              </TableHeader>
   426→              <TableBody>
   427→                {isLoading ? (
   428→                  Array.from({ length: 5 }).map((_, i) => (
   429→                    <TableRow key={i}>
   430→                      {Array.from({ length: 6 }).map((_, j) => (
   431→                        <TableCell key={j}>
   432→                          <Skeleton className="h-4 w-full" />
   433→                        </TableCell>
   434→                      ))}
   435→                    </TableRow>
   436→                  ))
   437→                ) : departures.length === 0 ? (
   438→                  <TableRow>
   439→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   440→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   441→                      No departures scheduled for today
   442→                    </TableCell>
   443→                  </TableRow>
   444→                ) : (
   445→                  departures.map((dep) => {
   446→                    const balance = getBalance(dep)
   447→                    const isPending = dep.status === 'checked_in'
   448→                    const isZeroBalance = balance === 0
   449→                    return (
   450→                      <TableRow key={dep.id}>
   451→                        <TableCell className="font-bold font-mono">{dep.room.number} <RoomTypeBedBadge typeName={dep.room.type.name} bedConfig={dep.room.type.bedConfig} typeCode={dep.room.type.code} pax={(dep.adults ?? 1) + (dep.children ?? 0)} inline /></TableCell>
   452→                        <TableCell>
   453→                          <div className="flex items-center gap-2">
   454→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   455→                            {dep.guest.vipLevel !== 'none' && (
   456→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   457→                                VIP
   458→                              </Badge>
   459→                            )}
   460→                          </div>
   461→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   462→                        </TableCell>
   463→                        <TableCell className="text-xs">
   464→                          <div>{formatDate(dep.checkOut)}</div>
   465→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   466→                        </TableCell>
   467→                        <TableCell className="text-right">
   468→                          <div className="flex items-center justify-end gap-1.5">
   469→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   470→                              {formatCurrency(balance)}
   471→                            </span>
   472→                            {/* Outstanding Balance Alert Badges */}
   473→                            {isPending && balance > 5000 && (
   474→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   475→                                HIGH
   476→                              </Badge>
   477→                            )}
   478→                            {isPending && balance > 0 && balance <= 5000 && (
   479→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   480→                                DUE
   481→                              </Badge>
   482→                            )}
   483→                          </div>
   484→                        </TableCell>
   485→                        <TableCell className="text-right">
   486→                          {dep.status === 'checked_out' ? (
   487→                            <div className="flex items-center justify-end gap-1">
   488→                              <BedSingle className="size-3 text-yellow-500" />
   489→                              <StatusBadge status="vacant_dirty" />
   490→                            </div>
   491→                          ) : (
   492→                            <StatusBadge status="checked_in" />
   493→                          )}
   494→                        </TableCell>
   495→                        <TableCell className="text-right">
   496→                          <div className="flex justify-end gap-1 flex-wrap">
   497→                            {isPending && (
   498→                              <>
   499→                                {/* Express Checkout (zero balance) */}
   500→                                {isZeroBalance && (
   501→                                  <Button
   502→                                    size="sm"
   503→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   504→                                    onClick={() => handleExpressCheckout(dep)}
   505→                                    disabled={checkoutMutation.isPending}
   506→                                  >
   507→                                    <Zap className="size-3 mr-0.5" /> Express
   508→                                  </Button>
   509→                                )}
   510→                                {/* Review Folio */}
   511→                                <Button
   512→                                  size="sm"
   513→                                  variant="outline"
   514→                                  className="text-xs h-7"
   515→                                  onClick={() => handleReviewFolio(dep)}
   516→                                >
   517→                                  <Eye className="size-3 mr-0.5" /> Review Folio
   518→                                </Button>
   519→                                <Button
   520→                                  size="sm"
   521→                                  variant="outline"
   522→                                  className="text-xs h-7"
   523→                                  onClick={() => handleNavigateToFolio(dep)}
   524→                                >
   525→                                  <FileText className="size-3 mr-0.5" /> Full Folio
   526→                                </Button>
   527→                                <Button
   528→                                  size="sm"
   529→                                  variant="outline"
   530→                                  className="text-xs h-7"
   531→                                  onClick={() => handleViewLedger(dep)}
   532→                                >
   533→                                  <BookOpen className="size-3 mr-0.5" /> View Ledger
   534→                                </Button>
   535→                                {/* Late Checkout */}
   536→                                <Button
   537→                                  size="sm"
   538→                                  variant="outline"
   539→                                  className="text-xs h-7"
   540→                                  onClick={() => handleRequestLateCheckout(dep)}
   541→                                >
   542→                                  <Clock className="size-3 mr-0.5" /> Late
   543→                                </Button>
   544→                                {/* Regular Checkout */}
   545→                                <Button
   546→                                  size="sm"
   547→                                  className="text-xs h-7"
   548→                                  onClick={() => handleQuickCheckout(dep)}
   549→                                >
   550→                                  <LogOut className="size-3 mr-0.5" /> Checkout
   551→                                </Button>
   552→                              </>
