   159→  const queryClient = useQueryClient()
   160→  const { settings } = useSettingsStore()
   161→  const today = getTodayString()
   162→
   163→  // ─── Step state ────────────────────────────────────────────────
   164→  const [currentStep, setCurrentStep] = useState(0)
   165→
   166→  // ─── Selection states ────────────────────────────────────────
   167→  const [selectedReservationId, setSelectedReservationId] = useState<string>('')
   168→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   169→  const [roomSearchFilter, setRoomSearchFilter] = useState('')
   170→
   171→  // ─── Registration card form ────────────────────────────────────
   172→  const [regTitle, setRegTitle] = useState('')
   173→  const [regFirstName, setRegFirstName] = useState('')
   174→  const [regLastName, setRegLastName] = useState('')
   175→  const [regContactNo, setRegContactNo] = useState('')
   176→  const [regEmail, setRegEmail] = useState('')
   177→  const [regAddress, setRegAddress] = useState('')
   178→  const [regNationality, setRegNationality] = useState('')
   179→  const [regIdType, setRegIdType] = useState('passport')
   180→  const [regIdNumber, setRegIdNumber] = useState('')
   181→  const [regCity, setRegCity] = useState('')
   182→  const [regCountry, setRegCountry] = useState('')
   183→
   184→  // ─── Stay preferences ─────────────────────────────────────────
   185→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   186→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   187→  const [pillowType, setPillowType] = useState('')
   188→  const [roomPreference, setRoomPreference] = useState('')
   189→  const [wakeupCall, setWakeupCall] = useState('')
   190→
   191→  // ─── Billing ──────────────────────────────────────────────────
   192→  const [depositAmount, setDepositAmount] = useState('')
   193→  const [paymentMethod, setPaymentMethod] = useState('')
   194→  const [billingNotes, setBillingNotes] = useState('')
   195→
   196→  // ─── Conflict dialog state ──────────────────────────────────
   197→  const [conflictDialogOpen, setConflictDialogOpen] = useState(false)
   198→  const [conflictInfo, setConflictInfo] = useState<{
   199→    confirmationNo: string
   200→    guestName: string
   201→    checkIn: string
   202→    checkOut: string
   203→    status: string
   204→    roomNumber: string
   205→  } | null>(null)
   206→
   207→  // ─── Key card dialog ───────────────────────────────────────────
   208→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   209→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   210→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   211→  const [keyCardGuestName, setKeyCardGuestName] = useState('')
   212→  const [keyCardCheckOut, setKeyCardCheckOut] = useState('')
   213→
   214→  // ─── Queries ─────────────────────────────────────────────────
   215→
   216→  // Fetch specific reservation by ID (if prop provided)
   217→  const { data: specificResData, isLoading: loadingSpecific } = useQuery({
   218→    queryKey: ['reservation', reservationId],
   219→    queryFn: async () => {
   220→      const res = await fetch(`/api/reservations/${reservationId}`)
   221→      if (!res.ok) throw new Error('Failed to fetch reservation')
   222→      return res.json()
   223→    },
   224→    enabled: !!reservationId,
   225→  })
   226→
   227→  // Fetch today's confirmed arrivals
   228→  const { data: arrivalsData, isLoading: loadingArrivals } = useQuery({
   229→    queryKey: ['arrivals', today],
   230→    queryFn: async () => {
   231→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   232→      const res = await fetch(`/api/reservations?${params.toString()}`)
   233→      if (!res.ok) throw new Error('Failed to fetch arrivals')
   234→      return res.json()
   235→    },
   236→    enabled: !reservationId,
   237→  })
   238→
   239→  // Fetch all rooms with types
   240→  const { data: roomsData, isLoading: loadingRooms } = useQuery({
   241→    queryKey: ['rooms', 'all'],
   242→    queryFn: async () => {
   243→      const res = await fetch('/api/rooms')
   244→      if (!res.ok) throw new Error('Failed to fetch rooms')
   245→      return res.json()
   246→    },
   247→  })
   248→
   249→  // ─── Derived data ──────────────────────────────────────────────
   250→
   251→  const specificReservation: ReservationInfo | null = specificResData?.reservation || null
   252→  const arrivals: ReservationInfo[] = arrivalsData?.reservations || []
   253→
   254→  // The reservation we're working with
   255→  const activeReservation: ReservationInfo | null = reservationId
   256→    ? specificReservation
   257→    : arrivals.find((a) => a.id === selectedReservationId) || null
   258→
   259→  // Room type data from API (not hardcoded)
   260→  const roomTypes: RoomTypeData[] = useMemo(() => {
   261→    if (!roomsData?.roomTypes) return []
   262→    return roomsData.roomTypes.map((rt: { id: string; name: string; code: string; ratePlans: Array<{ baseRate: number }> }) => ({
   263→      id: rt.id,
   264→      name: rt.name,
   265→      code: rt.code,
   266→      baseRate: rt.ratePlans?.[0]?.baseRate || 0,
   267→    }))
   268→  }, [roomsData])
   269→
   270→  // Available rooms (vacant_clean or inspected)
   271→  const allRooms: RoomInfo[] = roomsData?.rooms || []
   272→  const availableRooms = allRooms.filter(
   273→    (r: RoomInfo) => r.status === 'vacant_clean' || r.status === 'inspected'
   274→  )
   275→
   276→  // Filter available rooms by search
   277→  const filteredAvailableRooms = useMemo(() => {
   278→    let filtered = availableRooms
   279→    if (roomSearchFilter.trim()) {
   280→      const q = roomSearchFilter.toLowerCase()
   281→      filtered = filtered.filter(
   282→        (r: RoomInfo) =>
   283→          r.number.toLowerCase().includes(q) ||
   284→          r.type.name.toLowerCase().includes(q) ||
   285→          String(r.floor).includes(q) ||
   286→          (r.wing && r.wing.toLowerCase().includes(q))
   287→      )
   288→    }
   289→    return filtered
   290→  }, [availableRooms, roomSearchFilter])
   291→
   292→  // Group rooms by floor
   293→  const roomsByFloor = useMemo(() => {
   294→    const grouped = new Map<number, RoomInfo[]>()
   295→    for (const room of filteredAvailableRooms) {
   296→      const floor = room.floor
   297→      if (!grouped.has(floor)) grouped.set(floor, [])
   298→      grouped.get(floor)!.push(room)
   299→    }
   300→    return new Map([...grouped.entries()].sort((a, b) => a[0] - b[0]))
   301→  }, [filteredAvailableRooms])
   302→
   303→  // Get room type name by ID
   304→  const getRoomTypeName = (typeId: string) => {
   305→    const rt = roomTypes.find((t) => t.id === typeId)
   306→    return rt?.name || 'Unknown'
   307→  }
   308→
   309→  // Get room rate by type ID
   310→  const getRoomTypeRate = (typeId: string) => {
   311→    const rt = roomTypes.find((t) => t.id === typeId)
   312→    return rt?.baseRate || 0
   313→  }
   314→
   315→  // ─── Initialize form when reservation loads ───────────────────
   316→
   317→  React.useEffect(() => {
   318→    if (specificReservation && reservationId) {
   319→      setSelectedReservationId(specificReservation.id)
   320→      if (specificReservation.room?.id) {
   321→        setSelectedRoomId(specificReservation.room.id)
   322→      }
   323→      setCurrentStep(specificReservation.guest ? 1 : 0)
   324→    }
   325→  }, [specificReservation, reservationId])
   326→
   327→  // Pre-fill registration fields when guest is selected
   328→  React.useEffect(() => {
   329→    if (activeReservation?.guest) {
   330→      const g = activeReservation.guest
   331→      setRegTitle('')
   332→      setRegFirstName(g.firstName || '')
   333→      setRegLastName(g.lastName || '')
   334→      setRegContactNo(g.phone || '')
   335→      setRegEmail(g.email || '')
   336→      setRegNationality(g.nationality || '')
   337→      setRegIdType(g.idType || 'passport')
   338→      setRegIdNumber(g.idNumber || '')
   339→      setRegAddress(g.address || '')
   340→      setRegCity(g.city || '')
   341→      setRegCountry(g.country || '')
   342→    }
   343→  }, [activeReservation?.guest])
   344→
   345→  // Auto-set deposit when room is selected
   346→  React.useEffect(() => {
   347→    if (activeReservation?.roomRate && !depositAmount) {
   348→      setDepositAmount(String(activeReservation.roomRate))
   349→    }
   350→  }, [activeReservation?.roomRate, depositAmount])
   351→
   352→  // ─── Mutations ────────────────────────────────────────────────
   353→
   354→  const checkInMutation = useMutation({
   355→    mutationFn: async ({
   356→      reservationId: resId,
   357→      roomId,
   358→      specialRequests,
   359→    }: {
   360→      reservationId: string
   361→      roomId: string
   362→      specialRequests: string
   363→    }) => {
   364→      const res = await fetch(`/api/reservations/${resId}`, {
   365→        method: 'PATCH',
   366→        headers: { 'Content-Type': 'application/json' },
   367→        body: JSON.stringify({
   368→          roomId,
   369→          status: 'checked_in',
   370→          specialRequests,
   371→        }),
   372→      })
   373→      if (!res.ok) {
   374→        const errData = await res.json().catch(() => ({}))
   375→        const error = new Error(errData.error || 'Failed to check in guest') as Error & { conflict?: typeof errData.conflict }
   376→        error.conflict = errData.conflict
   377→        throw error
   378→      }
   379→      return res.json()
   380→    },
   381→    onSuccess: () => {
   382→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   383→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   384→      queryClient.invalidateQueries({ queryKey: ['reservations'] })
   385→      queryClient.invalidateQueries({ queryKey: ['reservation', reservationId] })
   386→      toast.success('Guest checked in successfully!')
   387→    },
   388→    onError: (error: Error & { conflict?: { confirmationNo: string; guestName: string; checkIn: string; checkOut: string; status: string; roomNumber: string } }) => {
   389→      if (error.conflict) {
   390→        setConflictInfo(error.conflict)
   391→        setConflictDialogOpen(true)
   392→      } else {
   393→        toast.error('Check-in failed', {
   394→          description: error.message || 'Unknown error',
   395→        })
   396→      }
   397→    },
   398→  })
   399→
   400→  // ─── Special Requests Builder ─────────────────────────────────
   401→
   402→  const buildSpecialRequests = (): string => {
   403→    const parts: string[] = []
   404→
   405→    if (activeReservation?.specialRequests) {
   406→      parts.push(activeReservation.specialRequests)
   407→    }
   408→    if (earlyCheckIn) {
   409→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   410→    }
   411→    if (checkOutTime !== settings.defaultCheckOut) {
   412→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   413→    }
   414→    if (roomPreference) {
   415→      const label = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   416→      parts.push(`[PREFERENCE] Room: ${label}`)
   417→    }
   418→    if (pillowType) {
   419→      const label = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   420→      parts.push(`[PREFERENCE] Pillow: ${label}`)
   421→    }
   422→    if (wakeupCall) {
   423→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   424→    }
   425→    if (billingNotes) {
   426→      parts.push(`[BILLING] ${billingNotes}`)
   427→    }
   428→    if (paymentMethod) {
   429→      parts.push(`[PAYMENT] Method: ${paymentMethod}`)
   430→    }
   431→    if (depositAmount && parseFloat(depositAmount) > 0) {
   432→      parts.push(`[DEPOSIT] ${formatCurrency(parseFloat(depositAmount))}`)
   433→    }
   434→
   435→    return parts.join(' | ')
   436→  }
   437→
   438→  // ─── Step Navigation ──────────────────────────────────────────
   439→
   440→  const canGoNext = (): boolean => {
   441→    switch (currentStep) {
   442→      case 0: return !!activeReservation
   443→      case 1: return true // Guest verification is info-only
   444→      case 2: return !!selectedRoomId
   445→      case 3: return true // Preferences are optional
   446→      case 4: return true // Billing is optional
   447→      case 5: return keyCardIssued
   448→      default: return false
   449→    }
   450→  }
   451→
   452→  const handleNext = () => {
   453→    if (currentStep === 5) {
   454→      onComplete?.()
   455→      return
   456→    }
   457→
   458→    // On step 4 → 5, execute check-in
   459→    if (currentStep === 4 && activeReservation && selectedRoomId) {
   460→      const requests = buildSpecialRequests()
   461→      checkInMutation.mutate(
   462→        {
   463→          reservationId: activeReservation.id,
   464→          roomId: selectedRoomId,
   465→          specialRequests: requests,

   467→        {
   468→          onSuccess: () => {

   470→            const room = availableRooms.find((r: RoomInfo) => r.id === selectedRoomId)
   471→            const roomNumber = room?.number || activeReservation?.room?.number || ''
   472→            const guestName = activeReservation.guest
   473→              ? `${activeReservation.guest.firstName} ${activeReservation.guest.lastName}`
   474→              : 'Guest'
   475→
   476→            setKeyCardRoomNumber(roomNumber)
   477→            setKeyCardGuestName(guestName)
   478→            setKeyCardCheckOut(formatDate(activeReservation.checkOut))
   479→            setKeyCardIssued(false)
   480→            setKeyCardDialogOpen(true)
   481→            setCurrentStep(5)
   482→          },
   483→        }
   484→      )
   485→      return
   486→    }
   487→
   488→    setCurrentStep((prev) => Math.min(prev + 1, 5))
   489→  }
   490→
   491→  const handleBack = () => {
   492→    setCurrentStep((prev) => Math.max(prev - 1, 0))
   493→  }
   494→
   495→  // ─── Render: Step Indicator ────────────────────────────────────
   496→
   497→  const renderStepIndicator = () => (
   498→    <div className="mb-4">
   499→      <div className="flex items-center justify-between overflow-x-auto pb-2">
   500→        {STEP_LABELS.map((label, idx) => (
   501→          <React.Fragment key={idx}>
   502→            <div className="flex flex-col items-center gap-1 min-w-0">
   503→              <div
   504→                className={cn(
   505→                  'flex size-7 shrink-0 items-center justify-center rounded-full border-2 text-[10px] font-bold transition-colors',
   506→                  idx < currentStep && 'border-green-500 bg-green-500 text-white',
   507→                  idx === currentStep && 'border-primary bg-primary text-primary-foreground',
   508→                  idx > currentStep && 'border-muted-foreground/30 bg-background text-muted-foreground'
   509→                )}
   510→              >
   511→                {idx < currentStep ? <Check className="size-4" /> : idx + 1}
   512→              </div>
   513→              <span
   514→                className={cn(
   515→                  'text-[10px] sm:text-xs font-medium whitespace-nowrap',
   516→                  idx === currentStep ? 'text-primary' : 'text-muted-foreground'
   517→                )}
   518→              >
   519→                {label}
   520→              </span>
   521→            </div>
   522→            {idx < STEP_LABELS.length - 1 && (
   523→              <div
   524→                className={cn(
   525→                  'flex-1 h-0.5 mx-1 min-w-[12px] max-w-[40px] transition-colors',
   526→                  idx < currentStep ? 'bg-green-500' : 'bg-muted-foreground/20'
   527→                )}
   528→              />
   529→            )}
   530→          </React.Fragment>
   531→        ))}
   532→      </div>
   533→    </div>
   534→  )
   535→
   536→  // ─── Render: Step 0 - Select Arrival ─────────────────────────
   537→
   538→  const renderSelectArrival = () => {
   539→    if (loadingArrivals) {
   540→      return (
   541→        <div className="space-y-3 p-4">
   542→          {Array.from({ length: 4 }).map((_, i) => (
   543→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   544→          ))}
   545→        </div>
   546→      )
   547→    }
   548→
   549→    if (arrivals.length === 0) {
   550→      return (
   551→        <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   552→          <UserCheck className="size-12 mb-3 opacity-30" />
   553→          <p className="text-sm font-medium">No arrivals scheduled for today</p>
   554→          <p className="text-xs mt-1">Check-in date: {formatDate(today)}</p>
   555→        </div>
   556→      )
   557→    }
   558→
   559→    return (
   560→      <div className="space-y-2">
   561→        <div className="flex items-center gap-2 mb-3">
