   183→   183→// ── Main Component ───────────────────────────────────────────
   184→   184→export function LostFoundView() {
   185→   185→  const queryClient = useQueryClient()
   186→   186→  const [statusFilter, setStatusFilter] = useState<string>('all')
   187→   187→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   188→   188→  const [searchQuery, setSearchQuery] = useState('')
   189→   189→  const [addOpen, setAddOpen] = useState(false)
   190→   190→  const [claimOpen, setClaimOpen] = useState(false)
   191→   191→  const [viewOpen, setViewOpen] = useState(false)
   192→   192→  const [selectedItem, setSelectedItem] = useState<LostFoundItem | null>(null)
   193→   193→
   194→   194→  // Add form state
   195→   195→  const [addForm, setAddForm] = useState({
   196→   196→    itemName: '',
   197→   197→    category: 'other',
   198→   198→    roomId: '',
   199→   199→    storageLocation: '',
   200→   200→    foundBy: '',
   201→   201→    description: '',
   202→   202→  })
   203→   203→
   204→   204→  // Claim form state
   205→   205→  const [claimForm, setClaimForm] = useState({
   206→   206→    claimedBy: '',
   207→   207→    identityVerified: false,
   208→   208→    claimAttachment: '' as string, // base64 of ID photo
   209→   209→  })
   210→   210→
   211→   211→  const { data, isLoading } = useQuery<LostFoundItem[]>({
   212→   212→    queryKey: ['lost-found'],
   213→   213→    queryFn: () => apiFetch('/api/housekeeping?section=lost-found'),
   214→   214→  })
   215→   215→
   216→   216→  const items = data || []
   217→   217→
   218→   218→  // Computed stats
   219→   219→  const totalItems = items.length
   220→   220→  const unclaimedItems = items.filter((i) => i.status === 'found').length
   221→   221→  const claimedThisMonth = items.filter((i) => {
   222→   222→    if (i.status !== 'claimed' || !i.claimDate) return false
   223→   223→    return new Date(i.claimDate).getMonth() === new Date().getMonth()
   224→   224→  }).length
   225→   225→
   226→   226→  // Filtered items
   227→   227→  const filteredItems = items.filter((item) => {
   228→   228→    if (statusFilter !== 'all' && item.status !== statusFilter) return false
   229→   229→    if (categoryFilter !== 'all' && item.category !== categoryFilter) return false
   230→   230→    if (searchQuery) {
   231→   231→      const q = searchQuery.toLowerCase()
   232→   232→      return (
   233→   233→        item.itemName.toLowerCase().includes(q) ||
   234→   234→        item.foundBy.toLowerCase().includes(q) ||
   235→   235→        item.roomId?.toLowerCase().includes(q) ||
   236→   236→        item.description?.toLowerCase().includes(q)
   237→   237→      )
   238→   238→    }
   239→   239→    return true
   240→   240→  })
   241→   241→
   242→   242→  const addMutation = useMutation({
   243→   243→    mutationFn: (params: typeof addForm) =>
   244→   244→      apiFetch('/api/housekeeping', {
   245→   245→        method: 'POST',
   246→   246→        headers: { 'Content-Type': 'application/json' },
   247→   247→        body: JSON.stringify({ action: 'create-lost-found', ...params }),
   248→   248→      }),
   249→   249→    onSuccess: () => {
   250→   250→      queryClient.invalidateQueries({ queryKey: ['lost-found'] })
   251→   251→      toast.success(`Found item "${addForm.itemName}" reported successfully`)
   252→   252→      setAddOpen(false)
   253→   253→      setAddForm({ itemName: '', category: 'other', roomId: '', storageLocation: '', foundBy: '', description: '' })
   254→   254→    },
   255→   255→    onError: (err: Error) => {
   256→   256→      toast.error(err.message || 'Failed to report item')
   257→   257→    },
   258→   258→  })
   259→   259→
   260→   260→  const claimMutation = useMutation({
   261→   261→    mutationFn: (params: { id: string; claimedBy: string; identityVerified: boolean; claimAttachment: string }) =>
   262→   262→      apiFetch('/api/housekeeping', {
   263→   263→        method: 'POST',
   264→   264→        headers: { 'Content-Type': 'application/json' },
   265→   265→        body: JSON.stringify({ action: 'claim-lost-found', ...params }),
   266→   266→      }),
   267→   267→    onSuccess: () => {
   268→   268→      queryClient.invalidateQueries({ queryKey: ['lost-found'] })
   269→   269→      toast.success(`Item "${selectedItem?.itemName}" claimed successfully`)
   270→   270→      setClaimOpen(false)
   271→   271→      setClaimForm({ claimedBy: '', identityVerified: false, claimAttachment: '' })
   272→   272→      setSelectedItem(null)
   273→   273→    },
   274→   274→    onError: (err: Error) => {
   275→   275→      toast.error(err.message || 'Failed to claim item')
   276→   276→    },
   277→   277→  })
   278→   278→
   279→   279→  if (isLoading) {
   280→   280→    return (
   281→   281→      <div className="space-y-4">
   282→   282→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   283→   283→          {Array.from({ length: 3 }).map((_, i) => (
   284→   284→            <div key={i} className="h-24 rounded-lg bg-muted animate-pulse" />
   285→   285→          ))}
   286→   286→        </div>
   287→   287→        <div className="h-96 rounded-lg bg-muted animate-pulse" />
   288→   288→      </div>
   289→   289→    )
   290→   290→  }
   291→   291→
   292→   292→  return (
   293→   293→    <div className="space-y-4">
   294→   294→      {/* Stats */}
   295→   295→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   296→   296→        {[
   297→   297→          { label: 'Total Items', value: totalItems, icon: PackageOpen, color: 'text-foreground' },
   298→   298→          { label: 'Unclaimed', value: unclaimedItems, icon: Package, color: 'text-yellow-600' },
   299→   299→          { label: 'Claimed This Month', value: claimedThisMonth, icon: Hand, color: 'text-green-600' },
   300→   300→        ].map((stat) => (
   301→   301→          <Card key={stat.label} className="p-2.5">
   302→   302→            <div className="flex items-center justify-between">
   303→   303→              <div>
   304→   304→                <p className="text-xs font-medium text-muted-foreground">{stat.label}</p>
   305→   305→                <p className="text-lg font-bold tracking-tight">{stat.value}</p>
   306→   306→              </div>
   307→   307→              <stat.icon className={cn('h-8 w-8 opacity-20', stat.color)} />
   308→   308→            </div>
   309→   309→          </Card>
   310→   310→        ))}
   311→   311→      </div>
   312→   312→
   313→   313→      {/* Filters + Add Button */}
   314→   314→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   315→   315→        <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   316→   316→          <div className="relative flex-1 sm:max-w-xs">
   317→   317→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   318→   318→            <Input
   319→   319→              placeholder="Search items..."
   320→   320→              className="pl-9"
   321→   321→              value={searchQuery}
   322→   322→              onChange={(e) => setSearchQuery(e.target.value)}
   323→   323→            />
   324→   324→          </div>
   325→   325→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   326→   326→            <SelectTrigger className="sm:w-[120px] h-7 text-xs">
   327→   327→              <Filter className="h-4 w-4 mr-2" />
   328→   328→              <SelectValue placeholder="Status" />
   329→   329→            </SelectTrigger>
   330→   330→            <SelectContent>
   331→   331→              <SelectItem value="all">All Statuses</SelectItem>
   332→   332→              <SelectItem value="found">Found</SelectItem>
   333→   333→              <SelectItem value="claimed">Claimed</SelectItem>
   334→   334→              <SelectItem value="donated">Donated</SelectItem>
   335→   335→              <SelectItem value="disposed">Disposed</SelectItem>
   336→   336→            </SelectContent>
   337→   337→          </Select>
   338→   338→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   339→   339→            <SelectTrigger className="sm:w-[120px] h-7 text-xs">
   340→   340→              <SelectValue placeholder="Category" />
   341→   341→            </SelectTrigger>
   342→   342→            <SelectContent>
   343→   343→              <SelectItem value="all">All Categories</SelectItem>
   344→   344→              {CATEGORIES.map((c) => (
   345→   345→                <SelectItem key={c} value={c}>{formatCategory(c)}</SelectItem>
   346→   346→              ))}
   347→   347→            </SelectContent>
   348→   348→          </Select>
   349→   349→        </div>
   350→   350→        <Button className="gap-2" onClick={() => setAddOpen(true)}>
   351→   351→          <Plus className="h-4 w-4" />
   352→   352→          Report Found Item
   353→   353→        </Button>
   354→   354→      </div>
   355→   355→
   356→   356→      {/* Items Table */}
   357→   357→      <Card className="py-0">
   358→   358→        <ScrollArea className="max-h-[480px]">
   359→   359→          <Table>
   360→   360→            <TableHeader>
   361→   361→              <TableRow>
   362→   362→                <TableHead className="w-[100px]">Date Found</TableHead>
   363→   363→                <TableHead>Item</TableHead>
   364→   364→                <TableHead className="hidden md:table-cell">Category</TableHead>
   365→   365→                <TableHead className="hidden sm:table-cell">Room</TableHead>
   366→   366→                <TableHead className="hidden lg:table-cell">Found By</TableHead>
   367→   367→                <TableHead>Status</TableHead>
   368→   368→                <TableHead className="hidden lg:table-cell">Storage</TableHead>
   369→   369→                <TableHead className="w-[80px]">Actions</TableHead>
   370→   370→              </TableRow>
   371→   371→            </TableHeader>
   372→   372→            <TableBody>
   373→   373→              {filteredItems.length === 0 ? (
   374→   374→                <TableRow>
   375→   375→                  <TableCell colSpan={8} className="h-24 text-center text-muted-foreground">
   376→   376→                    <PackageOpen className="h-8 w-8 mx-auto opacity-20 mb-2" />
   377→   377→                    <p>No items found</p>
   378→   378→                  </TableCell>
   379→   379→                </TableRow>
   380→   380→              ) : (
   381→   381→                filteredItems.map((item) => {
   382→   382→                  const StatusIcon = STATUS_ICON[item.status] || Package
   383→   383→                  return (
   384→   384→                    <TableRow key={item.id}>
   385→   385→                      <TableCell className="text-xs">
   386→   386→                        {format(new Date(item.foundDate), 'MMM d')}
   387→   387→                      </TableCell>
   388→   388→                      <TableCell>
   389→   389→                        <div className="flex items-center gap-2">
   390→   390→                          <StatusIcon className="h-4 w-4 text-muted-foreground shrink-0" />
   391→   391→                          <div>
   392→   392→                            <p className="font-medium text-xs">{item.itemName}</p>
   393→   393→                            {item.description && (
   394→   394→                              <p className="text-xs text-muted-foreground truncate max-w-[200px]">
   395→   395→                                {item.description}
   396→   396→                              </p>
   397→   397→                            )}
   398→   398→                          </div>
   399→   399→                        </div>
   400→   400→                      </TableCell>
   401→   401→                      <TableCell className="hidden md:table-cell text-xs">
   402→   402→                        {formatCategory(item.category)}
   403→   403→                      </TableCell>
   404→   404→                      <TableCell className="hidden sm:table-cell text-xs">
   405→   405→                        {item.roomId || '—'}
   406→   406→                      </TableCell>
   407→   407→                      <TableCell className="hidden lg:table-cell text-xs">
   408→   408→                        {item.foundBy}
   409→   409→                      </TableCell>

   411→   411→                        <StatusBadge status={item.status} />
   412→   412→                      </TableCell>
   413→   413→                      <TableCell className="hidden lg:table-cell text-xs">
   414→   414→                        {item.storageLocation || '—'}

   416→   416→                      <TableCell>
   417→   417→                        <div className="flex items-center gap-1">
   418→   418→                          <Button variant="ghost" size="sm" className="h-7 w-7 p-0 text-muted-foreground hover:text-foreground"
   419→   419→                            onClick={() => { setSelectedItem(item); setViewOpen(true) }} title="View Details">
   420→   420→                            <Eye className="h-3.5 w-3.5" />
   421→   421→                          </Button>
   422→   422→                          {item.status === 'found' && (
   423→   423→                            <Button variant="outline" size="sm" className="h-7 text-xs"
   424→   424→                              onClick={() => { setSelectedItem(item); setClaimOpen(true) }}>
   425→   425→                              Claim

   427→   427→                          )}
   428→   428→                        </div>
   429→   429→                      </TableCell>
   430→   430→                    </TableRow>
   431→   431→                  )
   432→   432→                })
   433→   433→              )}
   434→   434→            </TableBody>
   435→   435→          </Table>
   436→   436→        </ScrollArea>
   437→   437→      </Card>
   438→   438→
