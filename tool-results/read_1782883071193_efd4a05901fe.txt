   180→// ─── Order Card ────────────────────────────────────────────────────────
   181→function OrderCard({
   182→  order,
   183→  onStatusChange,
   184→}: {
   185→  order: RoomServiceOrder
   186→  onStatusChange: (id: string, status: RoomServiceOrder['status']) => void

   195→    <Card className="rounded-lg border">
   196→      <CardHeader className="pb-2 pt-4 px-4">

   206→          </span>
   207→        </div>

   229→        </div>
   230→
   231→        {order.specialInstructions && (
   232→          <div className="rounded-md bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 p-2 flex items-start gap-2">

   234→            <p className="text-xs text-amber-700 dark:text-amber-300">{order.specialInstructions}</p>
   235→          </div>
   236→        )}
   237→
   238→        {/* Action Buttons */}
   239→        {order.status !== 'cancelled' && order.status !== 'delivered' && (
   240→          <div className="flex gap-2 pt-1">
   241→            {nextStatus && (
   242→              <Button
   243→                size="sm"
   244→                className="flex-1 gap-1 text-[11px] h-7"
   245→                onClick={() => onStatusChange(order.id, nextStatus as RoomServiceOrder['status'])}

   247→                {nextStatus === 'preparing' && <ChefHat className="h-3.5 w-3.5" />}

   251→            )}
   252→            <Button
   253→              size="sm"
   254→              variant="destructive"
   255→              className="gap-1 text-[11px] h-7"
   256→              onClick={() => onStatusChange(order.id, 'cancelled')}

   260→            </Button>
   261→          </div>

   264→    </Card>

   266→}

   268→// ─── New Order Dialog ─────────────────────────────────────────────────
   269→function NewOrderDialog({
   270→  open,
   271→  onClose,
   272→  onSubmit,
   273→}: {
   274→  open: boolean
   275→  onClose: () => void
   276→  onSubmit: (order: Omit<RoomServiceOrder, 'id' | 'createdAt' | 'status' | 'total'> & { total: number }) => void
   277→}) {
   278→  const [selectedGuest, setSelectedGuest] = useState('')
   279→  const [orderItems, setOrderItems] = useState<{ name: string; quantity: number; price: number }[]>([])
   280→  const [specialInstructions, setSpecialInstructions] = useState('')
   281→  const [selectedMenuCategory, setSelectedMenuCategory] = useState('all')
   282→
   283→  const selectedGuestData = MOCK_GUESTS.find((g) => g.id === selectedGuest)
   284→
   285→  const menuCategories = ['all', 'Appetizer', 'Main Course', 'Beverage', 'Breakfast', 'Bread', 'Salad', 'Dessert']
   286→  const filteredMenu = MOCK_MENU_ITEMS.filter((m) => selectedMenuCategory === 'all' || m.category === selectedMenuCategory)
   287→
   288→  const handleAddItem = (menuItem: typeof MOCK_MENU_ITEMS[0]) => {
   289→    const existing = orderItems.find((i) => i.name === menuItem.name)
   290→    if (existing) {
   291→      setOrderItems(orderItems.map((i) => i.name === menuItem.name ? { ...i, quantity: i.quantity + 1 } : i))
   292→    } else {
   293→      setOrderItems([...orderItems, { name: menuItem.name, quantity: 1, price: menuItem.price }])
   294→    }
   295→  }
   296→
   297→  const handleRemoveItem = (name: string) => {
   298→    setOrderItems(orderItems.filter((i) => i.name !== name))
   299→  }
   300→
   301→  const handleQtyChange = (name: string, delta: number) => {
   302→    setOrderItems(orderItems.map((i) => {
   303→      if (i.name !== name) return i
   304→      const newQty = Math.max(0, i.quantity + delta)
   305→      return { ...i, quantity: newQty }
   306→    }).filter((i) => i.quantity > 0))
   307→  }
   308→
   309→  const total = orderItems.reduce((s, i) => s + i.price * i.quantity, 0)
   310→
   311→  const handleSubmit = () => {
   312→    if (!selectedGuest || orderItems.length === 0) {
   313→      toast.error('Please select a guest and add at least one item')
   314→      return
   315→    }
   316→    onSubmit({
   317→      roomNumber: selectedGuestData!.room,
   318→      floor: selectedGuestData!.floor,
   319→      guestName: selectedGuestData!.name,
   320→      phone: '+977-98' + Math.floor(10000000 + Math.random() * 90000000).toString(),
   321→      items: orderItems,
   322→      specialInstructions,
   323→      total,
   324→    })
   325→    setSelectedGuest('')
   326→    setOrderItems([])
   327→    setSpecialInstructions('')
   328→    onClose()
   329→  }
   330→
   331→  return (
   332→    <Dialog open={open} onOpenChange={onClose}>
   333→      <DialogContent className="sm:max-w-2xl max-h-[85vh] overflow-y-auto">
   334→        <DialogHeader>
   335→          <DialogTitle className="flex items-center gap-2">
   336→            <BellRing className="h-4 w-4" />
   337→            New Room Service Order
   338→          </DialogTitle>
   339→        </DialogHeader>
   340→        <div className="space-y-4">
   341→          {/* Guest Selection */}
   342→          <div>
   343→            <Label className="text-sm font-medium">Select Guest</Label>
   344→            <Select value={selectedGuest} onValueChange={setSelectedGuest}>
   345→              <SelectTrigger className="mt-1.5">
   346→                <SelectValue placeholder="Choose in-house guest..." />
   347→              </SelectTrigger>
   348→              <SelectContent>
   349→                {MOCK_GUESTS.map((g) => (
   350→                  <SelectItem key={g.id} value={g.id}>
   351→                    {g.name} — Room {g.room} (Floor {g.floor})
   352→                  </SelectItem>
   353→                ))}
   354→              </SelectContent>
   355→            </Select>
   356→          </div>
   357→
   358→          {/* Menu Items */}
   359→          <div>
   360→            <Label className="text-sm font-medium">Add Items from Menu</Label>
   361→            <Select value={selectedMenuCategory} onValueChange={setSelectedMenuCategory}>
   362→              <SelectTrigger className="mt-1.5 mb-2">
   363→                <SelectValue />
   364→              </SelectTrigger>
   365→              <SelectContent>
   366→                {menuCategories.map((cat) => (
   367→                  <SelectItem key={cat} value={cat}>
   368→                    {cat === 'all' ? 'All Categories' : cat}
   369→                  </SelectItem>
   370→                ))}
   371→              </SelectContent>
   372→            </Select>
   373→            <ScrollArea className="h-48">
   374→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-1.5 pr-2">
   375→                {filteredMenu.map((item) => (
   376→                  <button
   377→                    key={item.id}
   378→                    onClick={() => handleAddItem(item)}
   379→                    className="flex items-center justify-between rounded-md border p-2 text-left text-xs hover:bg-muted/50 transition-colors"
   380→                  >
   381→                    <span className="truncate">{item.name}</span>
   382→                    <span className="ml-2 flex items-center gap-1 flex-shrink-0">
   383→                      <span className="font-medium">{formatNPR(item.price)}</span>
   384→                      <Plus className="h-3 w-3 text-muted-foreground" />
   385→                    </span>
   386→                  </button>
   387→                ))}
   388→              </div>
   389→            </ScrollArea>
   390→          </div>
   391→
   392→          <Separator />
   393→
   394→          {/* Selected Items */}
   395→          {orderItems.length > 0 && (
   396→            <div>
   397→              <Label className="text-sm font-medium">Order Items</Label>
   398→              <div className="mt-2 space-y-1.5">
   399→                {orderItems.map((item) => (
   400→                  <div key={item.name} className="flex items-center justify-between rounded-md border p-2">
   401→                    <div className="flex-1 min-w-0">
   402→                      <p className="text-xs font-medium truncate">{item.name}</p>
   403→                      <p className="text-[10px] text-muted-foreground">{formatNPR(item.price)}</p>
   404→                    </div>
   405→                    <div className="flex items-center gap-1">
   406→                      <Button variant="outline" size="icon" className="h-6 w-6" onClick={() => handleQtyChange(item.name, -1)}>
   407→                        <span className="text-xs">−</span>
   408→                      </Button>
   409→                      <span className="w-6 text-center text-xs font-medium">{item.quantity}</span>
   410→                      <Button variant="outline" size="icon" className="h-6 w-6" onClick={() => handleQtyChange(item.name, 1)}>
   411→                        <span className="text-xs">+</span>
   412→                      </Button>
   413→                    </div>
   414→                    <span className="text-xs font-medium w-16 text-right">{formatNPR(item.price * item.quantity)}</span>
   415→                    <Button variant="ghost" size="icon" className="h-6 w-6 text-destructive" onClick={() => handleRemoveItem(item.name)}>
   416→                      <XCircle className="h-3 w-3" />
   417→                    </Button>
   418→                  </div>
   419→                ))}
   420→                <div className="flex justify-between pt-1 text-sm font-bold border-t mt-2 pt-2">
   421→                  <span>Total</span>
   422→                  <span>{formatNPR(total)}</span>
   423→                </div>
   424→              </div>
   425→            </div>
   426→          )}
   427→
   428→          {/* Special Instructions */}
   429→          <div>
   430→            <Label className="text-sm font-medium">Special Instructions</Label>
   431→            <Input
   432→              placeholder="Any dietary requirements or special requests..."
   433→              value={specialInstructions}
   434→              onChange={(e) => setSpecialInstructions(e.target.value)}

   475→      preparing: 'Order is now being prepared',
   476→      delivered: 'Order marked as delivered',
   477→      cancelled: 'Order has been cancelled',

   481→

   485→      id: `RS-${String(orders.length + 1).padStart(3, '0')}`,
   486→      status: 'received',
   487→      createdAt: new Date().toISOString(),
   488→    }
   489→    setOrders((prev) => [newOrder, ...prev])
   490→    toast.success(`Room service order placed for Room ${order.roomNumber}`)
   491→  }
   492→
   493→  if (isLoading && !data) {
   494→    return (
   495→      <div className="space-y-2">
   496→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-2">
   497→          {[1, 2, 3, 4].map((i) => (

   499→          ))}
   500→        </div>

   504→        </div>
   505→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-2">
   506→          <Skeleton className="h-60 rounded-lg" />

   525→            </SelectTrigger>

   527→              <SelectItem value="all">All Status</SelectItem>
   528→              <SelectItem value="received">Received</SelectItem>
   529→              <SelectItem value="preparing">Preparing</SelectItem>
   530→              <SelectItem value="delivered">Delivered</SelectItem>
   531→              <SelectItem value="cancelled">Cancelled</SelectItem>

   533→          </Select>
