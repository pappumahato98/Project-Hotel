   164→    default: return 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400 border-gray-200 dark:border-gray-700'

   256→// ── Task Detail Dialog ───────────────────────────────────────
   257→function TaskDetailDialog({ task, open, onOpenChange }: { task: HkTask | null; open: boolean; onOpenChange: (open: boolean) => void }) {
   258→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   259→  const queryClient = useQueryClient()
   260→
   261→  const updateStatusMutation = useMutation({
   262→    mutationFn: (body: { id: string; status: string; inspectedBy?: string }) =>
   263→      apiFetch('/api/housekeeping', { method: 'POST', body }),
   264→    onSuccess: () => {
   265→      toast.success('Task status updated')
   266→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   267→      queryClient.invalidateQueries({ queryKey: ['housekeeping-rooms'] })
   268→      onOpenChange(false)
   269→    },
   270→  })
   271→
   272→  if (!task) return null
   273→
   274→  const toggleCheck = (item: string) => {
   275→    setCheckedItems((prev) => ({ ...prev, [item]: !prev[item] }))
   276→  }
   277→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   278→
   279→  return (
   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→      <DialogContent className="sm:max-w-lg">
   282→        <DialogHeader>
   283→          <DialogTitle className="flex items-center gap-2">
   284→            Room {task.room.number} — {formatTaskType(task.taskType)}
   285→          </DialogTitle>
   286→          <DialogDescription>
   287→            Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   288→          </DialogDescription>
   289→        </DialogHeader>
   290→
   291→        <div className="space-y-4">
   292→          <div className="grid grid-cols-2 gap-3">
   293→            <div className="rounded-lg border p-3 space-y-1">
   294→              <p className="text-xs text-muted-foreground">Priority</p>
   295→              <Badge variant="outline" className={cn(priorityColor(task.priority))}>
   296→                {task.priority.charAt(0).toUpperCase() + task.priority.slice(1)}
   297→              </Badge>
   298→            </div>
   299→            <div className="rounded-lg border p-3 space-y-1">
   300→              <p className="text-xs text-muted-foreground">Status</p>
   301→              <StatusBadge status={task.status} />
   302→            </div>
   303→            <div className="rounded-lg border p-3 space-y-1">
   304→              <p className="text-xs text-muted-foreground">Assigned To</p>
   305→              <p className="text-sm font-medium">{task.assignedTo || 'Unassigned'}</p>
   306→            </div>
   307→            <div className="rounded-lg border p-3 space-y-1">
   308→              <p className="text-xs text-muted-foreground">Est. Time</p>
   309→              <p className="text-sm font-medium">{task.estimatedMinutes} min</p>
   310→            </div>
   311→          </div>
   312→
   313→          <div className="space-y-2">
   314→            <div className="flex items-center justify-between">
   315→              <p className="text-sm font-semibold">Cleaning Checklist</p>
   316→              <span className="text-xs text-muted-foreground">{checkedCount}/{CLEANING_CHECKLIST.length}</span>
   317→            </div>
   318→            <div className="rounded-lg border divide-y max-h-64 overflow-y-auto">
   319→              {CLEANING_CHECKLIST.map((item) => (
   320→                <label
   321→                  key={item}
   322→                  className="flex items-center gap-3 px-3 py-2 text-sm hover:bg-muted/50 cursor-pointer transition-colors"
   323→                >
   324→                  <Checkbox
   325→                    checked={checkedItems[item] || false}
   326→                    onCheckedChange={() => toggleCheck(item)}
   327→                  />
   328→                  <span className={checkedItems[item] ? 'line-through text-muted-foreground' : ''}>
   329→                    {item}
   330→                  </span>
   331→                </label>
   332→              ))}
   333→            </div>
   334→          </div>
   335→
   336→          {task.notes && (
   337→            <div className="rounded-lg border p-3">
   338→              <p className="text-xs text-muted-foreground mb-1">Notes</p>
   339→              <p className="text-sm">{task.notes}</p>
   340→            </div>
   341→          )}
   342→
   343→          {/* Status Actions */}
   344→          <div className="flex gap-2 pt-2">
   345→            {task.status === 'pending' && (
   346→              <Button
   347→                size="sm"
   348→                variant="outline"
   349→                className="flex-1"
   350→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'in_progress' })}
   351→                disabled={updateStatusMutation.isPending}
   352→              >
   353→                <Clock className="w-3.5 h-3.5 mr-1.5" /> Start Cleaning
   354→              </Button>
   355→            )}
   356→            {task.status === 'in_progress' && (
   357→              <Button
   358→                size="sm"
   359→                className="flex-1"
   360→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'cleaned' })}
   361→                disabled={updateStatusMutation.isPending}
   362→              >
   363→                <CheckCircle2 className="w-3.5 h-3.5 mr-1.5" /> Mark Cleaned
   364→              </Button>
   365→            )}
   366→            {task.status === 'cleaned' && (
   367→              <Button
   368→                size="sm"
   369→                className="flex-1"
   370→                onClick={() => updateStatusMutation.mutate({ id: task.id, status: 'inspected', inspectedBy: 'Admin' })}
   371→                disabled={updateStatusMutation.isPending}
   372→              >
   373→                <Eye className="w-3.5 h-3.5 mr-1.5" /> Mark Inspected
   374→              </Button>
   375→            )}
   376→          </div>
   377→        </div>
   378→      </DialogContent>
   379→    </Dialog>
   380→  )
   381→}

   383→// ── Attendant View ───────────────────────────────────────────
   384→function AttendantView({ tasks }: { tasks: HkTask[] }) {
   385→  const attendants = new Map<string, HkTask[]>()
   386→  tasks.forEach((task) => {
   387→    const name = task.assignedTo || 'Unassigned'
   388→    if (!attendants.has(name)) attendants.set(name, [])
   389→    attendants.get(name)!.push(task)
   390→  })
   391→
   392→  return (
   393→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   394→      {Array.from(attendants.entries()).map(([name, attTasks]) => (
   395→        <Card key={name}>
   396→          <CardContent className="p-3 space-y-2">
   397→            <div className="flex items-center gap-3">
   398→              <div className="flex h-9 w-9 items-center justify-center rounded-full bg-muted text-sm font-medium">
   399→                {name.charAt(0)}
   400→              </div>
   401→              <div>
   402→                <p className="text-sm font-semibold">{name}</p>
   403→                <p className="text-xs text-muted-foreground">{attTasks.length} tasks assigned</p>
   404→              </div>
   405→            </div>
   406→            {attTasks.map((task) => (
   407→              <div key={task.id} className="flex items-center justify-between rounded-md border p-2 text-sm">
   408→                <div className="flex items-center gap-2">
   409→                  <BedDouble className="h-3.5 w-3.5 text-muted-foreground" />
   410→                  <span className="font-medium">Room {task.room.number}</span>
   411→                </div>
   412→                <StatusBadge status={task.status} />
   413→              </div>
   414→            ))}
   415→          </CardContent>
   416→        </Card>
   417→      ))}
   418→    </div>
   419→  )
   420→}

  1038→      {/* Views */}

  1043→      {/* Task Detail Dialog (for Kanban/Attendant) */}
