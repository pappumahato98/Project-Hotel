     8→  LogOut, BedDouble, Receipt, CreditCard, AlertTriangle, CheckCircle2, Printer,

    10→} from 'lucide-react'

    36→// ─── Types ──────────────────────────────────────────────────────────────
    37→
    38→interface DepartureGuest {
    39→  id: string
    40→  firstName: string
    41→  lastName: string
    42→  vipLevel: string
    43→}
    44→
    45→interface DepartureRoom {
    46→  id: string
    47→  number: string
    48→  floor: number
    49→  type: { name: string; code: string; bedConfig?: string }
    50→}
    51→
    52→interface FolioLineItem {
    53→  id: string
    54→  description: string
    55→  amount: number
    56→  type: 'charge' | 'payment'
    57→  date: string
    58→}
    59→
    60→interface DepartureFolio {
    61→  id: string
    62→  balance: number
    63→  status: string
    64→  items?: FolioLineItem[]
    65→}
    66→
    67→interface Departure {
    68→  id: string
    69→  confirmationNo: string
    70→  status: string
    71→  checkIn: string
    72→  checkOut: string
    73→  roomRate: number
    74→  totalAmount: number
    75→  paidAmount: number
    76→  creditLimit: number
    77→  guest: DepartureGuest
    78→  room: DepartureRoom | null
    79→  adults?: number
    80→  children?: number
    81→  folios: DepartureFolio[]
    82→}
    83→
    84→type LateCheckoutOption = '14:00' | '16:00' | '18:00'
    85→type PaymentMethod = 'cash' | 'card' | 'bank_transfer'
    86→
    87→const LATE_CHECKOUT_OPTIONS: Record<LateCheckoutOption, { label: string; surchargePercent: number; timeLabel: string }> = {
    88→  '14:00': { label: '2:00 PM', surchargePercent: 25, timeLabel: '2:00 PM (+25% surcharge)' },
    89→  '16:00': { label: '4:00 PM', surchargePercent: 50, timeLabel: '4:00 PM (+50% surcharge)' },
    90→  '18:00': { label: '6:00 PM', surchargePercent: 100, timeLabel: '6:00 PM (+100% surcharge)' },
    91→}
    92→
    93→const PAYMENT_METHOD_LABELS: Record<PaymentMethod, string> = {
    94→  cash: 'Cash',
    95→  card: 'Credit/Debit Card',
    96→  bank_transfer: 'Bank Transfer',
    97→}
    98→
    99→// ─── Component ──────────────────────────────────────────────────────────
   100→
   101→export function DeparturesView() {
   102→  const queryClient = useQueryClient()
   103→  const { settings } = useSettingsStore()
   104→  const { preferences } = usePreferencesStore()
   105→  const { navigateTo } = useNavigationStore()
   106→  const today = getTodayString()
   107→
   108→  // Dialog states
   109→  const [checkoutDialogOpen, setCheckoutDialogOpen] = useState(false)
   110→  const [folioDialogOpen, setFolioDialogOpen] = useState(false)
   111→  const [lateCheckoutDialogOpen, setLateCheckoutDialogOpen] = useState(false)
   112→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   113→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false)
   114→
   115→  // Selected departure
   116→  const [selectedDeparture, setSelectedDeparture] = useState<Departure | null>(null)
   117→
   118→  // Late checkout
   119→  const [lateCheckoutOption, setLateCheckoutOption] = useState<LateCheckoutOption>('14:00')
   120→
   121→  // Payment
   122→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash')
   123→  const [paymentAmount, setPaymentAmount] = useState('')
   124→  const [paymentReference, setPaymentReference] = useState('')
   125→
   126→  // Receipt data
   127→  const [receiptData, setReceiptData] = useState<Departure | null>(null)
   128→
   129→  // Fetch today's departures (checked_in with checkOut = today)
   130→  const { data, isLoading } = useQuery({
   131→    queryKey: ['departures', today],
   132→    queryFn: async () => {
   133→      // Fetch both: today's departures AND overdue departures
   134→      const [todayData, overdueData] = await Promise.all([
   135→        apiFetch(`/api/reservations?status=checked_in&checkOutDate=${today}`),
   136→        // Fetch overdue: checked_in with checkOut < today
   137→        apiFetch(`/api/reservations?status=checked_in&checkOutBefore=${today}`),
   138→      ])
   139→      // Merge and deduplicate
   140→      const todayReservations = todayData?.reservations || []
   141→      const overdueReservations = overdueData?.reservations || []
   142→      const seen = new Set<string>()
   143→      const merged = [...todayReservations, ...overdueReservations].filter((r: { id: string }) => {
   144→        if (seen.has(r.id)) return false
   145→        seen.add(r.id)
   146→        return true
   147→      })
   148→      return { reservations: merged }
   149→    },
   150→    refetchInterval: 30000,
   151→  })
   152→
   153→  const departures: Departure[] = data?.reservations || []
   154→
   155→  // Stats
   156→  const overdueCount = departures.filter((d) => new Date(d.checkOut) < new Date(today)).length
   157→  const todayCount = departures.length - overdueCount
   158→  const totalDepartures = departures.length
   159→  const checkedOutCount = departures.filter((d) => d.status === 'checked_out').length
   160→  const pendingDepartures = departures.filter((d) => d.status === 'checked_in').length
   161→  const outstandingBalance = departures.reduce((sum, d) => {
   162→    return sum + (d.folios[0]?.balance || 0)
   163→  }, 0)
   164→
   165→  // Zero-balance pending guests (for express & batch checkout)
   166→  const zeroBalancePending = departures.filter(
   167→    (d) => d.status === 'checked_in' && (d.folios[0]?.balance || 0) === 0,
   168→  )
   169→
   170→  // ─── Mutations ─────────────────────────────────────────────────────
   171→
   172→  // Checkout mutation
   173→  const checkoutMutation = useMutation({
   174→    mutationFn: async (reservationId: string) => {
   175→      return apiFetch(`/api/reservations/${reservationId}`, {
   176→        method: 'PATCH',
   177→        headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify({ status: 'checked_out' }),
   179→      })
   180→    },
   181→    onSuccess: () => {
   182→      invalidate.afterCheckout(queryClient)
   183→    },
   184→  })
   185→
   186→  // Payment mutation
   187→  const paymentMutation = useMutation({
   188→    mutationFn: async ({ folioId, amount, method, reference }: { folioId: string; amount: number; method: PaymentMethod; reference: string }) => {
   189→      return apiFetch(`/api/folio/${folioId}`, {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ type: 'payment', paymentMethod: method, amount, reference }),
   193→      })
   194→    },
   195→    onSuccess: () => {
   196→      invalidate.afterFolioChange(queryClient)
   197→      setPaymentDialogOpen(false)
   198→      setPaymentAmount('')
   199→      setPaymentReference('')
   200→      toast.success('Payment recorded successfully')
   201→    },
   202→    onError: () => {
   203→      toast.error('Failed to record payment')
   204→    },
   205→  })
   206→
   207→  // Late checkout mutation
   208→  const lateCheckoutMutation = useMutation({
   209→    mutationFn: async ({ reservationId, newCheckOut, surcharge }: { reservationId: string; newCheckOut: string; surcharge: number }) => {
   210→      // First get the folio ID for this reservation
   211→      const reservation = departures.find((d) => d.id === reservationId)
   212→      const folioId = reservation?.folios[0]?.id
   213→
   214→      const result = await apiFetch(`/api/reservations/${reservationId}`, {
   215→        method: 'PATCH',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({ checkOut: newCheckOut }),
   218→      })
   219→      // Add surcharge to folio
   220→      if (surcharge > 0 && folioId) {
   221→        await apiFetch(`/api/folio/${folioId}`, {
   222→          method: 'POST',
   223→          headers: { 'Content-Type': 'application/json' },
   224→          body: JSON.stringify({
   225→            type: 'charge',
   226→            transactionType: 'late_checkout',
   227→            description: `Late checkout surcharge (${LATE_CHECKOUT_OPTIONS[lateCheckoutOption].label})`,
   228→            amount: surcharge,
   229→          }),
   230→        })
   231→      }
   232→      return result
   233→    },
   234→    onSuccess: () => {
   235→      invalidate.afterReservationChange(queryClient)
   236→      invalidate.afterFolioChange(queryClient)
   237→      setLateCheckoutDialogOpen(false)
   238→      toast.success('Late checkout request processed successfully')
   239→    },
   240→    onError: () => {
   241→      toast.error('Failed to process late checkout request')
   242→    },
   243→  })
   244→
   245→  // Batch checkout mutation
   246→  const batchCheckoutMutation = useMutation({
   247→    mutationFn: async (ids: string[]) => {
   248→      const results = []
   249→      for (let i = 0; i < ids.length; i++) {
   250→        const result = await apiFetch(`/api/reservations/${ids[i]}`, {
   251→          method: 'PATCH',
   252→          headers: { 'Content-Type': 'application/json' },
   253→          body: JSON.stringify({ status: 'checked_out' }),
   254→        })
   255→        results.push(result)
   256→        toast.success(`Checked out ${i + 1} of ${ids.length} rooms...`)
   257→      }
   258→      return results
   259→    },
   260→    onSuccess: (_data, ids) => {
   261→      invalidate.afterCheckout(queryClient)
   262→      toast.success(`Batch checkout complete! ${ids.length} rooms checked out.`)
   263→      toast.info(`${ids.length} rooms marked for housekeeping - Vacant Dirty`)
   264→    },
   265→    onError: () => {
   266→      toast.error('Batch checkout failed. Some rooms may not have been processed.')
   267→    },
   268→  })
   269→
   270→  // ─── Handlers ─────────────────────────────────────────────────────
   271→
   272→  const getBalance = (dep: Departure) => dep.folios[0]?.balance || 0
   273→
   274→  const handleExpressCheckout = (dep: Departure) => {
   275→    checkoutMutation.mutate(dep.id, {
   276→      onSuccess: () => {
   277→        toast.success(`Express checkout complete for Room ${dep.room?.number ?? 'N/A'}`)
   278→        toast.info(`Room ${dep.room?.number ?? 'N/A'} marked for housekeeping - Vacant Dirty`)
   279→      },
   280→    })
   281→  }
   282→
   283→  const handleReviewFolio = (dep: Departure) => {
   284→    setSelectedDeparture(dep)
   285→    setFolioDialogOpen(true)
   286→  }
   287→
   288→  const handleNavigateToFolio = (dep: Departure) => {
   289→    useFolioContextStore.getState().setFolioContext({
   290→      reservationId: dep.id,
   291→      guestId: dep.guest?.id || '',
   292→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   293→      roomNumber: dep.room?.number || '',
   294→      confirmationNo: dep.confirmationNo,
   295→    })
   296→    navigateTo('front-desk', 'folio')
   297→  }
   298→
   299→  const handleViewLedger = (dep: Departure) => {
   300→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   301→      guestId: dep.guest?.id || '',
   302→      guestName: `${dep.guest?.firstName || ''} ${dep.guest?.lastName || ''}`.trim(),
   303→    })
   304→    navigateTo('front-desk', 'guest-ledger')
   305→  }
   306→
   307→  const handleFolioSettleCheckout = () => {
   308→    setFolioDialogOpen(false)
   309→    setCheckoutDialogOpen(true)
   310→  }
   311→
   312→  const handleQuickCheckout = (dep: Departure) => {
   313→    setSelectedDeparture(dep)
   314→    setCheckoutDialogOpen(true)
   315→  }
   316→
   317→  const handleRequestLateCheckout = (dep: Departure) => {
   318→    setSelectedDeparture(dep)
   319→    setLateCheckoutOption('14:00')
   320→    setLateCheckoutDialogOpen(true)
   321→  }
   322→
   323→  const handleConfirmLateCheckout = () => {
   324→    if (!selectedDeparture) return
   325→    const option = LATE_CHECKOUT_OPTIONS[lateCheckoutOption]
   326→    const surcharge = Math.round(selectedDeparture.roomRate * option.surchargePercent / 100)
   327→
   328→    // Build new checkout datetime with the selected time
   329→    const currentCheckout = new Date(selectedDeparture.checkOut)
   330→    const [hours, minutes] = lateCheckoutOption.split(':').map(Number)
   331→    currentCheckout.setHours(hours, minutes, 0, 0)
   332→
   333→    lateCheckoutMutation.mutate({
   334→      reservationId: selectedDeparture.id,
   335→      newCheckOut: currentCheckout.toISOString(),
   336→      surcharge,
   337→    })
   338→  }
   339→
   340→  const handleOpenPaymentDialog = () => {
   341→    if (!selectedDeparture) return
   342→    const balance = getBalance(selectedDeparture)
   343→    setPaymentAmount(String(balance))
   344→    setPaymentMethod('cash')
   345→    setPaymentReference('')
   346→    setPaymentDialogOpen(true)
   347→  }
   348→
   349→  const handleRecordPayment = () => {
   350→    if (!selectedDeparture) return
   351→    const folioId = selectedDeparture.folios[0]?.id
   352→    if (!folioId) {
   353→      toast.error('No folio found for this reservation')
   354→      return
   355→    }
   356→    const amount = Number(paymentAmount)
   357→    if (isNaN(amount) || amount <= 0) {
   358→      toast.error('Please enter a valid payment amount')
   359→      return
   360→    }
   361→    paymentMutation.mutate({
   362→      folioId,
   363→      amount,
   364→      method: paymentMethod,
   365→      reference: paymentReference,
   366→    })
   367→  }
   368→
   369→  const confirmCheckout = () => {
   370→    if (!selectedDeparture) return
   371→    const roomNumber = selectedDeparture.room.number
   372→    checkoutMutation.mutate(selectedDeparture.id, {
   373→      onSuccess: () => {
   374→        setCheckoutDialogOpen(false)
   375→        setReceiptData(selectedDeparture)
   376→        setReceiptDialogOpen(true)
   377→        toast.info(`Room ${roomNumber} marked for housekeeping - Vacant Dirty`)
   378→      },
   379→    })
   380→  }
   381→
   382→  const handleBatchCheckout = () => {
   383→    const ids = zeroBalancePending.map((d) => d.id)
   384→    if (ids.length === 0) {
   385→      toast.info('No guests with zero balance to batch checkout')
   386→      return
   387→    }
   388→    batchCheckoutMutation.mutate(ids)
   389→  }
   390→
   391→  const handlePrintReceipt = () => {
   392→    toast.success('Receipt sent to printer')
   393→  }
   394→
   395→  const handleEmailReceipt = () => {
   396→    toast.success('Receipt emailed to guest')
   397→  }
   398→
   399→  // ─── Computed values ──────────────────────────────────────────────
   400→
   401→  const lateCheckoutFee = selectedDeparture
   402→    ? Math.round(selectedDeparture.roomRate * LATE_CHECKOUT_OPTIONS[lateCheckoutOption].surchargePercent / 100)
   403→    : 0
   404→
   405→  // ─── Render ────────────────────────────────────────────────────────
   406→
   407→  return (
   408→    <div className="flex flex-col gap-2">
   409→      {/* Header */}
   410→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   411→        <div>
   412→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Departures</h2>
   413→          <p className="text-xs text-muted-foreground">
   414→            {todayCount} scheduled for {formatDate(today)}{overdueCount > 0 && <span className="text-red-600 font-medium"> · {overdueCount} overdue</span>}
   415→          </p>
   416→        </div>
   417→        {zeroBalancePending.length > 0 && (
   418→          <Button
   419→            onClick={handleBatchCheckout}
   420→            className="bg-green-600 hover:bg-green-700 text-white shrink-0"
   421→            disabled={batchCheckoutMutation.isPending}
   422→          >
   423→            <Zap className="size-4 mr-1.5" />
   424→            {batchCheckoutMutation.isPending
   425→              ? `Processing ${batchCheckoutMutation.submittedAt ? '...' : ''}`
   426→              : `Batch Checkout All Ready (${zeroBalancePending.length})`}
   427→          </Button>
   428→        )}
   429→      </div>
   430→
   431→      {/* Departures List */}
   432→      <Card className="py-0">
   433→        <CardContent className="p-0">
   434→          <div className="max-h-[600px] overflow-y-auto">
   435→            <Table>
   436→              <TableHeader>
   437→                <TableRow>
   438→                  <TableHead className="w-[70px]">Room</TableHead>
   439→                  <TableHead>Guest</TableHead>
   440→                  <TableHead className="w-[100px]">Check-out</TableHead>
   441→                  <TableHead className="w-[110px] text-right">Folio Balance</TableHead>
   442→                  <TableHead className="w-[100px] text-right">Status</TableHead>
   443→                  <TableHead className="w-[280px] text-right">Actions</TableHead>
   444→                </TableRow>
   445→              </TableHeader>
   446→              <TableBody>
   447→                {isLoading ? (
   448→                  Array.from({ length: 5 }).map((_, i) => (
   449→                    <TableRow key={i}>
   450→                      {Array.from({ length: 6 }).map((_, j) => (
   451→                        <TableCell key={j}>
   452→                          <Skeleton className="h-4 w-full" />
   453→                        </TableCell>
   454→                      ))}
   455→                    </TableRow>
   456→                  ))
   457→                ) : departures.length === 0 ? (
   458→                  <TableRow>
   459→                    <TableCell colSpan={6} className="h-24 text-center text-muted-foreground">
   460→                      <LogOut className="size-8 mx-auto mb-2 text-muted-foreground/50" />
   461→                      No departures scheduled for today
   462→                    </TableCell>
   463→                  </TableRow>
   464→                ) : (
   465→                  departures.map((dep) => {
   466→                    const balance = getBalance(dep)
   467→                    const isPending = dep.status === 'checked_in'
   468→                    const isZeroBalance = balance === 0
   469→                    const isOverdue = new Date(dep.checkOut) < new Date(today)
   470→                    return (
   471→                      <TableRow key={dep.id} className={isOverdue ? 'bg-red-50/50 dark:bg-red-950/20' : ''}>
   472→                        <TableCell className="font-bold font-mono">{dep.room ? <>{dep.room.number} <RoomTypeBedBadge typeName={dep.room.type.name} bedConfig={dep.room.type.bedConfig} typeCode={dep.room.type.code} pax={(dep.adults ?? 1) + (dep.children ?? 0)} inline /></> : <span className="text-muted-foreground">Unassigned</span>}</TableCell>
   473→                        <TableCell>
   474→                          <div className="flex items-center gap-2">
   475→                            <span className="font-medium">{dep.guest.firstName} {dep.guest.lastName}</span>
   476→                            {isOverdue && (
   477→                              <Badge className="text-[10px] px-1 py-0 bg-red-100 text-red-700 dark:bg-red-950 dark:text-red-300 border-0">
   478→                                OVERDUE
   479→                              </Badge>
   480→                            )}
   481→                            {dep.guest.vipLevel !== 'none' && (
   482→                              <Badge className="text-[10px] px-1 py-0 bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300">
   483→                                VIP
   484→                              </Badge>
   485→                            )}
   486→                          </div>
   487→                          <p className="text-xs text-muted-foreground">{dep.confirmationNo}</p>
   488→                        </TableCell>
   489→                        <TableCell className="text-xs">
   490→                          <div>{formatDate(dep.checkOut)}</div>
   491→                          <div className="text-muted-foreground">{formatTime(dep.checkOut)}</div>
   492→                        </TableCell>
   493→                        <TableCell className="text-right">
   494→                          <div className="flex items-center justify-end gap-1.5">
   495→                            <span className={cn('font-medium', balance > 0 ? 'text-red-600' : 'text-green-600')}>
   496→                              {formatCurrency(balance)}
   497→                            </span>
   498→                            {/* Outstanding Balance Alert Badges */}
   499→                            {isPending && balance > 5000 && (
   500→                              <Badge className="bg-red-500 text-white text-[9px] px-1 py-0 animate-pulse border-0">
   501→                                HIGH
   502→                              </Badge>
   503→                            )}
   504→                            {isPending && balance > 0 && balance <= 5000 && (
   505→                              <Badge className="bg-orange-500 text-white text-[9px] px-1 py-0 border-0">
   506→                                DUE
   507→                              </Badge>
   508→                            )}
   509→                          </div>
   510→                        </TableCell>
   511→                        <TableCell className="text-right">
   512→                          {dep.status === 'checked_out' ? (
   513→                            <div className="flex items-center justify-end gap-1">
   514→                              <BedSingle className="size-3 text-yellow-500" />
   515→                              <StatusBadge status="vacant_dirty" />
   516→                            </div>
   517→                          ) : (
   518→                            <StatusBadge status="checked_in" />
   519→                          )}
   520→                        </TableCell>
   521→                        <TableCell className="text-right">
   522→                          <div className="flex justify-end gap-1 flex-wrap">
   523→                            {isPending && (
   524→                              <>
   525→                                {/* Express Checkout (zero balance) */}
   526→                                {isZeroBalance && (
   527→                                  <Button
   528→                                    size="sm"
   529→                                    className="text-xs h-7 bg-green-600 hover:bg-green-700 text-white"
   530→                                    onClick={() => handleExpressCheckout(dep)}
   531→                                    disabled={checkoutMutation.isPending}
   532→                                  >
   533→                                    <Zap className="size-3 mr-0.5" /> Express
