   159→   159→  const queryClient = useQueryClient()
   160→   160→  const { settings } = useSettingsStore()

   162→   162→  const today = getTodayString()

   168→   168→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   169→   169→
   170→   170→  // Selection states
   171→   171→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   172→   172→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   173→   173→
   174→   174→  // Check-in form states
   175→   175→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   176→   176→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   177→   177→  const [roomPreference, setRoomPreference] = useState('')
   178→   178→  const [pillowType, setPillowType] = useState('')
   179→   179→  const [wakeupCall, setWakeupCall] = useState('')
   180→   180→
   181→   181→  // Key card states
   182→   182→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   183→   183→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   184→   184→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   185→   185→
   186→   186→  // Walk-in form state

   188→   188→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   189→   189→
   190→   190→  // ─── Queries ────────────────────────────────────────────────────────
   191→   191→
   192→   192→  const { data, isLoading } = useQuery({
   193→   193→    queryKey: ['arrivals', today],
   194→   194→    queryFn: async () => {
   195→   195→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   196→   196→      return apiFetch(`/api/reservations?${params.toString()}`)
   197→   197→    },
   198→   198→    refetchInterval: 30000,
   199→   199→  })
   200→   200→
   201→   201→  const { data: roomsData } = useQuery({
   202→   202→    queryKey: ['rooms', 'vacant'],
   203→   203→    queryFn: () => apiFetch('/api/rooms'),
   204→   204→  })
   205→   205→
   206→   206→  const arrivals: Arrival[] = data?.reservations || []
   207→   207→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   208→   208→  const availableRooms = allRooms.filter(
   209→   209→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   210→   210→  )

   213→   213→
   214→   214→  const sortedArrivals = useMemo(() => {
   215→   215→    return [...arrivals].sort((a, b) => {
   216→   216→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   217→   217→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   218→   218→      if (aVip && !bVip) return -1
   219→   219→      if (!aVip && bVip) return 1
   220→   220→      return 0
   221→   221→    })
   222→   222→  }, [arrivals])
   223→   223→
   224→   224→  // ─── Stats ──────────────────────────────────────────────────────────
   225→   225→
   226→   226→  const totalArrivals = arrivals.length

   233→   233→  const assignRoomMutation = useMutation({
   234→   234→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   235→   235→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   236→   236→      if (specialRequests !== undefined) {
   237→   237→        payload.specialRequests = specialRequests
   238→   238→      }
   239→   239→      return apiFetch(`/api/reservations/${reservationId}`, {
   240→   240→        method: 'PATCH',
   241→   241→        headers: { 'Content-Type': 'application/json' },
   242→   242→        body: JSON.stringify(payload),
   243→   243→      })
   244→   244→    },
   245→   245→    onSuccess: (_data, variables) => {
   246→   246→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   247→   247→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   248→   248→      setRoomPickerOpen(false)
   249→   249→      setCheckInDialogOpen(false)
   250→   250→
   251→   251→      // Find the room number for key card dialog
   252→   252→      const room = availableRooms.find((r) => r.id === variables.roomId)
   253→   253→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   254→   254→      openKeyCardDialog(roomNumber)
   255→   255→    },
   256→   256→  })
   257→   257→
   258→   258→  const checkInMutation = useMutation({
   259→   259→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   260→   260→      const payload: Record<string, unknown> = { status: 'checked_in' }
   261→   261→      if (specialRequests !== undefined) {
   262→   262→        payload.specialRequests = specialRequests
   263→   263→      }
   264→   264→      return apiFetch(`/api/reservations/${reservationId}`, {
   265→   265→        method: 'PATCH',
   266→   266→        headers: { 'Content-Type': 'application/json' },
   267→   267→        body: JSON.stringify(payload),
   268→   268→      })
   269→   269→    },
   270→   270→    onSuccess: (_data, variables) => {
   271→   271→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   272→   272→      setCheckInDialogOpen(false)
   273→   273→
   274→   274→      const roomNumber = selectedArrival?.room?.number || ''
   275→   275→      openKeyCardDialog(roomNumber)
   276→   276→    },
   277→   277→  })
   278→   278→
   279→   279→  const walkInMutation = useMutation({
   280→   280→    mutationFn: async (form: WalkInForm) => {
   281→   281→      // 1. Create guest
   282→   282→      const guestData = await apiFetch('/api/guests', {
   283→   283→        method: 'POST',
   284→   284→        headers: { 'Content-Type': 'application/json' },
   285→   285→        body: JSON.stringify({
   286→   286→          firstName: form.firstName,
   287→   287→          lastName: form.lastName,
   288→   288→          phone: form.phone || null,
   289→   289→          email: form.email || null,
   290→   290→          nationality: form.nationality || null,
   291→   291→          idType: form.idType || null,
   292→   292→          idNumber: form.idNumber || null,
   293→   293→        }),
   294→   294→      })
   295→   295→      const guestId = guestData.guest.id
   296→   296→
   297→   297→      // 2. Find a vacant room
   298→   298→      const vacantRoom = availableRooms[0]
   299→   299→      if (!vacantRoom) throw new Error('No available rooms')
   300→   300→
   301→   301→      // 3. Create reservation as checked_in (walk-in)
   302→   302→      const checkOutDate = form.checkOutDate || (() => {
   303→   303→        const d = new Date()
   304→   304→        d.setDate(d.getDate() + 1)
   305→   305→        return d
   306→   306→      })()
   307→   307→
   308→   308→      const resData = await apiFetch('/api/reservations', {
   309→   309→        method: 'POST',
   310→   310→        headers: { 'Content-Type': 'application/json' },
   311→   311→        body: JSON.stringify({
   312→   312→          guestId,
   313→   313→          roomId: vacantRoom.id,
   314→   314→          adults: form.adults,
   315→   315→          children: form.children,
   316→   316→          checkIn: new Date().toISOString(),
   317→   317→          checkOut: checkOutDate.toISOString(),
   318→   318→          roomRate: form.roomRate,
   319→   319→          source: 'walk_in',
   320→   320→          reservationType: 'walk_in',
   321→   321→          guaranteed: false,
   322→   322→        }),
   323→   323→      })
   324→   324→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   325→   325→    },
   326→   326→    onSuccess: (result) => {
   327→   327→      queryClient.invalidateQueries({ queryKey: ['arrivals'] })
   328→   328→      queryClient.invalidateQueries({ queryKey: ['rooms'] })
   329→   329→      queryClient.invalidateQueries({ queryKey: ['guests'] })
   330→   330→      setWalkInDialogOpen(false)
   331→   331→      setWalkInForm(initialWalkInForm)
   332→   332→
   333→   333→      // Open key card dialog
   334→   334→      openKeyCardDialog(result.roomNumber)
   335→   335→    },
   336→   336→    onError: (error) => {
   337→   337→      toast.error('Walk-in check-in failed', {
   338→   338→        description: error instanceof Error ? error.message : 'Unknown error',
   339→   339→      })
   340→   340→    },
   341→   341→  })
   342→   342→
   343→   343→  // ─── Handlers ──────────────────────────────────────────────────────
   344→   344→

   378→   378→
   379→   379→  const buildSpecialRequests = (arrival: Arrival): string => {
   380→   380→    const parts: string[] = []
   381→   381→
   382→   382→    // Preserve existing requests
   383→   383→    if (arrival.specialRequests) {

   385→   385→    }
   386→   386→
   387→   387→    // Early check-in note

   389→   389→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   390→   390→    }
   391→   391→
   392→   392→    // Check-out time override
   393→   393→    if (checkOutTime !== settings.defaultCheckOut) {
   394→   394→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   395→   395→    }
   396→   396→
   397→   397→    // Room preference
   398→   398→    if (roomPreference) {
   399→   399→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   400→   400→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   401→   401→    }
   402→   402→
   403→   403→    // Pillow type
   404→   404→    if (pillowType) {
   405→   405→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   406→   406→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   407→   407→    }
   408→   408→
   409→   409→    // Wake-up call
   410→   410→    if (wakeupCall) {
   411→   411→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   412→   412→    }
   413→   413→
   414→   414→    return parts.join(' | ')
   415→   415→  }
   416→   416→
   417→   417→  const confirmCheckIn = () => {
   418→   418→    if (!selectedArrival) return
   419→   419→
   420→   420→    const updatedRequests = buildSpecialRequests(selectedArrival)
   421→   421→
   422→   422→    if (selectedRoomId) {
   423→   423→      assignRoomMutation.mutate({
   424→   424→        reservationId: selectedArrival.id,
   425→   425→        roomId: selectedRoomId,
   426→   426→        specialRequests: updatedRequests,
   427→   427→      })
   428→   428→    } else if (selectedArrival.room) {
   429→   429→      checkInMutation.mutate({
   430→   430→        reservationId: selectedArrival.id,
   431→   431→        specialRequests: updatedRequests,
   432→   432→      })
   433→   433→    }
   434→   434→  }
   435→   435→
   436→   436→  const openKeyCardDialog = (roomNumber: string) => {
   437→   437→    setKeyCardRoomNumber(roomNumber)
   438→   438→    setKeyCardIssued(false)
   439→   439→    setKeyCardProcessed(false)
   440→   440→    setKeyCardDialogOpen(true)
   441→   441→  }
   442→   442→
   443→   443→  const handleKeyCardConfirm = () => {

   446→   446→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   447→   447→      duration: 5000,
   448→   448→      icon: <Mail className="size-4" />,
   449→   449→    })
   450→   450→    setSelectedArrival(null)
   451→   451→    setSelectedRoomId('')
   452→   452→  }
   453→   453→
   454→   454→  const resetCheckInForm = () => {
   455→   455→    setEarlyCheckIn(false)
   456→   456→    setCheckOutTime(settings.defaultCheckOut)
   457→   457→    setRoomPreference('')
   458→   458→    setPillowType('')
   459→   459→    setWakeupCall('')
   460→   460→  }
   461→   461→
   462→   462→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   463→   463→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   464→   464→    setWalkInForm((prev) => ({
   465→   465→      ...prev,
   466→   466→      roomTypeId,
   467→   467→      roomRate: roomType?.baseRate || 0,
   468→   468→    }))
   469→   469→  }
   470→   470→
   471→   471→  // ─── Render ─────────────────────────────────────────────────────────
   472→   472→
   473→   473→  return (
