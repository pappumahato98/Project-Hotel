   161→// ─── Helpers ────────────────────────────────────────────────────────────
   162→
   163→function guestFullName(g: FolioGuest | null | undefined): string {
   164→  if (!g) return '—'
   165→  return `${g.firstName} ${g.lastName}`
   166→}
   167→
   168→function folioCharges(f: Folio): number {
   169→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   170→}
   171→
   172→function folioPayments(f: Folio): number {
   173→  return f.payments.reduce((s, p) => s + p.amount, 0)
   174→}
   175→
   176→function folioOutstanding(f: Folio): number {
   177→  return folioCharges(f) - folioPayments(f)
   178→}
   179→
   180→function isVoidedTransaction(t: FolioTransaction): boolean {
   181→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   182→}
   183→
   184→function isVoidedPayment(p: FolioPayment): boolean {
   185→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   186→}
   187→
   188→function useDebounce<T>(value: T, delay: number): T {
   189→  const [debounced, setDebounced] = useState(value)
   190→  useEffect(() => {
   191→    const timer = setTimeout(() => setDebounced(value), delay)
   192→    return () => clearTimeout(timer)
   193→  }, [value, delay])
   194→  return debounced
   195→}

  1586→          </TabsContent>
  1587→
  1588→          {/* Notes Tab */}
  1589→          <TabsContent value="notes" className="m-0">

  1591→              <div className="space-y-2">

  1612→        </Tabs>

  1614→    </div>
