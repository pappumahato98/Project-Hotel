   158→// ─── Component ──────────────────────────────────────────────────────────
   159→
   160→export function ArrivalsView() {
   161→  const queryClient = useQueryClient()
   162→  const { settings } = useSettingsStore()
   163→  const { navigateTo } = useNavigationStore()
   164→  const today = getTodayString()
   165→
   166→  // Dialog states
   167→  const [roomPickerOpen, setRoomPickerOpen] = useState(false)
   168→  const [checkInDialogOpen, setCheckInDialogOpen] = useState(false)
   169→  const [walkInDialogOpen, setWalkInDialogOpen] = useState(false)
   170→  const [keyCardDialogOpen, setKeyCardDialogOpen] = useState(false)
   171→
   172→  // Selection states
   173→  const [selectedArrival, setSelectedArrival] = useState<Arrival | null>(null)
   174→  const [selectedRoomId, setSelectedRoomId] = useState<string>('')
   175→
   176→  // Check-in form states
   177→  const [earlyCheckIn, setEarlyCheckIn] = useState(false)
   178→  const [checkOutTime, setCheckOutTime] = useState(settings.defaultCheckOut)
   179→  const [roomPreference, setRoomPreference] = useState('')
   180→  const [pillowType, setPillowType] = useState('')
   181→  const [wakeupCall, setWakeupCall] = useState('')
   182→
   183→  // Key card states
   184→  const [keyCardRoomNumber, setKeyCardRoomNumber] = useState('')
   185→  const [keyCardIssued, setKeyCardIssued] = useState(false)
   186→  const [keyCardProcessed, setKeyCardProcessed] = useState(false)
   187→
   188→  // Mobile detail view state
   189→  const [mobileShowDetail, setMobileShowDetail] = useState(false)
   190→
   191→  // Walk-in form state
   192→  const [walkInForm, setWalkInForm] = useState<WalkInForm>(initialWalkInForm)
   193→  const [walkInCheckOutOpen, setWalkInCheckOutOpen] = useState(false)
   194→
   195→  // ─── Queries ────────────────────────────────────────────────────────
   196→
   197→  const { data, isLoading } = useQuery({
   198→    queryKey: ['arrivals', today],
   199→    queryFn: async () => {
   200→      const params = new URLSearchParams({ status: 'confirmed', checkInDate: today })
   201→      return apiFetch(`/api/reservations?${params.toString()}`)
   202→    },
   203→    refetchInterval: 30000,
   204→  })
   205→
   206→  const { data: roomsData } = useQuery({
   207→    queryKey: ['rooms', 'vacant'],
   208→    queryFn: () => apiFetch('/api/rooms'),
   209→  })
   210→
   211→  const arrivals: Arrival[] = data?.reservations || []
   212→  const allRooms: AvailableRoom[] = roomsData?.rooms || []
   213→  const availableRooms = allRooms.filter(
   214→    (r) => r.number && r.type && (r.status === 'vacant_clean' || r.status === 'inspected')
   215→  )
   216→
   217→  // ─── VIP Priority Sorting ───────────────────────────────────────────
   218→
   219→  const sortedArrivals = useMemo(() => {
   220→    return [...arrivals].sort((a, b) => {
   221→      const aVip = a.guest?.vipLevel && a.guest.vipLevel !== 'none'
   222→      const bVip = b.guest?.vipLevel && b.guest.vipLevel !== 'none'
   223→      if (aVip && !bVip) return -1
   224→      if (!aVip && bVip) return 1
   225→      return 0
   226→    })
   227→  }, [arrivals])
   228→
   229→  // ─── Stats ──────────────────────────────────────────────────────────
   230→
   231→  const totalArrivals = arrivals.length
   232→  const vipArrivals = arrivals.filter((a) => a.guest?.vipLevel && a.guest.vipLevel !== 'none').length
   233→  const unassignedArrivals = arrivals.filter((a) => !a.room).length
   234→  const pendingArrivals = arrivals.filter((a) => a.status === 'confirmed').length
   235→
   236→  // ─── Mutations ──────────────────────────────────────────────────────
   237→
   238→  const assignRoomMutation = useMutation({
   239→    mutationFn: async ({ reservationId, roomId, specialRequests }: { reservationId: string; roomId: string; specialRequests?: string }) => {
   240→      const payload: Record<string, unknown> = { roomId, status: 'checked_in' }
   241→      if (specialRequests !== undefined) {
   242→        payload.specialRequests = specialRequests
   243→      }
   244→      return apiFetch(`/api/reservations/${reservationId}`, {
   245→        method: 'PATCH',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify(payload),
   248→      })
   249→    },
   250→    onSuccess: (_data, variables) => {
   251→      invalidate.afterReservationChange(queryClient)
   252→      setRoomPickerOpen(false)
   253→      setCheckInDialogOpen(false)
   254→
   255→      // Find the room number for key card dialog
   256→      const room = availableRooms.find((r) => r.id === variables.roomId)
   257→      const roomNumber = room?.number || selectedArrival?.room?.number || ''
   258→      openKeyCardDialog(roomNumber)
   259→    },
   260→  })
   261→
   262→  const checkInMutation = useMutation({
   263→    mutationFn: async ({ reservationId, specialRequests }: { reservationId: string; specialRequests?: string }) => {
   264→      const payload: Record<string, unknown> = { status: 'checked_in' }
   265→      if (specialRequests !== undefined) {
   266→        payload.specialRequests = specialRequests
   267→      }
   268→      return apiFetch(`/api/reservations/${reservationId}`, {
   269→        method: 'PATCH',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify(payload),
   272→      })
   273→    },
   274→    onSuccess: (_data, variables) => {
   275→      invalidate.afterCheckIn(queryClient)
   276→      setCheckInDialogOpen(false)
   277→
   278→      const roomNumber = selectedArrival?.room?.number || ''
   279→      openKeyCardDialog(roomNumber)
   280→    },
   281→  })
   282→
   283→  const walkInMutation = useMutation({
   284→    mutationFn: async (form: WalkInForm) => {
   285→      // 1. Create guest
   286→      const guestData = await apiFetch('/api/guests', {
   287→        method: 'POST',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify({
   290→          firstName: form.firstName,
   291→          lastName: form.lastName,
   292→          phone: form.phone || null,
   293→          email: form.email || null,
   294→          nationality: form.nationality || null,
   295→          idType: form.idType || null,
   296→          idNumber: form.idNumber || null,
   297→        }),
   298→      })
   299→      const guestId = guestData.guest.id
   300→
   301→      // 2. Find a vacant room
   302→      const vacantRoom = availableRooms[0]
   303→      if (!vacantRoom) throw new Error('No available rooms')
   304→
   305→      // 3. Create reservation as checked_in (walk-in)
   306→      const checkOutDate = form.checkOutDate || (() => {
   307→        const d = new Date()
   308→        d.setDate(d.getDate() + 1)
   309→        return d
   310→      })()
   311→
   312→      const resData = await apiFetch('/api/reservations', {
   313→        method: 'POST',
   314→        headers: { 'Content-Type': 'application/json' },
   315→        body: JSON.stringify({
   316→          guestId,
   317→          roomId: vacantRoom.id,
   318→          adults: form.adults,
   319→          children: form.children,
   320→          checkIn: new Date().toISOString(),
   321→          checkOut: checkOutDate.toISOString(),
   322→          roomRate: form.roomRate,
   323→          source: 'walk_in',
   324→          reservationType: 'walk_in',
   325→          guaranteed: false,
   326→        }),
   327→      })
   328→      return { roomNumber: vacantRoom.number, reservation: resData.reservation }
   329→    },
   330→    onSuccess: (result) => {
   331→      invalidate.afterCheckIn(queryClient)
   332→      setWalkInDialogOpen(false)
   333→      setWalkInForm(initialWalkInForm)
   334→
   335→      // Open key card dialog
   336→      openKeyCardDialog(result.roomNumber)
   337→    },
   338→    onError: (error) => {
   339→      toast.error('Walk-in check-in failed', {
   340→        description: error instanceof Error ? error.message : 'Unknown error',
   341→      })
   342→    },
   343→  })
   344→
   345→  // ─── Handlers ──────────────────────────────────────────────────────
   346→
   347→  const handleAssignRoom = (arrival: Arrival) => {
   348→    setSelectedArrival(arrival)
   349→    setRoomPickerOpen(true)
   350→  }
   351→
   352→  const handleQuickCheckIn = (arrival: Arrival) => {
   353→    if (!arrival.room) {
   354→      setSelectedArrival(arrival)
   355→      setRoomPickerOpen(true)
   356→      return
   357→    }
   358→    setSelectedArrival(arrival)
   359→    resetCheckInForm()
   360→    setCheckInDialogOpen(true)
   361→  }
   362→
   363→  const handleOpenWalkIn = () => {
   364→    setWalkInForm(initialWalkInForm)
   365→    setWalkInDialogOpen(true)
   366→  }
   367→
   368→  const handleNavigateToCheckIn = (arrival: Arrival) => {
   369→    useFrontDeskContextStore.getState().setPrefillReservationId(arrival.id)
   370→    navigateTo('front-desk', 'check-in')
   371→  }
   372→
   373→  const handleViewLedger = (arrival: Arrival) => {
   374→    useGuestLedgerContextStore.getState().setGuestLedgerContext({
   375→      guestId: arrival.guest?.id || '',
   376→      guestName: `${arrival.guest?.firstName || ''} ${arrival.guest?.lastName || ''}`.trim(),
   377→    })
   378→    navigateTo('front-desk', 'guest-ledger')
   379→  }
   380→
   381→  const buildSpecialRequests = (arrival: Arrival): string => {
   382→    const parts: string[] = []
   383→
   384→    // Preserve existing requests
   385→    if (arrival.specialRequests) {
   386→      parts.push(arrival.specialRequests)
   387→    }
   388→
   389→    // Early check-in note
   390→    if (earlyCheckIn) {
   391→      parts.push(`[EARLY CHECK-IN] Checked in before ${settings.defaultCheckIn}`)
   392→    }
   393→
   394→    // Check-out time override
   395→    if (checkOutTime !== settings.defaultCheckOut) {
   396→      parts.push(`[CHECK-OUT OVERRIDE] Preferred check-out: ${checkOutTime}`)
   397→    }
   398→
   399→    // Room preference
   400→    if (roomPreference) {
   401→      const prefLabel = ROOM_PREFERENCES.find((p) => p.value === roomPreference)?.label || roomPreference
   402→      parts.push(`[PREFERENCE] Room: ${prefLabel}`)
   403→    }
   404→
   405→    // Pillow type
   406→    if (pillowType) {
   407→      const pillowLabel = PILLOW_TYPES.find((p) => p.value === pillowType)?.label || pillowType
   408→      parts.push(`[PREFERENCE] Pillow: ${pillowLabel}`)
   409→    }
   410→
   411→    // Wake-up call
   412→    if (wakeupCall) {
   413→      parts.push(`[SERVICE] Wake-up call requested at ${wakeupCall}`)
   414→    }
   415→
   416→    return parts.join(' | ')
   417→  }
   418→
   419→  const confirmCheckIn = () => {
   420→    if (!selectedArrival) return
   421→
   422→    const updatedRequests = buildSpecialRequests(selectedArrival)
   423→
   424→    if (selectedRoomId) {
   425→      assignRoomMutation.mutate({
   426→        reservationId: selectedArrival.id,
   427→        roomId: selectedRoomId,
   428→        specialRequests: updatedRequests,
   429→      })
   430→    } else if (selectedArrival.room) {
   431→      checkInMutation.mutate({
   432→        reservationId: selectedArrival.id,
   433→        specialRequests: updatedRequests,
   434→      })
   435→    }
   436→  }
   437→
   438→  const openKeyCardDialog = (roomNumber: string) => {
   439→    setKeyCardRoomNumber(roomNumber)
   440→    setKeyCardIssued(false)
   441→    setKeyCardProcessed(false)
   442→    setKeyCardDialogOpen(true)
   443→  }
   444→
   445→  const handleKeyCardConfirm = () => {
   446→    setKeyCardProcessed(true)
   447→    setKeyCardDialogOpen(false)
   448→    toast.success(`Check-in complete! Welcome letter sent to Room ${keyCardRoomNumber}`, {
   449→      duration: 5000,
   450→      icon: <Mail className="size-4" />,
   451→    })
   452→    setSelectedArrival(null)
   453→    setSelectedRoomId('')
   454→  }
   455→
   456→  const resetCheckInForm = () => {
   457→    setEarlyCheckIn(false)
   458→    setCheckOutTime(settings.defaultCheckOut)
   459→    setRoomPreference('')
   460→    setPillowType('')
   461→    setWakeupCall('')
   462→  }
   463→
   464→  const handleWalkInRoomTypeChange = (roomTypeId: string) => {
   465→    const roomType = ROOM_TYPES.find((rt) => rt.id === roomTypeId)
   466→    setWalkInForm((prev) => ({
   467→      ...prev,
   468→      roomTypeId,
   469→      roomRate: roomType?.baseRate || 0,
   470→    }))
   471→  }
   472→
   473→  const handleSelectArrival = (arrival: Arrival) => {
   474→    setSelectedArrival(arrival)
   475→    setSelectedRoomId('')
   476→    setMobileShowDetail(true)
   477→  }
   478→
   479→  const handleBackToList = () => {
   480→    setMobileShowDetail(false)
   481→  }
   482→
   483→  // ─── Render ─────────────────────────────────────────────────────────
   484→
   485→  return (
   486→    <div className="flex flex-col gap-3">
   487→      {/* Header */}
   488→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 shrink-0">
   489→        <div>
   490→          <h2 className="text-sm font-semibold text-gray-700 dark:text-gray-200">Today&apos;s Arrivals</h2>
   491→          <p className="text-xs text-muted-foreground">
   492→            Guest check-ins scheduled for {formatDate(today)}
   493→          </p>
   494→        </div>
   495→        <div className="flex items-center gap-2 shrink-0">
   496→          <Badge variant="secondary" className="text-[10px]">
   497→            {totalArrivals} expected
   498→          </Badge>
   499→          {vipArrivals > 0 && (
   500→            <Badge variant="outline" className="text-[10px] border-amber-300 text-amber-700 dark:text-amber-400">
   501→              <Crown className="size-3 mr-0.5" />
   502→              {vipArrivals} VIP
   503→            </Badge>
   504→          )}
   505→          {unassignedArrivals > 0 && (
   506→            <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-600 dark:text-amber-400">
   507→              <AlertTriangle className="size-3 mr-0.5" />
   508→              {unassignedArrivals} unassigned
   509→            </Badge>
   510→          )}
   511→          <Button onClick={handleOpenWalkIn} size="sm" className="gap-1.5">
   512→            <Footprints className="size-3.5" />
   513→            <span className="hidden sm:inline">Walk-in</span>
   514→          </Button>
   515→        </div>
   516→      </div>
   517→
   518→      {/* ─── Split Screen Body ─── */}
   519→      <div className="h-[calc(100vh-14rem)] min-h-[400px] flex flex-col lg:flex-row gap-3 overflow-hidden">
   520→
   521→        {/* ─── Left Panel: Arrivals List ─── */}
   522→        <div className={cn(
   523→          'flex flex-col min-h-0 border rounded-lg bg-card overflow-hidden',
   524→          'w-full lg:w-[42%] xl:w-[38%]',
   525→          mobileShowDetail && selectedArrival && 'hidden lg:flex',
   526→        )}>
   527→          {/* List header */}
   528→          <div className="px-3 py-2 border-b shrink-0 flex items-center justify-between bg-muted/30">
   529→            <span className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   530→              Expected Arrivals
   531→            </span>
   532→            <span className="text-[10px] text-muted-foreground">{sortedArrivals.length} guests</span>
   533→          </div>
   534→          {/* List body */}
   535→          <div className="flex-1 overflow-y-auto">
   536→            {isLoading ? (
   537→              <div className="p-2 space-y-2">
   538→                {Array.from({ length: 4 }).map((_, i) => (
   539→                  <Skeleton key={i} className="h-16 w-full rounded-lg" />
   540→                ))}
   541→              </div>
   542→            ) : sortedArrivals.length === 0 ? (
   543→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   544→                <LogIn className="size-8 mb-2 opacity-50" />
   545→                <p className="text-sm">No arrivals scheduled for today</p>
   546→              </div>
   547→            ) : (
   548→              <div className="divide-y">
   549→                {sortedArrivals.map((arrival) => {
   550→                  const isUnassigned = !arrival.room
   551→                  const isVip = arrival.guest?.vipLevel && arrival.guest.vipLevel !== 'none'
   552→                  const isSelected = selectedArrival?.id === arrival.id
   553→                  return (
   554→                    <button
   555→                      key={arrival.id}
   556→                      type="button"
   557→                      onClick={() => handleSelectArrival(arrival)}
