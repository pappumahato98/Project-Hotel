   181→// ─── Helpers ────────────────────────────────────────────────────────────
   182→
   183→function guestFullName(g: FolioGuest | null | undefined): string {
   184→  if (!g) return '—'
   185→  return `${g.firstName} ${g.lastName}`
   186→}
   187→
   188→function folioCharges(f: Folio): number {
   189→  return f.transactions.reduce((s, t) => s + t.totalAmount, 0)
   190→}
   191→
   192→function folioPayments(f: Folio): number {
   193→  return f.payments.reduce((s, p) => s + p.amount, 0)
   194→}
   195→
   196→function folioOutstanding(f: Folio): number {
   197→  return folioCharges(f) - folioPayments(f)
   198→}
   199→
   200→function isVoidedTransaction(t: FolioTransaction): boolean {
   201→  return t.totalAmount === 0 && t.description.includes('[VOIDED')
   202→}
   203→
   204→function isVoidedPayment(p: FolioPayment): boolean {
   205→  return p.amount === 0 && p.reference?.includes('[VOIDED')
   206→}
   207→
   208→function useDebounce<T>(value: T, delay: number): T {
   209→  const [debounced, setDebounced] = useState(value)
   210→  useEffect(() => {
   211→    const timer = setTimeout(() => setDebounced(value), delay)
   212→    return () => clearTimeout(timer)
   213→  }, [value, delay])
   214→  return debounced
   215→}
