   214→   214→function TaskDetailDialog({ task, open, onOpenChange }: { task: HkTask | null; open: boolean; onOpenChange: (open: boolean) => void }) {
   215→   215→  const [checkedItems, setCheckedItems] = useState<Record<string, boolean>>({})
   216→   216→  const queryClient = useQueryClient()
   217→   217→
   218→   218→  const updateStatusMutation = useMutation({
   219→   219→    mutationFn: (body: { id: string; status: string; inspectedBy?: string }) =>
   220→   220→      apiFetch('/api/housekeeping', { method: 'POST', body }),
   221→   221→    onSuccess: () => {
   222→   222→      toast.success('Task status updated')
   223→   223→      queryClient.invalidateQueries({ queryKey: ['housekeeping-tasks'] })
   224→   224→      queryClient.invalidateQueries({ queryKey: ['housekeeping-rooms'] })
   225→   225→      onOpenChange(false)
   226→   226→    },
   227→   227→  })
   228→   228→
   229→   229→  if (!task) return null
   230→   230→
   231→   231→  const toggleCheck = (item: string) => {
   232→   232→    setCheckedItems((prev) => ({ ...prev, [item]: !prev[item] }))
   233→   233→  }
   234→   234→  const checkedCount = Object.values(checkedItems).filter(Boolean).length
   235→   235→
   236→   236→  return (
   237→   237→    <Dialog open={open} onOpenChange={onOpenChange}>
   238→   238→      <DialogContent className="sm:max-w-lg">
   239→   239→        <DialogHeader>
   240→   240→          <DialogTitle className="flex items-center gap-2">
   241→   241→            Room {task.room.number} — {formatTaskType(task.taskType)}
   242→   242→          </DialogTitle>
   243→   243→          <DialogDescription>
   244→   244→            Floor {task.room.floor}{task.room.wing ? ` · ${task.room.wing} Wing` : ''} · {task.room.type.name}
   245→   245→          </DialogDescription>
   246→   246→        </DialogHeader>
   247→   247→
   248→   248→        <div className="space-y-4">
   249→   249→          <div className="grid grid-cols-2 gap-3">
   250→   250→            <div className="rounded-lg border p-3 space-y-1">
   251→   251→              <p className="text-xs text-muted-foreground">Priority</p>
   252→   252→              <Badge variant="outline" className={cn(priorityColor(task.priority))}>
   253→   253→                {task.priority.charAt(0).toUpperCase() + task.priority.slice(1)}
   254→   254→              </Badge>
   255→   255→            </div>
   256→   256→            <div className="rounded-lg border p-3 space-y-1">
   257→   257→              <p className="text-xs text-muted-foreground">Status</p>
   258→   258→              <StatusBadge status={task.status} />
   259→   259→            </div>
   260→   260→            <div className="rounded-lg border p-3 space-y-1">
   261→   261→              <p className="text-xs text-muted-foreground">Assigned To</p>
   262→   262→              <p className="text-sm font-medium">{task.assignedTo || 'Unassigned'}</p>
   263→   263→            </div>
   264→   264→            <div className="rounded-lg border p-3 space-y-1">
   265→   265→              <p className="text-xs text-muted-foreground">Est. Time</p>
   266→   266→              <p className="text-sm font-medium">{task.estimatedMinutes} min</p>
   267→   267→            </div>
   268→   268→          </div>
   269→   269→
   270→   270→          <div className="space-y-2">
   271→   271→            <div className="flex items-center justify-between">
   272→   272→              <p className="text-sm font-semibold">Cleaning Checklist</p>
   273→   273→              <span className="text-xs text-muted-foreground">{checkedCount}/{CLEANING_CHECKLIST.length}</span>
   274→   274→            </div>
   275→   275→            <div className="rounded-lg border divide-y max-h-64 overflow-y-auto">
   276→   276→              {CLEANING_CHECKLIST.map((item) => (
   277→   277→                <label
   278→   278→                  key={item}

   298→   298→          )}
   299→   299→
   300→   300→          {/* Status Actions */}
   301→   301→          <div className="flex gap-2 pt-2">
   302→   302→            {task.status === 'pending' && (
   303→   303→              <Button
   304→   304→                size="sm"
   305→   305→                variant="outline"
   306→   306→                className="flex-1"

   314→   314→              <Button

   341→   341→function AttendantView({ tasks }: { tasks: HkTask[] }) {
   342→   342→  const attendants = new Map<string, HkTask[]>()
   343→   343→  tasks.forEach((task) => {
   344→   344→    const name = task.assignedTo || 'Unassigned'
   345→   345→    if (!attendants.has(name)) attendants.set(name, [])
   346→   346→    attendants.get(name)!.push(task)
   347→   347→  })
   348→   348→
   349→   349→  return (
   350→   350→    <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-3">
   351→   351→      {Array.from(attendants.entries()).map(([name, attTasks]) => (
   352→   352→        <Card key={name}>
   353→   353→          <CardContent className="p-3 space-y-2">
   354→   354→            <div className="flex items-center gap-3">
   355→   355→              <div className="flex h-9 w-9 items-center justify-center rounded-full bg-muted text-sm font-medium">
   356→   356→                {name.charAt(0)}
   357→   357→              </div>
   358→   358→              <div>
   359→   359→                <p className="text-sm font-semibold">{name}</p>
   360→   360→                <p className="text-xs text-muted-foreground">{attTasks.length} tasks assigned</p>
   361→   361→              </div>
   362→   362→            </div>
   363→   363→            {attTasks.map((task) => (
   364→   364→              <div key={task.id} className="flex items-center justify-between rounded-md border p-2 text-sm">
   365→   365→                <div className="flex items-center gap-2">
   366→   366→                  <BedDouble className="h-3.5 w-3.5 text-muted-foreground" />
   367→   367→                  <span className="font-medium">Room {task.room.number}</span>
   368→   368→                </div>
   369→   369→                <StatusBadge status={task.status} />
   370→   370→              </div>
   371→   371→            ))}
   372→   372→          </CardContent>
   373→   373→        </Card>
   374→   374→      ))}
   375→   375→    </div>
   376→   376→  )
